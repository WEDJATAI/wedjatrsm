     1→'use client'
     2→
     3→// ─── Vision Cameras tab — camera CRUD + detection settings + edge ────
     4→// Registering streams (rtsp/http) mapped to floor plans, live status
     5→// per camera, admin-tunable detection thresholds/timings, and the edge
     6→// ingest connection (endpoint + one-time-rotatable key).
     7→
     8→import { useState } from 'react'
     9→import { useMutation, useQueryClient } from '@tanstack/react-query'
    10→import type { UseMutationResult } from '@tanstack/react-query'
    11→import {
    12→  Cctv,
    13→  Copy,
    14→  KeyRound,
    15→  Loader2,
    16→  MonitorPlay,
    17→  Pencil,
    18→  Plus,
    19→  RotateCcw,
    20→  SlidersHorizontal,
    21→  Sparkles,
    22→  Trash2,
    23→} from 'lucide-react'
    24→import { toast } from 'sonner'
    25→
    26→import {
    27→  AlertDialog,
    28→  AlertDialogAction,
    29→  AlertDialogCancel,
    30→  AlertDialogContent,
    31→  AlertDialogDescription,
    32→  AlertDialogFooter,
    33→  AlertDialogHeader,
    34→  AlertDialogTitle,
    35→} from '@/components/ui/alert-dialog'
    36→import { Badge } from '@/components/ui/badge'
    37→import { Button } from '@/components/ui/button'
    38→import { Card, CardContent, CardDescription, CardHeader, CardTitle } from '@/components/ui/card'
    39→import {
    40→  Dialog,
    41→  DialogContent,
    42→  DialogDescription,
    43→  DialogFooter,
    44→  DialogHeader,
    45→  DialogTitle,
    46→} from '@/components/ui/dialog'
    47→import { Input } from '@/components/ui/input'
    48→import { Label } from '@/components/ui/label'
    49→import {
    50→  Select,
    51→  SelectContent,
    52→  SelectItem,
    53→  SelectTrigger,
    54→  SelectValue,
    55→} from '@/components/ui/select'
    56→import { Skeleton } from '@/components/ui/skeleton'
    57→import { Switch } from '@/components/ui/switch'
    58→import { apiFetch } from '@/lib/api'
    59→import { elapsedSince } from '@/lib/format'
    60→import { useI18n } from '@/lib/i18n'
    61→import type { FloorPlan, VisionCameraDTO, VisionConfigDTO } from '@/lib/types'
    62→import { cn } from '@/lib/utils'
    63→
    64→import { VisionErrorCard, VisionSkeletonGrid } from './vision-shared'
    65→import type { VisionIngestInfo } from './vision-view'
    66→import { cameraStatusClass } from './vision-utils'
    67→
    68→type SetupDemoMutation = UseMutationResult<
    69→  { results: { event_id: string; outcome: string; detail: string | null }[] },
    70→  Error,
    71→  void
    72→>
    73→
    74→type Props = {
    75→  cameras: VisionCameraDTO[]
    76→  isLoading: boolean
    77→  isError: boolean
    78→  refetch: () => void
    79→  config?: VisionConfigDTO
    80→  ingest?: VisionIngestInfo
    81→  configLoading: boolean
    82→  configError: boolean
    83→  configRefetch: () => void
    84→  floorPlans: FloorPlan[]
    85→  setupDemoMutation: SetupDemoMutation
    86→}
    87→
    88→const CONFIG_FIELDS = [
    89→  { key: 'highConfidence', unit: '0–1' },
    90→  { key: 'mediumConfidence', unit: '0–1' },
    91→  { key: 'vacancyDelaySeconds', unit: 's' },
    92→  { key: 'movementDedupeMinutes', unit: 'min' },
    93→  { key: 'movementCooldownMinutes', unit: 'min' },
    94→  { key: 'serviceDelayMinutes', unit: 'min' },
    95→  { key: 'maxEventAgeSeconds', unit: 's' },
    96→  { key: 'manualHoldMinutes', unit: 'min' },
    97→] as const
    98→
    99→type ConfigKey = (typeof CONFIG_FIELDS)[number]['key']
   100→type ConfigForm = Record<ConfigKey, string>
   101→
   102→export default function VisionCamerasTab({
   103→  cameras,
   104→  isLoading,
   105→  isError,
   106→  refetch,
   107→  config,
   108→  ingest,
   109→  configLoading,
   110→  configError,
   111→  configRefetch,
   112→  floorPlans,
   113→  setupDemoMutation,
   114→}: Props) {
   115→  const { t } = useI18n()
   116→  const queryClient = useQueryClient()
   117→
   118→  const [formOpen, setFormOpen] = useState(false)
   119→  const [editingCamera, setEditingCamera] = useState<VisionCameraDTO | null>(null)
   120→  const [deleteTarget, setDeleteTarget] = useState<VisionCameraDTO | null>(null)
   121→  const [rotateOpen, setRotateOpen] = useState(false)
   122→  const [revealedKey, setRevealedKey] = useState<string | null>(null)
   123→
   124→  // camera form fields
   125→  const [fCode, setFCode] = useState('')
   126→  const [fName, setFName] = useState('')
   127→  const [fUrl, setFUrl] = useState('')
   128→  const [fFloor, setFFloor] = useState<string>('none')
   129→
   130→  // detection-settings form (synced from the server config during render)
   131→  const [configForm, setConfigForm] = useState<ConfigForm | null>(null)
   132→  const [configSynced, setConfigSynced] = useState<string | null>(null)
   133→  const configFingerprint = config ? JSON.stringify(config) : null
   134→  if (config && configFingerprint !== configSynced) {
   135→    setConfigSynced(configFingerprint)
   136→    setConfigForm({
   137→      highConfidence: String(config.highConfidence),
   138→      mediumConfidence: String(config.mediumConfidence),
   139→      vacancyDelaySeconds: String(config.vacancyDelaySeconds),
   140→      movementDedupeMinutes: String(config.movementDedupeMinutes),
   141→      movementCooldownMinutes: String(config.movementCooldownMinutes),
   142→      serviceDelayMinutes: String(config.serviceDelayMinutes),
   143→      maxEventAgeSeconds: String(config.maxEventAgeSeconds),
   144→      manualHoldMinutes: String(config.manualHoldMinutes),
   145→    })
   146→  }
   147→
   148→  const invalidateCameras = async () => {
   149→    await Promise.all([
   150→      queryClient.invalidateQueries({ queryKey: ['vision-cameras'] }),
   151→      queryClient.invalidateQueries({ queryKey: ['vision-overview'] }),
   152→      queryClient.invalidateQueries({ queryKey: ['vision-zones'] }),
   153→    ])
   154→  }
   155→
   156→  const createCameraMutation = useMutation({
   157→    mutationFn: (body: { code: string; name: string; streamUrl?: string; floorPlanId?: number }) =>
   158→      apiFetch<{ camera: VisionCameraDTO }>('/api/vision/cameras', { body }),
   159→    onSuccess: async () => {
   160→      toast.success(t('vision.cameras.created'))
   161→      setFormOpen(false)
   162→      resetForm()
   163→      await invalidateCameras()
   164→    },
   165→    onError: (err: Error) => toast.error(err.message),
   166→  })
   167→
   168→  const updateCameraMutation = useMutation({
   169→    mutationFn: (vars: { id: number; body: Record<string, unknown> }) =>
   170→      apiFetch<{ camera: VisionCameraDTO }>(`/api/vision/cameras/${vars.id}`, {
   171→        method: 'PUT',
   172→        body: vars.body,
   173→      }),
   174→    onSuccess: async () => {
   175→      toast.success(t('vision.cameras.updated'))
   176→      await invalidateCameras()
   177→    },
   178→    onError: (err: Error) => toast.error(err.message),
   179→  })
   180→
   181→  const deleteCameraMutation = useMutation({
   182→    mutationFn: (id: number) => apiFetch<{ ok: boolean }>(`/api/vision/cameras/${id}`, { method: 'DELETE' }),
   183→    onSuccess: async () => {
   184→      toast.success(t('vision.cameras.deleted'))
   185→      setDeleteTarget(null)
   186→      await invalidateCameras()
   187→    },
   188→    onError: (err: Error) => toast.error(err.message),
   189→  })
   190→
   191→  const saveConfigMutation = useMutation({
   192→    mutationFn: (body: Partial<VisionConfigDTO>) =>
   193→      apiFetch<{ config: VisionConfigDTO }>('/api/vision/config', { method: 'PUT', body }),
   194→    onSuccess: async () => {
   195→      toast.success(t('vision.config.saved'))
   196→      await queryClient.invalidateQueries({ queryKey: ['vision-config'] })
   197→    },
   198→    onError: (err: Error) => toast.error(err.message),
   199→  })
   200→
   201→  const rotateKeyMutation = useMutation({
   202→    mutationFn: () =>
   203→      apiFetch<{ config: VisionConfigDTO; ingest: VisionIngestInfo }>('/api/vision/config', {
   204→        method: 'PUT',
   205→        body: { rotateIngestKey: true },
   206→      }),
   207→    onSuccess: (data) => {
   208→      toast.success(t('vision.edge.rotated'))
   209→      setRevealedKey(data.ingest.key)
   210→      setRotateOpen(false)
   211→      void queryClient.invalidateQueries({ queryKey: ['vision-config'] })
   212→    },
   213→    onError: (err: Error) => toast.error(err.message),
   214→  })
   215→
   216→  const resetForm = () => {
   217→    setFCode('')
   218→    setFName('')
   219→    setFUrl('')
   220→    setFFloor('none')
   221→    setEditingCamera(null)
   222→  }
   223→
   224→  const openAdd = () => {
   225→    resetForm()
   226→    setFormOpen(true)
   227→  }
   228→
   229→  const openEdit = (camera: VisionCameraDTO) => {
   230→    setEditingCamera(camera)
   231→    setFCode(camera.code)
   232→    setFName(camera.name)
   233→    setFUrl(camera.streamUrl ?? '')
   234→    setFFloor(camera.floorPlanId != null ? String(camera.floorPlanId) : 'none')
   235→    setFormOpen(true)
   236→  }
   237→
   238→  const submitCameraForm = () => {
   239→    if (fCode.trim() === '' || fName.trim() === '') {
   240→      toast.error(t('common.required'))
   241→      return
   242→    }
   243→    const floorPlanId = fFloor === 'none' ? undefined : Number(fFloor)
   244→    const streamUrl = fUrl.trim() === '' ? undefined : fUrl.trim()
   245→    if (editingCamera) {
   246→      updateCameraMutation.mutate({
   247→        id: editingCamera.id,
   248→        body: { name: fName.trim(), streamUrl, floorPlanId },
   249→      })
   250→    } else {
   251→      createCameraMutation.mutate({
   252→        code: fCode.trim(),
   253→        name: fName.trim(),
   254→        streamUrl,
   255→        floorPlanId,
   256→      })
   257→    }
   258→  }
   259→
   260→  const submitConfig = () => {
   261→    if (!configForm) return
   262→    const values: Partial<Record<ConfigKey, number>> = {}
   263→    for (const field of CONFIG_FIELDS) {
   264→      const raw = configForm[field.key].trim()
   265→      const num = Number(raw)
   266→      if (raw === '' || !Number.isFinite(num) || num < 0) {
   267→        toast.error(t('vision.config.numberInvalid'))
   268→        return
   269→      }
   270→      values[field.key] = num
   271→    }
   272→    if ((values.mediumConfidence as number) >= (values.highConfidence as number)) {
   273→      toast.error(t('vision.config.invalid'))
   274→      return
   275→    }
   276→    saveConfigMutation.mutate(values as Partial<VisionConfigDTO>)
   277→  }
   278→
   279→  const copyKey = async (key: string) => {
   280→    try {
   281→      await navigator.clipboard.writeText(key)
   282→      toast.success(t('vision.edge.copied'))
   283→    } catch {
   284→      toast.error(t('error.generic'))
   285→    }
   286→  }
   287→
   288→  if (isLoading) {
   289→    return <VisionSkeletonGrid count={6} />
   290→  }
   291→  if (isError) {
   292→    return <VisionErrorCard onRetry={refetch} />
   293→  }
   294→
   295→  return (
   296→    <section className="space-y-4">
   297→      {/* ── header row ── */}
   298→      <div className="flex flex-wrap items-center justify-between gap-2">
   299→        <p className="text-sm text-muted-foreground">{t('vision.cameras.subtitle')}</p>
   300→        <div className="flex flex-wrap items-center gap-2">
   301→          <Button variant="outline" className="h-11 gap-1.5" disabled={setupDemoMutation.isPending} onClick={() => setupDemoMutation.mutate()}>
   302→            {setupDemoMutation.isPending ? (
   303→              <Loader2 className="size-4 animate-spin" aria-hidden />
   304→            ) : (
   305→              <Sparkles className="size-4" aria-hidden />
   306→            )}
   307→            {t('vision.setupDemo')}
   308→          </Button>
   309→          <Button className="h-11 gap-1.5" onClick={openAdd}>
   310→            <Plus className="size-4" aria-hidden />
   311→            {t('vision.cameras.add')}
   312→          </Button>
   313→        </div>
   314→      </div>
   315→
   316→      {/* ── camera cards ── */}
   317→      {cameras.length === 0 ? (
   318→        <Card className="border-dashed">
   319→          <CardContent className="flex flex-col items-center gap-2 p-8 text-center">
   320→            <Cctv className="size-8 text-muted-foreground/40" aria-hidden />
   321→            <p className="font-medium">{t('vision.cameras.empty')}</p>
   322→            <p className="max-w-md text-sm text-muted-foreground">{t('vision.cameras.emptyHint')}</p>
   323→          </CardContent>
   324→        </Card>
   325→      ) : (
   326→        <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-3">
   327→          {cameras.map((camera) => (
   328→            <Card key={camera.id} className="gap-3">
   329→              <CardContent className="space-y-3 p-4">
   330→                <div className="flex items-start justify-between gap-2">
   331→                  <div className="min-w-0 space-y-1">
   332→                    <div className="flex flex-wrap items-center gap-2">
   333→                      <Badge variant="secondary" className="font-mono text-[10px] uppercase">
   334→                        {camera.code}
   335→                      </Badge>
   336→                      <Badge variant="outline" className={cameraStatusClass(camera.status)}>
   337→                        {t(`vision.cameras.status.${camera.status}`)}
   338→                      </Badge>
   339→                    </div>
   340→                    <p className="truncate font-semibold">{camera.name}</p>
   341→                    <p className="text-xs text-muted-foreground">
   342→                      {camera.floorPlanName ?? t('vision.cameras.floorPlanNone')} ·{' '}
   343→                      {t('vision.cameras.zoneCount', { n: camera.zoneCount })}
   344→                    </p>
   345→                  </div>
   346→                  <Switch
   347→                    checked={camera.active}
   348→                    aria-label={`${t('common.active')} — ${camera.name}`}
   349→                    onCheckedChange={(checked) =>
   350→                      updateCameraMutation.mutate({ id: camera.id, body: { active: checked } })
   351→                    }
   352→                  />
   353→                </div>
   354→                <div className="space-y-1 text-xs text-muted-foreground">
   355→                  <p>
   356→                    <span className="font-medium">{t('vision.cameras.lastSeen')}: </span>
   357→                    <span className="tabular-nums">
   358→                      {camera.lastSeenAt ? elapsedSince(camera.lastSeenAt) : t('vision.cameras.never')}
   359→                    </span>
   360→                  </p>
   361→                  {camera.lastError && (
   362→                    <p className="truncate text-rose-600" title={camera.lastError}>
   363→                      <span className="font-medium">{t('vision.cameras.lastError')}: </span>
   364→                      {camera.lastError}
   365→                    </p>
   366→                  )}
   367→                </div>
   368→                <div className="flex items-center justify-end gap-2">
   369→                  <Button
   370→                    variant="outline"
   371→                    size="sm"
   372→                    className="h-11 gap-1.5"
   373→                    onClick={() => openEdit(camera)}
   374→                  >
   375→                    <Pencil className="size-4" aria-hidden />
   376→                    {t('common.edit')}
   377→                  </Button>
   378→                  <Button
   379→                    variant="outline"
   380→                    size="sm"
   381→                    className="h-11 gap-1.5 border-rose-300 text-rose-700 hover:bg-rose-50 hover:text-rose-700"
   382→                    onClick={() => setDeleteTarget(camera)}
   383→                  >
   384→                    <Trash2 className="size-4" aria-hidden />
   385→                    {t('common.delete')}
   386→                  </Button>
   387→                </div>
   388→              </CardContent>
   389→            </Card>
   390→          ))}
   391→        </div>
   392→      )}
   393→
   394→      {/* ── detection settings + edge connection ── */}
   395→      <div className="grid items-start gap-4 lg:grid-cols-2">
   396→        <Card>
   397→          <CardHeader>
   398→            <CardTitle className="flex items-center gap-2 text-base">
   399→              <SlidersHorizontal className="size-4" aria-hidden />
   400→              {t('vision.config.title')}
   401→            </CardTitle>
   402→            <CardDescription>{t('vision.config.subtitle')}</CardDescription>
   403→          </CardHeader>
   404→          <CardContent className="space-y-4">
   405→            {configLoading ? (
   406→              <div className="space-y-3">
   407→                {CONFIG_FIELDS.map((f) => (
   408→                  <Skeleton key={f.key} className="h-11 w-full" />
   409→                ))}
   410→              </div>
   411→            ) : configError || !config || !configForm ? (
   412→              <div className="flex flex-col items-center gap-2 py-4 text-center">
   413→                <p className="text-sm text-muted-foreground">{t('vision.loadFailed')}</p>
   414→                <Button variant="outline" size="sm" onClick={configRefetch}>
   415→                  {t('common.retry')}
   416→                </Button>
   417→              </div>
   418→            ) : (
   419→              <>
   420→                <div className="grid gap-3 sm:grid-cols-2">
   421→                  {CONFIG_FIELDS.map((field) => (
   422→                    <div key={field.key} className="space-y-1">
   423→                      <Label htmlFor={`cfg-${field.key}`} className="text-xs">
   424→                        {t(`vision.config.${field.key}`)}
   425→                      </Label>
   426→                      <Input
   427→                        id={`cfg-${field.key}`}
   428→                        type="number"
   429→                        inputMode="decimal"
   430→                        step="any"
   431→                        min={0}
   432→                        className="h-11"
   433→                        value={configForm[field.key]}
   434→                        onChange={(e) =>
   435→                          setConfigForm((prev) =>
   436→                            prev ? { ...prev, [field.key]: e.target.value } : prev,
   437→                          )
   438→                        }
   439→                      />
   440→                      <p className="text-[11px] leading-snug text-muted-foreground">
   441→                        {t(`vision.config.${field.key}Hint`)}
   442→                      </p>
   443→                    </div>
   444→                  ))}
   445→                </div>
   446→                <Button
   447→                  className="h-11 w-full sm:w-auto sm:px-6"
   448→                  disabled={saveConfigMutation.isPending}
   449→                  onClick={submitConfig}
   450→                >
   451→                  {saveConfigMutation.isPending && (
   452→                    <Loader2 className="size-4 animate-spin" aria-hidden />
   453→                  )}
   454→                  {t('common.save')}
   455→                </Button>
   456→              </>
   457→            )}
   458→          </CardContent>
   459→        </Card>
   460→
   461→        {/* Edge connection */}
   462→        <Card>
   463→          <CardHeader>
   464→            <CardTitle className="flex items-center gap-2 text-base">
   465→              <KeyRound className="size-4" aria-hidden />
   466→              {t('vision.edge.title')}
   467→            </CardTitle>
   468→            <CardDescription>{t('vision.edge.subtitle')}</CardDescription>
   469→          </CardHeader>
   470→          <CardContent className="space-y-4">
   471→            {configLoading ? (
   472→              <Skeleton className="h-28 w-full" />
   473→            ) : configError || !ingest ? (
   474→              <div className="flex flex-col items-center gap-2 py-4 text-center">
   475→                <p className="text-sm text-muted-foreground">{t('vision.loadFailed')}</p>
   476→                <Button variant="outline" size="sm" onClick={configRefetch}>
   477→                  {t('common.retry')}
   478→                </Button>
   479→              </div>
   480→            ) : (
   481→              <>
   482→                <div className="space-y-2">
   483→                  <div className="flex items-center justify-between gap-2 text-xs">
   484→                    <span className="font-medium text-muted-foreground">{t('vision.edge.endpoint')}</span>
   485→                    <code className="rounded border bg-stone-100 px-2 py-0.5 font-mono text-[11px]">
   486→                      {ingest.endpoint}
   487→                    </code>
   488→                  </div>
   489→                  <div className="flex items-center justify-between gap-2 text-xs">
   490→                    <span className="font-medium text-muted-foreground">{t('vision.edge.key')}</span>
   491→                    <code className="max-w-[60%] truncate rounded border bg-stone-100 px-2 py-0.5 font-mono text-[11px]">
   492→                      {ingest.key}
   493→                    </code>
   494→                  </div>
   495→                  <p className="text-[11px] text-muted-foreground">{t('vision.edge.keyMasked')}</p>
   496→                </div>
   497→
   498→                <Button
   499→                  variant="outline"
   500→                  className="h-11 gap-1.5"
   501→                  disabled={rotateKeyMutation.isPending}
   502→                  onClick={() => setRotateOpen(true)}
   503→                >
   504→                  <RotateCcw className="size-4" aria-hidden />
   505→                  {t('vision.edge.rotate')}
   506→                </Button>
   507→
   508→                {revealedKey && (
   509→                  <div className="space-y-2 rounded-xl border border-amber-400 bg-amber-50 p-3">
   510→                    <div className="flex items-center justify-between gap-2">
   511→                      <code className="min-w-0 flex-1 break-all font-mono text-xs font-bold text-amber-900">
   512→                        {revealedKey}
   513→                      </code>
   514→                      <Button
   515→                        variant="outline"
   516→                        size="sm"
   517→                        className="h-11 shrink-0 gap-1.5"
   518→                        onClick={() => void copyKey(revealedKey)}
   519→                      >
   520→                        <Copy className="size-4" aria-hidden />
   521→                        {t('vision.edge.copyKey')}
   522→                      </Button>
   523→                    </div>
   524→                    <p className="flex items-start gap-1.5 text-[11px] font-semibold text-amber-800">
   525→                      <MonitorPlay className="mt-0.5 size-3.5 shrink-0" aria-hidden />
   526→                      {t('vision.edge.storeSafe')}
   527→                    </p>
   528→                  </div>
   529→                )}
   530→                <p className="text-[11px] leading-snug text-muted-foreground">
   531→                  {t('vision.edge.authNote')}
   532→                </p>
   533→              </>
   534→            )}
   535→          </CardContent>
   536→        </Card>
   537→      </div>
   538→
   539→      {/* ── add / edit camera dialog ── */}
   540→      <Dialog open={formOpen} onOpenChange={(o) => { if (!o) { setFormOpen(false); resetForm() } }}>
   541→        <DialogContent className="sm:max-w-md">
   542→          <DialogHeader>
   543→            <DialogTitle>{editingCamera ? t('vision.cameras.edit') : t('vision.cameras.add')}</DialogTitle>
   544→            <DialogDescription>
   545→              {editingCamera ? t('vision.cameras.editDesc') : t('vision.cameras.codeHint')}
   546→            </DialogDescription>
   547→          </DialogHeader>
   548→          <div className="space-y-3">
