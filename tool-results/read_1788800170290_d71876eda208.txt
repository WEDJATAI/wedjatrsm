    65→    65→const clampPercent = (value: number, min: number, max: number) =>
    66→    66→  Math.min(max, Math.max(min, value))

    69→    69→  const parsed = Number(value)
    70→    70→  return Number.isFinite(parsed) ? parsed : fallback

    85→    85→  const [addTableOpen, setAddTableOpen] = useState(false)
    86→    86→  const [addTableName, setAddTableName] = useState('')

    88→    88→
    89→    89→  const [editOpen, setEditOpen] = useState(false)

   109→   109→  // ── Mutations ────────────────────────────────────────────────────────
   110→   110→
   111→   111→  const createPlanMutation = useMutation({
   112→   112→    mutationFn: (name: string) =>
   113→   113→      apiFetch<{ floorPlan: FloorPlan }>('/api/floorplans', { method: 'POST', body: { name } }),
   114→   114→    onSuccess: ({ floorPlan }) => {
   115→   115→      toast.success(`Floor plan "${floorPlan.name}" created`)

   121→   121→    onError: (mutationError) => toast.error(mutationError.message || 'Failed to create floor plan'),
   122→   122→  })
   123→   123→
   124→   124→  const togglePlanMutation = useMutation({
   125→   125→    mutationFn: ({ id, active }: { id: number; active: boolean }) =>
   126→   126→      apiFetch(`/api/floorplans/${id}`, { method: 'PUT', body: { active } }),
   127→   127→    onMutate: async ({ id, active }) => {
   128→   128→      await queryClient.cancelQueries({ queryKey: ['floorplans'] })
   129→   129→      const previous = queryClient.getQueryData<{ floorPlans: FloorPlan[] }>(['floorplans'])
   130→   130→      if (previous) {
   131→   131→        queryClient.setQueryData(['floorplans'], {
   132→   132→          floorPlans: previous.floorPlans.map((plan) =>
   133→   133→            plan.id === id ? { ...plan, active } : plan,
   134→   134→          ),
   135→   135→        })
   136→   136→      }
   137→   137→      return { previous }
   138→   138→    },
   139→   139→    onError: (mutationError, _variables, context) => {
   140→   140→      toast.error(mutationError.message || 'Failed to update floor plan')
   141→   141→      if (context?.previous) {
   142→   142→        queryClient.setQueryData(['floorplans'], context.previous)
   143→   143→      }
   144→   144→    },
   145→   145→    onSettled: () => {
   146→   146→      void queryClient.invalidateQueries({ queryKey: ['floorplans'] })
   147→   147→    },
   148→   148→  })
   149→   149→
   150→   150→  const addTableMutation = useMutation({
   151→   151→    mutationFn: (input: {
   152→   152→      floorPlanId: number
   153→   153→      name: string
   154→   154→      capacity: number
   155→   155→      positionX: number
   156→   156→      positionY: number

   158→   158→    onSuccess: ({ table }) => {
   159→   159→      toast.success(`Table "${table.name}" added`)
   160→   160→      void queryClient.invalidateQueries({ queryKey: ['floorplans'] })

   162→   162→      setAddTableName('')
   163→   163→      setAddTableCapacity('2')
   164→   164→    },
   165→   165→    onError: (mutationError) => toast.error(mutationError.message || 'Failed to add table'),
   166→   166→  })
   167→   167→
   168→   168→  const updateTableMutation = useMutation({
   169→   169→    mutationFn: ({
   170→   170→      id,
   171→   171→      body,
   172→   172→    }: {
   173→   173→      id: number
   174→   174→      body: Record<string, unknown>
   175→   175→      successMessage?: string
   176→   176→      closeDialog?: boolean

   178→   178→    onSuccess: (_result, variables) => {
   179→   179→      toast.success(variables.successMessage ?? 'Table updated')
   180→   180→      void queryClient.invalidateQueries({ queryKey: ['floorplans'] })

   182→   182→        setEditOpen(false)
   183→   183→        setEditingTable(null)
   184→   184→      }
   185→   185→    },
   186→   186→    onError: (mutationError) => toast.error(mutationError.message || 'Failed to update table'),
   187→   187→  })

   190→   190→
   191→   191→  function openTableDialog(table: RestaurantTable) {

   193→   193→    setEditCapacity(String(table.capacity))
   194→   194→    setEditX(String(round1(table.positionX)))
   195→   195→    setEditY(String(round1(table.positionY)))

   201→   201→  // ── Pointer-based dragging (mouse + touch) ───────────────────────────
   202→   202→
   203→   203→  function beginDrag(event: React.PointerEvent<HTMLDivElement>, table: RestaurantTable) {
   204→   204→    if (event.pointerType === 'mouse' && event.button !== 0) return
   205→   205→    const rect = canvasRef.current?.getBoundingClientRect()
   206→   206→    if (!rect) return
   207→   207→    event.preventDefault()
   208→   208→    try {
   209→   209→      event.currentTarget.setPointerCapture(event.pointerId)
   210→   210→    } catch {
   211→   211→      // Pointer already released — ignore.
   212→   212→    }
   213→   213→    const pointerX = ((event.clientX - rect.left) / rect.width) * 100
   214→   214→    const pointerY = ((event.clientY - rect.top) / rect.height) * 100
   215→   215→    const state: DragState = {
   216→   216→      table,
   217→   217→      pointerId: event.pointerId,
   218→   218→      startX: event.clientX,
   219→   219→      startY: event.clientY,
   220→   220→      offX: pointerX - table.positionX,
   221→   221→      offY: pointerY - table.positionY,
   222→   222→      x: table.positionX,
   223→   223→      y: table.positionY,
   224→   224→      movedPx: 0,
   225→   225→    }
   226→   226→    dragRef.current = state
   227→   227→    setDrag(state)
   228→   228→  }
   229→   229→
   230→   230→  function moveDrag(event: React.PointerEvent<HTMLDivElement>) {
   231→   231→    const current = dragRef.current
   232→   232→    if (!current || current.pointerId !== event.pointerId) return
   233→   233→    const rect = canvasRef.current?.getBoundingClientRect()
   234→   234→    if (!rect) return
   235→   235→    const pointerX = ((event.clientX - rect.left) / rect.width) * 100
   236→   236→    const pointerY = ((event.clientY - rect.top) / rect.height) * 100
   237→   237→    const next: DragState = {
   238→   238→      ...current,
   239→   239→      x: clampPercent(pointerX - current.offX, 2, 98),
   240→   240→      y: clampPercent(pointerY - current.offY, 2, 98),
   241→   241→      movedPx: Math.max(
   242→   242→        current.movedPx,
   243→   243→        Math.hypot(event.clientX - current.startX, event.clientY - current.startY),
   244→   244→      ),
   245→   245→    }
   246→   246→    dragRef.current = next
   247→   247→    setDrag(next)
   248→   248→  }
   249→   249→
   250→   250→  function endDrag(event: React.PointerEvent<HTMLDivElement>, commit: boolean) {
   251→   251→    const current = dragRef.current
   252→   252→    if (!current || current.pointerId !== event.pointerId) return
   253→   253→    dragRef.current = null
   254→   254→    setDrag(null)
   255→   255→    if (!commit) return
   256→   256→    if (current.movedPx < 6) {
   257→   257→      // Treated as a click → open the table editor.
   258→   258→      openTableDialog(current.table)
   259→   259→      return
   260→   260→    }
   261→   261→    updateTableMutation.mutate({
   262→   262→      id: current.table.id,
   263→   263→      body: { positionX: round1(current.x), positionY: round1(current.y) },
   264→   264→      successMessage: 'Position saved',
   265→   265→    })
   266→   266→  }

   268→   268→  // ── Form submit handlers ─────────────────────────────────────────────
   269→   269→
   270→   270→  function handleCreatePlan() {
   271→   271→    const name = newPlanName.trim()
   272→   272→    if (!name) {
   273→   273→      toast.error('Floor plan name is required')
   274→   274→      return
   275→   275→    }
   276→   276→    createPlanMutation.mutate(name)
   277→   277→  }
   278→   278→
   279→   279→  function handleAddTable() {
   280→   280→    if (!selectedPlan) return
   281→   281→    const name = addTableName.trim()
   282→   282→    if (!name) {
   283→   283→      toast.error('Table name is required')
   284→   284→      return
   285→   285→    }
   286→   286→    const parsedCapacity = Number(addTableCapacity)
   287→   287→    const capacity = Math.max(
   288→   288→      1,
   289→   289→      Math.round(Number.isFinite(parsedCapacity) && parsedCapacity > 0 ? parsedCapacity : 2),
   290→   290→    )
   291→   291→    // Spawn at a center-ish random spot (30-70%).
   292→   292→    const positionX = round1(30 + Math.random() * 40)
   293→   293→    const positionY = round1(30 + Math.random() * 40)
   294→   294→    addTableMutation.mutate({ floorPlanId: selectedPlan.id, name, capacity, positionX, positionY })
   295→   295→  }
   296→   296→
   297→   297→  function handleSaveTable() {
   298→   298→    if (!editingTable) return
   299→   299→    const name = editName.trim()
   300→   300→    if (!name) {
   301→   301→      toast.error('Table name is required')
   302→   302→      return
   303→   303→    }
   304→   304→    const parsedCapacity = Number(editCapacity)
   305→   305→    const capacity = Math.max(
   306→   306→      1,
   307→   307→      Math.round(
   308→   308→        Number.isFinite(parsedCapacity) && parsedCapacity > 0 ? parsedCapacity : editingTable.capacity,
   309→   309→      ),
   310→   310→    )
   311→   311→    const body: Record<string, unknown> = {
   312→   312→      name,
   313→   313→      capacity,
   314→   314→      positionX: clampPercent(round1(toNumber(editX, editingTable.positionX)), 0, 100),
   315→   315→      positionY: clampPercent(round1(toNumber(editY, editingTable.positionY)), 0, 100),
   316→   316→    }
   317→   317→    // 'occupied' is managed automatically by open orders — never send it.
   318→   318→    if (editStatus === 'free' || editStatus === 'reserved') {
   319→   319→      body.status = editStatus
   320→   320→    }
   321→   321→    updateTableMutation.mutate({
   322→   322→      id: editingTable.id,
   323→   323→      body,
   324→   324→      successMessage: 'Table updated',
   325→   325→      closeDialog: true,
   326→   326→    })
   327→   327→  }
   328→   328→
   329→   329→  function handleRemoveTable() {
   330→   330→    if (!editingTable) return
   331→   331→    updateTableMutation.mutate({
   332→   332→      id: editingTable.id,
   333→   333→      body: { active: false },
   334→   334→      successMessage: 'Table removed',
   335→   335→      closeDialog: true,
   336→   336→    })
   337→   337→  }
   338→   338→
   339→   339→  const creatingPlan = createPlanMutation.isPending
   340→   340→  const addingTable = addTableMutation.isPending
   341→   341→  const savingTable = updateTableMutation.isPending
   342→   342→
   343→   343→  return (
   344→   344→    <div className="mx-auto max-w-7xl space-y-6 p-4 sm:p-6">
   345→   345→      {/* Header */}
   346→   346→      <div className="flex flex-wrap items-center justify-between gap-3">
   347→   347→        <div>
   348→   348→          <h1 className="text-2xl font-bold tracking-tight">Floor Plans</h1>
   349→   349→          <p className="text-sm text-muted-foreground">
   350→   350→            Arrange tables per dining area — positions sync live to the POS table map.
   351→   351→          </p>
   352→   352→        </div>
   353→   353→        <Dialog open={newPlanOpen} onOpenChange={setNewPlanOpen}>
   354→   354→          <DialogTrigger asChild>
   355→   355→            <Button size="lg">
   356→   356→              <Plus className="size-4" />
   357→   357→              New Floor Plan
   358→   358→            </Button>
   359→   359→          </DialogTrigger>
   360→   360→          <DialogContent className="sm:max-w-sm">
   361→   361→            <DialogHeader>
   362→   362→              <DialogTitle>New floor plan</DialogTitle>
   363→   363→              <DialogDescription>
   364→   364→                Create a new dining area, e.g. Terrace or Mezzanine.
   365→   365→              </DialogDescription>
   366→   366→            </DialogHeader>
   367→   367→            <div className="grid gap-1.5">
   368→   368→              <Label htmlFor="plan-name">Name</Label>
   369→   369→              <Input
   370→   370→                id="plan-name"
   371→   371→                value={newPlanName}
   372→   372→                placeholder="e.g. Main Hall"
   373→   373→                onChange={(event) => setNewPlanName(event.target.value)}
   374→   374→                onKeyDown={(event) => {
   375→   375→                  if (event.key === 'Enter') handleCreatePlan()
   376→   376→                }}
   377→   377→              />
   378→   378→            </div>
   379→   379→            <DialogFooter>
   380→   380→              <Button variant="outline" onClick={() => setNewPlanOpen(false)}>
   381→   381→                Cancel
   382→   382→              </Button>
   383→   383→              <Button onClick={handleCreatePlan} disabled={creatingPlan || !newPlanName.trim()}>
   384→   384→                {creatingPlan ? <Loader2 className="size-4 animate-spin" /> : <Plus className="size-4" />}
   385→   385→                Create
   386→   386→              </Button>
   387→   387→            </DialogFooter>
   388→   388→          </DialogContent>
   389→   389→        </Dialog>
   390→   390→      </div>
   391→   391→
   392→   392→      {isError && (
   393→   393→        <div className="rounded-lg border border-destructive/40 bg-destructive/10 px-4 py-3 text-sm text-destructive">
   394→   394→          {error instanceof Error ? error.message : 'Failed to load floor plans'}
   395→   395→        </div>
   396→   396→      )}
   397→   397→
   398→   398→      {isLoading ? (
   399→   399→        <div className="grid gap-6 lg:grid-cols-[280px_1fr]">
   400→   400→          <div className="space-y-2">
   401→   401→            {Array.from({ length: 4 }).map((_, index) => (
   402→   402→              <Skeleton key={index} className="h-16 w-full" />
   403→   403→            ))}
   404→   404→          </div>
   405→   405→          <div className="space-y-3">
   406→   406→            <Skeleton className="h-8 w-48" />
   407→   407→            <Skeleton className="h-[460px] w-full rounded-xl" />
   408→   408→          </div>
   409→   409→        </div>
   410→   410→      ) : floorPlans.length === 0 ? (
   411→   411→        <Card className="flex flex-col items-center justify-center gap-3 border-dashed p-12 py-20 text-center">
   412→   412→          <Map className="size-14 text-muted-foreground/40" />
   413→   413→          <div className="space-y-1">
   414→   414→            <p className="font-semibold">No floor plans yet</p>
   415→   415→            <p className="text-sm text-muted-foreground">Create your first floor plan</p>
   416→   416→          </div>
   417→   417→          <Button onClick={() => setNewPlanOpen(true)}>
   418→   418→            <Plus className="size-4" />
   419→   419→            New Floor Plan
   420→   420→          </Button>
   421→   421→        </Card>
   422→   422→      ) : (
   423→   423→        <div className="grid gap-6 lg:grid-cols-[280px_1fr]">
   424→   424→          {/* Floor plan list */}
   425→   425→          <div className="space-y-2 self-start">
   426→   426→            <p className="px-1 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   427→   427→              Dining areas
   428→   428→            </p>
   429→   429→            {floorPlans.map((plan) => {
   430→   430→              const activeTableCount = plan.tables.filter((table) => table.active).length
   431→   431→              const isSelected = selectedPlan !== null && plan.id === selectedPlan.id
   432→   432→              return (
   433→   433→                <Card
   434→   434→                  key={plan.id}
   435→   435→                  role="button"
   436→   436→                  tabIndex={0}
   437→   437→                  onClick={() => setSelectedPlanId(plan.id)}
   438→   438→                  onKeyDown={(event) => {
   439→   439→                    if (event.key === 'Enter' || event.key === ' ') {
   440→   440→                      event.preventDefault()
   441→   441→                      setSelectedPlanId(plan.id)
   442→   442→                    }
   443→   443→                  }}
   444→   444→                  className={cn(
   445→   445→                    'cursor-pointer p-3 transition-colors hover:bg-accent/40',
   446→   446→                    isSelected && 'border-primary bg-primary/5',
   447→   447→                    !plan.active && 'opacity-60',
   448→   448→                  )}
   449→   449→                >
   450→   450→                  <div className="flex items-center justify-between gap-3">
   451→   451→                    <div className="min-w-0">
   452→   452→                      <p className="truncate text-sm font-semibold leading-tight">{plan.name}</p>
   453→   453→                      <p className="mt-0.5 text-xs text-muted-foreground">
   454→   454→                        {activeTableCount} tables
   455→   455→                      </p>
   456→   456→                    </div>
   457→   457→                    <div className="flex shrink-0 items-center" onClick={(e) => e.stopPropagation()}>
   458→   458→                      <Switch
   459→   459→                        checked={plan.active}
   460→   460→                        onCheckedChange={(checked) =>
