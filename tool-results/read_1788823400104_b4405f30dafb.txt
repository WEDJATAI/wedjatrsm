     4→import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query'

    27→  Dialog,

    33→} from '@/components/ui/dialog'

    47→
    48→type CartPanelProps = {
    49→  order: Order | null
    50→  orderLoading?: boolean
    51→  draft: DraftItem[]
    52→  table: { id: number | null; name: string }
    53→  onDraftChange: (draft: DraftItem[]) => void
    54→  onSend: () => void
    55→  onPay: () => void
    56→  /** Open the pre-payment guest check (pos-view flushes the draft first). */
    57→  onPrintCheck?: () => void
    58→  onCancel: () => void
    59→  canCancel?: boolean
    60→  sending?: boolean
    61→  userRole?: string
    62→}
    63→
    64→const STATUS_CHIP: Record<string, string> = {
    65→  new: 'bg-zinc-100 text-zinc-500',
    66→  preparing: 'bg-amber-100 text-amber-700 animate-pulse',
    67→  ready: 'bg-emerald-100 text-emerald-700',
    68→  served: 'bg-muted text-muted-foreground',
    69→}
    70→
    71→export default function CartPanel({
    72→  order,
    73→  orderLoading = false,
    74→  draft,
    75→  table,
    76→  onDraftChange,
    77→  onSend,
    78→  onPay,
    79→  onPrintCheck,
    80→  onCancel,
    81→  canCancel = false,
    82→  sending = false,
    83→  userRole,
    84→}: CartPanelProps) {
    85→  const queryClient = useQueryClient()
    86→  const { t, lang } = useI18n()
    87→  const orderId = order?.id ?? null
    88→
    89→  const [editing, setEditing] = useState<DraftItem | null>(null)
    90→  const [editQty, setEditQty] = useState('1')
    91→  const [editNotes, setEditNotes] = useState('')
    92→  const [editCourse, setEditCourse] = useState('main')
    93→
    94→  const [discountOpen, setDiscountOpen] = useState(false)
    95→  const [discountTab, setDiscountTab] = useState<'percent' | 'fixed'>('percent')
    96→  const [percentInput, setPercentInput] = useState('')
    97→  const [fixedInput, setFixedInput] = useState('')
    98→
    99→  // ── PIN-gated item deletion state ─────────────────────────────────
   100→  // Sent items may only be removed with the admin's 6-digit PIN; the
   101→  // dialog stays open on a wrong PIN so the user can retry.
   102→  const [pinDialogItem, setPinDialogItem] = useState<OrderItem | null>(null)
   103→  const [pinInput, setPinInput] = useState('')
   104→  const [pinWrong, setPinWrong] = useState(false)
   105→
   106→  // ── Item transfer ("Move items") state ────────────────────────────
   107→  const [moveMode, setMoveMode] = useState(false)
   108→  const [moveSelected, setMoveSelected] = useState<Set<number>>(() => new Set())
   109→  const [moveDialogOpen, setMoveDialogOpen] = useState(false)
   110→
   111→  const totals = computeCartTotals(order, draft)
   112→  const noItems = (order?.items.length ?? 0) + draft.length === 0
   113→
   114→  // Open orders (for the move-items target picker) — only fetched while the
   115→  // picker dialog is open; the query key matches the floor's live query.
   116→  const { data: openOrdersData, isLoading: openOrdersLoading } = useQuery({
   117→    queryKey: ['orders', 'open'],
   118→    queryFn: () => fetcher<{ orders: Order[] }>('/api/orders?status=open'),
   119→    enabled: moveDialogOpen,
   120→  })
   121→  const otherOpenOrders = (openOrdersData?.orders ?? []).filter((o) => o.id !== orderId)
   122→
   123→  // ── Server mutations (sent items) ─────────────────────────────────
   124→  const markServed = useMutation({
   125→    mutationFn: (itemId: number) =>
   126→      apiFetch<{ item: OrderItem }>(`/api/order-items/${itemId}`, {
   127→        method: 'PUT',
   128→        body: { status: 'served' },
   129→      }),
   130→    onSuccess: async (data) => {
   131→      toast.success(
   132→        t('pos.markedServedToast', {
   133→          name: data.item.product
   134→            ? localizedName(data.item.product.name, data.item.product.nameAr, lang)
   135→            : t('pos.item'),
   136→        }),
   137→      )
   138→      await queryClient.invalidateQueries({ queryKey: ['pos-order'] })
   139→      await queryClient.invalidateQueries({ queryKey: ['orders'] })
   140→    },
   141→    onError: (err: Error) => toast.error(err.message),
   142→  })
   143→
   144→  // Item removal is PIN-gated (PUT /api/orders/[id] { removeItemIds, removePin }):
   145→  // the dialog collects the 6-digit PIN and the API verifies it server-side
   146→  // (403 with a descriptive message on a wrong or missing PIN).
   147→  const removeItem = useMutation({
   148→    mutationFn: ({ itemId, pin }: { itemId: number; pin: string }) => {
   149→      if (orderId == null) throw new Error(t('pos.noActiveOrder'))
   150→      return apiFetch<{ order: Order }>(`/api/orders/${orderId}`, {
   151→        method: 'PUT',
   152→        body: { removeItemIds: [itemId], removePin: pin },
   153→      })
   154→    },
   155→    onSuccess: async ({ order: updated }) => {
   156→      queryClient.setQueryData(['pos-order', updated.id], { order: updated })
   157→      await queryClient.invalidateQueries({ queryKey: ['orders'] })
   158→      await queryClient.invalidateQueries({ queryKey: ['floorplans'] })
   159→      await queryClient.invalidateQueries({ queryKey: ['tables-status'] })
   160→      toast.success(t('pos.itemRemovedToast'))
   161→      closePinDialog()
   162→    },
   163→    onError: (err: Error) => {
   164→      // The API message is descriptive (403 wrong/missing PIN); the dialog
   165→      // stays open with the inline wrong-PIN hint so the user can retry.
   166→      toast.error(err.message)
   167→      setPinWrong(true)
   168→    },
   169→  })
   170→
   171→  // ── PIN dialog helpers ────────────────────────────────────────────
   172→  const openPinDialog = (item: OrderItem) => {
   173→    setPinDialogItem(item)
   174→    setPinInput('')
   175→    setPinWrong(false)
   176→  }
   177→
   178→  const closePinDialog = () => {
   179→    setPinDialogItem(null)
   180→    setPinInput('')
   181→    setPinWrong(false)
   182→  }
   183→
   184→  const confirmPin = () => {
   185→    if (!pinDialogItem || pinInput.length < DELETE_PIN_LENGTH || removeItem.isPending) return
   186→    removeItem.mutate({ itemId: pinDialogItem.id, pin: pinInput })
   187→  }
   188→
   189→  const applyDiscount = useMutation({
   190→    mutationFn: (discountAmount: number) => {
   191→      if (orderId == null) throw new Error(t('pos.noActiveOrder'))
   192→      return apiFetch<{ order: Order }>(`/api/orders/${orderId}`, {
   193→        method: 'PUT',
   194→        body: { discountAmount },
   195→      })
   196→    },
   197→    onSuccess: async ({ order: updated }) => {
   198→      queryClient.setQueryData(['pos-order', updated.id], { order: updated })
   199→      await queryClient.invalidateQueries({ queryKey: ['orders'] })
   200→      await queryClient.invalidateQueries({ queryKey: ['floorplans'] })
   201→      await queryClient.invalidateQueries({ queryKey: ['tables-status'] })
   202→      toast.success(t('pos.discountUpdatedToast'))
   203→      setDiscountOpen(false)
   204→    },
   205→    onError: (err: Error) => toast.error(err.message),
   206→  })
   207→
   208→  // ── Item transfer mutation (move sent items to another open order) ──
   209→  const transferItems = useMutation({
   210→    mutationFn: (vars: {
   211→      sourceId: number
   212→      itemIds: number[]
   213→      targetOrderId: number
   214→      targetLabel: string
   215→    }) =>
   216→      apiFetch<{ source: Order; target: Order }>(`/api/orders/${vars.sourceId}/transfer-items`, {
   217→        method: 'POST',
   218→        body: { itemIds: vars.itemIds, targetOrderId: vars.targetOrderId },
   219→      }),
   220→    onSuccess: async (_data, vars) => {
   221→      toast.success(
   222→        t('pos.itemsMovedToast', { n: vars.itemIds.length, target: vars.targetLabel }),
   223→      )
   224→      await Promise.all([
   225→        queryClient.invalidateQueries({ queryKey: ['orders'] }),
   226→        queryClient.invalidateQueries({ queryKey: ['orders', 'open'] }),
   227→        queryClient.invalidateQueries({ queryKey: ['floorplans'] }),
   228→        queryClient.invalidateQueries({ queryKey: ['tables-status'] }),
   229→        queryClient.invalidateQueries({ queryKey: ['pos-order', vars.sourceId] }),
   230→        queryClient.invalidateQueries({ queryKey: ['pos-order', vars.targetOrderId] }),
   231→      ])
   232→      exitMoveMode()
   233→    },
   234→    onError: (err: Error) => toast.error(err.message),
   235→  })
   236→
   237→  // ── Move-items helpers ───────────────────────────────────────────
   238→  const enterMoveMode = () => {
   239→    setMoveMode(true)
   240→    setMoveSelected(new Set())
   241→  }
   242→
   243→  const exitMoveMode = () => {
   244→    setMoveMode(false)
   245→    setMoveSelected(new Set())
   246→    setMoveDialogOpen(false)
   247→  }
   248→
   249→  const toggleMoveItem = (itemId: number) => {
   250→    setMoveSelected((prev) => {
   251→      const next = new Set(prev)
   252→      if (next.has(itemId)) next.delete(itemId)
   253→      else next.add(itemId)
   254→      return next
   255→    })
   256→  }
   257→
   258→  // Only ids that still exist on the (live-polled) order — stale ids are dropped.
   259→  const selectedMoveIds = (order?.items ?? []).filter((i) => moveSelected.has(i.id)).map((i) => i.id)
   260→  const moveCount = selectedMoveIds.length
   261→
   262→  const pickMoveTarget = (target: Order) => {
   263→    if (orderId == null || moveCount === 0) return
   264→    transferItems.mutate({
   265→      sourceId: orderId,
   266→      itemIds: selectedMoveIds,
   267→      targetOrderId: target.id,
   268→      targetLabel: target.table?.name ?? t('common.takeaway'),
   269→    })
   270→  }
   271→
   272→  // ── Draft helpers ─────────────────────────────────────────────────
   273→  const updateDraft = (key: string, patch: Partial<DraftItem>) =>
   274→    onDraftChange(draft.map((d) => (d.key === key ? { ...d, ...patch } : d)))
   275→
   276→  const changeQty = (d: DraftItem, delta: number) => {
   277→    const next = round2(d.quantity + delta)
   278→    if (next <= 0) onDraftChange(draft.filter((x) => x.key !== d.key))
   279→    else updateDraft(d.key, { quantity: next })
   280→  }
   281→
   282→  const openEdit = (d: DraftItem) => {
   283→    setEditing(d)
   284→    setEditQty(String(d.quantity))
   285→    setEditNotes(d.notes)
   286→    setEditCourse(d.course)
   287→  }
   288→
   289→  const saveEdit = () => {
   290→    if (!editing) return
   291→    const qty = round2(parseFloat(editQty))
   292→    if (!Number.isFinite(qty) || qty <= 0) return
   293→    updateDraft(editing.key, { quantity: qty, notes: editNotes.trim(), course: editCourse })
   294→    setEditing(null)
   295→  }
   296→
   297→  // Seed discount dialog inputs from the current order discount when opening.
   298→  const openDiscountDialog = () => {
   299→    const current = order?.discountAmount ?? 0
   300→    const subtotal = order?.subtotalAmount ?? 0
   301→    setPercentInput(current > 0 && subtotal > 0 ? String(round2((current / subtotal) * 100)) : '')
   302→    setFixedInput(current > 0 ? String(current) : '')
   303→    setDiscountOpen(true)
   304→  }
   305→
   306→  // Effective discount from the dialog inputs (clamped to [0, order subtotal]).
   307→  const orderSubtotal = round2(order?.subtotalAmount ?? 0)
   308→  const previewDiscount = (() => {
   309→    if (!order) return 0
   310→    const raw =
   311→      discountTab === 'percent'
   312→        ? (orderSubtotal * (parseFloat(percentInput) || 0)) / 100
   313→        : parseFloat(fixedInput) || 0
   314→    if (!Number.isFinite(raw) || raw < 0) return 0
   315→    return round2(Math.min(raw, orderSubtotal))
   316→  })()
   317→  const previewBase = round2(orderSubtotal - previewDiscount)
   318→  const previewVat = round2(previewBase * TAX_RATE)
   319→  const previewServiceTax = round2(previewBase * SERVICE_TAX_RATE)
   320→
   321→  return (
   322→    <div className="flex min-h-0 flex-1 flex-col">
   323→      {/* Header */}
   324→      <div className="flex shrink-0 items-center justify-between gap-2 border-b border-[#E2E2E0] px-4 py-3">
   325→        <div className="min-w-0">
   326→          <p className="truncate text-base font-bold">{table.name}</p>
   327→          {(order?.user?.name || userRole) && (
   328→            <p className="text-[11px] text-muted-foreground">
   329→              {t('pos.server')}:{' '}
   330→              {order?.user?.name ?? (userRole ? t(`role.${userRole}`) : '')}
   331→            </p>
   332→          )}
   333→        </div>
   334→        {order ? (
   335→          <Badge className="shrink-0 bg-[#714B67] text-white hover:bg-[#714B67]">
   336→            {t('common.order')} #{order.id}
   337→          </Badge>
   338→        ) : (
   339→          <Badge variant="secondary" className="shrink-0">
   340→            {t('pos.notSent')}
   341→          </Badge>
   342→        )}
   343→      </div>
   344→
   345→      {/* Body */}
   346→      <div className="rms-scroll min-h-0 flex-1 overflow-y-auto px-4">
   347→        {orderLoading && !order && draft.length === 0 ? (
   348→          <div className="space-y-3 py-4">
   349→            <Skeleton className="h-14 w-full" />
   350→            <Skeleton className="h-14 w-full" />
   351→            <Skeleton className="h-14 w-full" />
   352→          </div>
   353→        ) : noItems ? (
   354→          <div className="flex h-full min-h-[160px] flex-col items-center justify-center gap-2 py-8 text-muted-foreground">
   355→            <ShoppingBag className="size-10 opacity-30" />
   356→            <p className="text-sm">{t('pos.tapToAdd')}</p>
   357→          </div>
   358→        ) : (
   359→          <>
   360→            {/* Sent to kitchen */}
   361→            {order && order.items.length > 0 && (
   362→              <section className="py-2">
   363→                <div className="flex items-center justify-between gap-2">
   364→                  <h3 className="py-1 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   365→                    {t('pos.sentItems')} ({order.items.length})
   366→                  </h3>
   367→                  <div className="flex shrink-0 items-center gap-1">
   368→                    {!moveMode && (
   369→                      <Button
   370→                        variant="outline"
   371→                        size="sm"
   372→                        className="h-8 gap-1 px-2 text-xs border-[#714B67]/40 text-[#714B67] hover:bg-[#714B67]/10 hover:text-[#714B67]"
   373→                        title={t('pos.transferItems')}
   374→                        onClick={enterMoveMode}
   375→                      >
   376→                        <ArrowLeftRight className="size-3.5" /> {t('pos.transferItems')}
   377→                      </Button>
   378→                    )}
   379→                    {moveMode ? (
   380→                      <span className="text-[11px] text-muted-foreground">{t('pos.moveModeHint')}</span>
   381→                    ) : (
   382→                      canCancel && (
   383→                        <Button
   384→                          variant="ghost"
   385→                          size="sm"
   386→                          className="h-8 px-2 text-xs text-destructive hover:bg-destructive/10 hover:text-destructive"
   387→                          title={t('pos.cancelOrder')}
   388→                          onClick={onCancel}
   389→                        >
   390→                          <Ban className="size-3.5" /> {t('pos.cancelOrder')}
   391→                        </Button>
   392→                      )
   393→                    )}
   394→                  </div>
   395→                </div>
   396→                {order.items.map((item) => (
   397→                  <SentItemRow
   398→                    key={item.id}
   399→                    item={item}
   400→                    onServed={() => markServed.mutate(item.id)}
   401→                    onRemove={() => openPinDialog(item)}
   402→                    servedPending={markServed.isPending && markServed.variables === item.id}
   403→                    removePending={removeItem.isPending && removeItem.variables?.itemId === item.id}
   404→                    moveMode={moveMode}
   405→                    moveSelected={moveSelected.has(item.id)}
   406→                    onToggleMove={() => toggleMoveItem(item.id)}
   407→                  />
   408→                ))}
   409→              </section>
   410→            )}
   411→
   412→            {/* New items (draft) */}
   413→            {draft.length > 0 && (
   414→              <section className="py-2">
   415→                <h3 className="py-1 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   416→                  {t('pos.newItemsNotSent')} ({draft.length})
   417→                </h3>
   418→                {draft.map((d) => (
   419→                  <DraftRow
   420→                    key={d.key}
   421→                    item={d}
   422→                    onQty={(delta) => changeQty(d, delta)}
   423→                    onEdit={() => openEdit(d)}
   424→                    onRemove={() => onDraftChange(draft.filter((x) => x.key !== d.key))}
   425→                  />
   426→                ))}
   427→              </section>
   428→            )}
   429→          </>
   430→        )}
   431→      </div>
   432→
   433→      {/* Footer */}
   434→      <div className="shrink-0 space-y-1.5 border-t border-[#E2E2E0] p-4">
   435→        <SummaryRow label={t('money.subtotal')} value={formatCurrency(totals.subtotal)} />
   436→        <div className="flex items-center justify-between text-sm">
   437→          <span className="flex items-center gap-1 text-muted-foreground">
   438→            {t('money.discount')}
   439→            <Button
   440→              variant="ghost"
   441→              size="icon"
   442→              className="size-7 text-muted-foreground"
   443→              disabled={!order}
   444→              title={order ? t('pos.applyDiscount') : t('pos.sendFirst')}
   445→              onClick={openDiscountDialog}
   446→            >
   447→              <Pencil className="size-3.5" />
   448→            </Button>
   449→          </span>
   450→          <span className="tabular-nums text-muted-foreground">
   451→            − {formatCurrency(totals.discount)}
   452→          </span>
   453→        </div>
   454→        <SummaryRow label={t('money.tax')} value={formatCurrency(totals.tax)} />
   455→        <SummaryRow label={t('money.serviceTax')} value={formatCurrency(totals.serviceTax)} />
   456→        <div className="flex items-center justify-between border-t border-[#E2E2E0] pt-2">
   457→          <span className="text-sm font-semibold">{t('money.total')}</span>
   458→          <span className="text-lg font-bold tabular-nums">{formatCurrency(totals.total)}</span>
   459→        </div>
   460→        {order && order.paidAmount > 0 && (
   461→          <>
   462→            <SummaryRow
   463→              label={t('money.paid')}
   464→              value={formatCurrency(order.paidAmount)}
   465→              valueClassName="text-emerald-600"
   466→            />
   467→            <SummaryRow
   468→              label={t('money.remaining')}
   469→              value={formatCurrency(Math.max(0, order.remainingAmount))}
   470→              valueClassName="font-bold text-amber-600"
   471→            />
   472→          </>
   473→        )}
   474→
   475→        {moveMode ? (
   476→          /* ── Item-transfer footer ── */
   477→          <div className="flex gap-2 pt-1">
   478→            <Button
   479→              variant="outline"
   480→              className="h-14 flex-1 rounded-xl border-[#E2E2E0] text-sm"
   481→              onClick={exitMoveMode}
   482→              disabled={transferItems.isPending}
   483→            >
   484→              {t('common.cancel')}
   485→            </Button>
   486→            <Button
   487→              className="h-14 flex-[1.6] rounded-xl bg-[#714B67] text-base font-semibold text-white hover:bg-[#714B67]/90"
   488→              disabled={moveCount === 0 || transferItems.isPending}
   489→              onClick={() => setMoveDialogOpen(true)}
   490→            >
   491→              {transferItems.isPending ? (
   492→                <Loader2 className="animate-spin" />
   493→              ) : (
   494→                <ArrowLeftRight className="size-5" />
   495→              )}
   496→              <span className="truncate">{t('pos.moveNItems', { n: moveCount })}</span>
   497→            </Button>
   498→          </div>
   499→        ) : (
   500→          /* ── Normal actions ── */
   501→          <>
   502→            <Button
   503→              variant="secondary"
   504→              className="h-11 w-full rounded-xl"
   505→              disabled={draft.length === 0 || sending}
   506→              onClick={onSend}
   507→            >
   508→              {sending ? <Loader2 className="animate-spin" /> : <Send />} {t('pos.sendToKitchen')}
   509→              {draft.length > 0 && (
   510→                <Badge className="ms-1 h-5 min-w-5 rounded-full px-1.5 tabular-nums">
   511→                  {draft.reduce((n, d) => n + d.quantity, 0)}
   512→                </Badge>
   513→              )}
   514→            </Button>
   515→            <div className="flex gap-2">
   516→              <Button
   517→                variant="outline"
   518→                className="h-14 flex-1 rounded-xl border-[#E2E2E0] text-sm"
   519→                disabled={!order || sending || !onPrintCheck}
   520→                title={!order ? t('pos.printCheckDisabled') : t('pos.printCheckHint')}
   521→                onClick={onPrintCheck}
   522→              >
   523→                <Printer />
   524→                <span className="hidden sm:inline">{t('pos.printCheck')}</span>
   525→              </Button>
   526→              <Button

   766→              }}

   768→                if (e.key === 'Enter') {

   800→              {t('pos.pinConfirm')}
