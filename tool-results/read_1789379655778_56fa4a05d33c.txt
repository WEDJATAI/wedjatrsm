   412→  /** Append (or merge into) a draft line. Merging happens ONLY when the
   413→   *  product + course match, the line has no notes AND the modifier
   414→   *  signature is identical (same option ids in the same order) — otherwise
   415→   *  the optioned item becomes its own line. `price` stays the BASE product
   416→   *  price; `modifiers` carries the deltas (see lineUnitPrice). */
   417→  const appendDraftLine = (p: Product, quantity: number, modifiers?: SelectedModifier[]) => {
   418→    const sig = modifierSignature(modifiers)
   419→    setDraft((prev) => {
   420→      const course = guessCourse(p)
   421→      const existing = prev.find(
   422→        (d) =>
   423→          d.productId === p.id &&
   424→          !d.notes &&
   425→          d.course === course &&
   426→          modifierSignature(d.modifiers) === sig,
   427→      )
   428→      if (existing) {
   429→        return prev.map((d) =>
   430→          d.key === existing.key ? { ...d, quantity: round2(d.quantity + quantity) } : d,
   431→        )
   432→      }
   433→      return [
   434→        ...prev,
   435→        {
   436→          key: newDraftKey(),
   437→          productId: p.id,
   438→          name: p.name,
   439→          nameAr: p.nameAr ?? null,
   440→          price: p.price,
   441→          quantity,
   442→          notes: '',
   443→          course,
   444→          modifiers,
   445→        },
   446→      ]
   447→    })
   448→  }

   474→  const sendToKitchen = async (): Promise<Order | null> => {
   475→    if (draft.length === 0 || sending) return order
   476→    setSending(true)
   477→    try {
   478→      let result: { order: Order }
   479→      // Multi-table seating (Seat Party): create the order spanning ALL the
   480→      // selected tables (tableIds — primary first). Single-table and takeaway
   481→      // paths keep the classic tableId payload byte-identical.
   482→      const seatingList = seatingTables ?? null
   483→      const multiSeating = (seatingList?.length ?? 0) > 1
   484→      if (!order && multiSeating && seatingList) {
   485→        result = await apiFetch<{ order: Order }>('/api/orders', {
   486→          method: 'POST',
   487→          body: {
   488→            tableIds: seatingList.map((tb) => tb.id),
   489→            items: draftToPayload(draft),
   490→            guests: clampGuests(guestsDraft),
   491→          },
   492→        })
   493→      } else if (!order) {
   494→        result = await apiFetch<{ order: Order }>('/api/orders', {
   495→          method: 'POST',
   496→          body: {
   497→            tableId: selectedTable?.id ?? null,
   498→            items: draftToPayload(draft),
   499→            guests: clampGuests(guestsDraft),
   500→          },
   501→        })
   502→      } else {
   503→        result = await apiFetch<{ order: Order }>(`/api/orders/${order.id}`, {
   504→          method: 'PUT',
   505→          body: { addItems: draftToPayload(draft) },
   506→        })
   507→      }
   508→      setDraft([])
   509→      queryClient.setQueryData(['pos-order', result.order.id], { order: result.order })

   519→      toast.error(err instanceof Error ? err.message : t('pos.sendFailedToast'))
   520→      return null
   521→    } finally {
   522→      setSending(false)

   526→  const handlePay = async () => {
   527→    if (sending) return

   530→      current = await sendToKitchen()

   533→    if (!current) return
   534→    setPayOrder(current)
   535→    setPayOpen(true)
   536→  }

   539→  // then open the check modal with the fresh order state.
   540→  const handlePrintCheck = async () => {
   541→    if (sending) return
   542→    let current: Order | null = order
   543→    if (draft.length > 0) {
   544→      current = await sendToKitchen()
   545→      if (!current) return // sending failed → abort printing
   546→    }
   547→    if (!current) return
   548→    setCheckOrder(current)
   549→    setCheckOpen(true)
   550→  }

   555→      stashDraft() // normally empty — the draft was flushed by sendToKitchen
   556→      setReceiptOrder(updated)
   557→      setMode('tables')
   558→      setSelectedTable(null)
   559→      setActiveOrderId(null)

   561→      setGuestsDraft(2)

   563→      setSeatingTables(null)

   571→        queryClient.invalidateQueries({ queryKey: ['orders'] }),

   585→    setSelectedTable(null)
   586→    setActiveOrderId(null)

   588→    setGuestsDraft(2)

   590→    setSeatingTables(null)

   594→  }

   603→  // the payment modal (rendered in tables mode too; the modal hides its own

   605→  const handleSettleDeferred = (deferredOrder: Order) => {
   606→    setPayOrder(deferredOrder)
   607→    setPayOpen(true)

   612→  const handleOrderDeferred = async (_deferredOrder: Order) => {

   614→    await invalidateShared()

   617→  const cancelMutation = useMutation({

   621→      toast.success(t('pos.cancelToast'))

   641→
   642→  /** Enter order mode for a live order — mirrors openTakeawayOrder /
   643→   *  applySelectTable semantics. Hash-driven, so it never pushes history. */
   644→  const restoreLiveOrder = (restored: Order) => {
   645→    setSelectedTable(

   647→        ? { id: restored.table.id, name: restored.table.name }
   648→        : { id: null, name: t('common.takeaway') },

   659→    toast.success(t('pos.orderRestoredToast', { order: restored.id }))
   660→  }
   661→
   662→  /** Claim the right to run a deep restore for `target`. Returns the epoch, or
   663→   *  null when the identical restore is already in flight (popstate +
   664→   *  hashchange double fire). A different target supersedes the old one. */
   665→  const claimNavRestore = (target: string): number | null => {
   666→    if (navInFlightRef.current === target) return null
   667→    navInFlightRef.current = target
   668→    navEpochRef.current += 1
   669→    return navEpochRef.current
   670→  }
   671→
   672→  /** Release the restore guards — only the owning epoch may clear them. */
   673→  const releaseNavRestore = (epoch: number) => {
   674→    if (epoch !== navEpochRef.current) return
   675→    navInFlightRef.current = null
   676→  }
   677→
   678→  /** True when the hash still points at `target` (mid-flight re-check). */
   679→  const hashStill = (target: string) => {
   680→    const now = currentNav()
   681→    return now != null && now.view === 'pos' && now.sub === target
   682→  }
   683→
   684→  /** Failed restore → toast + a safe floor entry. */
   685→  const cannotReopen = () => {
   686→    toast.error(t('pos.cannotReopenToast'))
   687→    replaceNav('pos')
   688→    resetToTables()
   689→  }
   690→
   691→  /** Fetch an order imperatively (null on error). */
   692→  const fetchOrderOrNull = async (id: number): Promise<Order | null> => {
   693→    try {
   694→      const data = await fetcher<{ order: Order }>(`/api/orders/${id}`)
   695→      return data.order ?? null
   696→    } catch {
   697→      return null
   698→    }
   699→  }
   700→
   701→  /** Restore the live-order screen for '#/pos/order/{id}'. */
   702→  const restoreOrderById = async (id: number, target: string) => {
   703→    const epoch = claimNavRestore(target)
   704→    if (epoch == null) return
   705→    stashDraft() // leaving the current context — never lose unsent items
   706→    try {
   707→      const fetched = await fetchOrderOrNull(id)
   708→      // The user navigated again mid-flight → abort silently.
   709→      if (epoch !== navEpochRef.current || !hashStill(target)) return
   710→      if (fetched && fetched.status === 'open') {
   711→        restoreLiveOrder(fetched)
   712→      } else {
   713→        cannotReopen() // closed / cancelled / merged away / fetch error
   714→      }
   715→    } finally {
   716→      releaseNavRestore(epoch)
   717→    }
   718→  }
   719→
   720→  /** Restore the screen for '#/pos/table/{id}': its live order when the table
   721→   *  got occupied meanwhile (hash fixed to order/{id}), or the free-table
   722→   *  draft screen — applySelectTable pops any stashed draft for it. */
   723→  const restoreTableById = async (id: number, target: string) => {
   724→    const epoch = claimNavRestore(target)
   725→    if (epoch == null) return
   726→    stashDraft()
   727→    try {
   728→      // Same data source as the floor screen (TableSelect's floorplans query).
   729→      const data = await fetcher<{ floorPlans: FloorPlan[] }>('/api/floorplans')
   730→      if (epoch !== navEpochRef.current || !hashStill(target)) return
   731→      queryClient.setQueryData(['floorplans'], data)
   732→      const table =
   733→        data.floorPlans.flatMap((fp) => fp.tables).find((tb) => tb.id === id) ?? null
   734→      if (!table) {
   735→        cannotReopen()
   736→        return
   737→      }
   738→      if (table.openOrderId != null) {
   739→        // Occupied meanwhile → its live order screen; fix the hash to it.
   740→        const orderTarget = `order/${table.openOrderId}`
   741→        replaceNav('pos', orderTarget)
   742→        const fetched = await fetchOrderOrNull(table.openOrderId)
   743→        if (epoch !== navEpochRef.current || !hashStill(orderTarget)) return
   744→        if (fetched && fetched.status === 'open') restoreLiveOrder(fetched)
   745→        else cannotReopen()
   746→        return
   747→      }
   748→      // Free table → draft mode (restores are push-suppressed inside).
   749→      withNavSuppression(() => applySelectTable(table, 2))
   750→    } catch {
   751→      // Floorplans fetch failed — only bail to the floor when the hash still
   752→      // points at this target (never clobber another view's hash).
   753→      if (epoch === navEpochRef.current && hashStill(target)) cannotReopen()
   754→    } finally {
   755→      releaseNavRestore(epoch)
   756→    }
   757→  }
   758→
   759→  /** onNav handler — browser back/forward + hand-edited hashes. Idempotent:
   760→   *  navigating to the state we are already in is a no-op (checked against
   761→   *  both the rendered state and the synchronous appliedNavRef mirror). */
   762→  const handleNavEvent = (nav: NavHash | null) => {
   763→    if (nav == null || nav.view !== 'pos') return // another view owns the hash
   764→    const sub = nav.sub
   765→    if (sub == null) {
   766→      // Floor — leave order mode exactly like the in-app back button
   767→      // (stash + reset + replaceNav('#/pos'), itself idempotent here).
   768→      if (mode === 'order' || appliedNavRef.current != null) resetToTables()
   769→      return
   770→    }
   771→    if (sub.startsWith('order/')) {
   772→      const id = Number.parseInt(sub.slice('order/'.length), 10)
   773→      if (!Number.isInteger(id) || id <= 0) return
   774→      if (appliedNavRef.current === sub) return // idempotent re-entry
   775→      if (mode === 'order' && activeOrderId === id) return
   776→      void restoreOrderById(id, sub)
   777→      return
   778→    }
   779→    if (sub.startsWith('table/')) {
   780→      const id = Number.parseInt(sub.slice('table/'.length), 10)
   781→      if (!Number.isInteger(id) || id <= 0) return
   782→      if (appliedNavRef.current === sub) return
   783→      if (mode === 'order' && selectedTable?.id === id && activeOrderId == null) return
   784→      void restoreTableById(id, sub)
   785→      return
   786→    }
   787→    if (sub === 'takeaway') {
   788→      if (appliedNavRef.current === sub) return
   789→      if (mode === 'order' && selectedTable?.id == null && activeOrderId == null) return
   790→      const epoch = claimNavRestore(sub)
   791→      if (epoch == null) return
   792→      try {
   793→        stashDraft()
   794→        // startTakeaway semantics with pushes suppressed; pops the
   795→        // 'takeaway' stash (toast when items come back).
   796→        withNavSuppression(() => startTakeaway())
   797→      } finally {
   798→        releaseNavRestore(epoch)
   799→      }
   800→    }
   801→  }
   802→
   803→  // Always dispatch nav events to the handler of the LATEST render (single
   804→  // subscription created below, no stale closures).
   805→  useEffect(() => {
   806→    navHandlerRef.current = handleNavEvent
   807→  })
   808→
   809→  // Single subscription for the component's lifetime (idempotent handler).
   810→  useEffect(() => {
   811→    const unsubscribe = onNav((nav) => navHandlerRef.current(nav))
   812→    return unsubscribe
   813→  }, [])
   814→
   815→  // Mount-time deep-link restore: refresh/bookmark on '#/pos/order/N',
   816→  // '#/pos/table/N' or '#/pos/takeaway' reopens that screen (pushes are
   817→  // suppressed — the hash already points here). Floor is the default state.
   818→  useEffect(() => {
   819→    const nav = currentNav()
   820→    if (nav && nav.view === 'pos' && nav.sub) navHandlerRef.current(nav)
   821→  }, [])
   822→
   823→  // Re-assert the deep hash when this view becomes active again: page.tsx
   824→  // pushes a shallow '#/pos' when returning from another view, and this
   825→  // replaces it with the real deep hash of the preserved order screen.
   826→  useEffect(() => {
   827→    const wasActive = prevActiveRef.current
   828→    prevActiveRef.current = active
   829→    // Fire only on the hidden → visible flip (wasActive false, now true)
   830→    // while an order screen is preserved.
   831→    if (wasActive || !active || mode !== 'order') return
   832→    const sub = activeOrderId
   833→      ? `order/${activeOrderId}`
   834→      : selectedTable?.id != null
   835→        ? `table/${selectedTable.id}`
   836→        : 'takeaway'
   837→    replaceNav('pos', sub)
   838→  }, [active, mode, activeOrderId, selectedTable?.id])
   839→
   840→  // ── Render ───────────────────────────────────────────────────────
   841→  if (mode === 'tables') {
   842→    return (
   843→      <>
   844→        <TableSelect
   845→          onSelectTable={selectTable}
   846→          onTakeaway={startTakeaway}
   847→          onOpenTakeawayOrder={openTakeawayOrder}
   848→          transferOrderId={transferOrderId}
   849→          onTransferDone={handleTransferDone}
   850→          onSeatParty={handleSeatParty}
   851→          onSettleDeferred={handleSettleDeferred}
   852→          onMyShift={() => setMyShiftOpen(true)}
   853→        />
   854→        {/* R8: per-server shift closeout — sales, tips & payment mix today */}
   855→        <MyShiftSheet open={myShiftOpen} onOpenChange={setMyShiftOpen} />
   856→        {/* Guests quick dialog — shown BEFORE entering order mode on a free
   857→            table (or after a Seat Party selection). */}
   858→        <GuestsDialog
   859→          open={guestsDialogOpen}
   860→          value={guestsValue}
   861→          pending={updateGuests.isPending}
   862→          onValueChange={setGuestsValue}
   863→          onConfirm={confirmGuests}
   864→          onCancel={cancelGuestsDialog}
   865→        />
   866→        {/* Deferred-check settlement (chips on the floor) — same payment modal
   867→            as order mode; hides its defer button for already-deferred orders. */}
   868→        {payOrder && (
   869→          <PaymentModal
   870→            order={payOrder}
   871→            open={payOpen}
   872→            onOpenChange={(o) => {
   873→              setPayOpen(o)
   874→              if (!o) setPayOrder(null)
   875→            }}
   876→            onSuccess={(updated, closed) => void handlePaySuccess(updated, closed)}
   877→            onDeferred={(deferredOrder) => void handleOrderDeferred(deferredOrder)}
   878→          />
   879→        )}
   880→        {/* Receipt after a full payment — the screen resets to the floor while
   881→            the receipt stays on top (independent of the payment modal). */}
   882→        {receiptOrder && (
   883→          <ReceiptModal
   884→            order={receiptOrder}
   885→            open={!!receiptOrder}
   886→            onOpenChange={(o) => {
   887→              if (!o) setReceiptOrder(null)
   888→            }}
   889→            onClose={() => setReceiptOrder(null)}
   890→          />
   891→        )}
   892→      </>
   893→    )
   894→  }
   895→
   896→  return (
   897→    <div className="flex h-full min-h-[560px] flex-col">
   898→      {/* ── Top bar (Odoo order chrome) ── */}
   899→      <header className="flex min-h-16 shrink-0 flex-wrap items-center gap-2 border-b border-[#E2E2E0] bg-white px-3 py-2 shadow-sm sm:px-4">
   900→        <Button
   901→          variant="ghost"
   902→          className="h-11 rounded-xl px-3 text-[#714B67] hover:bg-[#714B67]/10"
   903→          onClick={resetToTables}
   904→          title={t('pos.backToTables')}
   905→        >
   906→          <ChevronLeft className="size-5 rtl:rotate-180" />
   907→          <span>{t('pos.title')}</span>
   908→        </Button>
   909→        <div className="flex min-w-0 flex-1 flex-wrap items-center gap-x-2 gap-y-1">
   910→          <span className="truncate text-xl font-bold">{selectedTable?.name ?? t('pos.order')}</span>
   911→          {mergedExtraCount > 0 && (
   912→            <Badge className="gap-1 border-violet-300 bg-violet-100 text-violet-800 hover:bg-violet-100">
   913→              <Combine className="size-3" aria-hidden />
   914→              {t('pos.mergedTablesBadge', { n: mergedExtraCount })}
   915→            </Badge>
   916→          )}
   917→          {order && (
   918→            <Badge className="bg-[#714B67] text-white hover:bg-[#714B67]">
   919→              {t('common.order')} #{order.id}
   920→            </Badge>
   921→          )}
   922→          {order && (
   923→            <span className="text-xs tabular-nums text-muted-foreground" data-tick={elapsedTick}>
   924→              {elapsedSince(order.createdAt)}
   925→            </span>
   926→          )}
   927→          {preparingCount > 0 && (
   928→            <Badge className="border-amber-300 bg-amber-100 text-amber-800 hover:bg-amber-100">
   929→              {t('pos.preparingCount', { n: preparingCount })}
   930→            </Badge>
   931→          )}
   932→          {readyCount > 0 && (
   933→            <Badge className="border-emerald-300 bg-emerald-100 text-emerald-800 hover:bg-emerald-100">
   934→              {t('pos.readyCount', { n: readyCount })}
   935→            </Badge>
   936→          )}
   937→        </div>
   938→        <div className="flex items-center gap-2">
   939→          {/* Guests: N — edit while drafting (sent with the order) or on the live order */}
   940→          <Button
   941→            variant="outline"
   942→            className="h-11 rounded-xl border-[#714B67]/40 text-[#714B67] hover:bg-[#714B67]/10 hover:text-[#714B67]"
   943→            onClick={openGuestsEdit}
   944→            title={t('pos.editGuests')}
   945→          >
   946→            <Users />
   947→            <span className="hidden sm:inline">{t('common.guests')}:</span>
   948→            <span className="font-semibold tabular-nums">{order ? order.guests : guestsDraft}</span>
   949→          </Button>
   950→          <Button
   951→            variant="outline"
   952→            className="h-11 rounded-xl border-[#714B67]/40 text-[#714B67] hover:bg-[#714B67]/10 hover:text-[#714B67]"
   953→            disabled={!order}
   954→            onClick={startTransfer}
   955→            title={order ? t('pos.transferHint') : t('pos.transferDisabledHint')}
   956→          >
   957→            <ArrowLeftRight />
   958→            <span className="hidden md:inline">{t('pos.transfer')}</span>
   959→          </Button>
   960→          {order && canCancel && (
   961→            <Button
   962→              variant="outline"
   963→              className="h-11 rounded-xl border-destructive/40 text-destructive hover:bg-destructive/10 hover:text-destructive"
   964→              onClick={() => setCancelOpen(true)}

   966→              <Ban />
   967→              <span className="hidden md:inline">{t('common.cancel')}</span>
   968→            </Button>

   971→      </header>

   976→        <div className="flex max-h-[60vh] w-full shrink-0 flex-col border-t border-[#E2E2E0] bg-white md:max-h-none md:w-[380px] md:border-l md:border-t-0 lg:w-[420px]">
