    61→/** Odoo 17-style tile silhouettes (BINDING — mirrors the POS floor screen). */
    62→const SHAPE_TILE_CLASSES: Record<string, string> = {
    63→  square: 'aspect-square rounded-2xl',
    64→  round: 'aspect-square rounded-full',
    65→  rectangle: 'h-24 rounded-2xl',
    66→  oval: 'h-24 rounded-full',
    67→}
    68→
    69→function shapeTileClasses(shape: string): string {
    70→  return SHAPE_TILE_CLASSES[shape] ?? SHAPE_TILE_CLASSES.square
    71→}
    72→
    73→/** Occupied-tile duration heat: <15m emerald · 15–44 amber · 45–89 orange · ≥90 rose. */
    74→function occupiedHeatClasses(mins: number): { surface: string; amount: string } {
    75→  if (mins < 15)
    76→    return { surface: 'bg-emerald-50 border-emerald-200 text-emerald-900', amount: 'text-emerald-700' }
    77→  if (mins < 45)
    78→    return { surface: 'bg-amber-50 border-amber-200 text-amber-900', amount: 'text-amber-700' }
    79→  if (mins < 90)
    80→    return { surface: 'bg-orange-100 border-orange-300 text-orange-900', amount: 'text-orange-700' }
    81→  return { surface: 'bg-rose-100 border-rose-300 text-rose-900', amount: 'text-rose-700' }
    82→}
    83→
    84→/** Tile surface per live status: free = white, reserved = amber ring, occupied = duration heat. */
    85→function tableSurfaceClasses(
    86→  table: RestaurantTable,
    87→  mins: number,
    88→): { surface: string; amount: string | null } {
    89→  if (table.status === 'occupied') {
    90→    const heat = occupiedHeatClasses(mins)
    91→    return { surface: heat.surface, amount: heat.amount }
    92→  }
    93→  if (table.status === 'reserved')
    94→    return { surface: 'bg-amber-50/70 border-amber-300 ring-1 ring-amber-400', amount: null }
    95→  if (table.status === 'free') return { surface: 'bg-white border-[#E2E2E0] shadow-sm', amount: null }
    96→  // unknown status — neutral stone fallback (keeps tiles readable)
    97→  return { surface: 'border-stone-300 bg-stone-50 text-stone-700', amount: null }
    98→}
    99→
   100→/** Tiny shape glyph used next to each Select option. */
   101→function ShapeGlyph({ shape, className }: { shape: string; className?: string }) {
   102→  const cls =
   103→    shape === 'round'
   104→      ? 'size-3.5 rounded-full'
   105→      : shape === 'oval'
   106→        ? 'h-3 w-5 rounded-full'
   107→        : shape === 'rectangle'
   108→          ? 'h-3 w-5 rounded-[3px]'
   109→          : 'size-3.5 rounded-[3px]'
   110→  return <span className={cn('inline-block shrink-0 border-2 border-current', cls, className)} aria-hidden />
   111→}
   112→
   113→const clampPercent = (value: number, min: number, max: number) =>
   114→  Math.min(max, Math.max(min, value))
   115→const round1 = (value: number) => Math.round(value * 10) / 10
   116→const toNumber = (value: string, fallback: number) => {
   117→  const parsed = Number(value)
   118→  return Number.isFinite(parsed) ? parsed : fallback
   119→}

   258→  // ── Pointer-based dragging (mouse + touch) ───────────────────────────

   260→  function beginDrag(event: React.PointerEvent<HTMLDivElement>, table: RestaurantTable) {
   261→    if (event.pointerType === 'mouse' && event.button !== 0) return
   262→    const rect = canvasRef.current?.getBoundingClientRect()
   263→    if (!rect) return
   264→    event.preventDefault()
   265→    try {
   266→      event.currentTarget.setPointerCapture(event.pointerId)
   267→    } catch {
   268→      // Pointer already released — ignore.
   269→    }
   270→    const pointerX = ((event.clientX - rect.left) / rect.width) * 100
   271→    const pointerY = ((event.clientY - rect.top) / rect.height) * 100
   272→    const state: DragState = {
   273→      table,
   274→      pointerId: event.pointerId,
   275→      startX: event.clientX,
   276→      startY: event.clientY,
   277→      offX: pointerX - table.positionX,
   278→      offY: pointerY - table.positionY,
   279→      x: table.positionX,
   280→      y: table.positionY,
   281→      movedPx: 0,
   282→    }
   283→    dragRef.current = state
   284→    setDrag(state)
   285→  }
   286→
   287→  function moveDrag(event: React.PointerEvent<HTMLDivElement>) {
   288→    const current = dragRef.current
   289→    if (!current || current.pointerId !== event.pointerId) return
   290→    const rect = canvasRef.current?.getBoundingClientRect()
   291→    if (!rect) return
   292→    const pointerX = ((event.clientX - rect.left) / rect.width) * 100
   293→    const pointerY = ((event.clientY - rect.top) / rect.height) * 100
   294→    const next: DragState = {
   295→      ...current,
   296→      x: clampPercent(pointerX - current.offX, 2, 98),
   297→      y: clampPercent(pointerY - current.offY, 2, 98),
   298→      movedPx: Math.max(
   299→        current.movedPx,
   300→        Math.hypot(event.clientX - current.startX, event.clientY - current.startY),
   301→      ),
   302→    }
   303→    dragRef.current = next
   304→    setDrag(next)
   305→  }
   306→
   307→  function endDrag(event: React.PointerEvent<HTMLDivElement>, commit: boolean) {
   308→    const current = dragRef.current
   309→    if (!current || current.pointerId !== event.pointerId) return
   310→    dragRef.current = null
   311→    setDrag(null)
   312→    if (!commit) return
   313→    if (current.movedPx < 6) {
   314→      // Treated as a click → open the table editor.
   315→      openTableDialog(current.table)
   316→      return
   317→    }
   318→    updateTableMutation.mutate({
   319→      id: current.table.id,
   320→      body: { positionX: round1(current.x), positionY: round1(current.y) },
   321→      successMessage: t('admin.positionSaved'),
   322→    })
   323→  }

   336→  function handleAddTable() {

   352→      floorPlanId: selectedPlan.id,
   353→      name,
   354→      capacity,
   355→      shape: addTableShape,

   357→      positionY,

   361→  function handleSaveTable() {
   362→    if (!editingTable) return
   363→    const name = editName.trim()
   364→    if (!name) {
   365→      toast.error(t('admin.tableNameRequired'))
   366→      return
   367→    }
   368→    const parsedCapacity = Number(editCapacity)
   369→    const capacity = Math.max(
   370→      1,

   372→        Number.isFinite(parsedCapacity) && parsedCapacity > 0 ? parsedCapacity : editingTable.capacity,

   377→      capacity,

   384→      body.status = editStatus
   385→    }
   386→    updateTableMutation.mutate({
   387→      id: editingTable.id,
   388→      body,
   389→      successMessage: t('admin.tableUpdated'),
   390→      closeDialog: true,
   391→    })
   392→  }
