     4→import type { LucideIcon } from 'lucide-react'

    27→} from '@/components/ui/dialog'

    29→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs'

    32→import { PAYMENT_METHODS, TIP_PRESETS } from '@/lib/constants'
    33→import { formatCurrency, formatQty } from '@/lib/format'
    34→import { localizedName, useI18n } from '@/lib/i18n'

    39→
    40→type PaymentModalProps = {
    41→  order: Order
    42→  open: boolean
    43→  onOpenChange: (open: boolean) => void
    44→  onSuccess: (order: Order, closed: boolean) => void
    45→  /** Called after the check is deferred (pay later, client name recorded).
    46→   *  The modal closes itself; the parent resets its order screen state. */
    47→  onDeferred?: (order: Order) => void
    48→}
    49→
    50→type PaymentResult = {
    51→  order: Order
    52→  paidAmount: number
    53→  remaining: number
    54→  closed: boolean
    55→}
    56→
    57→type EditableRow = {
    58→  id: string
    59→  method: string
    60→  amount: string
    61→  reference: string
    62→}
    63→
    64→type SubmitRow = { method: string; amount: number; reference?: string }
    65→
    66→type SplitTab = 'single' | 'equal' | 'items' | 'custom'
    67→
    68→const METHOD_META: Record<string, { labelKey: string; icon: LucideIcon }> = {
    69→  cash: { labelKey: 'status.payment.cash', icon: Banknote },
    70→  card: { labelKey: 'status.payment.card', icon: CreditCard },
    71→  other: { labelKey: 'status.payment.other', icon: MoreHorizontal },
    72→}
    73→
    74→export default function PaymentModal({ order, open, onOpenChange, onSuccess, onDeferred }: PaymentModalProps) {
    75→  const { t, lang } = useI18n()
    76→  const [tab, setTab] = useState<SplitTab>('single')
    77→  const [submitting, setSubmitting] = useState(false)
    78→  const [checkOpen, setCheckOpen] = useState(false)
    79→
    80→  // Defer-payment state (check stays open under the client's name)
    81→  const [deferOpen, setDeferOpen] = useState(false)
    82→  const [deferName, setDeferName] = useState('')
    83→  const [deferSubmitting, setDeferSubmitting] = useState(false)
    84→
    85→  // Which payment row the method tiles currently target.
    86→  const [activeIdx, setActiveIdx] = useState(0)
    87→
    88→  // R8: tips — gratuity ON TOP of the bill, attached PER PAYMENT ROW
    89→  // (single tab → one tip; equal/items/custom → per-payer tips). Keyed per
    90→  // tab+row so every payer's tip survives row switching; the preset
    91→  // selection and the custom input text are tracked separately per row.
    92→  const [tipSel, setTipSel] = useState<Record<string, number | 'custom'>>({})
    93→  const [tipCustom, setTipCustom] = useState<Record<string, string>>({})
    94→
    95→  const [singleRow, setSingleRow] = useState<EditableRow>({ id: 'single', method: 'cash', amount: '', reference: '' })
    96→  const [customRows, setCustomRows] = useState<EditableRow[]>([])
    97→
    98→  const [eqPayers, setEqPayers] = useState(2)
    99→  const [eqMethods, setEqMethods] = useState<Record<number, string>>({})
   100→
   101→  const [itPayers, setItPayers] = useState(2)
   102→  const [itMethods, setItMethods] = useState<Record<number, string>>({})
   103→  const [assignments, setAssignments] = useState<Record<number, number>>({})
   104→
   105→  const remaining = round2(Math.max(0, order.remainingAmount))
   106→
   107→  // Reset state whenever the modal opens for (a new) order.
   108→  useEffect(() => {
   109→    if (!open) return
   110→    const rem = round2(Math.max(0, order.remainingAmount))
   111→    setTab('single')
   112→    setSingleRow({ id: 'single', method: 'cash', amount: String(rem), reference: '' })
   113→    setCustomRows([{ id: newDraftKey(), method: 'cash', amount: String(rem), reference: '' }])
   114→    setEqPayers(2)
   115→    setEqMethods({})
   116→    setItPayers(2)
   117→    setItMethods({})
   118→    setAssignments({})
   119→    setSubmitting(false)
   120→    setActiveIdx(0)
   121→    setTipSel({})
   122→    setTipCustom({})
   123→    setCheckOpen(false)
   124→    setDeferOpen(false)
   125→    setDeferName('')
   126→    setDeferSubmitting(false)
   127→  }, [open, order.id, order.remainingAmount])
   128→
   129→  // ── Derived payment rows per tab ──────────────────────────────────
   130→  const eqAmounts = useMemo(() => {
   131→    const part = round2(remaining / eqPayers)
   132→    const amounts = Array.from({ length: eqPayers }, () => part)
   133→    const others = round2(part * (eqPayers - 1))
   134→    amounts[eqPayers - 1] = round2(Math.max(0, remaining - others))
   135→    return amounts
   136→  }, [remaining, eqPayers])
   137→
   138→  const itemLines = useMemo(
   139→    () =>
   140→      order.items.map((it) => ({
   141→        id: it.id,
   142→        label: `${formatQty(it.quantity)} × ${
   143→          it.product ? localizedName(it.product.name, it.product.nameAr, lang) : t('pos.item')
   144→        }`,
   145→        total: round2(it.quantity * it.unitPrice),
   146→      })),
   147→    [order.items, t, lang],
   148→  )
   149→
   150→  const itAmounts = useMemo(() => {
   151→    const subtotal = round2(order.subtotalAmount)
   152→    const rawTotals = new Array<number>(itPayers).fill(0)
   153→    for (const line of itemLines) {
   154→      const payer = assignments[line.id] ?? 0
   155→      if (payer < itPayers) rawTotals[payer] = round2(rawTotals[payer] + line.total)
   156→    }
   157→    const amounts: number[] = []
   158→    let allocated = 0
   159→    for (let i = 0; i < itPayers; i++) {
   160→      if (i === itPayers - 1) {
   161→        amounts.push(round2(Math.max(0, remaining - allocated)))
   162→      } else {
   163→        const share = subtotal > 0 ? round2((rawTotals[i] / subtotal) * remaining) : 0
   164→        amounts.push(share)
   165→        allocated = round2(allocated + share)
   166→      }
   167→    }
   168→    return amounts
   169→  }, [itemLines, assignments, itPayers, remaining, order.subtotalAmount])
   170→
   171→  const submitRows: SubmitRow[] = useMemo(() => {
   172→    if (tab === 'single') {
   173→      return [toSubmitRow(singleRow)]
   174→    }
   175→    if (tab === 'equal') {
   176→      return eqAmounts.map((amount, i) => ({
   177→        method: eqMethods[i] ?? 'cash',
   178→        amount,
   179→      }))
   180→    }
   181→    if (tab === 'items') {
   182→      return itAmounts.map((amount, i) => ({
   183→        method: itMethods[i] ?? 'cash',
   184→        amount,
   185→      }))
   186→    }
   187→    return customRows.map(toSubmitRow)
   188→  }, [tab, singleRow, customRows, eqAmounts, eqMethods, itAmounts, itMethods])
   189→
   190→  const sum = round2(submitRows.reduce((s, r) => s + r.amount, 0))
   191→  const diff = round2(sum - remaining)
   192→  const exceeds = diff > 0.01
   193→  const exact = Math.abs(diff) <= 0.01
   194→
   195→  // ── Active row / method tiles ─────────────────────────────────────
   196→  const rowCount =
   197→    tab === 'single' ? 1 : tab === 'equal' ? eqPayers : tab === 'items' ? itPayers : customRows.length
   198→  const safeActiveIdx = Math.min(activeIdx, Math.max(0, rowCount - 1))
   199→
   200→  // ── Tips (R8) ────────────────────────────────────────────────
   201→  // Tips sit ON TOP of the bill: `remaining` stays bill-only and the
   202→  // validation below is unchanged (Σ row amounts ≤ remaining as before).
   203→
   204→  /** Stable per-row key for the tip state ('single' | 'eq:0' | 'it:1' | 'c:<rowId>'). */
   205→  const tipKeyFor = (index: number): string => {
   206→    if (tab === 'single') return 'single'
   207→    if (tab === 'equal') return `eq:${index}`
   208→    if (tab === 'items') return `it:${index}`
   209→    return `c:${customRows[index]?.id ?? index}`
   210→  }
   211→
   212→  const activeTipKey = tipKeyFor(safeActiveIdx)
   213→  const activeSel = tipSel[activeTipKey]
   214→  const activeCustomText = tipCustom[activeTipKey] ?? ''
   215→
   216→  const activeRowAmount = useMemo(
   217→    () => round2(submitRows[safeActiveIdx]?.amount ?? 0),
   218→    [submitRows, safeActiveIdx],
   219→  )
   220→
   221→  /** Tip for a submit-row index: {value, invalid} — presets are % of that row's amount. */
   222→  const rowTip = (index: number): { value: number; invalid: boolean } => {
   223→    const key = tipKeyFor(index)
   224→    const sel = tipSel[key]
   225→    if (sel == null || sel === 0) return { value: 0, invalid: false }
   226→    if (sel === 'custom') {
   227→      const parsed = parseTipInput(tipCustom[key] ?? '')
   228→      return parsed == null ? { value: 0, invalid: true } : { value: parsed, invalid: false }
   229→    }
   230→    return { value: round2((submitRows[index]?.amount ?? 0) * (sel / 100)), invalid: false }
   231→  }
   232→
   233→  const activeTip = rowTip(safeActiveIdx)
   234→  const tipsTotal = round2(submitRows.reduce((s, _r, i) => s + rowTip(i).value, 0))
   235→  const tipInvalid = submitRows.some((_r, i) => rowTip(i).invalid)
   236→
   237→  const canSubmit = !submitting && remaining > 0 && sum > 0 && !exceeds && !tipInvalid
   238→
   239→  const activeRowLabel =
   240→    tab === 'single'
   241→      ? t('pos.payment')
   242→      : tab === 'equal' || tab === 'items'
   243→        ? t('pos.payer', { n: safeActiveIdx + 1 })
   244→        : t('pos.paymentN', { n: safeActiveIdx + 1 })
   245→
   246→  const currentMethodAt = (index: number): string => {
   247→    if (tab === 'single') return singleRow.method
   248→    if (tab === 'equal') return eqMethods[index] ?? 'cash'
   249→    if (tab === 'items') return itMethods[index] ?? 'cash'
   250→    return customRows[index]?.method ?? 'cash'
   251→  }
   252→
   253→  const setMethodAt = (index: number, m: string) => {
   254→    if (tab === 'single') {
   255→      setSingleRow((r) => ({ ...r, method: m }))
   256→    } else if (tab === 'equal') {
   257→      setEqMethods((prev) => ({ ...prev, [index]: m }))
   258→    } else if (tab === 'items') {
   259→      setItMethods((prev) => ({ ...prev, [index]: m }))
   260→    } else {
   261→      setCustomRows((rows) => rows.map((r, i) => (i === index ? { ...r, method: m } : r)))
   262→    }
   263→  }
   264→
   265→  const handleTabChange = (v: string) => {
   266→    setTab(v as SplitTab)
   267→    setActiveIdx(0)
   268→  }
   269→
   270→  // ── Guest check rows mirroring the current split configuration ────
   271→  const checkRows: CheckSplitRow[] = useMemo(() => {
   272→    if (tab === 'single') {
   273→      const amount = parseAmount(singleRow.amount) || remaining
   274→      return [{ label: t('pos.fullBill'), amount, method: singleRow.method }]
   275→    }
   276→    if (tab === 'equal') {
   277→      return eqAmounts.map((amount, i) => ({
   278→        label: t('pos.partOf', { i: i + 1, n: eqPayers }),
   279→        amount,
   280→        method: eqMethods[i] ?? 'cash',
   281→      }))
   282→    }
   283→    if (tab === 'items') {
   284→      return itAmounts.map((amount, i) => ({
   285→        label: t('pos.payer', { n: i + 1 }),
   286→        amount,
   287→        method: itMethods[i] ?? 'cash',
   288→      }))
   289→    }
   290→    return customRows.map((r, i) => ({
   291→      label: t('pos.paymentN', { n: i + 1 }),
   292→      amount: parseAmount(r.amount),
   293→      method: r.method,
   294→    }))
   295→  }, [tab, singleRow, customRows, eqAmounts, eqMethods, eqPayers, itAmounts, itMethods, remaining, t])
   296→
   297→  // ── Handlers ──────────────────────────────────────────────────────
   298→  const clampPayers = (raw: number, min: number, max: number) => {
   299→    const n = Math.round(Number.isFinite(raw) ? raw : min)
   300→    return Math.min(max, Math.max(min, n))
   301→  }
   302→
   303→  const setAssignment = (itemId: number, payer: number) =>
   304→    setAssignments((prev) => ({ ...prev, [itemId]: payer }))
   305→
   306→  const addCustomRow = () => {
   307→    const current = round2(customRows.reduce((s, r) => s + parseAmount(r.amount), 0))
   308→    const next = round2(Math.max(0, remaining - current))
   309→    setCustomRows((rows) => [...rows, { id: newDraftKey(), method: 'cash', amount: String(next), reference: '' }])
   310→  }
   311→
   312→  const handleSubmit = async () => {
   313→    // R8: each payment row carries its own tip (≥ 0, round2) on top of the
   314→    // amount — the API persists it but never counts it toward paidAmount.
   315→    const payments = submitRows
   316→      .map((r, i) => ({ ...r, amount: round2(r.amount), tip: Math.max(0, round2(rowTip(i).value)) }))
   317→      .filter((r) => r.amount > 0)
   318→      .map((r) => ({
   319→        method: r.method,
   320→        amount: r.amount,
   321→        tip: r.tip,
   322→        ...(r.method === 'card' && r.reference?.trim() ? { reference: r.reference.trim() } : {}),
   323→      }))
   324→    if (payments.length === 0) return
   325→    setSubmitting(true)
   326→    try {
   327→      const result = await apiFetch<PaymentResult>(`/api/orders/${order.id}/payments`, {
   328→        method: 'POST',
   329→        body: { payments },
   330→      })
   331→      if (result.closed) {
   332→        toast.success(t('pos.paidClosedToast'))
   333→        onSuccess(result.order, true)
   334→        onOpenChange(false)
   335→      } else {
   336→        toast.success(t('pos.paymentRecordedToast', { amount: formatCurrency(result.remaining) }))
   337→        onSuccess(result.order, false)
   338→        onOpenChange(false)
   339→      }
   340→    } catch (err) {
   341→      toast.error(err instanceof Error ? err.message : t('pos.paymentFailedToast'))
   342→    } finally {
   343→      setSubmitting(false)
   344→    }
   345→  }
   346→
   347→  // ── Defer payment (client pays later, check tracked by name) ──────
   348→  const canDefer = order.status === 'open' && remaining > 0
   349→
   350→  const handleDefer = async () => {
   351→    const clientName = deferName.trim()
   352→    if (!clientName) {
   353→      toast.error(t('pos.deferredNameRequired'))
   354→      return
   355→    }
   356→    if (deferSubmitting) return
   357→    setDeferSubmitting(true)
   358→    try {
   359→      const result = await apiFetch<{ order: Order }>(`/api/orders/${order.id}/defer`, {
   360→        method: 'POST',
   361→        body: { clientName },
   362→      })
   363→      toast.success(t('pos.deferredToast', { order: order.id, client: clientName }))
   364→      setDeferOpen(false)
   365→      onDeferred?.(result.order)
   366→      onOpenChange(false)
   367→    } catch (err) {
   368→      toast.error(err instanceof Error ? err.message : t('pos.paymentFailedToast'))
   369→    } finally {
   370→      setDeferSubmitting(false)
   371→    }
   372→  }
   373→
   374→  const singleExceeds = tab === 'single' && parseAmount(singleRow.amount) > round2(remaining + 0.01)
   375→
   376→  return (
   377→    <>
   378→      <Dialog open={open} onOpenChange={onOpenChange}>
   379→        <DialogContent className="max-h-[92dvh] overflow-y-auto sm:max-w-lg">
   380→          <DialogHeader>
   381→            <DialogTitle>
   382→              {t('pos.payment')} · {order.table?.name ?? t('common.takeaway')} ·{' '}
   383→              {t('common.order')} #{order.id}
   384→            </DialogTitle>
   385→            <DialogDescription>{t('pos.paymentDesc')}</DialogDescription>
   386→          </DialogHeader>
   387→
   388→          {/* Summary */}
   389→          <div className="grid grid-cols-3 gap-2 rounded-xl border border-[#E2E2E0] bg-white p-3 text-center shadow-sm">
   390→            <div>
   391→              <p className="text-xs text-muted-foreground">{t('money.total')}</p>
   392→              <p className="text-sm font-semibold tabular-nums">{formatCurrency(order.totalAmount)}</p>
   393→            </div>
   394→            <div>
   395→              <p className="text-xs text-muted-foreground">{t('money.paid')}</p>
   396→              <p className="text-sm font-semibold tabular-nums text-emerald-600">
   397→                {formatCurrency(order.paidAmount)}
   398→              </p>
   399→            </div>
   400→            <div>
   401→              <p className="text-xs text-muted-foreground">{t('money.remaining')}</p>
   402→              <p className="text-primary text-2xl font-bold tabular-nums">
   403→                {formatCurrency(remaining)}
   404→              </p>
   405→            </div>
   406→          </div>
   407→
   408→          {/* Method tiles — apply to the active row below */}
   409→          <div className="space-y-1.5">
   410→            <p className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   411→              {t('pos.methodAppliesTo')} <span className="text-[#714B67]">{activeRowLabel}</span>
   412→            </p>
   413→            <div className="grid grid-cols-3 gap-2">
   414→              {PAYMENT_METHODS.map((m) => {
   415→                const meta = METHOD_META[m] ?? { labelKey: `status.payment.${m}`, icon: MoreHorizontal }
   416→                const Icon = meta.icon
   417→                const active = currentMethodAt(safeActiveIdx) === m
   418→                return (
   419→                  <button
   420→                    key={m}
   421→                    type="button"
   422→                    onClick={() => setMethodAt(safeActiveIdx, m)}
   423→                    aria-pressed={active}
   424→                    className={cn(
   425→                      'flex h-16 flex-col items-center justify-center gap-1 rounded-xl border-2 text-sm font-semibold transition active:scale-95',
   426→                      active
   427→                        ? 'border-[#714B67] bg-[#714B67]/10 text-[#714B67]'
   428→                        : 'border-[#E2E2E0] bg-white text-stone-600 hover:border-[#714B67]/40',
   429→                    )}
   430→                  >
   431→                    <Icon className="size-6" aria-hidden />
   432→                    {t(meta.labelKey)}
   433→                  </button>
   434→                )
   435→              })}
   436→            </div>
   437→          </div>
   438→
   439→          <Tabs value={tab} onValueChange={handleTabChange}>
   440→            <TabsList className="grid h-10 w-full grid-cols-4">
   441→              <TabsTrigger value="single" className="text-xs sm:text-sm">
   442→                {t('pos.tabSingle')}
   443→              </TabsTrigger>
   444→              <TabsTrigger value="equal" className="text-xs sm:text-sm">
   445→                {t('pos.tabEqual')}
   446→              </TabsTrigger>
   447→              <TabsTrigger value="items" className="text-xs sm:text-sm">
   448→                {t('pos.tabItems')}
   449→              </TabsTrigger>
   450→              <TabsTrigger value="custom" className="text-xs sm:text-sm">
   451→                {t('pos.tabCustom')}
   452→              </TabsTrigger>
   453→            </TabsList>
   454→
   455→            {/* ── Single ── */}
   456→            <TabsContent value="single" className="space-y-2 pt-3">
   457→              <PayRow
   458→                label={t('pos.payment')}
   459→                method={singleRow.method}
   460→                active
   461→                amountStr={singleRow.amount}
   462→                onAmountChange={(v) => setSingleRow((r) => ({ ...r, amount: v }))}
   463→                reference={singleRow.reference}
   464→                onReferenceChange={(v) => setSingleRow((r) => ({ ...r, reference: v }))}
   465→                editable
   466→                error={
   467→                  singleExceeds
   468→                    ? t('pos.exceedsBy', {
   469→                        amount: formatCurrency(round2(parseAmount(singleRow.amount) - remaining)),
   470→                      })
   471→                    : undefined
   472→                }
   473→              />
   474→            </TabsContent>
   475→
   476→            {/* ── Equal Split ── */}
   477→            <TabsContent value="equal" className="space-y-3 pt-3">
   478→              <div className="flex items-center gap-3 rounded-xl border border-[#E2E2E0] bg-white p-2.5">
   479→                <Users className="size-4 text-muted-foreground" />
   480→                <span className="flex-1 text-sm font-medium">{t('pos.splitBetween')}</span>
   481→                <Input
   482→                  type="number"
   483→                  min={2}
   484→                  max={12}
   485→                  value={eqPayers}
   486→                  onChange={(e) => setEqPayers(clampPayers(parseInt(e.target.value, 10) || 2, 2, 12))}
   487→                  className="h-10 w-16 text-center tabular-nums"
   488→                />
   489→                <span className="text-sm text-muted-foreground">{t('pos.payers')}</span>
   490→              </div>
   491→              {eqAmounts.map((amount, i) => (
   492→                <PayRow
   493→                  key={i}
   494→                  label={t('pos.payer', { n: i + 1 })}
   495→                  method={eqMethods[i] ?? 'cash'}
   496→                  active={safeActiveIdx === i}
   497→                  onActivate={() => setActiveIdx(i)}
   498→                  amount={amount}
   499→                />
   500→              ))}
   501→              <p className="text-center text-xs text-muted-foreground">
   502→                {t('pos.partsTotal')}{' '}
   503→                <span className="font-semibold tabular-nums">{formatCurrency(sum)}</span>
   504→              </p>
   505→            </TabsContent>
   506→
   507→            {/* ── By Items ── */}
   508→            <TabsContent value="items" className="space-y-3 pt-3">
   509→              <div className="flex items-center gap-3 rounded-xl border border-[#E2E2E0] bg-white p-2.5">
   510→                <Users className="size-4 text-muted-foreground" />
   511→                <span className="flex-1 text-sm font-medium">{t('pos.assignItemsTo')}</span>
   512→                <Input
   513→                  type="number"

   815→function toSubmitRow(r: EditableRow): SubmitRow {
   816→  return {
   817→    method: r.method,
   818→    amount: parseAmount(r.amount),
   819→    ...(r.method === 'card' && r.reference.trim() ? { reference: r.reference.trim() } : {}),
   820→  }
   821→}

   823→/** Parse a custom tip input: null = invalid (blocks submit), '' = 0, else ≥ 0 rounded. */
   824→function parseTipInput(s: string): number | null {
   825→  if (s.trim() === '') return 0
   826→  const v = Number(s)
   827→  if (!Number.isFinite(v) || v < 0) return null
   828→  return round2(v)
   829→}

   831→function PayRow({
   832→  label,
   833→  method,

   835→  onActivate,
   836→  amount,
   837→  amountStr,
   838→  onAmountChange,
   839→  reference,
   840→  onReferenceChange,
   841→  onRemove,
