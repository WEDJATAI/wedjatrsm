    73→type ProductForm = {
    74→  name: string
    75→  categoryId: string // '' = no category
    76→  price: string
    77→  cost: string
    78→  isSellable: boolean
    79→  isStockable: boolean
    80→  sku: string
    81→  imageUrl: string
    82→  lowStockThreshold: string
    83→  stock: string // create only
    84→}
    85→
    86→type ProductFormErrors = Partial<
    87→  Record<'name' | 'price' | 'cost' | 'lowStockThreshold' | 'stock', string>
    88→>
    89→
    90→const EMPTY_PRODUCT_FORM: ProductForm = {
    91→  name: '',
    92→  categoryId: '',
    93→  price: '',
    94→  cost: '',
    95→  isSellable: true,
    96→  isStockable: false,
    97→  sku: '',
    98→  imageUrl: '',
    99→  lowStockThreshold: '10',
   100→  stock: '',
   101→}
   102→
   103→function toProductForm(p: Product): ProductForm {
   104→  return {
   105→    name: p.name,
   106→    categoryId: p.categoryId === null ? '' : String(p.categoryId),
   107→    price: String(p.price),
   108→    cost: String(p.cost),
   109→    isSellable: p.isSellable,
   110→    isStockable: p.isStockable,
   111→    sku: p.sku ?? '',
   112→    imageUrl: p.imageUrl ?? '',
   113→    lowStockThreshold: String(p.lowStockThreshold),
   114→    stock: String(p.stock),
   115→  }
   116→}
   117→
   118→function validateProductForm(
   119→  form: ProductForm,
   120→  isCreate: boolean,
   121→  t: (key: string) => string,
   122→): ProductFormErrors {
   123→  const errors: ProductFormErrors = {}
   124→  if (form.name.trim() === '') errors.name = t('admin.nameRequired')
   125→  const price = Number(form.price)
   126→  if (form.price.trim() === '' || !Number.isFinite(price) || price < 0) {
   127→    errors.price = t('admin.priceMin')
   128→  }
   129→  if (form.cost.trim() !== '') {
   130→    const cost = Number(form.cost)
   131→    if (!Number.isFinite(cost) || cost < 0) errors.cost = t('admin.costMin')
   132→  }
   133→  if (form.isStockable) {
   134→    if (form.lowStockThreshold.trim() !== '') {
   135→      const threshold = Number(form.lowStockThreshold)
   136→      if (!Number.isFinite(threshold) || threshold < 0) {
   137→        errors.lowStockThreshold = t('admin.mustBe0')
   138→      }
   139→    }
   140→    if (isCreate && form.stock.trim() !== '') {
   141→      const stock = Number(form.stock)
   142→      if (!Number.isFinite(stock) || stock < 0) errors.stock = t('admin.mustBe0')
   143→    }
   144→  }
   145→  return errors
   146→}

   179→function stockColorClass(product: Product): string | undefined {
   180→  if (!product.isStockable) return undefined
   181→  if (product.stock <= product.lowStockThreshold) {
   182→    return 'font-medium text-rose-600 dark:text-rose-400'
   183→  }
   184→  if (product.stock <= product.lowStockThreshold * 2) {
   185→    return 'font-medium text-amber-600 dark:text-amber-500'
   186→  }
   187→  return undefined
   188→}
   189→
   190→function MarginCell({ product }: { product: Product }) {
   191→  if (product.price <= 0) return <span className="text-muted-foreground">—</span>
   192→  const margin = ((product.price - product.cost) / product.price) * 100
   193→  const color =
   194→    margin > 40
   195→      ? 'text-emerald-600 dark:text-emerald-400'
   196→      : margin >= 20
   197→        ? 'text-amber-600 dark:text-amber-500'
   198→        : 'text-rose-600 dark:text-rose-400'
   199→  return <span className={cn('text-xs font-medium tabular-nums', color)}>{Math.round(margin)}%</span>
   200→}
   201→
   202→function StockCell({ product }: { product: Product }) {
   203→  if (!product.isStockable) return <span className="text-muted-foreground">—</span>
   204→  return <span className={cn('tabular-nums', stockColorClass(product))}>{formatQty(product.stock)}</span>
   205→}

   230→function ActiveBadge({ active, label }: { active: boolean; label: { active: string; inactive: string } }) {
   231→  if (!active)
   232→    return (
   233→      <Badge variant="outline" className="text-muted-foreground">
   234→        {label.inactive}
   235→      </Badge>
   236→    )
   237→  return (
   238→    <Badge
   239→      variant="outline"
   240→      className="border-emerald-600/30 bg-emerald-50 text-emerald-700 dark:border-emerald-500/30 dark:bg-emerald-500/10 dark:text-emerald-400"
   241→    >
   242→      {label.active}
   243→    </Badge>
   244→  )
   245→}

   284→export default function ProductsView() {
   285→  const { t } = useI18n()
   286→  const queryClient = useQueryClient()
   287→
   288→  const [search, setSearch] = useState('')
   289→  const [categoryFilter, setCategoryFilter] = useState('all')
   290→  const [showInactive, setShowInactive] = useState(false)
   291→
   292→  const [formOpen, setFormOpen] = useState(false)
   293→  const [editing, setEditing] = useState<Product | null>(null)
   294→  const [form, setForm] = useState<ProductForm>(EMPTY_PRODUCT_FORM)
   295→
   296→  const [deleteTarget, setDeleteTarget] = useState<Product | null>(null)
   297→
   298→  const productsQuery = useQuery({
   299→    queryKey: ['products', { all: showInactive, category: categoryFilter }],
   300→    queryFn: () =>
   301→      fetcher<{ products: Product[] }>(
   302→        `/api/products?${new URLSearchParams({
   303→          ...(showInactive ? { all: '1' } : {}),
   304→          ...(categoryFilter !== 'all' ? { category: categoryFilter } : {}),
   305→        }).toString()}`,
   306→      ),
   307→  })
   308→
   309→  const categoriesQuery = useQuery({
   310→    queryKey: ['categories', 'all'],
   311→    queryFn: () => fetcher<{ categories: Category[] }>('/api/categories?all=1'),
   312→  })
   313→
   314→  const categories = useMemo(
   315→    () =>
   316→      [...(categoriesQuery.data?.categories ?? [])].sort(
   317→        (a, b) => a.displayOrder - b.displayOrder || a.name.localeCompare(b.name),
   318→      ),
   319→    [categoriesQuery.data],
   320→  )
   321→
   322→  const products = useMemo(() => {
   323→    const items = productsQuery.data?.products ?? []
   324→    const q = search.trim().toLowerCase()
   325→    const filtered =
   326→      q === ''
   327→        ? items
   328→        : items.filter(
   329→            (p) => p.name.toLowerCase().includes(q) || (p.sku ?? '').toLowerCase().includes(q),
   330→          )
   331→    return [...filtered].sort((a, b) => a.name.localeCompare(b.name))
   332→  }, [productsQuery.data, search])
   333→
   334→  const isCreate = editing === null
   335→  const errors = validateProductForm(form, isCreate, t)
   336→  const hasErrors = Object.values(errors).some(Boolean)
   337→
   338→  const saveMutation = useMutation({
   339→    mutationFn: async () => {
   340→      const payload: Record<string, unknown> = {
   341→        name: form.name.trim(),
   342→        price: Number(form.price),
   343→        cost: form.cost.trim() === '' ? 0 : Number(form.cost),
   344→        isSellable: form.isSellable,
   345→        isStockable: form.isStockable,
   346→      }
   347→      if (form.categoryId !== '') payload.categoryId = Number(form.categoryId)
   348→      if (form.sku.trim() !== '') payload.sku = form.sku.trim()
   349→      if (form.imageUrl.trim() !== '') payload.imageUrl = form.imageUrl.trim()
   350→      if (form.isStockable) {
   351→        payload.lowStockThreshold =
   352→          form.lowStockThreshold.trim() === '' ? 0 : Number(form.lowStockThreshold)
   353→        if (editing === null) {
   354→          payload.stock = form.stock.trim() === '' ? 0 : Number(form.stock)
   355→        }
   356→      }
   357→      return editing === null
   358→        ? apiFetch<{ product: Product }>('/api/products', { method: 'POST', body: payload })
   359→        : apiFetch<{ product: Product }>(`/api/products/${editing.id}`, {
   360→            method: 'PUT',
   361→            body: payload,
   362→          })
   363→    },
   364→    onSuccess: () => {
   365→      void queryClient.invalidateQueries({ queryKey: ['products'] })
   366→      toast.success(editing === null ? t('admin.productCreated') : t('admin.productUpdated'))
   367→      setFormOpen(false)
   368→    },
   369→    onError: (err: Error) => toast.error(err.message),
   370→  })
   371→
   372→  const deleteMutation = useMutation({
   373→    mutationFn: (id: number) =>
   374→      apiFetch<{ ok: boolean }>(`/api/products/${id}`, { method: 'DELETE' }),
   375→    onSuccess: () => {
   376→      void queryClient.invalidateQueries({ queryKey: ['products'] })
   377→      toast.success(t('admin.productDeactivated'))
   378→      setDeleteTarget(null)
   379→    },
   380→    onError: (err: Error) => toast.error(err.message),
   381→  })
   382→
   383→  function openCreate() {
   384→    setEditing(null)
   385→    setForm(EMPTY_PRODUCT_FORM)
   386→    setFormOpen(true)
   387→  }
   388→
   389→  function openEdit(product: Product) {
   390→    setEditing(product)
   391→    setForm(toProductForm(product))
   392→    setFormOpen(true)
   393→  }
   394→
   395→  const hasActiveFilters = search.trim() !== '' || categoryFilter !== 'all'
   396→
   397→  return (
   398→    <div className="mx-auto w-full max-w-7xl space-y-6 p-4 sm:p-6">
   399→      {/* Header */}
   400→      <div className="flex flex-wrap items-start justify-between gap-4">
   401→        <div className="space-y-1">
   402→          <h1 className="text-2xl font-bold">{t('nav.products')}</h1>
   403→          <p className="text-muted-foreground text-sm">{t('admin.productsSubtitle')}</p>
   404→        </div>
   405→        <Button className="h-11" onClick={openCreate}>
   406→          <Plus /> {t('admin.newProduct')}
   407→        </Button>
   408→      </div>
   409→
   410→      {/* Toolbar */}
   411→      <Card className="p-3">
   412→        <div className="flex flex-wrap items-center gap-2">
   413→          <div className="relative w-full sm:w-72">
   414→            <Search
   415→              className="pointer-events-none absolute top-1/2 start-3 size-4 -translate-y-1/2 text-muted-foreground"
   416→              aria-hidden
   417→            />
   418→            <Input
   419→              value={search}
   420→              onChange={(e) => setSearch(e.target.value)}
   421→              placeholder={t('admin.searchProduct')}
   422→              className="h-11 ps-9"
   423→              aria-label={t('admin.searchProduct')}
   424→            />
   425→          </div>
   426→          <Select value={categoryFilter} onValueChange={setCategoryFilter}>
   427→            <SelectTrigger className="h-11 w-full sm:w-52" aria-label={t('admin.allCategories')}>
   428→              <SelectValue placeholder={t('admin.allCategories')} />
   429→            </SelectTrigger>
   430→            <SelectContent>
   431→              <SelectItem value="all">{t('admin.allCategories')}</SelectItem>
   432→              {categories.map((c) => (
   433→                <SelectItem
   434→                  key={c.id}
   435→                  value={String(c.id)}
   436→                  className={c.active ? undefined : 'text-muted-foreground'}
   437→                >
   438→                  {c.name}
   439→                  {c.active ? '' : ` ${t('admin.inactiveSuffix')}`}
   440→                </SelectItem>
   441→              ))}
   442→            </SelectContent>
   443→          </Select>
   444→          <div className="flex h-11 items-center gap-2 rounded-md border px-3">
   445→            <Switch
   446→              id="products-show-inactive"
   447→              checked={showInactive}
   448→              onCheckedChange={setShowInactive}
   449→            />
   450→            <Label htmlFor="products-show-inactive" className="cursor-pointer text-sm">
   451→              {t('admin.showInactive')}
   452→            </Label>
   453→          </div>
   454→        </div>
   455→      </Card>
   456→
   457→      {/* Content */}
   458→      <Card className="p-4">
   459→        {productsQuery.isLoading ? (
   460→          <div className="space-y-2">
   461→            {Array.from({ length: 8 }, (_, i) => (
   462→              <div key={i} className="flex items-center gap-4 rounded-lg border p-3">
   463→                <div className="flex-1 space-y-2">
   464→                  <Skeleton className="h-4 w-44" />
   465→                  <Skeleton className="h-3 w-24" />
   466→                </div>
   467→                <Skeleton className="h-4 w-16" />
   468→                <Skeleton className="size-11 rounded-md" />
   469→              </div>
   470→            ))}
   471→          </div>
   472→        ) : productsQuery.isError ? (
   473→          <div className="flex flex-col items-center justify-center gap-3 py-12 text-center">
   474→            <TriangleAlert className="size-10 text-destructive" aria-hidden />
   475→            <div>
   476→              <p className="font-medium">{t('admin.loadProductsFailed')}</p>
   477→              <p className="text-muted-foreground text-sm">
   478→                {productsQuery.error?.message ?? t('common.error')}

   493→            </div>

   496→                <Plus /> {t('admin.newProduct')}
   497→              </Button>
   498→            )}
   499→          </div>
   500→        ) : (
   501→          <>
   502→            {/* Desktop / tablet table */}
   503→            <div className="hidden md:block">
   504→              <div className="rms-scroll max-h-[520px] overflow-y-auto">
   505→                <Table>
   506→                  <TableHeader>

   510→                      <TableHead className="text-end">{t('common.price')}</TableHead>
   511→                      <TableHead className="hidden text-end lg:table-cell">{t('common.cost')}</TableHead>

   537→                        <TableCell className="text-end font-medium tabular-nums">
   538→                          {formatCurrency(p.price)}

   540→                        <TableCell className="hidden text-end text-muted-foreground tabular-nums lg:table-cell">
   541→                          {formatCurrency(p.cost)}
   542→                        </TableCell>
   543→                        <TableCell className="hidden lg:table-cell">
