    36→import { localizedName, useI18n } from '@/lib/i18n'

    41→
    42→type PaymentModalProps = {
    43→  order: Order
    44→  open: boolean
    45→  onOpenChange: (open: boolean) => void
    46→  onSuccess: (order: Order, closed: boolean) => void
    47→  /** Called after the check is deferred (pay later, client name recorded).
    48→   *  The modal closes itself; the parent resets its order screen state. */
    49→  onDeferred?: (order: Order) => void
    50→}
    51→
    52→type PaymentResult = {
    53→  order: Order
    54→  paidAmount: number
    55→  remaining: number
    56→  closed: boolean
    57→}
    58→
    59→type EditableRow = {
    60→  id: string
    61→  method: string
    62→  amount: string
    63→  reference: string
    64→}
    65→
    66→type SubmitRow = { method: string; amount: number; reference?: string }
    67→
    68→type SplitTab = 'single' | 'equal' | 'items' | 'custom'
    69→
    70→const METHOD_META: Record<string, { labelKey: string; icon: LucideIcon }> = {
    71→  cash: { labelKey: 'status.payment.cash', icon: Banknote },
    72→  card: { labelKey: 'status.payment.card', icon: CreditCard },
    73→  other: { labelKey: 'status.payment.other', icon: MoreHorizontal },
    74→}
    75→
    76→export default function PaymentModal({ order, open, onOpenChange, onSuccess, onDeferred }: PaymentModalProps) {
    77→  const { t, lang, isRTL } = useI18n()
    78→  const [tab, setTab] = useState<SplitTab>('single')
    79→  const [submitting, setSubmitting] = useState(false)
    80→  const [checkOpen, setCheckOpen] = useState(false)
    81→
    82→  // Defer-payment state (check stays open under the client's name)
    83→  const [deferOpen, setDeferOpen] = useState(false)
    84→  const [deferName, setDeferName] = useState('')
    85→  const [deferSubmitting, setDeferSubmitting] = useState(false)
    86→
    87→  // Which payment row the method tiles currently target.
    88→  const [activeIdx, setActiveIdx] = useState(0)
    89→
    90→  // R8: tips — gratuity ON TOP of the bill, attached PER PAYMENT ROW
    91→  // (single tab → one tip; equal/items/custom → per-payer tips). Keyed per
    92→  // tab+row so every payer's tip survives row switching; the preset
    93→  // selection and the custom input text are tracked separately per row.
    94→  const [tipSel, setTipSel] = useState<Record<string, number | 'custom'>>({})
    95→  const [tipCustom, setTipCustom] = useState<Record<string, string>>({})
    96→
    97→  const [singleRow, setSingleRow] = useState<EditableRow>({ id: 'single', method: 'cash', amount: '', reference: '' })
    98→  const [customRows, setCustomRows] = useState<EditableRow[]>([])
    99→
   100→  const [eqPayers, setEqPayers] = useState(2)
   101→  const [eqMethods, setEqMethods] = useState<Record<number, string>>({})
   102→
   103→  const [itPayers, setItPayers] = useState(2)
   104→  const [itMethods, setItMethods] = useState<Record<number, string>>({})
   105→  const [assignments, setAssignments] = useState<Record<number, number>>({})
   106→
   107→  const remaining = round2(Math.max(0, order.remainingAmount))
   108→
   109→  // Reset state whenever the modal opens for (a new) order.
   110→  useEffect(() => {
   111→    if (!open) return
   112→    const rem = round2(Math.max(0, order.remainingAmount))
   113→    setTab('single')
   114→    setSingleRow({ id: 'single', method: 'cash', amount: String(rem), reference: '' })
   115→    setCustomRows([{ id: newDraftKey(), method: 'cash', amount: String(rem), reference: '' }])
   116→    setEqPayers(2)
   117→    setEqMethods({})
   118→    setItPayers(2)
   119→    setItMethods({})
   120→    setAssignments({})
   121→    setSubmitting(false)
   122→    setActiveIdx(0)
   123→    setTipSel({})
   124→    setTipCustom({})
   125→    setCheckOpen(false)
   126→    setDeferOpen(false)
   127→    setDeferName('')
   128→    setDeferSubmitting(false)
   129→  }, [open, order.id, order.remainingAmount])
   130→
   131→  // ── Derived payment rows per tab ──────────────────────────────────
   132→  const eqAmounts = useMemo(() => {
   133→    const part = round2(remaining / eqPayers)
   134→    const amounts = Array.from({ length: eqPayers }, () => part)
   135→    const others = round2(part * (eqPayers - 1))
   136→    amounts[eqPayers - 1] = round2(Math.max(0, remaining - others))
   137→    return amounts
   138→  }, [remaining, eqPayers])
   139→
   140→  const itemLines = useMemo(
   141→    () =>
   142→      order.items.map((it) => ({
   143→        id: it.id,
   144→        label: `${formatQty(it.quantity)} × ${
   145→          it.product ? localizedName(it.product.name, it.product.nameAr, lang) : t('pos.item')
   146→        }`,
   147→        total: round2(it.quantity * it.unitPrice),
   148→      })),
   149→    [order.items, t, lang],
   150→  )
   151→
   152→  const itAmounts = useMemo(() => {
   153→    const subtotal = round2(order.subtotalAmount)
   154→    const rawTotals = new Array<number>(itPayers).fill(0)
   155→    for (const line of itemLines) {
   156→      const payer = assignments[line.id] ?? 0
   157→      if (payer < itPayers) rawTotals[payer] = round2(rawTotals[payer] + line.total)
   158→    }
   159→    const amounts: number[] = []
   160→    let allocated = 0
   161→    for (let i = 0; i < itPayers; i++) {
   162→      if (i === itPayers - 1) {
   163→        amounts.push(round2(Math.max(0, remaining - allocated)))
   164→      } else {
   165→        const share = subtotal > 0 ? round2((rawTotals[i] / subtotal) * remaining) : 0
   166→        amounts.push(share)
   167→        allocated = round2(allocated + share)
   168→      }
   169→    }
   170→    return amounts
   171→  }, [itemLines, assignments, itPayers, remaining, order.subtotalAmount])
   172→
   173→  const submitRows: SubmitRow[] = useMemo(() => {
   174→    if (tab === 'single') {
   175→      return [toSubmitRow(singleRow)]
   176→    }
   177→    if (tab === 'equal') {
   178→      return eqAmounts.map((amount, i) => ({
   179→        method: eqMethods[i] ?? 'cash',
   180→        amount,
   181→      }))
   182→    }
   183→    if (tab === 'items') {
   184→      return itAmounts.map((amount, i) => ({
   185→        method: itMethods[i] ?? 'cash',
   186→        amount,
   187→      }))
   188→    }
   189→    return customRows.map(toSubmitRow)
   190→  }, [tab, singleRow, customRows, eqAmounts, eqMethods, itAmounts, itMethods])
   191→
   192→  const sum = round2(submitRows.reduce((s, r) => s + r.amount, 0))
   193→  const diff = round2(sum - remaining)
   194→  const exceeds = diff > 0.01
   195→  const exact = Math.abs(diff) <= 0.01
   196→
   197→  // ── Active row / method tiles ─────────────────────────────────────
   198→  const rowCount =
   199→    tab === 'single' ? 1 : tab === 'equal' ? eqPayers : tab === 'items' ? itPayers : customRows.length
   200→  const safeActiveIdx = Math.min(activeIdx, Math.max(0, rowCount - 1))
   201→
   202→  // ── Tips (R8) ────────────────────────────────────────────────
   203→  // Tips sit ON TOP of the bill: `remaining` stays bill-only and the
   204→  // validation below is unchanged (Σ row amounts ≤ remaining as before).
   205→
   206→  /** Stable per-row key for the tip state ('single' | 'eq:0' | 'it:1' | 'c:<rowId>'). */
   207→  const tipKeyFor = (index: number): string => {
   208→    if (tab === 'single') return 'single'
   209→    if (tab === 'equal') return `eq:${index}`
   210→    if (tab === 'items') return `it:${index}`
   211→    return `c:${customRows[index]?.id ?? index}`
   212→  }
   213→
   214→  const activeTipKey = tipKeyFor(safeActiveIdx)
   215→  const activeSel = tipSel[activeTipKey]
   216→  const activeCustomText = tipCustom[activeTipKey] ?? ''
   217→
   218→  const activeRowAmount = useMemo(
   219→    () => round2(submitRows[safeActiveIdx]?.amount ?? 0),
   220→    [submitRows, safeActiveIdx],
   221→  )
   222→
   223→  /** Tip for a submit-row index: {value, invalid} — presets are % of that row's amount. */
   224→  const rowTip = (index: number): { value: number; invalid: boolean } => {
   225→    const key = tipKeyFor(index)
   226→    const sel = tipSel[key]
   227→    if (sel == null || sel === 0) return { value: 0, invalid: false }
   228→    if (sel === 'custom') {
   229→      const parsed = parseTipInput(tipCustom[key] ?? '')
   230→      return parsed == null ? { value: 0, invalid: true } : { value: parsed, invalid: false }
   231→    }
   232→    return { value: round2((submitRows[index]?.amount ?? 0) * (sel / 100)), invalid: false }
   233→  }
   234→
   235→  const activeTip = rowTip(safeActiveIdx)
   236→  const tipsTotal = round2(submitRows.reduce((s, _r, i) => s + rowTip(i).value, 0))
   237→  const tipInvalid = submitRows.some((_r, i) => rowTip(i).invalid)
   238→
   239→  const canSubmit = !submitting && remaining > 0 && sum > 0 && !exceeds && !tipInvalid
   240→
   241→  const activeRowLabel =
   242→    tab === 'single'
   243→      ? t('pos.payment')
   244→      : tab === 'equal' || tab === 'items'
   245→        ? t('pos.payer', { n: safeActiveIdx + 1 })
   246→        : t('pos.paymentN', { n: safeActiveIdx + 1 })
   247→
   248→  const currentMethodAt = (index: number): string => {
   249→    if (tab === 'single') return singleRow.method
   250→    if (tab === 'equal') return eqMethods[index] ?? 'cash'
   251→    if (tab === 'items') return itMethods[index] ?? 'cash'
   252→    return customRows[index]?.method ?? 'cash'
   253→  }
   254→
   255→  const setMethodAt = (index: number, m: string) => {
   256→    if (tab === 'single') {
   257→      setSingleRow((r) => ({ ...r, method: m }))
   258→    } else if (tab === 'equal') {
   259→      setEqMethods((prev) => ({ ...prev, [index]: m }))
   260→    } else if (tab === 'items') {
   261→      setItMethods((prev) => ({ ...prev, [index]: m }))
   262→    } else {
   263→      setCustomRows((rows) => rows.map((r, i) => (i === index ? { ...r, method: m } : r)))
   264→    }
   265→  }
   266→
   267→  const handleTabChange = (v: string) => {
   268→    setTab(v as SplitTab)
   269→    setActiveIdx(0)
   270→  }
   271→
   272→  // ── Guest check rows mirroring the current split configuration ────
   273→  const checkRows: CheckSplitRow[] = useMemo(() => {
   274→    if (tab === 'single') {
   275→      const amount = parseAmount(singleRow.amount) || remaining
   276→      return [{ label: t('pos.fullBill'), amount, method: singleRow.method }]
   277→    }
   278→    if (tab === 'equal') {
   279→      return eqAmounts.map((amount, i) => ({
   280→        label: t('pos.partOf', { i: i + 1, n: eqPayers }),
   281→        amount,
   282→        method: eqMethods[i] ?? 'cash',
   283→      }))
   284→    }
   285→    if (tab === 'items') {
   286→      return itAmounts.map((amount, i) => ({
   287→        label: t('pos.payer', { n: i + 1 }),
   288→        amount,
   289→        method: itMethods[i] ?? 'cash',
   290→      }))
   291→    }
   292→    return customRows.map((r, i) => ({
   293→      label: t('pos.paymentN', { n: i + 1 }),
   294→      amount: parseAmount(r.amount),
   295→      method: r.method,
   296→    }))
   297→  }, [tab, singleRow, customRows, eqAmounts, eqMethods, eqPayers, itAmounts, itMethods, remaining, t])
   298→
   299→  // ── Handlers ──────────────────────────────────────────────────────
   300→  const clampPayers = (raw: number, min: number, max: number) => {
   301→    const n = Math.round(Number.isFinite(raw) ? raw : min)
   302→    return Math.min(max, Math.max(min, n))
   303→  }
   304→
   305→  const setAssignment = (itemId: number, payer: number) =>
   306→    setAssignments((prev) => ({ ...prev, [itemId]: payer }))
   307→
   308→  const addCustomRow = () => {
   309→    const current = round2(customRows.reduce((s, r) => s + parseAmount(r.amount), 0))
   310→    const next = round2(Math.max(0, remaining - current))
   311→    setCustomRows((rows) => [...rows, { id: newDraftKey(), method: 'cash', amount: String(next), reference: '' }])
   312→  }
   313→
   314→  const handleSubmit = async () => {
   315→    // R8: each payment row carries its own tip (≥ 0, round2) on top of the
   316→    // amount — the API persists it but never counts it toward paidAmount.
   317→    const payments = submitRows
   318→      .map((r, i) => ({ ...r, amount: round2(r.amount), tip: Math.max(0, round2(rowTip(i).value)) }))
   319→      .filter((r) => r.amount > 0)
   320→      .map((r) => ({
   321→        method: r.method,
   322→        amount: r.amount,
   323→        tip: r.tip,
   324→        ...(r.method === 'card' && r.reference?.trim() ? { reference: r.reference.trim() } : {}),
   325→      }))
   326→    if (payments.length === 0) return
   327→    setSubmitting(true)
   328→    try {
   329→      const result = await apiFetch<PaymentResult>(`/api/orders/${order.id}/payments`, {
   330→        method: 'POST',
   331→        body: { payments },
   332→      })
   333→      if (result.closed) {
   334→        toast.success(t('pos.paidClosedToast'))
   335→        onSuccess(result.order, true)
   336→        onOpenChange(false)
   337→      } else {
   338→        toast.success(t('pos.paymentRecordedToast', { amount: formatCurrency(result.remaining) }))
   339→        onSuccess(result.order, false)
   340→        onOpenChange(false)
   341→      }
   342→    } catch (err) {
   343→      toast.error(err instanceof Error ? err.message : t('pos.paymentFailedToast'))
   344→    } finally {
   345→      setSubmitting(false)
   346→    }
   347→  }
   348→
   349→  // ── Defer payment (client pays later, check tracked by name) ──────
   350→  const canDefer = order.status === 'open' && remaining > 0
   351→
   352→  const handleDefer = async () => {
   353→    const clientName = deferName.trim()
   354→    if (!clientName) {
   355→      toast.error(t('pos.deferredNameRequired'))
   356→      return
   357→    }
   358→    if (deferSubmitting) return
   359→    setDeferSubmitting(true)
   360→    try {
   361→      const result = await apiFetch<{ order: Order }>(`/api/orders/${order.id}/defer`, {
   362→        method: 'POST',
   363→        body: { clientName },
   364→      })
   365→      toast.success(t('pos.deferredToast', { order: order.id, client: clientName }))
   366→      setDeferOpen(false)
   367→      onDeferred?.(result.order)
   368→      onOpenChange(false)
   369→    } catch (err) {
   370→      toast.error(err instanceof Error ? err.message : t('pos.paymentFailedToast'))
   371→    } finally {
   372→      setDeferSubmitting(false)
   373→    }
   374→  }
   375→
   376→  const singleExceeds = tab === 'single' && parseAmount(singleRow.amount) > round2(remaining + 0.01)
   377→
   378→  return (
   379→    <>
   380→      <Dialog open={open} onOpenChange={onOpenChange}>
   381→        <DialogContent className="max-h-[92dvh] overflow-y-auto sm:max-w-lg">
   382→          <DialogHeader>
   383→            <DialogTitle>
   384→              {t('pos.payment')} · {order.table?.name ?? t('common.takeaway')} ·{' '}
   385→              {t('common.order')} #{order.id}
   386→            </DialogTitle>
   387→            <DialogDescription>{t('pos.paymentDesc')}</DialogDescription>
   388→          </DialogHeader>
   389→
   390→          {/* Summary */}
   391→          <div className="grid grid-cols-3 gap-2 rounded-xl border border-[#E2E2E0] bg-white p-3 text-center shadow-sm">
   392→            <div>
   393→              <p className="text-xs text-muted-foreground">{t('money.total')}</p>
   394→              <p className="text-sm font-semibold tabular-nums">{formatCurrency(order.totalAmount)}</p>
   395→            </div>
   396→            <div>
   397→              <p className="text-xs text-muted-foreground">{t('money.paid')}</p>
   398→              <p className="text-sm font-semibold tabular-nums text-emerald-600">
   399→                {formatCurrency(order.paidAmount)}
   400→              </p>
   401→            </div>
   402→            <div>
   403→              <p className="text-xs text-muted-foreground">{t('money.remaining')}</p>
   404→              <p className="text-primary text-2xl font-bold tabular-nums">
   405→                {formatCurrency(remaining)}
   406→              </p>
   407→            </div>
   408→          </div>
   409→
   410→          {/* Method tiles — apply to the active row below */}
   411→          <div className="space-y-1.5">
   412→            <p className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   413→              {t('pos.methodAppliesTo')} <span className="text-[#714B67]">{activeRowLabel}</span>
   414→            </p>
   415→            <div className="grid grid-cols-3 gap-2">
   416→              {PAYMENT_METHODS.map((m) => {
   417→                const meta = METHOD_META[m] ?? { labelKey: `status.payment.${m}`, icon: MoreHorizontal }
   418→                const Icon = meta.icon
   419→                const active = currentMethodAt(safeActiveIdx) === m
   420→                return (
   421→                  <button
   422→                    key={m}
   423→                    type="button"
   424→                    onClick={() => setMethodAt(safeActiveIdx, m)}
   425→                    aria-pressed={active}
   426→                    className={cn(
   427→                      'flex h-16 flex-col items-center justify-center gap-1 rounded-xl border-2 text-sm font-semibold transition active:scale-95',
   428→                      active
   429→                        ? 'border-[#714B67] bg-[#714B67]/10 text-[#714B67]'
   430→                        : 'border-[#E2E2E0] bg-white text-stone-600 hover:border-[#714B67]/40',
   431→                    )}
   432→                  >
   433→                    <Icon className="size-6" aria-hidden />
   434→                    {t(meta.labelKey)}
   435→                  </button>
   436→                )
   437→              })}
   438→            </div>
   439→          </div>
   440→
   441→          <Tabs value={tab} onValueChange={handleTabChange}>
   442→            <TabsList className="grid h-10 w-full grid-cols-4">
   443→              <TabsTrigger value="single" className="text-xs sm:text-sm">
   444→                {t('pos.tabSingle')}
   445→              </TabsTrigger>
   446→              <TabsTrigger value="equal" className="text-xs sm:text-sm">
   447→                {t('pos.tabEqual')}
   448→              </TabsTrigger>
   449→              <TabsTrigger value="items" className="text-xs sm:text-sm">
   450→                {t('pos.tabItems')}
   451→              </TabsTrigger>
   452→              <TabsTrigger value="custom" className="text-xs sm:text-sm">
   453→                {t('pos.tabCustom')}
   454→              </TabsTrigger>
   455→            </TabsList>
   456→
   457→            {/* ── Single ── */}
   458→            <TabsContent value="single" className="space-y-2 pt-3">
   459→              <PayRow
   460→                label={t('pos.payment')}
   461→                method={singleRow.method}
   462→                active
   463→                amountStr={singleRow.amount}
   464→                onAmountChange={(v) => setSingleRow((r) => ({ ...r, amount: v }))}
   465→                reference={singleRow.reference}
   466→                onReferenceChange={(v) => setSingleRow((r) => ({ ...r, reference: v }))}
   467→                editable
   468→                error={
   469→                  singleExceeds
   470→                    ? t('pos.exceedsBy', {
   471→                        amount: formatCurrency(round2(parseAmount(singleRow.amount) - remaining)),
   472→                      })
   473→                    : undefined
   474→                }
   475→              />
   476→            </TabsContent>
   477→
   478→            {/* ── Equal Split ── */}
   479→            <TabsContent value="equal" className="space-y-3 pt-3">
   480→              <div className="flex items-center gap-3 rounded-xl border border-[#E2E2E0] bg-white p-2.5">
   481→                <Users className="size-4 text-muted-foreground" />
   482→                <span className="flex-1 text-sm font-medium">{t('pos.splitBetween')}</span>
   483→                <Input
   484→                  type="number"
   485→                  min={2}
   486→                  max={12}
   487→                  value={eqPayers}
   488→                  onChange={(e) => {
   489→                    const next = clampPayers(parseInt(e.target.value, 10) || 2, 2, 12)
   490→                    setEqPayers(next)
   491→                    setActiveIdx((i) => Math.min(i, next - 1))
   492→                  }}
   493→                  className="h-10 w-16 text-center tabular-nums"
   494→                />
   495→                <span className="text-sm text-muted-foreground">{t('pos.payers')}</span>
   496→              </div>
   497→              {/* R11: customer-by-customer navigation — advances the payer the
   498→                  method tiles + tips target. */}
   499→              <PayerNav count={eqPayers} activeIdx={safeActiveIdx} onSelect={setActiveIdx} isRTL={isRTL} />
   500→              {eqAmounts.map((amount, i) => (
   501→                <PayRow
   502→                  key={i}
   503→                  label={t('pos.payer', { n: i + 1 })}
   504→                  method={eqMethods[i] ?? 'cash'}
   505→                  active={safeActiveIdx === i}
   506→                  onActivate={() => setActiveIdx(i)}
   507→                  amount={amount}
   508→                />
   509→              ))}
   510→              <p className="text-center text-xs text-muted-foreground">
   511→                {t('pos.partsTotal')}{' '}
   512→                <span className="font-semibold tabular-nums">{formatCurrency(sum)}</span>
   513→              </p>
   514→            </TabsContent>
   515→

   834→/** Parse a custom tip input: null = invalid (blocks submit), '' = 0, else ≥ 0 rounded. */
   835→function parseTipInput(s: string): number | null {
   836→  if (s.trim() === '') return 0
   837→  const v = Number(s)
   838→  if (!Number.isFinite(v) || v < 0) return null
   839→  return round2(v)
   840→}

   906→function PayRow({
   907→  label,
   908→  method,
   909→  active = false,
   910→  onActivate,
   911→  amount,

   914→  reference,

   918→  error,

   921→  method: string
   922→  /** Whether the method tiles currently target this row. */
   923→  active?: boolean
   924→  onActivate?: () => void
   925→  amount?: number
   926→  amountStr?: string
   927→  onAmountChange?: (v: string) => void
   928→  reference?: string

   940→      className={cn(
   941→        'cursor-pointer rounded-xl border p-2.5 transition-colors',

   998→        <Input
   999→          value={reference ?? ''}
  1000→          onChange={(e) => onReferenceChange(e.target.value)}
  1001→          placeholder={t('pos.refOptional')}
  1002→          className="mt-1.5 h-10"
