    44→
    45→type TableSelectProps = {
    46→  onSelectTable: (table: RestaurantTable) => void
    47→  onTakeaway: () => void
    48→  onOpenTakeawayOrder: (order: Order) => void
    49→  /** When set, start directly in transfer step 2 (destination selection) for this order. */
    50→  transferOrderId?: number | null
    51→  /** Called after a transfer completes (pos-view resets its order screen state). */
    52→  onTransferDone?: () => void
    53→  /** Seat Party tool result — the free tables selected for the party, in
    54→   *  selection order (primary table first). TableSelect exits the tool itself. */
    55→  onSeatParty: (tables: RestaurantTable[]) => void
    56→  /** Deferred-check chip clicked — pos-view opens the payment modal to settle it. */
    57→  onSettleDeferred: (order: Order) => void
    58→}
    59→
    60→type ToolKind = 'transfer' | 'merge' | 'seat'
    61→type SourcePick = { orderId: number }
    62→
    63→type TableInteraction = 'normal' | 'eligible' | 'ineligible'
    64→
    65→/** Per-shape tile silhouette classes — Odoo 17-style tiles. The floor grid
    66→ *  keeps its responsive layout (tiles fill the grid cell width); only the
    67→ *  silhouette changes (square/round/rectangle/oval). The admin floor-plan
    68→ *  editor mirrors these exact strings. */
    69→const SHAPE_TILE_CLASSES: Record<string, string> = {
    70→  square: 'aspect-square rounded-2xl',
    71→  round: 'aspect-square rounded-full',
    72→  rectangle: 'h-24 rounded-2xl',
    73→  oval: 'h-24 rounded-full',
    74→}
    75→
    76→function shapeClasses(shape: string): string {
    77→  return SHAPE_TILE_CLASSES[shape] ?? SHAPE_TILE_CLASSES.square
    78→}
    79→
    80→export default function TableSelect({
    81→  onSelectTable,
    82→  onTakeaway,
    83→  onOpenTakeawayOrder,
    84→  transferOrderId,
    85→  onTransferDone,
    86→  onSeatParty,
    87→  onSettleDeferred,
    88→}: TableSelectProps) {
    89→  const queryClient = useQueryClient()
    90→  const { t } = useI18n()
    91→
    92→  const [floorIdx, setFloorIdx] = useState(0)
    93→  // When pos-view passes transferOrderId (Transfer clicked on the order
    94→  // screen), this component mounts fresh in transfer step 2 (destination
    95→  // selection) — pos-view only sets it right before switching to tables view,
    96→  // so mount-time initialization is sufficient.
    97→  const [tool, setTool] = useState<ToolKind | null>(transferOrderId != null ? 'transfer' : null)
    98→  const [source, setSource] = useState<SourcePick | null>(
    99→    transferOrderId != null ? { orderId: transferOrderId } : null,
   100→  )
   101→  // Seat Party: the free tables tapped so far, in selection order (the first
   102→  // one becomes the primary table of the merged seating).
   103→  const [seatSelection, setSeatSelection] = useState<RestaurantTable[]>([])
   104→  // Paid/deferred tile tap-to-clear confirmation dialog target.
   105→  const [clearTarget, setClearTarget] = useState<RestaurantTable | null>(null)
   106→
   107→  const { data: floorPlanData, isLoading } = useQuery({
   108→    queryKey: ['floorplans'],
   109→    queryFn: () => fetcher<{ floorPlans: FloorPlan[] }>('/api/floorplans'),
   110→    refetchInterval: 3000, // LIVE table statuses
   111→  })
   112→
   113→  const { data: openOrdersData } = useQuery({
   114→    queryKey: ['orders', 'open'],
   115→    queryFn: () => fetcher<{ orders: Order[] }>('/api/orders?status=open'),
   116→    refetchInterval: 3000,
   117→  })
   118→
   119→  // Outstanding deferred checks (client left, payment pending) — settle chips.
   120→  const { data: deferredOrdersData } = useQuery({
   121→    queryKey: ['orders', 'deferred'],
   122→    queryFn: () => fetcher<{ orders: Order[] }>('/api/orders?status=deferred'),
   123→    refetchInterval: 3000,
   124→    enabled: !tool,
   125→  })
   126→
   127→  const floorPlans = floorPlanData?.floorPlans ?? []
   128→  const openOrders = openOrdersData?.orders ?? []
   129→  const deferredOrders = deferredOrdersData?.orders ?? []
   130→  const takeawayOrders = useMemo(
   131→    () => openOrders.filter((o) => o.tableId === null),
   132→    [openOrders],
   133→  )
   134→
   135→  const idx = floorPlans.length > 0 ? Math.min(floorIdx, floorPlans.length - 1) : 0
   136→  const plan = floorPlans[idx] ?? null
   137→
   138→  const sourceOrder = source ? openOrders.find((o) => o.id === source.orderId) ?? null : null
   139→  const sourceLabel = source ? sourceOrder?.table?.name ?? t('common.takeaway') : null
   140→
   141→  // ── Manual table turnover (two clicks): paid/deferred → bus → amber
   142→  // 'dirty' (needs cleaning) → clean → free. Audit-logged server-side.
   143→  const clearTableMutation = useMutation({
   144→    mutationFn: (vars: { id: number; name: string; phase: 'bus' | 'clean' }) =>
   145→      apiFetch<{ table: RestaurantTable }>(`/api/tables/${vars.id}/clear`, {
   146→        method: 'POST',
   147→        body: { action: vars.phase },
   148→      }),
   149→    onSuccess: async (_data, vars) => {
   150→      toast.success(
   151→        vars.phase === 'bus'
   152→          ? t('pos.tableBussedToast', { table: vars.name })
   153→          : t('pos.tableCleanedToast', { table: vars.name }),
   154→      )
   155→      setClearTarget(null)
   156→      await Promise.all([
   157→        queryClient.invalidateQueries({ queryKey: ['floorplans'] }),
   158→        queryClient.invalidateQueries({ queryKey: ['tables-status'] }),
   159→        queryClient.invalidateQueries({ queryKey: ['orders'] }),
   160→      ])
   161→    },
   162→    onError: (err: Error) => toast.error(err.message),
   163→  })
   164→
   165→  // ── Transfer & merge mutations ───────────────────────────────────
   166→  const invalidateAfterMove = async (...orderIds: number[]) => {
   167→    await Promise.all([
   168→      queryClient.invalidateQueries({ queryKey: ['floorplans'] }),
   169→      queryClient.invalidateQueries({ queryKey: ['orders'] }),
   170→      queryClient.invalidateQueries({ queryKey: ['orders', 'open'] }),
   171→      queryClient.invalidateQueries({ queryKey: ['tables-status'] }),
   172→      ...orderIds.map((id) => queryClient.invalidateQueries({ queryKey: ['pos-order', id] })),
   173→    ])
   174→  }
   175→
   176→  const transferMutation = useMutation({
   177→    mutationFn: (vars: { orderId: number; tableId: number; tableName: string }) =>
   178→      apiFetch<{ order: Order }>(`/api/orders/${vars.orderId}/transfer`, {
   179→        method: 'POST',
   180→        body: { tableId: vars.tableId },
   181→      }),
   182→    onSuccess: async (_data, vars) => {
   183→      toast.success(t('pos.transferToast', { order: vars.orderId, table: vars.tableName }))
   184→      await invalidateAfterMove(vars.orderId)
   185→      exitTool()
   186→      onTransferDone?.()
   187→    },
   188→    onError: (err: Error) => toast.error(err.message),
   189→  })
   190→
   191→  const mergeMutation = useMutation({
   192→    mutationFn: (vars: { sourceId: number; targetId: number; targetLabel: string }) =>
   193→      apiFetch<{ order: Order }>(`/api/orders/${vars.targetId}/merge`, {
   194→        method: 'POST',
   195→        body: { sourceOrderId: vars.sourceId },
   196→      }),
   197→    onSuccess: async (_data, vars) => {
   198→      toast.success(t('pos.mergeToast', { table: vars.targetLabel }))
   199→      await invalidateAfterMove(vars.sourceId, vars.targetId)
   200→      exitTool()
   201→    },
   202→    onError: (err: Error) => toast.error(err.message),
   203→  })
   204→
   205→  const busy = transferMutation.isPending || mergeMutation.isPending
   206→
   207→  // ── Tool mode helpers ────────────────────────────────────────────
   208→  const exitTool = () => {
   209→    setTool(null)
   210→    setSource(null)
   211→    setSeatSelection([])
   212→  }
   213→
   214→  const startTransferTool = () => {
   215→    setTool('transfer')
   216→    setSource(null)
   217→    setSeatSelection([])
   218→  }
   219→
   220→  const startMergeTool = () => {
   221→    setTool('merge')
   222→    setSource(null)
   223→    setSeatSelection([])
   224→  }
   225→
   226→  const startSeatTool = () => {
   227→    setTool('seat')
   228→    setSource(null)
   229→    setSeatSelection([])
   230→  }
   231→
   232→  // Paid/deferred/dirty tables hold no open order — they await the manual
   233→  // bussing/cleaning clicks (amber turnover flow).
   234→  const isCleanupState = (t2: RestaurantTable) =>
   235→    t2.status === 'paid' || t2.status === 'deferred' || t2.status === 'dirty'
   236→
   237→  const tableInteraction = (t2: RestaurantTable): TableInteraction => {
   238→    if (!tool || busy) return busy ? 'ineligible' : 'normal'
   239→    if (tool === 'seat') {
   240→      // Only free tables can host a new party.
   241→      return t2.status === 'free' && t2.openOrderId == null ? 'normal' : 'ineligible'
   242→    }
   243→    if (tool === 'transfer') {
   244→      if (source == null) return t2.openOrderId != null ? 'eligible' : 'ineligible'
   245→      // Destination: free/reserved tables with no open order (paid/deferred/
   246→      // dirty tables await their turnover clicks, they are not destinations).
   247→      return t2.openOrderId == null && t2.status !== 'occupied' && !isCleanupState(t2)
   248→        ? 'eligible'
   249→        : 'ineligible'
   250→    }
   251→    // Merge: source = any occupied table; target = another open order's table.
   252→    if (source == null) return t2.openOrderId != null ? 'eligible' : 'ineligible'
   253→    return t2.openOrderId != null && t2.openOrderId !== source.orderId ? 'eligible' : 'ineligible'
   254→  }
   255→
   256→  const takeawayInteraction = (o: Order): TableInteraction => {
   257→    if (!tool || busy) return busy ? 'ineligible' : 'normal'
   258→    if (tool === 'seat') return 'ineligible'
   259→    if (tool === 'transfer') {
   260→      // Takeaway orders are valid transfer sources, never destinations.
   261→      return source == null ? 'eligible' : 'ineligible'
   262→    }
   263→    if (source == null) return 'eligible'
   264→    return o.id !== source.orderId ? 'eligible' : 'ineligible'
   265→  }
   266→
   267→  const handleTableClick = (t2: RestaurantTable) => {
   268→    if (busy) return
   269→    if (tool === 'seat') {
   270→      if (t2.status !== 'free' || t2.openOrderId != null) return
   271→      setSeatSelection((prev) => {
   272→        if (prev.some((s) => s.id === t2.id)) return prev.filter((s) => s.id !== t2.id)
   273→        if (prev.length >= MAX_SEATING_TABLES) return prev // max 4 tables per seating
   274→        return [...prev, t2]
   275→      })
   276→      return
   277→    }
   278→    if (tool === 'transfer') {
   279→      if (source == null) {
   280→        if (t2.openOrderId == null) return
   281→        setSource({ orderId: t2.openOrderId })
   282→        return
   283→      }
   284→      if (t2.openOrderId != null || t2.status === 'occupied' || isCleanupState(t2)) return
   285→      transferMutation.mutate({ orderId: source.orderId, tableId: t2.id, tableName: t2.name })
   286→      return
   287→    }
   288→    if (tool === 'merge') {
   289→      if (source == null) {
   290→        if (t2.openOrderId == null) return
   291→        setSource({ orderId: t2.openOrderId })
   292→        return
   293→      }
   294→      if (t2.openOrderId == null || t2.openOrderId === source.orderId) return
   295→      mergeMutation.mutate({ sourceId: source.orderId, targetId: t2.openOrderId, targetLabel: t2.name })
   296→      return
   297→    }
   298→    // Normal mode — paid/deferred tiles open the bussing dialog (→ amber
   299→    // 'dirty'), dirty tiles the cleaning dialog (→ free).
   300→    if (isCleanupState(t2) && t2.openOrderId == null) {
   301→      setClearTarget(t2)
   302→      return
   303→    }
   304→    onSelectTable(t2)
   305→  }
   306→
   307→  const handleTakeawayClick = (o: Order) => {
   308→    if (busy) return
   309→    if (tool === 'seat') return
   310→    if (tool === 'transfer') {
   311→      if (source == null) setSource({ orderId: o.id })
   312→      return
   313→    }
   314→    if (tool === 'merge') {
   315→      if (source == null) {
   316→        setSource({ orderId: o.id })
   317→      } else if (o.id !== source.orderId) {
   318→        mergeMutation.mutate({
   319→          sourceId: source.orderId,
   320→          targetId: o.id,
   321→          targetLabel: t('common.takeaway'),
   322→        })
   323→      }
   324→      return
   325→    }
   326→    onOpenTakeawayOrder(o)
   327→  }
   328→
   329→  // Seat Party: hand the selection (in order) to pos-view, then exit the tool.
   330→  const continueSeat = () => {
   331→    if (seatSelection.length === 0) return
   332→    const selected = seatSelection
   333→    exitTool()
   334→    onSeatParty(selected)
   335→  }
   336→
   337→  // ── Banner text (transfer/merge — seat has its own banner) ────────
   338→  const bannerText = busy
   339→    ? tool === 'transfer'
   340→      ? t('pos.transferring')
   341→      : t('pos.merging')
   342→    : tool === 'transfer'
   343→      ? source
   344→        ? sourceLabel
   345→          ? t('pos.transferDest', { order: source.orderId, label: sourceLabel })
   346→          : t('pos.transferDestNoLabel', { order: source.orderId })
   347→        : t('pos.transferPick')
   348→      : source
   349→        ? t('pos.mergeTarget', {
   350→            source: sourceLabel ?? `#${source.orderId}`,
   351→          })
   352→        : t('pos.mergePick')
   353→
   354→  const tables = plan?.tables ?? []
   355→
   356→  // ── Hall stats (current floor) + Seat Party availability ─────────
   357→  const hallFreeCount = tables.filter(
   358→    (tb) => tb.status === 'free' && tb.openOrderId == null,
   359→  ).length
   360→  const hallOccupiedCount = tables.filter(
   361→    (tb) => tb.status === 'occupied' || tb.openOrderId != null,
   362→  ).length
   363→  const hallSeats = tables.reduce((sum, tb) => sum + (tb.capacity ?? 0), 0)
   364→  const hallDirtyCount = tables.filter(
   365→    (tb) => tb.status === 'dirty' && tb.openOrderId == null,
   366→  ).length
   367→  const hasFreeTable = hallFreeCount > 0
   368→  const seatSeats = seatSelection.reduce((sum, tb) => sum + (tb.capacity ?? 0), 0)
   369→
   370→  return (
   371→    <div className="rms-scroll h-full overflow-y-auto">
   372→      <div className="mx-auto flex w-full max-w-6xl flex-col">
   373→        {/* ── Control bar ── */}
   374→        <div className="flex flex-wrap items-center justify-between gap-3 border-b border-[#E2E2E0] bg-white px-4 py-3">
   375→          {/* Floor switcher */}
   376→          <div className="flex items-center gap-2">
   377→            <Button
   378→              variant="outline"
   379→              size="icon"
   380→              className="h-11 w-11 rounded-xl"
   381→              disabled={idx <= 0}
   382→              onClick={() => setFloorIdx(idx - 1)}
   383→              aria-label={t('pos.prevFloor')}
   384→            >
   385→              <ChevronLeft className="size-5 rtl:rotate-180" />
   386→            </Button>
   387→            <span className="min-w-[110px] text-center text-lg font-bold">
   388→              {plan?.name ?? t('pos.floors')}
   389→            </span>
   390→            <Button
   391→              variant="outline"
   392→              size="icon"
   393→              className="h-11 w-11 rounded-xl"
   394→              disabled={idx >= floorPlans.length - 1}
   395→              onClick={() => setFloorIdx(idx + 1)}
   396→              aria-label={t('pos.nextFloor')}
   397→            >
   398→              <ChevronRight className="size-5 rtl:rotate-180" />
   399→            </Button>
   400→          </div>
   401→
   402→          {/* Actions */}
   403→          <div className="flex flex-wrap items-center gap-2">
   404→            <Button
   405→              variant="outline"
   406→              className="h-11 rounded-xl"
   407→              disabled={openOrders.length === 0 || !!tool}
   408→              onClick={startTransferTool}
   409→            >
   410→              <ArrowLeftRight className="text-[#714B67]" />
   411→              <span className="hidden sm:inline">{t('pos.transfer')}</span>
   412→            </Button>
   413→            <Button
   414→              variant="outline"
   415→              className="h-11 rounded-xl"
   416→              disabled={openOrders.length < 2 || !!tool}
   417→              onClick={startMergeTool}
   418→            >
   419→              <Combine className="text-[#714B67]" />
   420→              <span className="hidden sm:inline">{t('pos.merge')}</span>
   421→            </Button>
   422→            <Button
   423→              variant="outline"
   424→              className="h-11 rounded-xl border-violet-400 text-violet-700 hover:bg-violet-50 hover:text-violet-700"
   425→              disabled={!hasFreeTable || !!tool}
   426→              onClick={startSeatTool}
   427→              title={t('pos.seatParty')}
   428→            >
   429→              <Armchair className="text-violet-700" />
   430→              <span className="hidden sm:inline">{t('pos.seatParty')}</span>
   431→            </Button>
   432→            <Button
   433→              className="h-11 rounded-xl bg-[#714B67] text-white hover:bg-[#714B67]/90"
   434→              disabled={!!tool}
   435→              onClick={onTakeaway}
   436→            >
   437→              <Plus /> {t('pos.newTakeaway')}
   438→            </Button>
   439→          </div>
   440→        </div>
   441→
   442→        {/* ── Seat Party banner (violet) ── */}
   443→        {tool === 'seat' ? (
   444→          <div
   445→            className="flex flex-wrap items-center gap-2 bg-violet-600 px-4 py-3 text-white sm:gap-3"
   446→            role="status"
   447→          >
   448→            <Armchair className="size-5 shrink-0" aria-hidden />
   449→            <div className="min-w-0 flex-1">
   450→              <p className="text-sm font-semibold">{t('pos.seatPick')}</p>
   451→              <p className="text-xs font-medium tabular-nums text-white/80">
   452→                {t('pos.seatTables', { n: seatSelection.length, seats: seatSeats })}
   453→              </p>
   454→            </div>
   455→            <Button
   456→              className="h-11 shrink-0 rounded-xl bg-white px-4 text-violet-700 hover:bg-violet-50 hover:text-violet-700"
   457→              disabled={seatSelection.length === 0}
   458→              onClick={continueSeat}
   459→              title={seatSelection.length === 0 ? t('pos.seatNeedOne') : t('pos.seatContinue')}
   460→            >
   461→              {t('pos.seatContinue')}
   462→            </Button>
   463→            <Button
   464→              variant="ghost"
   465→              size="icon"
   466→              className="size-11 shrink-0 rounded-full text-white hover:bg-white/20 hover:text-white"
   467→              onClick={exitTool}
   468→              aria-label={t('pos.seatCancel')}
   469→              title={t('common.cancel')}
   470→            >
   471→              <X className="size-5" />
   472→            </Button>
   473→          </div>
   474→        ) : (
   475→          /* ── Transfer/merge banner ── */
   476→          tool && (
   477→            <div
   478→              className="flex items-center gap-3 bg-[#714B67] px-4 py-3 text-white"
   479→              role="status"
   480→            >
   481→              {busy ? (
   482→                <Loader2 className="size-5 shrink-0 animate-spin" aria-hidden />
   483→              ) : tool === 'transfer' ? (
   484→                <ArrowLeftRight className="size-5 shrink-0" aria-hidden />
   485→              ) : (
   486→                <Combine className="size-5 shrink-0" aria-hidden />
   487→              )}
   488→              <p className="min-w-0 flex-1 text-sm font-semibold">{bannerText}</p>
   489→              <Button
   490→                variant="ghost"
   491→                size="icon"
   492→                className="size-11 shrink-0 rounded-full text-white hover:bg-white/20 hover:text-white"
   493→                onClick={exitTool}
   494→                disabled={busy}
   495→                aria-label={t('pos.cancelTool')}
   496→                title={t('common.cancel')}
   497→              >
   498→                <X className="size-5" />
   499→              </Button>
   500→            </div>
   501→          )
   502→        )}
   503→
   504→        {/* ── Hall panel: tables grid + takeaway + deferred checks ── */}
   505→        <div className="p-4 sm:p-6">
   506→          <div className="relative rounded-2xl border border-[#E2E2E0] bg-[radial-gradient(circle,#ece7dc_1px,transparent_1px)] [background-size:22px_22px] p-4 sm:p-6">
   507→            {/* Hall stats bar — current floor */}
   508→            <div className="mb-4 flex flex-wrap items-center gap-x-4 gap-y-1 text-xs font-medium text-stone-500">
   509→              <span className="tabular-nums">{t('admin.hallStatsTables', { n: tables.length })}</span>
   510→              <span className="tabular-nums font-semibold text-emerald-700">
   511→                {t('admin.hallStatsFree', { n: hallFreeCount })}
   512→              </span>
   513→              <span className="tabular-nums font-semibold text-amber-700">
   514→                {t('admin.hallStatsOccupied', { n: hallOccupiedCount })}
   515→              </span>
   516→              {hallDirtyCount > 0 && (
   517→                <span className="tabular-nums font-semibold text-amber-600">
   518→                  {t('admin.hallStatsDirty', { n: hallDirtyCount })}
   519→                </span>
   520→              )}
   521→              <span className="flex items-center gap-1 tabular-nums">
   522→                <Users className="size-3.5" aria-hidden />
   523→                {hallSeats} {t('common.seats')}
   524→              </span>
   525→            </div>
   526→

   629→            {/* Entrance marker — centered on the panel's bottom border */}

   642→      <AlertDialog
   643→        open={clearTarget != null}
   644→        onOpenChange={(o) => {
   645→          if (!o && !clearTableMutation.isPending) setClearTarget(null)

   666→            <AlertDialogCancel disabled={clearTableMutation.isPending}>

   668→            </AlertDialogCancel>
   669→            <AlertDialogAction
   670→              className={cn(
   671→                clearTarget?.status === 'dirty'
   672→                  ? 'bg-emerald-600 text-white hover:bg-emerald-700'
   673→                  : 'bg-amber-600 text-white hover:bg-amber-700',
   674→              )}
   675→              disabled={clearTableMutation.isPending}
   676→              onClick={(e) => {
   677→                e.preventDefault()
   678→                if (clearTarget) {
   679→                  clearTableMutation.mutate({
   680→                    id: clearTarget.id,
   681→                    name: clearTarget.name,
   682→                    phase: clearTarget.status === 'dirty' ? 'clean' : 'bus',
   683→                  })
   684→                }
   685→              }}
   686→            >
   687→              {clearTableMutation.isPending && <Loader2 className="size-4 animate-spin" />}{' '}
   688→              {clearTarget?.status === 'dirty' ? t('pos.markCleanedFree') : t('pos.sendToCleaning')}
   689→            </AlertDialogAction>
   690→          </AlertDialogFooter>
   691→        </AlertDialogContent>
   692→      </AlertDialog>
   693→    </div>
   694→  )
   695→}

   697→function TableTile({
   698→  table,
   699→  interaction,
   700→  selected,
   701→  onClick,
   702→}: {
   703→  table: RestaurantTable
   704→  interaction: TableInteraction
   705→  /** Seat Party: this free table is part of the current selection. */
   706→  selected?: boolean
   707→  onClick: () => void
   708→}) {
   709→  const { t } = useI18n()
   710→  const occupied = table.status === 'occupied' || table.openOrderId != null
   711→  const reserved = table.status === 'reserved' && table.openOrderId == null
   712→  // Round5/6: bill settled (awaiting bussing) / deferred check (client left)
   713→  // / dirty = bussed, amber, awaiting the cleaning click.
   714→  const paid = table.status === 'paid' && table.openOrderId == null
   715→  const deferred = table.status === 'deferred' && table.openOrderId == null
   716→  const dirty = table.status === 'dirty' && table.openOrderId == null
   717→  // Round/oval silhouettes get extra horizontal padding so the centered
   718→  // content stays inside the circle (overflow-hidden + truncation clip it).
   719→  const isRound = table.shape === 'round' || table.shape === 'oval'
   720→  // Corner badges: on circles the top corner is clipped by the silhouette,
   721→  // so they sit at the top-center edge instead.
   722→  const cornerBadgePos = isRound ? 'left-1/2 top-1 -translate-x-1/2' : 'end-1 top-1'
   723→
   724→  // Odoo duration "heat": occupied tables tint by how long guests are seated.
   725→  const mins =
   726→    occupied && table.openOrderSince
   727→      ? Math.max(0, (Date.now() - new Date(table.openOrderSince).getTime()) / 60000)
   728→      : 0
   729→

   731→    ? mins < 15
   732→      ? {
   733→          tile: 'border-emerald-200 bg-emerald-50',

   735→          amount: 'text-emerald-700',
   736→          meta: 'text-emerald-600',
   737→        }

   740→            tile: 'border-amber-200 bg-amber-50',
   741→            text: 'text-amber-900',
   742→            amount: 'text-amber-700',
   743→            meta: 'text-amber-600',
   744→          }

   747→              tile: 'border-orange-300 bg-orange-100',
   748→              text: 'text-orange-900',
   749→              amount: 'text-orange-700',
   750→              meta: 'text-orange-600',

   757→            }

   762→      type="button"
   763→      disabled={interaction === 'ineligible'}
   764→      onClick={onClick}
   765→      className={cn(

   770→          ? cn(heat.tile, heat.text)
   771→          : paid
   772→            ? 'border-emerald-600 bg-emerald-600 text-white'
   773→            : deferred
   774→              ? 'border-violet-400 bg-violet-100 text-violet-900 ring-1 ring-violet-400'
   775→              : dirty
   776→                ? 'border-amber-500 bg-amber-200 text-amber-900 ring-2 ring-amber-400'
   777→                : reserved
   778→                  ? 'border-amber-300 bg-amber-50/70 text-amber-900 ring-1 ring-amber-400'
   779→                  : 'border-[#E2E2E0] bg-white text-stone-500',
   780→        interaction === 'ineligible' && 'cursor-not-allowed opacity-40',
   781→        interaction === 'eligible' && 'ring-2 ring-[#714B67] ring-offset-1',
   782→        selected && 'ring-2 ring-violet-600 ring-offset-1',
   783→      )}
   784→    >
   785→      {/* FREE — clean white tile with capacity */}

   787→        <>
   788→          <p className="max-w-full truncate text-base font-bold leading-tight text-stone-500">
   789→            {table.name}
   790→          </p>
   791→          <p className="flex items-center gap-1 text-xs text-stone-400">
   792→            <Users className="size-3.5" aria-hidden />

   797→

   799→      {occupied && (

   801→          <p className="max-w-full truncate text-base font-bold leading-tight">{table.name}</p>
