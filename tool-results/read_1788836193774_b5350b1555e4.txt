    65→function daysAgo(n: number): string {

    71→function parseDay(value: string): Date {
    72→  // 'YYYY-MM-DD' parsed as local midnight (avoids UTC off-by-one)
    73→  return new Date(`${value}T00:00:00`)
    74→}

    76→function dayLocale(): string {
    77→  return formatLocale() === 'ar' ? 'ar-EG-u-nu-latn' : 'en-GB'
    78→}

    80→function shortDay(dateStr: string): string {
    81→  return parseDay(dateStr).toLocaleDateString(dayLocale(), {
    82→    day: '2-digit',
    83→    month: 'short',
    84→  })

   112→function ChartTooltip({
   113→  active,
   114→  payload,
   115→  label,
   116→  detail,
   117→}: {
   118→  active?: boolean
   119→  payload?: TooltipEntry[]
   120→  label?: string | number
   121→  detail?: (entry: TooltipEntry) => string
   122→}) {

   125→    <div className="rounded-lg border bg-popover px-3 py-2 text-xs shadow-md">

   128→        <p key={i} className="text-muted-foreground">

   183→/** Build the printable Z-Report paper. Same print mechanism as the POS
   184→ *  receipt/check modals: thermal monospace, 320px, dashed rules, flex rows. */
   185→function buildZReportHtml(
   186→  report: ZReport,
   187→  t: Translate,
   188→  restaurantName: string,
   189→  restaurantNameAr: string,
   190→): string {
   191→  const row = (l: string, r: string, cls = '') =>
   192→    `<div class="r ${cls}"><span>${escapeHtml(l)}</span><span>${escapeHtml(r)}</span></div>`
   193→  const dashed = '<div class="dashed"></div>'
   194→  const section = (title: string) => `<p class="sec">${escapeHtml(title)}</p>`
   195→
   196→  const lines: string[] = []
   197→  lines.push(`<h3>${escapeHtml(restaurantName)}</h3>`)
   198→  lines.push(`<p class="arn" dir="rtl">${escapeHtml(restaurantNameAr)}</p>`)
   199→  lines.push(`<p>${escapeHtml(t('admin.zreportTitle'))}</p>`)
   200→  lines.push(dashed)
   201→  lines.push(row(t('admin.zreportDate'), formatDate(parseDay(report.date))))
   202→  lines.push(dashed)
   203→  lines.push(row(t('admin.zreportOrdersClosed'), String(report.ordersClosed)))
   204→  lines.push(row(t('admin.zreportCovers'), String(report.covers)))
   205→  lines.push(row(t('admin.zreportGross'), formatCurrency(report.grossSubtotal)))
   206→  lines.push(row(t('admin.zreportDiscounts'), formatCurrency(report.discounts)))
   207→  lines.push(row(t('admin.zreportVat'), formatCurrency(report.vat)))
   208→  lines.push(row(t('admin.zreportService'), formatCurrency(report.serviceTax)))
   209→  lines.push(row(t('admin.zreportNet'), formatCurrency(report.netTotal), 'bold'))
   210→  lines.push(row(t('admin.zreportAvgCheck'), formatCurrency(report.avgCheck)))
   211→  lines.push(row(t('admin.zreportCancelled'), String(report.cancelledCount)))
   212→  lines.push(dashed)
   213→  lines.push(section(t('admin.zreportPayments')))
   214→  for (const m of report.paymentsByMethod) {
   215→    lines.push(row(`${t(`status.payment.${m.method}`)} (${m.count})`, formatCurrency(m.amount)))
   216→  }
   217→  if (report.paymentsByMethod.length === 0) lines.push('<p class="muted">—</p>')
   218→  lines.push(
   219→    row(
   220→      t('admin.zreportPaymentsTotal'),
   221→      formatCurrency(report.paymentsTotal),
   222→      'bold',
   223→    ),
   224→  )
   225→  lines.push(dashed)
   226→  lines.push(row(t('admin.zreportDeferredSettled'), formatCurrency(report.deferredSettled)))
   227→  lines.push(row(t('admin.zreportDeferredOutstanding'), formatCurrency(report.deferredOutstanding)))
   228→  lines.push(dashed)
   229→  lines.push(section(t('admin.zreportByWaiter')))
   230→  for (const w of report.byWaiter) {
   231→    lines.push(row(`${w.name} (${w.orders})`, formatCurrency(w.net)))
   232→  }
   233→  if (report.byWaiter.length === 0) lines.push('<p class="muted">—</p>')
   234→  return lines.join('\n')
   235→}

   266→  const { restaurantName, restaurantNameAr } = useAppSettings()
   267→  // report is fetched manually: draft date in state, submitted date drives the query
   268→  const [date, setDate] = useState(todayLocal)
   269→  const [queryDate, setQueryDate] = useState<string | null>(null)
   270→
   271→  const zreportQuery = useQuery({
   272→    queryKey: ['zreport', queryDate],
   273→    enabled: queryDate != null,
   274→    queryFn: () => fetcher<{ report: ZReport }>(`/api/reports/zreport?date=${queryDate}`),
   275→  })
   276→
   277→  const report = zreportQuery.data?.report
   278→  const noData = report != null && report.ordersClosed === 0 && report.paymentsTotal === 0
   279→  const paymentsCount = (report?.paymentsByMethod ?? []).reduce((sum, m) => sum + m.count, 0)
   280→
   281→  function handlePrint() {
   282→    if (!report) return
   283→    const html = buildZReportHtml(report, t, restaurantName, restaurantNameAr)
   284→    const w = window.open('', '_blank', 'width=380,height=760')
   285→    if (!w) {
   286→      window.alert(t('pos.receiptPopupBlocked'))
   287→      return
   288→    }
   289→    w.document.write(
   290→      `<html dir="${isRTL ? 'rtl' : 'ltr'}" lang="${lang}"><head><meta charset="utf-8"><title>${escapeHtml(
   291→        t('admin.zreportTitle'),
   292→      )}</title><style>body{font-family:monospace;font-size:13px;padding:24px;width:320px} .r{display:flex;justify-content:space-between} .dashed{border-top:1px dashed #000;margin:8px 0} h3,p{margin:2px 0;text-align:center} .sec{font-weight:bold;margin:6px 0 2px} .muted{color:#555} .bold{font-weight:bold} .arn{font-weight:bold;direction:rtl;margin:2px 0}</style></head><body>${html}</body></html>`,
   293→    )
   294→    w.document.close()
   295→    w.focus()
   296→    w.print()
   297→  }
   298→
   299→  return (
   300→    <Card>
   301→      <CardHeader>
   302→        <CardTitle>{t('admin.zreportTitle')}</CardTitle>
   303→        <CardDescription>{t('admin.zreportSubtitle')}</CardDescription>

   309→            <Label htmlFor="zreport-date" className="text-xs text-muted-foreground">
   310→              {t('admin.zreportDate')}
   311→            </Label>

   313→              id="zreport-date"
   314→              type="date"
   315→              value={date}

   319→            />
   320→          </div>

   322→            className="h-11"
   323→            onClick={() => setQueryDate(date)}
   324→            disabled={zreportQuery.isFetching}
   325→          >
   326→            <CalendarDays />
