     1→'use client'
     2→
     3→import { useState } from 'react'
     4→import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query'
     5→import {
     6→  ArrowLeftRight,
     7→  Ban,
     8→  Check,
     9→  CreditCard,
    10→  Loader2,
    11→  Minus,
    12→  Pencil,
    13→  Plus,
    14→  Printer,
    15→  Send,
    16→  ShoppingBag,
    17→  StickyNote,
    18→  Trash2,
    19→  Users,
    20→  X,
    21→} from 'lucide-react'
    22→
    23→import { Badge } from '@/components/ui/badge'
    24→import { Button } from '@/components/ui/button'
    25→import {
    26→  Dialog,
    27→  DialogContent,
    28→  DialogDescription,
    29→  DialogFooter,
    30→  DialogHeader,
    31→  DialogTitle,
    32→} from '@/components/ui/dialog'
    33→import { Input } from '@/components/ui/input'
    34→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select'
    35→import { Skeleton } from '@/components/ui/skeleton'
    36→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs'
    37→import { Textarea } from '@/components/ui/textarea'
    38→import { toast } from 'sonner'
    39→import { apiFetch, fetcher } from '@/lib/api'
    40→import { COURSES, TAX_RATE } from '@/lib/constants'
    41→import { formatCurrency, formatQty } from '@/lib/format'
    42→import { useI18n, localizedName } from '@/lib/i18n'
    43→import { cn } from '@/lib/utils'
    44→import type { Order, OrderItem } from '@/lib/types'
    45→import { computeCartTotals, round2, type DraftItem } from './pos-utils'
    46→
    47→type CartPanelProps = {
    48→  order: Order | null
    49→  orderLoading?: boolean
    50→  draft: DraftItem[]
    51→  table: { id: number | null; name: string }
    52→  onDraftChange: (draft: DraftItem[]) => void
    53→  onSend: () => void
    54→  onPay: () => void
    55→  /** Open the pre-payment guest check (pos-view flushes the draft first). */
    56→  onPrintCheck?: () => void
    57→  onCancel: () => void
    58→  canCancel?: boolean
    59→  sending?: boolean
    60→  userRole?: string
    61→}
    62→
    63→const STATUS_CHIP: Record<string, string> = {
    64→  new: 'bg-zinc-100 text-zinc-500',
    65→  preparing: 'bg-amber-100 text-amber-700 animate-pulse',
    66→  ready: 'bg-emerald-100 text-emerald-700',
    67→  served: 'bg-muted text-muted-foreground',
    68→}
    69→
    70→export default function CartPanel({
    71→  order,
    72→  orderLoading = false,
    73→  draft,
    74→  table,
    75→  onDraftChange,
    76→  onSend,
    77→  onPay,
    78→  onPrintCheck,
    79→  onCancel,
    80→  canCancel = false,
    81→  sending = false,
    82→  userRole,
    83→}: CartPanelProps) {
    84→  const queryClient = useQueryClient()
    85→  const { t, lang } = useI18n()
    86→  const orderId = order?.id ?? null
    87→
    88→  const [editing, setEditing] = useState<DraftItem | null>(null)
    89→  const [editQty, setEditQty] = useState('1')
    90→  const [editNotes, setEditNotes] = useState('')
    91→  const [editCourse, setEditCourse] = useState('main')
    92→
    93→  const [discountOpen, setDiscountOpen] = useState(false)
    94→  const [discountTab, setDiscountTab] = useState<'percent' | 'fixed'>('percent')
    95→  const [percentInput, setPercentInput] = useState('')
    96→  const [fixedInput, setFixedInput] = useState('')
    97→
    98→  // ── Item transfer ("Move items") state ────────────────────────────
    99→  const [moveMode, setMoveMode] = useState(false)
   100→  const [moveSelected, setMoveSelected] = useState<Set<number>>(() => new Set())
   101→  const [moveDialogOpen, setMoveDialogOpen] = useState(false)
   102→
   103→  const totals = computeCartTotals(order, draft)
   104→  const noItems = (order?.items.length ?? 0) + draft.length === 0
   105→
   106→  // Open orders (for the move-items target picker) — only fetched while the
   107→  // picker dialog is open; the query key matches the floor's live query.
   108→  const { data: openOrdersData, isLoading: openOrdersLoading } = useQuery({
   109→    queryKey: ['orders', 'open'],
   110→    queryFn: () => fetcher<{ orders: Order[] }>('/api/orders?status=open'),
   111→    enabled: moveDialogOpen,
   112→  })
   113→  const otherOpenOrders = (openOrdersData?.orders ?? []).filter((o) => o.id !== orderId)
   114→
   115→  // ── Server mutations (sent items) ─────────────────────────────────
   116→  const markServed = useMutation({
   117→    mutationFn: (itemId: number) =>
   118→      apiFetch<{ item: OrderItem }>(`/api/order-items/${itemId}`, {
   119→        method: 'PUT',
   120→        body: { status: 'served' },
   121→      }),
   122→    onSuccess: async (data) => {
   123→      toast.success(
   124→        t('pos.markedServedToast', {
   125→          name: data.item.product
   126→            ? localizedName(data.item.product.name, data.item.product.nameAr, lang)
   127→            : t('pos.item'),
   128→        }),
   129→      )
   130→      await queryClient.invalidateQueries({ queryKey: ['pos-order'] })
   131→      await queryClient.invalidateQueries({ queryKey: ['orders'] })
   132→    },
   133→    onError: (err: Error) => toast.error(err.message),
   134→  })
   135→
   136→  const removeItem = useMutation({
   137→    mutationFn: (itemId: number) => {
   138→      if (orderId == null) throw new Error(t('pos.noActiveOrder'))
   139→      return apiFetch<{ order: Order }>(`/api/orders/${orderId}`, {
   140→        method: 'PUT',
   141→        body: { removeItemIds: [itemId] },
   142→      })
   143→    },
   144→    onSuccess: async ({ order: updated }) => {
   145→      queryClient.setQueryData(['pos-order', updated.id], { order: updated })
   146→      await queryClient.invalidateQueries({ queryKey: ['orders'] })
   147→      await queryClient.invalidateQueries({ queryKey: ['floorplans'] })
   148→      await queryClient.invalidateQueries({ queryKey: ['tables-status'] })
   149→      toast.success(t('pos.itemRemovedToast'))
   150→    },
   151→    onError: (err: Error) => toast.error(err.message),
   152→  })
   153→
   154→  const applyDiscount = useMutation({
   155→    mutationFn: (discountAmount: number) => {
   156→      if (orderId == null) throw new Error(t('pos.noActiveOrder'))
   157→      return apiFetch<{ order: Order }>(`/api/orders/${orderId}`, {
   158→        method: 'PUT',
   159→        body: { discountAmount },
   160→      })
   161→    },
   162→    onSuccess: async ({ order: updated }) => {
   163→      queryClient.setQueryData(['pos-order', updated.id], { order: updated })
   164→      await queryClient.invalidateQueries({ queryKey: ['orders'] })
   165→      await queryClient.invalidateQueries({ queryKey: ['floorplans'] })
   166→      await queryClient.invalidateQueries({ queryKey: ['tables-status'] })
   167→      toast.success(t('pos.discountUpdatedToast'))
   168→      setDiscountOpen(false)
   169→    },
   170→    onError: (err: Error) => toast.error(err.message),
   171→  })
   172→
   173→  // ── Item transfer mutation (move sent items to another open order) ──
   174→  const transferItems = useMutation({
   175→    mutationFn: (vars: {
   176→      sourceId: number
   177→      itemIds: number[]
   178→      targetOrderId: number
   179→      targetLabel: string
   180→    }) =>
   181→      apiFetch<{ source: Order; target: Order }>(`/api/orders/${vars.sourceId}/transfer-items`, {
   182→        method: 'POST',
   183→        body: { itemIds: vars.itemIds, targetOrderId: vars.targetOrderId },
   184→      }),
   185→    onSuccess: async (_data, vars) => {
   186→      toast.success(
   187→        t('pos.itemsMovedToast', { n: vars.itemIds.length, target: vars.targetLabel }),
   188→      )
   189→      await Promise.all([
   190→        queryClient.invalidateQueries({ queryKey: ['orders'] }),
   191→        queryClient.invalidateQueries({ queryKey: ['orders', 'open'] }),
   192→        queryClient.invalidateQueries({ queryKey: ['floorplans'] }),
   193→        queryClient.invalidateQueries({ queryKey: ['tables-status'] }),
   194→        queryClient.invalidateQueries({ queryKey: ['pos-order', vars.sourceId] }),
   195→        queryClient.invalidateQueries({ queryKey: ['pos-order', vars.targetOrderId] }),
   196→      ])
   197→      exitMoveMode()
   198→    },
   199→    onError: (err: Error) => toast.error(err.message),
   200→  })
   201→
   202→  // ── Move-items helpers ───────────────────────────────────────────
   203→  const enterMoveMode = () => {
   204→    setMoveMode(true)
   205→    setMoveSelected(new Set())
   206→  }
   207→
   208→  const exitMoveMode = () => {
   209→    setMoveMode(false)
   210→    setMoveSelected(new Set())
   211→    setMoveDialogOpen(false)
   212→  }
   213→
   214→  const toggleMoveItem = (itemId: number) => {
   215→    setMoveSelected((prev) => {
   216→      const next = new Set(prev)
   217→      if (next.has(itemId)) next.delete(itemId)
   218→      else next.add(itemId)
   219→      return next
   220→    })
   221→  }
   222→
   223→  // Only ids that still exist on the (live-polled) order — stale ids are dropped.
   224→  const selectedMoveIds = (order?.items ?? []).filter((i) => moveSelected.has(i.id)).map((i) => i.id)
   225→  const moveCount = selectedMoveIds.length
   226→
   227→  const pickMoveTarget = (target: Order) => {
   228→    if (orderId == null || moveCount === 0) return
   229→    transferItems.mutate({
   230→      sourceId: orderId,
   231→      itemIds: selectedMoveIds,
   232→      targetOrderId: target.id,
   233→      targetLabel: target.table?.name ?? t('common.takeaway'),
   234→    })
   235→  }
   236→
   237→  // ── Draft helpers ─────────────────────────────────────────────────
   238→  const updateDraft = (key: string, patch: Partial<DraftItem>) =>
   239→    onDraftChange(draft.map((d) => (d.key === key ? { ...d, ...patch } : d)))
   240→
   241→  const changeQty = (d: DraftItem, delta: number) => {
   242→    const next = round2(d.quantity + delta)
   243→    if (next <= 0) onDraftChange(draft.filter((x) => x.key !== d.key))
   244→    else updateDraft(d.key, { quantity: next })
   245→  }
   246→
   247→  const openEdit = (d: DraftItem) => {
   248→    setEditing(d)
   249→    setEditQty(String(d.quantity))
   250→    setEditNotes(d.notes)
   251→    setEditCourse(d.course)
   252→  }
   253→
   254→  const saveEdit = () => {
   255→    if (!editing) return
   256→    const qty = round2(parseFloat(editQty))
   257→    if (!Number.isFinite(qty) || qty <= 0) return
   258→    updateDraft(editing.key, { quantity: qty, notes: editNotes.trim(), course: editCourse })
   259→    setEditing(null)
   260→  }
   261→
   262→  // Seed discount dialog inputs from the current order discount when opening.
   263→  const openDiscountDialog = () => {
   264→    const current = order?.discountAmount ?? 0
   265→    const subtotal = order?.subtotalAmount ?? 0
   266→    setPercentInput(current > 0 && subtotal > 0 ? String(round2((current / subtotal) * 100)) : '')
   267→    setFixedInput(current > 0 ? String(current) : '')
   268→    setDiscountOpen(true)
   269→  }
   270→
   271→  // Effective discount from the dialog inputs (clamped to [0, order subtotal]).
   272→  const orderSubtotal = round2(order?.subtotalAmount ?? 0)
   273→  const previewDiscount = (() => {
   274→    if (!order) return 0
   275→    const raw =
   276→      discountTab === 'percent'
   277→        ? (orderSubtotal * (parseFloat(percentInput) || 0)) / 100
   278→        : parseFloat(fixedInput) || 0
   279→    if (!Number.isFinite(raw) || raw < 0) return 0
   280→    return round2(Math.min(raw, orderSubtotal))
   281→  })()
   282→  const previewBase = round2(orderSubtotal - previewDiscount)
   283→  const previewTax = round2(previewBase * TAX_RATE)
   284→
   285→  return (
   286→    <div className="flex min-h-0 flex-1 flex-col">
   287→      {/* Header */}
   288→      <div className="flex shrink-0 items-center justify-between gap-2 border-b border-[#E2E2E0] px-4 py-3">
   289→        <div className="min-w-0">
   290→          <p className="truncate text-base font-bold">{table.name}</p>
   291→          {(order?.user?.name || userRole) && (
   292→            <p className="text-[11px] text-muted-foreground">
   293→              {t('pos.server')}:{' '}
   294→              {order?.user?.name ?? (userRole ? t(`role.${userRole}`) : '')}
   295→            </p>
   296→          )}
   297→        </div>
   298→        {order ? (
   299→          <Badge className="shrink-0 bg-[#714B67] text-white hover:bg-[#714B67]">
   300→            {t('common.order')} #{order.id}
   301→          </Badge>
   302→        ) : (
   303→          <Badge variant="secondary" className="shrink-0">
   304→            {t('pos.notSent')}
   305→          </Badge>
   306→        )}
   307→      </div>
   308→
   309→      {/* Body */}
   310→      <div className="rms-scroll min-h-0 flex-1 overflow-y-auto px-4">
   311→        {orderLoading && !order && draft.length === 0 ? (
   312→          <div className="space-y-3 py-4">
   313→            <Skeleton className="h-14 w-full" />
   314→            <Skeleton className="h-14 w-full" />
   315→            <Skeleton className="h-14 w-full" />
   316→          </div>
   317→        ) : noItems ? (
   318→          <div className="flex h-full min-h-[160px] flex-col items-center justify-center gap-2 py-8 text-muted-foreground">
   319→            <ShoppingBag className="size-10 opacity-30" />
   320→            <p className="text-sm">{t('pos.tapToAdd')}</p>
   321→          </div>
   322→        ) : (
   323→          <>
   324→            {/* Sent to kitchen */}
   325→            {order && order.items.length > 0 && (
   326→              <section className="py-2">
   327→                <div className="flex items-center justify-between gap-2">
   328→                  <h3 className="py-1 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   329→                    {t('pos.sentItems')} ({order.items.length})
   330→                  </h3>
   331→                  <div className="flex shrink-0 items-center gap-1">
   332→                    {!moveMode && (
   333→                      <Button
   334→                        variant="outline"
   335→                        size="sm"
   336→                        className="h-8 gap-1 px-2 text-xs border-[#714B67]/40 text-[#714B67] hover:bg-[#714B67]/10 hover:text-[#714B67]"
   337→                        title={t('pos.transferItems')}
   338→                        onClick={enterMoveMode}
   339→                      >
   340→                        <ArrowLeftRight className="size-3.5" /> {t('pos.transferItems')}
   341→                      </Button>
   342→                    )}
   343→                    {moveMode ? (
   344→                      <span className="text-[11px] text-muted-foreground">{t('pos.moveModeHint')}</span>
   345→                    ) : (
   346→                      canCancel && (
   347→                        <Button
   348→                          variant="ghost"
   349→                          size="sm"
   350→                          className="h-8 px-2 text-xs text-destructive hover:bg-destructive/10 hover:text-destructive"
   351→                          title={t('pos.cancelOrder')}
   352→                          onClick={onCancel}
   353→                        >
   354→                          <Ban className="size-3.5" /> {t('pos.cancelOrder')}
   355→                        </Button>
   356→                      )
   357→                    )}
   358→                  </div>
   359→                </div>
   360→                {order.items.map((item) => (
   361→                  <SentItemRow
   362→                    key={item.id}
   363→                    item={item}
   364→                    onServed={() => markServed.mutate(item.id)}
   365→                    onRemove={() => removeItem.mutate(item.id)}
   366→                    servedPending={markServed.isPending && markServed.variables === item.id}
   367→                    removePending={removeItem.isPending && removeItem.variables === item.id}
   368→                    moveMode={moveMode}
   369→                    moveSelected={moveSelected.has(item.id)}
   370→                    onToggleMove={() => toggleMoveItem(item.id)}
   371→                  />
   372→                ))}
   373→              </section>
   374→            )}
   375→
   376→            {/* New items (draft) */}
   377→            {draft.length > 0 && (
   378→              <section className="py-2">
   379→                <h3 className="py-1 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   380→                  {t('pos.newItemsNotSent')} ({draft.length})
   381→                </h3>
   382→                {draft.map((d) => (
   383→                  <DraftRow
   384→                    key={d.key}
   385→                    item={d}
   386→                    onQty={(delta) => changeQty(d, delta)}
   387→                    onEdit={() => openEdit(d)}
   388→                    onRemove={() => onDraftChange(draft.filter((x) => x.key !== d.key))}
   389→                  />
   390→                ))}
   391→              </section>
   392→            )}
   393→          </>
   394→        )}
   395→      </div>
   396→
   397→      {/* Footer */}
   398→      <div className="shrink-0 space-y-1.5 border-t border-[#E2E2E0] p-4">
   399→        <SummaryRow label={t('money.subtotal')} value={formatCurrency(totals.subtotal)} />
   400→        <div className="flex items-center justify-between text-sm">
   401→          <span className="flex items-center gap-1 text-muted-foreground">
   402→            {t('money.discount')}
   403→            <Button
   404→              variant="ghost"
   405→              size="icon"
   406→              className="size-7 text-muted-foreground"
   407→              disabled={!order}
   408→              title={order ? t('pos.applyDiscount') : t('pos.sendFirst')}
   409→              onClick={openDiscountDialog}
   410→            >
   411→              <Pencil className="size-3.5" />
   412→            </Button>
   413→          </span>
   414→          <span className="tabular-nums text-muted-foreground">
   415→            − {formatCurrency(totals.discount)}
   416→          </span>
   417→        </div>
   418→        <SummaryRow label={t('money.tax')} value={formatCurrency(totals.tax)} />
   419→        <div className="flex items-center justify-between border-t border-[#E2E2E0] pt-2">
   420→          <span className="text-sm font-semibold">{t('money.total')}</span>
   421→          <span className="text-lg font-bold tabular-nums">{formatCurrency(totals.total)}</span>
   422→        </div>
   423→        {order && order.paidAmount > 0 && (
   424→          <>
   425→            <SummaryRow
   426→              label={t('money.paid')}
   427→              value={formatCurrency(order.paidAmount)}
   428→              valueClassName="text-emerald-600"
   429→            />
   430→            <SummaryRow
   431→              label={t('money.remaining')}
   432→              value={formatCurrency(Math.max(0, order.remainingAmount))}
   433→              valueClassName="font-bold text-amber-600"
   434→            />
   435→          </>
   436→        )}
   437→
   438→        {moveMode ? (
   439→          /* ── Item-transfer footer ── */
   440→          <div className="flex gap-2 pt-1">
   441→            <Button
   442→              variant="outline"
   443→              className="h-14 flex-1 rounded-xl border-[#E2E2E0] text-sm"
   444→              onClick={exitMoveMode}
   445→              disabled={transferItems.isPending}
   446→            >
   447→              {t('common.cancel')}
   448→            </Button>
   449→            <Button
   450→              className="h-14 flex-[1.6] rounded-xl bg-[#714B67] text-base font-semibold text-white hover:bg-[#714B67]/90"
   451→              disabled={moveCount === 0 || transferItems.isPending}
   452→              onClick={() => setMoveDialogOpen(true)}
   453→            >
   454→              {transferItems.isPending ? (
   455→                <Loader2 className="animate-spin" />
   456→              ) : (
   457→                <ArrowLeftRight className="size-5" />
   458→              )}
   459→              <span className="truncate">{t('pos.moveNItems', { n: moveCount })}</span>
   460→            </Button>
   461→          </div>
   462→        ) : (
   463→          /* ── Normal actions ── */
   464→          <>
   465→            <Button
   466→              variant="secondary"
   467→              className="h-11 w-full rounded-xl"
   468→              disabled={draft.length === 0 || sending}
   469→              onClick={onSend}
   470→            >
   471→              {sending ? <Loader2 className="animate-spin" /> : <Send />} {t('pos.sendToKitchen')}
   472→              {draft.length > 0 && (
   473→                <Badge className="ms-1 h-5 min-w-5 rounded-full px-1.5 tabular-nums">
   474→                  {draft.reduce((n, d) => n + d.quantity, 0)}
   475→                </Badge>
   476→              )}
   477→            </Button>
   478→            <div className="flex gap-2">
   479→              <Button
   480→                variant="outline"
   481→                className="h-14 flex-1 rounded-xl border-[#E2E2E0] text-sm"
   482→                disabled={!order || sending || !onPrintCheck}
   483→                title={!order ? t('pos.printCheckDisabled') : t('pos.printCheckHint')}
   484→                onClick={onPrintCheck}
   485→              >
   486→                <Printer />
   487→                <span className="hidden sm:inline">{t('pos.printCheck')}</span>
   488→              </Button>
   489→              <Button
   490→                className="h-14 flex-[1.6] rounded-xl bg-emerald-600 text-base font-semibold text-white hover:bg-emerald-700"
   491→                disabled={noItems || sending}
   492→                onClick={onPay}
   493→              >
   494→                {sending ? <Loader2 className="animate-spin" /> : <CreditCard />}
   495→                <span className="truncate">
   496→                  {t('pos.payment')} · {formatCurrency(totals.total)}
   497→                </span>
   498→              </Button>
   499→            </div>
   500→          </>
   501→        )}
   502→      </div>
   503→
   504→      {/* Move-items target picker dialog */}
   505→      <Dialog open={moveDialogOpen} onOpenChange={(o) => !o && setMoveDialogOpen(false)}>
   506→        <DialogContent className="sm:max-w-md">
   507→          <DialogHeader>
   508→            <DialogTitle className="flex items-center gap-2">
   509→              <ArrowLeftRight className="size-5 text-[#714B67]" /> {t('pos.moveItemsTitle')}
   510→            </DialogTitle>
   511→            <DialogDescription>{t('pos.moveItemsDesc', { n: moveCount })}</DialogDescription>
   512→          </DialogHeader>
   513→          <div className="rms-scroll max-h-[50dvh] space-y-2 overflow-y-auto">
   514→            {openOrdersLoading ? (
   515→              <div className="space-y-2">
   516→                <Skeleton className="h-16 w-full" />
   517→                <Skeleton className="h-16 w-full" />
   518→              </div>
   519→            ) : otherOpenOrders.length === 0 ? (
   520→              <p className="py-6 text-center text-sm text-muted-foreground">
   521→                {t('pos.noOtherOpenOrders')}
   522→              </p>
   523→            ) : (
   524→              otherOpenOrders.map((o) => (
   525→                <button
   526→                  key={o.id}
   527→                  type="button"
   528→                  disabled={transferItems.isPending}
   529→                  onClick={() => pickMoveTarget(o)}

   719→function SentItemRow({
   720→  item,
   721→  onServed,
   722→  onRemove,
   723→  servedPending,
   724→  removePending,
   725→  moveMode,
   726→  moveSelected,
   727→  onToggleMove,
   728→}: {
   729→  item: OrderItem
   730→  onServed: () => void
   731→  onRemove: () => void
   732→  servedPending: boolean
   733→  removePending: boolean
   734→  moveMode: boolean
   735→  moveSelected: boolean
   736→  onToggleMove: () => void
   737→}) {
   738→  const { t, lang } = useI18n()
   739→  const chip = STATUS_CHIP[item.status] ?? STATUS_CHIP.served
   740→  const lineTotal = formatCurrency(round2(item.quantity * item.unitPrice))
   741→  return (
   742→    <div
   743→      onClick={moveMode ? onToggleMove : undefined}
   744→      role={moveMode ? 'button' : undefined}
   745→      aria-pressed={moveMode ? moveSelected : undefined}
   746→      className={cn(
   747→        'flex items-start gap-2 border-b border-border/60 py-2.5 last:border-b-0',
   748→        moveMode && 'cursor-pointer rounded-lg transition-colors',
   749→        moveMode && moveSelected && 'bg-[#714B67]/10',
   750→      )}
   751→    >
   752→      {moveMode && (
   753→        <span
   754→          className={cn(
   755→            'mt-1 flex size-5 shrink-0 items-center justify-center rounded border',
   756→            moveSelected
   757→              ? 'border-[#714B67] bg-[#714B67] text-white'
   758→              : 'border-[#E2E2E0] bg-white',
   759→          )}
   760→          aria-hidden
   761→        >
   762→          {moveSelected && <Check className="size-3.5" />}
   763→        </span>
   764→      )}
   765→      <span
   766→        className={cn(
   767→          'mt-0.5 shrink-0 rounded-full px-2 py-0.5 text-[10px] font-semibold uppercase tracking-wide',
   768→          chip,
   769→        )}
   770→      >
   771→        {t(`status.item.${item.status}`)}
   772→      </span>
   773→      <div className="min-w-0 flex-1" title={item.notes ?? undefined}>
   774→        <p className="truncate text-sm font-medium">
   775→          {formatQty(item.quantity)} ×{' '}
   776→          {item.product
   777→            ? localizedName(item.product.name, item.product.nameAr, lang)
   778→            : t('pos.item')}
   779→        </p>
   780→        {item.notes && (
   781→          <p className="flex items-center gap-1 truncate text-xs text-amber-600">
   782→            <StickyNote className="size-3 shrink-0" />
   783→            <span className="truncate">{item.notes}</span>
   784→          </p>
   785→        )}
   786→      </div>
   787→      <div className="flex shrink-0 flex-col items-end gap-1">
   788→        <span className="text-sm font-semibold tabular-nums">{lineTotal}</span>
   789→        {!moveMode && (
   790→          <div className="flex gap-1">
   791→            {item.status === 'ready' && (
   792→              <Button
   793→                variant="ghost"
   794→                size="sm"
   795→                className="h-8 gap-1 px-2 text-xs text-emerald-700 hover:bg-emerald-100 hover:text-emerald-800"
   796→                disabled={servedPending}
   797→                onClick={onServed}
   798→                title={t('pos.markServed')}
   799→              >
   800→                {servedPending ? (
   801→                  <Loader2 className="size-3.5 animate-spin" />
   802→                ) : (
   803→                  <Check className="size-3.5" />
   804→                )}
   805→                {t('status.item.served')}
   806→              </Button>
   807→            )}
   808→            <Button
   809→              variant="ghost"
   810→              size="icon"
   811→              className="size-8 text-muted-foreground hover:text-destructive"
   812→              disabled={removePending}
   813→              onClick={onRemove}
   814→              title={t('pos.removeItem')}
   815→            >
   816→              {removePending ? (
   817→                <Loader2 className="size-4 animate-spin" />
   818→              ) : (
   819→                <Trash2 className="size-4" />
   820→              )}
   821→            </Button>
   822→          </div>
   823→        )}
   824→      </div>
   825→    </div>
