   133→   133→    }

   151→   151→  const clampGuests = useCallback(

   178→   178→  const invalidateShared = useCallback(

   191→   191→  /** Enter order mode for a table (occupied → live order, free → fresh draft). */

   308→   308→
   309→   309→  const addProduct = (p: Product) => {
   310→   310→    setDraft((prev) => {

   315→   315→          d.key === existing.key ? { ...d, quantity: round2(d.quantity + 1) } : d,
   316→   316→        )
   317→   317→      }

   319→   319→        ...prev,
   320→   320→        {
   321→   321→          key: newDraftKey(),
   322→   322→          productId: p.id,
   323→   323→          name: p.name,
   324→   324→          nameAr: p.nameAr ?? null,
   325→   325→          price: p.price,
   326→   326→          quantity: 1,
   327→   327→          notes: '',
   328→   328→          course,
   329→   329→        },
   330→   330→      ]
   331→   331→    })

   334→   334→  const draftToPayload = (d: DraftItem[]) =>

   342→   342→  const sendToKitchen = async (): Promise<Order | null> => {

   344→   344→    setSending(true)
   345→   345→    try {
   346→   346→      let result: { order: Order }

   350→   350→      const seatingList = seatingTables ?? null
   351→   351→      const multiSeating = (seatingList?.length ?? 0) > 1
   352→   352→      if (!order && multiSeating && seatingList) {
   353→   353→        result = await apiFetch<{ order: Order }>('/api/orders', {
   354→   354→          method: 'POST',
   355→   355→          body: {
   356→   356→            tableIds: seatingList.map((tb) => tb.id),
   357→   357→            items: draftToPayload(draft),
   358→   358→            guests: clampGuests(guestsDraft),
   359→   359→          },
   360→   360→        })
   361→   361→      } else if (!order) {
   362→   362→        result = await apiFetch<{ order: Order }>('/api/orders', {
   363→   363→          method: 'POST',
   364→   364→          body: {
   365→   365→            tableId: selectedTable?.id ?? null,
   366→   366→            items: draftToPayload(draft),
   367→   367→            guests: clampGuests(guestsDraft),
   368→   368→          },
   369→   369→        })
   370→   370→      } else {
   371→   371→        result = await apiFetch<{ order: Order }>(`/api/orders/${order.id}`, {
   372→   372→          method: 'PUT',
   373→   373→          body: { addItems: draftToPayload(draft) },
   374→   374→        })
   375→   375→      }
   376→   376→      setDraft([])
   377→   377→      queryClient.setQueryData(['pos-order', result.order.id], { order: result.order })
   378→   378→      setActiveOrderId(result.order.id)
   379→   379→      await invalidateShared()
   380→   380→      toast.success(
   381→   381→        multiSeating && !order && seatingList
   382→   382→          ? t('pos.seatedToast', { n: seatingList.length })
   383→   383→          : t('pos.orderSentToast'),
   384→   384→      )
   385→   385→      return result.order
   386→   386→    } catch (err) {
   387→   387→      toast.error(err instanceof Error ? err.message : t('pos.sendFailedToast'))
   388→   388→      return null
   389→   389→    } finally {
   390→   390→      setSending(false)
   391→   391→    }
   392→   392→  }
   393→   393→
   394→   394→  const handlePay = async () => {
   395→   395→    if (sending) return
   396→   396→    let current: Order | null = order
   397→   397→    if (draft.length > 0) {
   398→   398→      current = await sendToKitchen()
   399→   399→      if (!current) return // sending failed → abort payment
   400→   400→    }
   401→   401→    if (!current) return
   402→   402→    setPayOrder(current)
   403→   403→    setPayOpen(true)
   404→   404→  }
   405→   405→
   406→   406→  // Print Check from the cart panel: flush the draft first (same pattern as pay),
   407→   407→  // then open the check modal with the fresh order state.
   408→   408→  const handlePrintCheck = async () => {
   409→   409→    if (sending) return
   410→   410→    let current: Order | null = order
   411→   411→    if (draft.length > 0) {
   412→   412→      current = await sendToKitchen()
   413→   413→      if (!current) return // sending failed → abort printing
   414→   414→    }
   415→   415→    if (!current) return
   416→   416→    setCheckOrder(current)
   417→   417→    setCheckOpen(true)
   418→   418→  }
   419→   419→
   420→   420→  const handlePaySuccess = async (updated: Order, closed: boolean) => {
   421→   421→    if (closed) {
   422→   422→      // Payment modal closes itself; show the receipt, then reset the screen.
   423→   423→      setReceiptOrder(updated)
   424→   424→      setMode('tables')
   425→   425→      setSelectedTable(null)
   426→   426→      setActiveOrderId(null)
   427→   427→      setDraft([])
   428→   428→      setGuestsDraft(2)
   429→   429→      setPendingSeatTables(null)
   430→   430→      setSeatingTables(null)
   431→   431→      readyRef.current = null
   432→   432→      await invalidateShared()
   433→   433→    } else {
   434→   434→      queryClient.setQueryData(['pos-order', updated.id], { order: updated })
   435→   435→      await Promise.all([
   436→   436→        queryClient.invalidateQueries({ queryKey: ['orders'] }),
   437→   437→        queryClient.invalidateQueries({ queryKey: ['orders', 'deferred'] }),
   438→   438→        queryClient.invalidateQueries({ queryKey: ['floorplans'] }),
   439→   439→      ])
   440→   440→    }
   441→   441→  }
   442→   442→
   443→   443→  // Transfer clicked on the order screen → go to the floor screen with the
   444→   444→  // order preselected (destination step). TableSelect reports completion.
   445→   445→  const startTransfer = () => {
   446→   446→    if (!order) return
   447→   447→    setTransferOrderId(order.id)
   448→   448→    setMode('tables')
   449→   449→    setSelectedTable(null)
   450→   450→    setActiveOrderId(null)
   451→   451→    setDraft([])
   452→   452→    setGuestsDraft(2)
   453→   453→    setPendingSeatTables(null)
   454→   454→    setSeatingTables(null)
   455→   455→    readyRef.current = null
   456→   456→  }

   569→   569→          <span className="truncate text-xl font-bold">{selectedTable?.name ?? t('pos.order')}</span>
   570→   570→          {mergedExtraCount > 0 && (
   571→   571→            <Badge className="gap-1 border-violet-300 bg-violet-100 text-violet-800 hover:bg-violet-100">
   572→   572→              <Combine className="size-3" aria-hidden />

   575→   575→          )}
   576→   576→          {order && (
   577→   577→            <Badge className="bg-[#714B67] text-white hover:bg-[#714B67]">
   578→   578→              {t('common.order')} #{order.id}
   579→   579→            </Badge>
   580→   580→          )}

   582→   582→            <span className="text-xs tabular-nums text-muted-foreground" data-tick={elapsedTick}>
   583→   583→              {elapsedSince(order.createdAt)}
   584→   584→            </span>
   585→   585→          )}
   586→   586→          {preparingCount > 0 && (
   587→   587→            <Badge className="border-amber-300 bg-amber-100 text-amber-800 hover:bg-amber-100">
   588→   588→              {t('pos.preparingCount', { n: preparingCount })}
   589→   589→            </Badge>
   590→   590→          )}
   591→   591→          {readyCount > 0 && (
   592→   592→            <Badge className="border-emerald-300 bg-emerald-100 text-emerald-800 hover:bg-emerald-100">
   593→   593→              {t('pos.readyCount', { n: readyCount })}
   594→   594→            </Badge>
   595→   595→          )}
   596→   596→        </div>
   597→   597→        <div className="flex items-center gap-2">
   598→   598→          {/* Guests: N — edit while drafting (sent with the order) or on the live order */}
   599→   599→          <Button
   600→   600→            variant="outline"
   601→   601→            className="h-11 rounded-xl border-[#714B67]/40 text-[#714B67] hover:bg-[#714B67]/10 hover:text-[#714B67]"
   602→   602→            onClick={openGuestsEdit}
   603→   603→            title={t('pos.editGuests')}
   604→   604→          >
   605→   605→            <Users />
   606→   606→            <span className="hidden sm:inline">{t('common.guests')}:</span>
   607→   607→            <span className="font-semibold tabular-nums">{order ? order.guests : guestsDraft}</span>
   608→   608→          </Button>
   609→   609→          <Button
   610→   610→            variant="outline"
   611→   611→            className="h-11 rounded-xl border-[#714B67]/40 text-[#714B67] hover:bg-[#714B67]/10 hover:text-[#714B67]"
   612→   612→            disabled={!order}
   613→   613→            onClick={startTransfer}
   614→   614→            title={order ? t('pos.transferHint') : t('pos.transferDisabledHint')}
   615→   615→          >
   616→   616→            <ArrowLeftRight />
   617→   617→            <span className="hidden md:inline">{t('pos.transfer')}</span>
   618→   618→          </Button>
   619→   619→          {order && canCancel && (
   620→   620→            <Button
   621→   621→              variant="outline"
   622→   622→              className="h-11 rounded-xl border-destructive/40 text-destructive hover:bg-destructive/10 hover:text-destructive"
   623→   623→              onClick={() => setCancelOpen(true)}
   624→   624→            >
   625→   625→              <Ban />
   626→   626→              <span className="hidden md:inline">{t('common.cancel')}</span>
   627→   627→            </Button>
   628→   628→          )}
   629→   629→        </div>
   630→   630→      </header>
   631→   631→
   632→   632→      {/* ── Body: product grid + cart ── */}
   633→   633→      <div className="flex min-h-0 flex-1 flex-col md:flex-row">
   634→   634→        <ProductGrid products={products} onAdd={addProduct} className="min-h-0 flex-1" />
   635→   635→        <div className="flex max-h-[60vh] w-full shrink-0 flex-col border-t border-[#E2E2E0] bg-white md:max-h-none md:w-[380px] md:border-l md:border-t-0 lg:w-[420px]">
   636→   636→          <CartPanel
   637→   637→            order={order}
   638→   638→            orderLoading={orderLoading}
   639→   639→            draft={draft}
   640→   640→            table={selectedTable ?? { id: null, name: t('pos.order') }}
   641→   641→            onDraftChange={setDraft}
   642→   642→            onSend={() => void sendToKitchen()}
   643→   643→            onPay={() => void handlePay()}
   644→   644→            onPrintCheck={() => void handlePrintCheck()}
   645→   645→            onCancel={() => setCancelOpen(true)}
   646→   646→            canCancel={canCancel}
   647→   647→            sending={sending}
   648→   648→            userRole={user?.role ?? ''}
   649→   649→          />
   650→   650→        </div>
   651→   651→      </div>
   652→   652→
   653→   653→      {/* ── Guests edit dialog (order mode) ── */}
   654→   654→      <GuestsDialog

   659→   659→        onConfirm={confirmGuests}
   660→   660→        onCancel={cancelGuestsDialog}
   661→   661→      />

   663→   663→      {/* ── Cancel order confirmation ── */}

   666→   666→          <AlertDialogHeader>
   667→   667→            <AlertDialogTitle>{t('pos.cancelTitle')}</AlertDialogTitle>

   670→   670→                order: order?.id ?? 0,

   672→   672→                total: formatCurrency(order?.totalAmount ?? 0),
   673→   673→                items: order?.items.length ?? 0,
   674→   674→              })}
   675→   675→            </AlertDialogDescription>
   676→   676→          </AlertDialogHeader>
   677→   677→          <AlertDialogFooter>
   678→   678→            <AlertDialogCancel disabled={cancelMutation.isPending}>
   679→   679→              {t('pos.keepOrder')}

   690→   690→              {t('pos.cancelOrder')}

   693→   693→        </AlertDialogContent>

   703→   703→            if (!o) setPayOrder(null)

   705→   705→          onSuccess={(updated, closed) => void handlePaySuccess(updated, closed)}
   706→   706→          onDeferred={(deferredOrder) => void handleOrderDeferred(deferredOrder)}

   709→   709→
   710→   710→      {/* ── Pre-payment guest check (from the cart panel) ── */}

   717→   717→            setCheckOpen(o)

   719→   719→          }}
   720→   720→        />
   721→   721→      )}
   722→   722→
   723→   723→      {/* ── Receipt — shown after an order is fully paid, independent of the payment modal ── */}

   725→   725→        <ReceiptModal
   726→   726→          order={receiptOrder}
   727→   727→          open={!!receiptOrder}

   731→   731→          onClose={() => setReceiptOrder(null)}
   732→   732→        />
   733→   733→      )}
   734→   734→    </div>
   735→   735→  )
   736→   736→}

   738→   738→/** Quick guests picker — stepper + 1-8 chips (min MIN_GUESTS, max MAX_GUESTS). */
   739→   739→function GuestsDialog({
   740→   740→  open,
   741→   741→  value,
   742→   742→  pending,
   743→   743→  onValueChange,
   744→   744→  onConfirm,
   745→   745→  onCancel,
   746→   746→}: {
   747→   747→  open: boolean
   748→   748→  value: number
   749→   749→  pending: boolean
   750→   750→  onValueChange: (v: number) => void
   751→   751→  onConfirm: () => void
   752→   752→  onCancel: () => void
   753→   753→}) {
   754→   754→  const { t } = useI18n()
   755→   755→
   756→   756→  const step = (delta: number) => {
   757→   757→    const next = Math.min(MAX_GUESTS, Math.max(MIN_GUESTS, Math.round(value + delta)))
   758→   758→    onValueChange(next)
   759→   759→  }
   760→   760→
   761→   761→  return (
   762→   762→    <Dialog open={open} onOpenChange={(o) => !o && onCancel()}>
   763→   763→      <DialogContent className="sm:max-w-sm">
   764→   764→        <DialogHeader>
   765→   765→          <DialogTitle className="flex items-center gap-2">
   766→   766→            <Users className="size-5 text-[#714B67]" /> {t('pos.editGuests')}
   767→   767→          </DialogTitle>
   768→   768→          <DialogDescription>{t('pos.guestsDialogDesc')}</DialogDescription>
   769→   769→        </DialogHeader>
   770→   770→        <div className="space-y-4">
   771→   771→          <div className="flex items-center justify-center gap-4">
   772→   772→            <Button
   773→   773→              variant="outline"
   774→   774→              size="icon"
   775→   775→              className="size-12 rounded-xl"
   776→   776→              disabled={value <= MIN_GUESTS}
   777→   777→              onClick={() => step(-1)}
   778→   778→              aria-label={t('pos.fewerGuests')}
   779→   779→            >
   780→   780→              <Minus />
   781→   781→            </Button>
   782→   782→            <span className="w-16 text-center text-4xl font-bold tabular-nums">{value}</span>
   783→   783→            <Button
   784→   784→              variant="outline"
   785→   785→              size="icon"
   786→   786→              className="size-12 rounded-xl"
   787→   787→              disabled={value >= MAX_GUESTS}
   788→   788→              onClick={() => step(1)}
   789→   789→              aria-label={t('pos.moreGuests')}
   790→   790→            >
   791→   791→              <Plus />
   792→   792→            </Button>
   793→   793→          </div>
   794→   794→          <div className="grid grid-cols-4 gap-1.5">
   795→   795→            {GUEST_QUICK_CHIPS.map((n) => (
   796→   796→              <button
   797→   797→                key={n}
   798→   798→                type="button"
   799→   799→                onClick={() => onValueChange(n)}
   800→   800→                className={cn(
   801→   801→                  'h-11 rounded-full border text-sm font-semibold tabular-nums transition-colors',
   802→   802→                  value === n
   803→   803→                    ? 'border-[#714B67] bg-[#714B67] text-white'
   804→   804→                    : 'border-[#E2E2E0] bg-white text-stone-600 hover:border-[#714B67]/40',
   805→   805→                )}

   807→   807→                {n}
   808→   808→              </button>
   809→   809→            ))}
   810→   810→          </div>
   811→   811→        </div>
   812→   812→        <DialogFooter>
   813→   813→          <Button variant="outline" onClick={onCancel} disabled={pending}>
   814→   814→            {t('common.cancel')}
   815→   815→          </Button>
   816→   816→          <Button
   817→   817→            className="bg-[#714B67] text-white hover:bg-[#714B67]/90"
   818→   818→            onClick={onConfirm}
   819→   819→            disabled={pending}
   820→   820→          >
   821→   821→            {pending ? <Loader2 className="animate-spin" /> : <Check />} {t('common.confirm')}
   822→   822→          </Button>
   823→   823→        </DialogFooter>
   824→   824→      </DialogContent>
   825→   825→    </Dialog>
   826→   826→  )
   827→   827→}
