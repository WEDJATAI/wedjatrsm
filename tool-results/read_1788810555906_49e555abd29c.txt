     1→     1→'use client'
     2→     2→
     3→     3→import { useEffect, useMemo, useState } from 'react'
     4→     4→import type { LucideIcon } from 'lucide-react'
     5→     5→import {
     6→     6→  AlertCircle,
     7→     7→  Banknote,
     8→     8→  Check,
     9→     9→  CreditCard,
    10→    10→  Loader2,
    11→    11→  MoreHorizontal,
    12→    12→  Plus,
    13→    13→  Printer,
    14→    14→  Users,
    15→    15→  X,
    16→    16→} from 'lucide-react'
    17→    17→
    18→    18→import { Button } from '@/components/ui/button'
    19→    19→import {
    20→    20→  Dialog,
    21→    21→  DialogContent,
    22→    22→  DialogDescription,
    23→    23→  DialogHeader,
    24→    24→  DialogTitle,
    25→    25→} from '@/components/ui/dialog'
    26→    26→import { Input } from '@/components/ui/input'
    27→    27→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs'
    28→    28→import { toast } from 'sonner'
    29→    29→import { apiFetch } from '@/lib/api'
    30→    30→import { PAYMENT_METHODS } from '@/lib/constants'
    31→    31→import { formatCurrency, formatQty } from '@/lib/format'
    32→    32→import { localizedName, useI18n } from '@/lib/i18n'
    33→    33→import { cn } from '@/lib/utils'
    34→    34→import type { Order } from '@/lib/types'
    35→    35→import CheckModal, { type CheckSplitRow } from './check-modal'
    36→    36→import { newDraftKey, parseAmount, round2 } from './pos-utils'
    37→    37→
    38→    38→type PaymentModalProps = {
    39→    39→  order: Order
    40→    40→  open: boolean
    41→    41→  onOpenChange: (open: boolean) => void
    42→    42→  onSuccess: (order: Order, closed: boolean) => void
    43→    43→}
    44→    44→
    45→    45→type PaymentResult = {
    46→    46→  order: Order
    47→    47→  paidAmount: number
    48→    48→  remaining: number
    49→    49→  closed: boolean
    50→    50→}
    51→    51→
    52→    52→type EditableRow = {
    53→    53→  id: string
    54→    54→  method: string
    55→    55→  amount: string
    56→    56→  reference: string
    57→    57→}
    58→    58→
    59→    59→type SubmitRow = { method: string; amount: number; reference?: string }
    60→    60→
    61→    61→type SplitTab = 'single' | 'equal' | 'items' | 'custom'
    62→    62→
    63→    63→const METHOD_META: Record<string, { labelKey: string; icon: LucideIcon }> = {
    64→    64→  cash: { labelKey: 'status.payment.cash', icon: Banknote },
    65→    65→  card: { labelKey: 'status.payment.card', icon: CreditCard },
    66→    66→  other: { labelKey: 'status.payment.other', icon: MoreHorizontal },
    67→    67→}
    68→    68→
    69→    69→export default function PaymentModal({ order, open, onOpenChange, onSuccess }: PaymentModalProps) {
    70→    70→  const { t, lang } = useI18n()
    71→    71→  const [tab, setTab] = useState<SplitTab>('single')
    72→    72→  const [submitting, setSubmitting] = useState(false)
    73→    73→  const [checkOpen, setCheckOpen] = useState(false)
    74→    74→
    75→    75→  // Which payment row the method tiles currently target.
    76→    76→  const [activeIdx, setActiveIdx] = useState(0)
    77→    77→
    78→    78→  const [singleRow, setSingleRow] = useState<EditableRow>({ id: 'single', method: 'cash', amount: '', reference: '' })
    79→    79→  const [customRows, setCustomRows] = useState<EditableRow[]>([])
    80→    80→
    81→    81→  const [eqPayers, setEqPayers] = useState(2)
    82→    82→  const [eqMethods, setEqMethods] = useState<Record<number, string>>({})
    83→    83→
    84→    84→  const [itPayers, setItPayers] = useState(2)
    85→    85→  const [itMethods, setItMethods] = useState<Record<number, string>>({})
    86→    86→  const [assignments, setAssignments] = useState<Record<number, number>>({})
    87→    87→
    88→    88→  const remaining = round2(Math.max(0, order.remainingAmount))
    89→    89→
    90→    90→  // Reset state whenever the modal opens for (a new) order.
    91→    91→  useEffect(() => {
    92→    92→    if (!open) return
    93→    93→    const rem = round2(Math.max(0, order.remainingAmount))
    94→    94→    setTab('single')
    95→    95→    setSingleRow({ id: 'single', method: 'cash', amount: String(rem), reference: '' })
    96→    96→    setCustomRows([{ id: newDraftKey(), method: 'cash', amount: String(rem), reference: '' }])
    97→    97→    setEqPayers(2)
    98→    98→    setEqMethods({})
    99→    99→    setItPayers(2)
   100→   100→    setItMethods({})
   101→   101→    setAssignments({})
   102→   102→    setSubmitting(false)
   103→   103→    setActiveIdx(0)
   104→   104→    setCheckOpen(false)
   105→   105→  }, [open, order.id, order.remainingAmount])
   106→   106→
   107→   107→  // ── Derived payment rows per tab ──────────────────────────────────
   108→   108→  const eqAmounts = useMemo(() => {
   109→   109→    const part = round2(remaining / eqPayers)
   110→   110→    const amounts = Array.from({ length: eqPayers }, () => part)
   111→   111→    const others = round2(part * (eqPayers - 1))
   112→   112→    amounts[eqPayers - 1] = round2(Math.max(0, remaining - others))
   113→   113→    return amounts
   114→   114→  }, [remaining, eqPayers])
   115→   115→
   116→   116→  const itemLines = useMemo(
   117→   117→    () =>
   118→   118→      order.items.map((it) => ({
   119→   119→        id: it.id,
   120→   120→        label: `${formatQty(it.quantity)} × ${
   121→   121→          it.product ? localizedName(it.product.name, it.product.nameAr, lang) : t('pos.item')
   122→   122→        }`,
   123→   123→        total: round2(it.quantity * it.unitPrice),
   124→   124→      })),
   125→   125→    [order.items, t, lang],
   126→   126→  )
   127→   127→
   128→   128→  const itAmounts = useMemo(() => {
   129→   129→    const subtotal = round2(order.subtotalAmount)
   130→   130→    const rawTotals = new Array<number>(itPayers).fill(0)
   131→   131→    for (const line of itemLines) {
   132→   132→      const payer = assignments[line.id] ?? 0
   133→   133→      if (payer < itPayers) rawTotals[payer] = round2(rawTotals[payer] + line.total)
   134→   134→    }
   135→   135→    const amounts: number[] = []
   136→   136→    let allocated = 0
   137→   137→    for (let i = 0; i < itPayers; i++) {
   138→   138→      if (i === itPayers - 1) {
   139→   139→        amounts.push(round2(Math.max(0, remaining - allocated)))
   140→   140→      } else {
   141→   141→        const share = subtotal > 0 ? round2((rawTotals[i] / subtotal) * remaining) : 0
   142→   142→        amounts.push(share)
   143→   143→        allocated = round2(allocated + share)
   144→   144→      }
   145→   145→    }
   146→   146→    return amounts
   147→   147→  }, [itemLines, assignments, itPayers, remaining, order.subtotalAmount])
   148→   148→
   149→   149→  const submitRows: SubmitRow[] = useMemo(() => {
   150→   150→    if (tab === 'single') {
   151→   151→      return [toSubmitRow(singleRow)]
   152→   152→    }
   153→   153→    if (tab === 'equal') {
   154→   154→      return eqAmounts.map((amount, i) => ({
   155→   155→        method: eqMethods[i] ?? 'cash',
   156→   156→        amount,
   157→   157→      }))
   158→   158→    }
   159→   159→    if (tab === 'items') {
   160→   160→      return itAmounts.map((amount, i) => ({
   161→   161→        method: itMethods[i] ?? 'cash',
   162→   162→        amount,
   163→   163→      }))
   164→   164→    }
   165→   165→    return customRows.map(toSubmitRow)
   166→   166→  }, [tab, singleRow, customRows, eqAmounts, eqMethods, itAmounts, itMethods])
   167→   167→
   168→   168→  const sum = round2(submitRows.reduce((s, r) => s + r.amount, 0))
   169→   169→  const diff = round2(sum - remaining)
   170→   170→  const exceeds = diff > 0.01
   171→   171→  const exact = Math.abs(diff) <= 0.01
   172→   172→  const canSubmit = !submitting && remaining > 0 && sum > 0 && !exceeds
   173→   173→
   174→   174→  // ── Active row / method tiles ─────────────────────────────────────
   175→   175→  const rowCount =
   176→   176→    tab === 'single' ? 1 : tab === 'equal' ? eqPayers : tab === 'items' ? itPayers : customRows.length
   177→   177→  const safeActiveIdx = Math.min(activeIdx, Math.max(0, rowCount - 1))
   178→   178→
   179→   179→  const activeRowLabel =
   180→   180→    tab === 'single'
   181→   181→      ? t('pos.payment')
   182→   182→      : tab === 'equal' || tab === 'items'
   183→   183→        ? t('pos.payer', { n: safeActiveIdx + 1 })
   184→   184→        : t('pos.paymentN', { n: safeActiveIdx + 1 })
   185→   185→
   186→   186→  const currentMethodAt = (index: number): string => {
   187→   187→    if (tab === 'single') return singleRow.method
   188→   188→    if (tab === 'equal') return eqMethods[index] ?? 'cash'
   189→   189→    if (tab === 'items') return itMethods[index] ?? 'cash'
   190→   190→    return customRows[index]?.method ?? 'cash'
   191→   191→  }
   192→   192→
   193→   193→  const setMethodAt = (index: number, m: string) => {
   194→   194→    if (tab === 'single') {
   195→   195→      setSingleRow((r) => ({ ...r, method: m }))
   196→   196→    } else if (tab === 'equal') {
   197→   197→      setEqMethods((prev) => ({ ...prev, [index]: m }))
   198→   198→    } else if (tab === 'items') {
   199→   199→      setItMethods((prev) => ({ ...prev, [index]: m }))
   200→   200→    } else {
   201→   201→      setCustomRows((rows) => rows.map((r, i) => (i === index ? { ...r, method: m } : r)))
   202→   202→    }
   203→   203→  }
   204→   204→
   205→   205→  const handleTabChange = (v: string) => {
   206→   206→    setTab(v as SplitTab)
   207→   207→    setActiveIdx(0)
   208→   208→  }
   209→   209→
   210→   210→  // ── Guest check rows mirroring the current split configuration ────
   211→   211→  const checkRows: CheckSplitRow[] = useMemo(() => {
   212→   212→    if (tab === 'single') {
   213→   213→      const amount = parseAmount(singleRow.amount) || remaining
   214→   214→      return [{ label: t('pos.fullBill'), amount, method: singleRow.method }]
   215→   215→    }
   216→   216→    if (tab === 'equal') {
   217→   217→      return eqAmounts.map((amount, i) => ({
   218→   218→        label: t('pos.partOf', { i: i + 1, n: eqPayers }),
   219→   219→        amount,
   220→   220→        method: eqMethods[i] ?? 'cash',
   221→   221→      }))
   222→   222→    }
   223→   223→    if (tab === 'items') {
   224→   224→      return itAmounts.map((amount, i) => ({
   225→   225→        label: t('pos.payer', { n: i + 1 }),
   226→   226→        amount,
   227→   227→        method: itMethods[i] ?? 'cash',
   228→   228→      }))
   229→   229→    }
   230→   230→    return customRows.map((r, i) => ({
   231→   231→      label: t('pos.paymentN', { n: i + 1 }),
   232→   232→      amount: parseAmount(r.amount),
   233→   233→      method: r.method,
   234→   234→    }))
   235→   235→  }, [tab, singleRow, customRows, eqAmounts, eqMethods, eqPayers, itAmounts, itMethods, remaining, t])
   236→   236→
   237→   237→  // ── Handlers ──────────────────────────────────────────────────────
   238→   238→  const clampPayers = (raw: number, min: number, max: number) => {
   239→   239→    const n = Math.round(Number.isFinite(raw) ? raw : min)
   240→   240→    return Math.min(max, Math.max(min, n))
   241→   241→  }
   242→   242→
   243→   243→  const setAssignment = (itemId: number, payer: number) =>
   244→   244→    setAssignments((prev) => ({ ...prev, [itemId]: payer }))
   245→   245→
   246→   246→  const addCustomRow = () => {
   247→   247→    const current = round2(customRows.reduce((s, r) => s + parseAmount(r.amount), 0))
   248→   248→    const next = round2(Math.max(0, remaining - current))
   249→   249→    setCustomRows((rows) => [...rows, { id: newDraftKey(), method: 'cash', amount: String(next), reference: '' }])
   250→   250→  }
   251→   251→
   252→   252→  const handleSubmit = async () => {
   253→   253→    const payments = submitRows
   254→   254→      .map((r) => ({ ...r, amount: round2(r.amount) }))
   255→   255→      .filter((r) => r.amount > 0)
   256→   256→      .map((r) => ({
   257→   257→        method: r.method,
   258→   258→        amount: r.amount,
   259→   259→        ...(r.method === 'card' && r.reference?.trim() ? { reference: r.reference.trim() } : {}),
   260→   260→      }))
   261→   261→    if (payments.length === 0) return
   262→   262→    setSubmitting(true)
   263→   263→    try {
   264→   264→      const result = await apiFetch<PaymentResult>(`/api/orders/${order.id}/payments`, {
   265→   265→        method: 'POST',
   266→   266→        body: { payments },
   267→   267→      })
   268→   268→      if (result.closed) {
   269→   269→        toast.success(t('pos.paidClosedToast'))
   270→   270→        onSuccess(result.order, true)
   271→   271→        onOpenChange(false)
   272→   272→      } else {
   273→   273→        toast.success(t('pos.paymentRecordedToast', { amount: formatCurrency(result.remaining) }))
   274→   274→        onSuccess(result.order, false)
   275→   275→        onOpenChange(false)
   276→   276→      }
   277→   277→    } catch (err) {
   278→   278→      toast.error(err instanceof Error ? err.message : t('pos.paymentFailedToast'))
   279→   279→    } finally {
   280→   280→      setSubmitting(false)
   281→   281→    }
   282→   282→  }
   283→   283→
   284→   284→  const singleExceeds = tab === 'single' && parseAmount(singleRow.amount) > round2(remaining + 0.01)
   285→   285→
   286→   286→  return (
   287→   287→    <>
   288→   288→      <Dialog open={open} onOpenChange={onOpenChange}>
   289→   289→        <DialogContent className="max-h-[92dvh] overflow-y-auto sm:max-w-lg">
   290→   290→          <DialogHeader>
   291→   291→            <DialogTitle>
   292→   292→              {t('pos.payment')} · {order.table?.name ?? t('common.takeaway')} ·{' '}
   293→   293→              {t('common.order')} #{order.id}
   294→   294→            </DialogTitle>
   295→   295→            <DialogDescription>{t('pos.paymentDesc')}</DialogDescription>
   296→   296→          </DialogHeader>
   297→   297→
   298→   298→          {/* Summary */}
   299→   299→          <div className="grid grid-cols-3 gap-2 rounded-xl border border-[#E2E2E0] bg-white p-3 text-center shadow-sm">
   300→   300→            <div>
   301→   301→              <p className="text-xs text-muted-foreground">{t('money.total')}</p>
   302→   302→              <p className="text-sm font-semibold tabular-nums">{formatCurrency(order.totalAmount)}</p>
   303→   303→            </div>
   304→   304→            <div>
   305→   305→              <p className="text-xs text-muted-foreground">{t('money.paid')}</p>
   306→   306→              <p className="text-sm font-semibold tabular-nums text-emerald-600">
   307→   307→                {formatCurrency(order.paidAmount)}
   308→   308→              </p>
   309→   309→            </div>
   310→   310→            <div>
   311→   311→              <p className="text-xs text-muted-foreground">{t('money.remaining')}</p>
   312→   312→              <p className="text-primary text-2xl font-bold tabular-nums">
   313→   313→                {formatCurrency(remaining)}
   314→   314→              </p>
   315→   315→            </div>
   316→   316→          </div>
   317→   317→
   318→   318→          {/* Method tiles — apply to the active row below */}
   319→   319→          <div className="space-y-1.5">
   320→   320→            <p className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   321→   321→              {t('pos.methodAppliesTo')} <span className="text-[#714B67]">{activeRowLabel}</span>
   322→   322→            </p>
   323→   323→            <div className="grid grid-cols-3 gap-2">
   324→   324→              {PAYMENT_METHODS.map((m) => {
   325→   325→                const meta = METHOD_META[m] ?? { labelKey: `status.payment.${m}`, icon: MoreHorizontal }
   326→   326→                const Icon = meta.icon
   327→   327→                const active = currentMethodAt(safeActiveIdx) === m
   328→   328→                return (
   329→   329→                  <button
   330→   330→                    key={m}
   331→   331→                    type="button"
   332→   332→                    onClick={() => setMethodAt(safeActiveIdx, m)}
   333→   333→                    aria-pressed={active}
   334→   334→                    className={cn(
   335→   335→                      'flex h-16 flex-col items-center justify-center gap-1 rounded-xl border-2 text-sm font-semibold transition active:scale-95',
   336→   336→                      active
   337→   337→                        ? 'border-[#714B67] bg-[#714B67]/10 text-[#714B67]'
   338→   338→                        : 'border-[#E2E2E0] bg-white text-stone-600 hover:border-[#714B67]/40',
   339→   339→                    )}
   340→   340→                  >
   341→   341→                    <Icon className="size-6" aria-hidden />
   342→   342→                    {t(meta.labelKey)}
   343→   343→                  </button>
   344→   344→                )
   345→   345→              })}
   346→   346→            </div>
   347→   347→          </div>
   348→   348→
   349→   349→          <Tabs value={tab} onValueChange={handleTabChange}>
   350→   350→            <TabsList className="grid h-10 w-full grid-cols-4">
   351→   351→              <TabsTrigger value="single" className="text-xs sm:text-sm">
   352→   352→                {t('pos.tabSingle')}
   353→   353→              </TabsTrigger>
   354→   354→              <TabsTrigger value="equal" className="text-xs sm:text-sm">
   355→   355→                {t('pos.tabEqual')}
   356→   356→              </TabsTrigger>
   357→   357→              <TabsTrigger value="items" className="text-xs sm:text-sm">
   358→   358→                {t('pos.tabItems')}
   359→   359→              </TabsTrigger>
   360→   360→              <TabsTrigger value="custom" className="text-xs sm:text-sm">
   361→   361→                {t('pos.tabCustom')}
   362→   362→              </TabsTrigger>
   363→   363→            </TabsList>
   364→   364→
   365→   365→            {/* ── Single ── */}
   366→   366→            <TabsContent value="single" className="space-y-2 pt-3">
   367→   367→              <PayRow
   368→   368→                label={t('pos.payment')}
   369→   369→                method={singleRow.method}
   370→   370→                active
   371→   371→                amountStr={singleRow.amount}
   372→   372→                onAmountChange={(v) => setSingleRow((r) => ({ ...r, amount: v }))}
   373→   373→                reference={singleRow.reference}
   374→   374→                onReferenceChange={(v) => setSingleRow((r) => ({ ...r, reference: v }))}
   375→   375→                editable
   376→   376→                error={
   377→   377→                  singleExceeds
   378→   378→                    ? t('pos.exceedsBy', {
   379→   379→                        amount: formatCurrency(round2(parseAmount(singleRow.amount) - remaining)),
   380→   380→                      })
   381→   381→                    : undefined
   382→   382→                }
   383→   383→              />
   384→   384→            </TabsContent>
   385→   385→
   386→   386→            {/* ── Equal Split ── */}
   387→   387→            <TabsContent value="equal" className="space-y-3 pt-3">
   388→   388→              <div className="flex items-center gap-3 rounded-xl border border-[#E2E2E0] bg-white p-2.5">
   389→   389→                <Users className="size-4 text-muted-foreground" />
   390→   390→                <span className="flex-1 text-sm font-medium">{t('pos.splitBetween')}</span>
   391→   391→                <Input
   392→   392→                  type="number"
   393→   393→                  min={2}
   394→   394→                  max={12}
   395→   395→                  value={eqPayers}
   396→   396→                  onChange={(e) => setEqPayers(clampPayers(parseInt(e.target.value, 10) || 2, 2, 12))}
   397→   397→                  className="h-10 w-16 text-center tabular-nums"
   398→   398→                />
   399→   399→                <span className="text-sm text-muted-foreground">{t('pos.payers')}</span>
   400→   400→              </div>
   401→   401→              {eqAmounts.map((amount, i) => (
   402→   402→                <PayRow
   403→   403→                  key={i}
   404→   404→                  label={t('pos.payer', { n: i + 1 })}
   405→   405→                  method={eqMethods[i] ?? 'cash'}
   406→   406→                  active={safeActiveIdx === i}
   407→   407→                  onActivate={() => setActiveIdx(i)}
   408→   408→                  amount={amount}
   409→   409→                />
   410→   410→              ))}
   411→   411→              <p className="text-center text-xs text-muted-foreground">
   412→   412→                {t('pos.partsTotal')}{' '}
   413→   413→                <span className="font-semibold tabular-nums">{formatCurrency(sum)}</span>
   414→   414→              </p>
   415→   415→            </TabsContent>
   416→   416→
   417→   417→            {/* ── By Items ── */}
   418→   418→            <TabsContent value="items" className="space-y-3 pt-3">
   419→   419→              <div className="flex items-center gap-3 rounded-xl border border-[#E2E2E0] bg-white p-2.5">
   420→   420→                <Users className="size-4 text-muted-foreground" />
   421→   421→                <span className="flex-1 text-sm font-medium">{t('pos.assignItemsTo')}</span>
   422→   422→                <Input
   423→   423→                  type="number"
   424→   424→                  min={2}
   425→   425→                  max={6}
   426→   426→                  value={itPayers}
   427→   427→                  onChange={(e) => setItPayers(clampPayers(parseInt(e.target.value, 10) || 2, 2, 6))}
   428→   428→                  className="h-10 w-16 text-center tabular-nums"
   429→   429→                />
   430→   430→                <span className="text-sm text-muted-foreground">{t('pos.payers')}</span>
   431→   431→              </div>
   432→   432→
   433→   433→              <div className="space-y-1.5">
   434→   434→                {itemLines.map((line) => {
   435→   435→                  const active = Math.min(assignments[line.id] ?? 0, itPayers - 1)
   436→   436→                  return (
   437→   437→                    <div

   541→   541→                </>

   571→   571→      <CheckModal order={order} open={checkOpen} onOpenChange={setCheckOpen} rows={checkRows} />
   572→   572→    </>
   573→   573→  )
