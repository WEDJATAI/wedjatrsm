    95→  // ── Round 7: history-aware navigation (browser/OS back button) ────
    96→  // Unsent drafts stashed per context (table:N / takeaway) so going back
    97→  // never loses items that were never sent to the kitchen.
    98→  const draftStashRef = useRef<Map<string, DraftItem[]>>(new Map())

   101→  // onNav idempotency checks hold even before React re-renders — a single
   102→  // back press can fire both popstate and hashchange.
   103→  const appliedNavRef = useRef<string | null>(null)
   104→  // True only while a restore applies state through an entry helper: navPush()

   106→  // history entries (otherwise Back would bounce forward).
   107→  const suppressNavPushRef = useRef(false)
   108→  // In-flight restore target + epoch: rejects duplicate restores (the
   109→  // popstate/hashchange double fire) and aborts stale ones when the user
   110→  // navigates again mid-flight.
   111→  const navInFlightRef = useRef<string | null>(null)
   112→  const navEpochRef = useRef(0)
   113→  // Latest onNav handler (single subscription, always-fresh state).
   114→  const navHandlerRef = useRef<(nav: NavHash | null) => void>(() => {})
   115→  // Previous `active` value — detects the hidden → visible flip used to

   144→    mode === 'order' && activeOrderId != null ? (orderData?.order ?? null) : null
   145→
   146→  // ── Products ─────────────────────────────────────────────────────
   147→  const { data: productsData } = useQuery({
   148→    queryKey: ['pos-products'],
   149→    queryFn: () => fetcher<{ products: Product[] }>('/api/products?sellable=1'),

   155→  // ── Newly-ready items → toast (once) ─────────────────────────────
   156→  // PosView stays mounted while another view is active (page.tsx keeps it
   157→  // hidden), so only toast while the POS is actually on screen — the ref
   158→  // still tracks readiness so nothing double-fires on return.
   159→  const readyRef = useRef<Set<number> | null>(null)
   160→  useEffect(() => {
   161→    if (!order) {
   162→      readyRef.current = null
   163→      return
   164→    }
   165→    const readyIds = new Set(order.items.filter((i) => i.status === 'ready').map((i) => i.id))
   166→    const prev = readyRef.current

   168→      const newlyReady = order.items.filter((i) => i.status === 'ready' && !prev.has(i.id))
   169→      if (newlyReady.length > 0 && active) {

   172→            n: newlyReady.length,

   176→      }
   177→    }
   178→    readyRef.current = readyIds

   181→  // ── Helpers ──────────────────────────────────────────────────────
   182→  const clampGuests = useCallback(
   183→    (raw: number) => {
   184→      const n = Math.round(Number.isFinite(raw) ? raw : 2)
   185→      return Math.min(MAX_GUESTS, Math.max(MIN_GUESTS, n))
   186→    },
   187→    [],
   188→  )
   189→
   190→  // ── Round 7: nav + draft-stash helpers ───────────────────────────
   191→
   192→  /** pushNav wrapper — silent while a restore applies state (restores must
   193→   *  not create history entries); a user-initiated push supersedes any
   194→   *  in-flight restore (aborts it via the epoch). */
   195→  const navPush = (view: string, sub?: string | null) => {
   196→    if (suppressNavPushRef.current) return
   197→    if (navInFlightRef.current != null) navEpochRef.current += 1
   198→    pushNav(view, sub)
   199→  }
   200→
   201→  /** Stash key for a draft context (per-table, or takeaway). */
   202→  const draftStashKey = (table: { id: number | null } | null) =>
   203→    table?.id != null ? `table:${table.id}` : 'takeaway'
   204→
   205→  /** Stash the unsent draft of the current context before leaving it. */
   206→  const stashDraft = () => {
   207→    if (draft.length > 0) {
   208→      draftStashRef.current.set(draftStashKey(selectedTable), draft)
   209→    }
   210→  }
   211→
   212→  /** Pop a stashed draft for `key`: restored (with a toast) when non-empty;
   213→   *  the entry is removed either way so stashes never leak. */
   214→  const restoreStash = (key: string, label: string) => {
   215→    const stashed = draftStashRef.current.get(key)
   216→    draftStashRef.current.delete(key)
   217→    if (stashed?.length) {
   218→      setDraft(stashed)
   219→      toast.success(t('pos.draftRestoredToast', { table: label }))
   220→    }
   221→  }
   222→
   223→  /** Run a state-applying entry helper with nav pushes suppressed. */
   224→  const withNavSuppression = (fn: () => void) => {
   225→    suppressNavPushRef.current = true
   226→    try {
   227→      fn()
   228→    } finally {
   229→      suppressNavPushRef.current = false
   230→    }
   231→  }
   232→
   233→  /** Leave order mode for the floor: stash any unsent draft, full state
   234→   *  reset, then rewrite the current history entry to '#/pos' (replace, never
   235→   *  push — the in-app back button must not spam history). Every exit path
   236→   *  goes through here, so the browser back button and the in-app button
   237→   *  land the user on the floor with the same draft preservation. */
   238→  const resetToTables = () => {
   239→    stashDraft()
   240→    setMode('tables')
   241→    setSelectedTable(null)
   242→    setActiveOrderId(null)
   243→    setDraft([])
   244→    setGuestsDraft(2)
   245→    setGuestsDialogOpen(false)
   246→    setGuestsTarget(null)
   247→    setPayOpen(false)

   249→    setCheckOpen(false)
   250→    setCheckOrder(null)

   252→    setTransferOrderId(null)

   254→    setSeatingTables(null)
   255→    readyRef.current = null
   256→    appliedNavRef.current = null
   257→    replaceNav('pos')
   258→  }

   260→  const invalidateShared = useCallback(
   261→    async () => {
   262→      await Promise.all([
   263→        queryClient.invalidateQueries({ queryKey: ['orders'] }),
   264→        queryClient.invalidateQueries({ queryKey: ['orders', 'deferred'] }),
   265→        queryClient.invalidateQueries({ queryKey: ['floorplans'] }),
   266→        queryClient.invalidateQueries({ queryKey: ['tables-status'] }),
   267→        queryClient.invalidateQueries({ queryKey: ['pos-products'] }),
   268→      ])
   269→    },
   270→    [queryClient],
   271→  )

   273→  /** Enter order mode for a table (occupied → live order, free → fresh draft). */
   274→  const applySelectTable = (table: RestaurantTable, guests: number) => {
   275→    setSelectedTable({ id: table.id, name: table.name })
   276→    setMode('order')
   277→    setDraft([])
   278→    setTransferOrderId(null)
   279→    setGuestsDraft(clampGuests(guests))
   280→    setSeatingTables([{ id: table.id, name: table.name }])
   281→    readyRef.current = null
   282→    if (table.openOrderId) {
   283→      setActiveOrderId(table.openOrderId)
   284→    } else {
   285→      setActiveOrderId(null)
   286→    }
   287→    appliedNavRef.current = table.openOrderId ? `order/${table.openOrderId}` : `table/${table.id}`
   288→    // Entering an order screen adds a history entry (occupied → live order
   289→    // hash, free → table draft hash); silent during popstate restores.
   290→    navPush('pos', table.openOrderId ? `order/${table.openOrderId}` : `table/${table.id}`)
   291→    // Re-entering a table pops its stashed unsent draft — occupied or free:
   292→    // a waiter may have backed out of this table with items not yet sent.
   293→    restoreStash(`table:${table.id}`, table.name)
   294→  }
   295→
   296→  // Table clicked on the floor: occupied tables open their order directly;
   297→  // a FREE table first asks for the number of guests (quick dialog), then
   298→  // enters order mode with that guest count baked into the draft.
   299→  const selectTable = (table: RestaurantTable) => {
   300→    if (table.openOrderId != null) {
   301→      applySelectTable(table, table.openOrderGuests ?? 2)
   302→      return
   303→    }
   304→    setGuestsTarget({ kind: 'new-table', table })
   305→    setGuestsValue(2)
   306→    setGuestsDialogOpen(true)
   307→  }
   308→
   309→  const startTakeaway = () => {
   310→    setSelectedTable({ id: null, name: t('common.takeaway') })
   311→    setMode('order')
   312→    setDraft([])
   313→    setActiveOrderId(null)
   314→    setTransferOrderId(null)
   315→    setGuestsDraft(1) // takeaway orders default to a single guest — no dialog
   316→    setSeatingTables(null) // no tables — tableIds stays unset
   317→    readyRef.current = null
   318→    appliedNavRef.current = 'takeaway'
   319→    navPush('pos', 'takeaway')
   320→    restoreStash('takeaway', t('common.takeaway'))
   321→  }
   322→
   323→  const openTakeawayOrder = (o: Order) => {
   324→    setSelectedTable({ id: null, name: t('common.takeaway') })
   325→    setMode('order')
   326→    setDraft([])
   327→    setTransferOrderId(null)
   328→    setGuestsDraft(o.guests ?? 1)
   329→    setSeatingTables(null) // no tables — tableIds stays unset
   330→    readyRef.current = null
   331→    queryClient.setQueryData(['pos-order', o.id], { order: o })
   332→    setActiveOrderId(o.id)
   333→    appliedNavRef.current = `order/${o.id}`
   334→    navPush('pos', `order/${o.id}`)
   335→  }
   336→
   337→  // ── Seat Party (merged seating from the beginning) ────────────────
   338→  // TableSelect hands over the selected free tables → ask for the guest
   339→  // count (seeded to the total capacity), then enter order mode with the
   340→  // full seating list. The first table is the primary table.
   341→  const handleSeatParty = (tables: RestaurantTable[]) => {
   342→    if (tables.length === 0) return
   343→    setPendingSeatTables(tables)
   344→    const totalSeats = tables.reduce((sum, tb) => sum + (tb.capacity ?? 0), 0)
   345→    setGuestsValue(clampGuests(totalSeats))
   346→    setGuestsTarget({ kind: 'seat-party' })
   347→    setGuestsDialogOpen(true)
   348→  }
   349→
   350→  // ── Guests dialog handlers ───────────────────────────────────────
   351→  const cancelGuestsDialog = () => {
   352→    if (guestsTarget?.kind === 'seat-party') setPendingSeatTables(null)
   353→    setGuestsDialogOpen(false)
   354→    setGuestsTarget(null)
   355→  }
   356→
   357→  const openGuestsEdit = () => {
   358→    setGuestsValue(order ? order.guests : guestsDraft)

   363→  const updateGuests = useMutation({
   364→    mutationFn: (vars: { id: number; guests: number }) =>
   365→      apiFetch<{ order: Order }>(`/api/orders/${vars.id}`, {

   367→        body: { guests: vars.guests },
   368→      }),

   372→        queryClient.invalidateQueries({ queryKey: ['floorplans'] }),

   375→      toast.success(t('pos.guestsUpdatedToast'))
   376→    },
   377→    onError: (err: Error) => toast.error(err.message),
   378→  })
   379→
   380→  const confirmGuests = () => {
   381→    const value = clampGuests(guestsValue)
   382→    if (guestsTarget?.kind === 'new-table') {
   383→      applySelectTable(guestsTarget.table, value)
   384→    } else if (guestsTarget?.kind === 'seat-party') {
   385→      const seatTables = pendingSeatTables ?? []
   386→      const primary = seatTables[0]
   387→      if (primary) {
   388→        // Enter order mode on the primary table, then widen the seating list
   389→        // to the full party selection (batched setStates — last write wins).
   390→        applySelectTable(primary, value)
   391→        setSeatingTables(seatTables.map((tb) => ({ id: tb.id, name: tb.name })))
   392→      }
   393→      setPendingSeatTables(null)
   394→    } else if (guestsTarget?.kind === 'draft') {
   395→      setGuestsDraft(value)
   396→    } else if (guestsTarget?.kind === 'order' && order) {
   397→      updateGuests.mutate({ id: order.id, guests: value })
   398→    }
   399→    setGuestsDialogOpen(false)
   400→    setGuestsTarget(null)
   401→  }

   403→  const addProduct = (p: Product) => {
   404→    setDraft((prev) => {
   405→      const course = guessCourse(p)
   406→      const existing = prev.find((d) => d.productId === p.id && !d.notes && d.course === course)
   407→      if (existing) {
   408→        return prev.map((d) =>
   409→          d.key === existing.key ? { ...d, quantity: round2(d.quantity + 1) } : d,
   410→        )
   411→      }
   412→      return [
   413→        ...prev,
   414→        {
   415→          key: newDraftKey(),
   416→          productId: p.id,
   417→          name: p.name,
   418→          nameAr: p.nameAr ?? null,
   419→          price: p.price,
   420→          quantity: 1,
   421→          notes: '',
   422→          course,
   423→        },
   424→      ]
   425→    })
   426→  }
   427→
   428→  const draftToPayload = (d: DraftItem[]) =>
   429→    d.map((d) => ({
   430→      productId: d.productId,
   431→      quantity: d.quantity,
   432→      notes: d.notes.trim() || undefined,
   433→      course: d.course,
   434→    }))
   435→
   436→  const sendToKitchen = async (): Promise<Order | null> => {
   437→    if (draft.length === 0 || sending) return order
   438→    setSending(true)
   439→    try {
   440→      let result: { order: Order }
   441→      // Multi-table seating (Seat Party): create the order spanning ALL the
   442→      // selected tables (tableIds — primary first). Single-table and takeaway
   443→      // paths keep the classic tableId payload byte-identical.
   444→      const seatingList = seatingTables ?? null
   445→      const multiSeating = (seatingList?.length ?? 0) > 1
   446→      if (!order && multiSeating && seatingList) {
   447→        result = await apiFetch<{ order: Order }>('/api/orders', {
   448→          method: 'POST',
   449→          body: {
   450→            tableIds: seatingList.map((tb) => tb.id),
   451→            items: draftToPayload(draft),
   452→            guests: clampGuests(guestsDraft),
   453→          },
   454→        })
   455→      } else if (!order) {
   456→        result = await apiFetch<{ order: Order }>('/api/orders', {
   457→          method: 'POST',
   458→          body: {
   459→            tableId: selectedTable?.id ?? null,
   460→            items: draftToPayload(draft),
   461→            guests: clampGuests(guestsDraft),
   462→          },
   463→        })
   464→      } else {
   465→        result = await apiFetch<{ order: Order }>(`/api/orders/${order.id}`, {
   466→          method: 'PUT',
   467→          body: { addItems: draftToPayload(draft) },
   468→        })
   469→      }
   470→      setDraft([])
   471→      queryClient.setQueryData(['pos-order', result.order.id], { order: result.order })
   472→      setActiveOrderId(result.order.id)
   473→      await invalidateShared()
   474→      toast.success(
   475→        multiSeating && !order && seatingList
   476→          ? t('pos.seatedToast', { n: seatingList.length })
   477→          : t('pos.orderSentToast'),
   478→      )
   479→      return result.order
   480→    } catch (err) {
   481→      toast.error(err instanceof Error ? err.message : t('pos.sendFailedToast'))
   482→      return null
   483→    } finally {
   484→      setSending(false)
   485→    }
   486→  }
   487→
   488→  const handlePay = async () => {
   489→    if (sending) return
   490→    let current: Order | null = order
   491→    if (draft.length > 0) {
   492→      current = await sendToKitchen()
   493→      if (!current) return // sending failed → abort payment
   494→    }
   495→    if (!current) return
   496→    setPayOrder(current)
   497→    setPayOpen(true)
   498→  }
   499→
   500→  // Print Check from the cart panel: flush the draft first (same pattern as pay),
   501→  // then open the check modal with the fresh order state.
   502→  const handlePrintCheck = async () => {
   503→    if (sending) return
   504→    let current: Order | null = order
   505→    if (draft.length > 0) {
   506→      current = await sendToKitchen()
   507→      if (!current) return // sending failed → abort printing
   508→    }
   509→    if (!current) return
   510→    setCheckOrder(current)
   511→    setCheckOpen(true)
   512→  }
   513→
   514→  const handlePaySuccess = async (updated: Order, closed: boolean) => {
   515→    if (closed) {
   516→      // Payment modal closes itself; show the receipt, then reset the screen.
   517→      stashDraft() // normally empty — the draft was flushed by sendToKitchen
   518→      setReceiptOrder(updated)
   519→      setMode('tables')
   520→      setSelectedTable(null)
   521→      setActiveOrderId(null)
   522→      setDraft([])
   523→      setGuestsDraft(2)
   524→      setPendingSeatTables(null)
   525→      setSeatingTables(null)
   526→      readyRef.current = null
   527→      appliedNavRef.current = null
   528→      replaceNav('pos') // paid check → floor entry, no extra history
   529→      await invalidateShared()
   530→    } else {
   531→      queryClient.setQueryData(['pos-order', updated.id], { order: updated })
   532→      await Promise.all([
   533→        queryClient.invalidateQueries({ queryKey: ['orders'] }),
   534→        queryClient.invalidateQueries({ queryKey: ['orders', 'deferred'] }),
   535→        queryClient.invalidateQueries({ queryKey: ['floorplans'] }),
   536→      ])
   537→    }
   538→  }
   539→
   540→  // Transfer clicked on the order screen → go to the floor screen with the
   541→  // order preselected (destination step). TableSelect reports completion.
   542→  const startTransfer = () => {
   543→    if (!order) return
   544→    stashDraft() // never lose unsent items while detouring through the floor
   545→    setTransferOrderId(order.id)
   546→    setMode('tables')
   547→    setSelectedTable(null)
   548→    setActiveOrderId(null)
   549→    setDraft([])
   550→    setGuestsDraft(2)
   551→    setPendingSeatTables(null)
   552→    setSeatingTables(null)
   553→    readyRef.current = null
   554→    appliedNavRef.current = null // transfer destination step lives on the floor
   555→    replaceNav('pos') // replace, never push — no history spam
   556→  }
   557→
   558→  const handleTransferDone = () => {
   559→    setTransferOrderId(null)
   560→    resetToTables()
   561→  }
   562→
   563→  // ── Deferred checks ─────────────────────────────────────────────
   564→  // Chip clicked on the floor → settle the outstanding deferred check via
   565→  // the payment modal (rendered in tables mode too; the modal hides its own
   566→  // defer button for already-deferred orders).
   567→  const handleSettleDeferred = (deferredOrder: Order) => {
   568→    setPayOrder(deferredOrder)
   569→    setPayOpen(true)
   570→  }
   571→
   572→  // Defer completed inside the payment modal (modal closes itself + fires
   573→  // the toast) → reset the order screen and refresh everything.
   574→  const handleOrderDeferred = async (_deferredOrder: Order) => {
   575→    resetToTables()
   576→    await invalidateShared()
   577→  }
   578→
   579→  const cancelMutation = useMutation({
   580→    mutationFn: (id: number) =>
   581→      apiFetch<{ order: Order }>(`/api/orders/${id}/cancel`, { method: 'POST' }),
   582→    onSuccess: async () => {
   583→      toast.success(t('pos.cancelToast'))
   584→      await invalidateShared()
   585→      resetToTables()
   586→    },
   587→    onError: (err: Error) => toast.error(err.message),
   588→  })
   589→
   590→  const canCancel =
   591→    !!order && !!user && (user.role === 'admin' || (order.userId != null && order.userId === user.id))
   592→
   593→  const preparingCount = order?.items.filter((i) => i.status === 'preparing').length ?? 0
   594→  const readyCount = order?.items.filter((i) => i.status === 'ready').length ?? 0
   595→
   596→  // Merged-seating badge: extra tables joined to this order (live order field
   597→  // wins once the order exists; draft mode falls back to the seating list).
   598→  const mergedExtraCount = order
   599→    ? (order.extraTableIds?.length ?? 0)
   600→    : Math.max(0, (seatingTables?.length ?? 1) - 1)
   601→
   602→  // ── Round 7: popstate-driven restores (browser/OS back & forward) ──
   603→
   604→  /** Enter order mode for a live order — mirrors openTakeawayOrder /
   605→   *  applySelectTable semantics. Hash-driven, so it never pushes history. */
   606→  const restoreLiveOrder = (restored: Order) => {
   607→    setSelectedTable(
   608→      restored.table
   609→        ? { id: restored.table.id, name: restored.table.name }
   610→        : { id: null, name: t('common.takeaway') },
   611→    )
   612→    setMode('order')
   613→    setDraft([])
   614→    setTransferOrderId(null)
   615→    setGuestsDraft(clampGuests(restored.guests ?? 1))
   616→    setSeatingTables(restored.table ? [{ id: restored.table.id, name: restored.table.name }] : null)
   617→    readyRef.current = null
   618→    appliedNavRef.current = `order/${restored.id}`
   619→    queryClient.setQueryData(['pos-order', restored.id], { order: restored })
   620→    setActiveOrderId(restored.id)
   621→    toast.success(t('pos.orderRestoredToast', { order: restored.id }))
   622→  }
   623→
   624→  /** Claim the right to run a deep restore for `target`. Returns the epoch, or
   625→   *  null when the identical restore is already in flight (popstate +
   626→   *  hashchange double fire). A different target supersedes the old one. */
   627→  const claimNavRestore = (target: string): number | null => {
   628→    if (navInFlightRef.current === target) return null
   629→    navInFlightRef.current = target
   630→    navEpochRef.current += 1
   631→    return navEpochRef.current
   632→  }
   633→
   634→  /** Release the restore guards — only the owning epoch may clear them. */
   635→  const releaseNavRestore = (epoch: number) => {
   636→    if (epoch !== navEpochRef.current) return
   637→    navInFlightRef.current = null
   638→  }
   639→
   640→  /** True when the hash still points at `target` (mid-flight re-check). */
   641→  const hashStill = (target: string) => {
   642→    const now = currentNav()
   643→    return now != null && now.view === 'pos' && now.sub === target
   644→  }
   645→
   646→  /** Failed restore → toast + a safe floor entry. */
   647→  const cannotReopen = () => {
   648→    toast.error(t('pos.cannotReopenToast'))
   649→    replaceNav('pos')
   650→    resetToTables()
   651→  }
   652→
   653→  /** Fetch an order imperatively (null on error). */
   654→  const fetchOrderOrNull = async (id: number): Promise<Order | null> => {
   655→    try {
   656→      const data = await fetcher<{ order: Order }>(`/api/orders/${id}`)
   657→      return data.order ?? null
   658→    } catch {
   659→      return null
   660→    }
   661→  }
   662→
   663→  /** Restore the live-order screen for '#/pos/order/{id}'. */
   664→  const restoreOrderById = async (id: number, target: string) => {
   665→    const epoch = claimNavRestore(target)
   666→    if (epoch == null) return
   667→    stashDraft() // leaving the current context — never lose unsent items
   668→    try {
   669→      const fetched = await fetchOrderOrNull(id)
   670→      // The user navigated again mid-flight → abort silently.
   671→      if (epoch !== navEpochRef.current || !hashStill(target)) return
   672→      if (fetched && fetched.status === 'open') {
   673→        restoreLiveOrder(fetched)
   674→      } else {
   675→        cannotReopen() // closed / cancelled / merged away / fetch error
   676→      }
   677→    } finally {
   678→      releaseNavRestore(epoch)
   679→    }
   680→  }
   681→
   682→  /** Restore the screen for '#/pos/table/{id}': its live order when the table
   683→   *  got occupied meanwhile (hash fixed to order/{id}), or the free-table
   684→   *  draft screen — applySelectTable pops any stashed draft for it. */
   685→  const restoreTableById = async (id: number, target: string) => {
   686→    const epoch = claimNavRestore(target)
   687→    if (epoch == null) return
   688→    stashDraft()
   689→    try {
   690→      // Same data source as the floor screen (TableSelect's floorplans query).
   691→      const data = await fetcher<{ floorPlans: FloorPlan[] }>('/api/floorplans')
   692→      if (epoch !== navEpochRef.current || !hashStill(target)) return
   693→      queryClient.setQueryData(['floorplans'], data)
   694→      const table =
   695→        data.floorPlans.flatMap((fp) => fp.tables).find((tb) => tb.id === id) ?? null
   696→      if (!table) {
   697→        cannotReopen()
   698→        return
   699→      }
   700→      if (table.openOrderId != null) {
   701→        // Occupied meanwhile → its live order screen; fix the hash to it.
   702→        const orderTarget = `order/${table.openOrderId}`
   703→        replaceNav('pos', orderTarget)
   704→        const fetched = await fetchOrderOrNull(table.openOrderId)
   705→        if (epoch !== navEpochRef.current || !hashStill(orderTarget)) return
   706→        if (fetched && fetched.status === 'open') restoreLiveOrder(fetched)
   707→        else cannotReopen()
   708→        return
   709→      }
   710→      // Free table → draft mode (restores are push-suppressed inside).
   711→      withNavSuppression(() => applySelectTable(table, 2))
   712→    } catch {
   713→      // Floorplans fetch failed — only bail to the floor when the hash still
   714→      // points at this target (never clobber another view's hash).
   715→      if (epoch === navEpochRef.current && hashStill(target)) cannotReopen()
   716→    } finally {
   717→      releaseNavRestore(epoch)
   718→    }
   719→  }
   720→
   721→  /** onNav handler — browser back/forward + hand-edited hashes. Idempotent:
   722→   *  navigating to the state we are already in is a no-op (checked against
   723→   *  both the rendered state and the synchronous appliedNavRef mirror). */
   724→  const handleNavEvent = (nav: NavHash | null) => {
   725→    if (nav == null || nav.view !== 'pos') return // another view owns the hash
   726→    const sub = nav.sub
   727→    if (sub == null) {
   728→      // Floor — leave order mode exactly like the in-app back button
   729→      // (stash + reset + replaceNav('#/pos'), itself idempotent here).
   730→      if (mode === 'order' || appliedNavRef.current != null) resetToTables()
   731→      return
   732→    }
   733→    if (sub.startsWith('order/')) {
   734→      const id = Number.parseInt(sub.slice('order/'.length), 10)
   735→      if (!Number.isInteger(id) || id <= 0) return
   736→      if (appliedNavRef.current === sub) return // idempotent re-entry
   737→      if (mode === 'order' && activeOrderId === id) return
   738→      void restoreOrderById(id, sub)
   739→      return
   740→    }
   741→    if (sub.startsWith('table/')) {
   742→      const id = Number.parseInt(sub.slice('table/'.length), 10)
   743→      if (!Number.isInteger(id) || id <= 0) return
   744→      if (appliedNavRef.current === sub) return
   745→      if (mode === 'order' && selectedTable?.id === id && activeOrderId == null) return
   746→      void restoreTableById(id, sub)
   747→      return
   748→    }
   749→    if (sub === 'takeaway') {
   750→      if (appliedNavRef.current === sub) return
   751→      if (mode === 'order' && selectedTable?.id == null && activeOrderId == null) return
   752→      const epoch = claimNavRestore(sub)
   753→      if (epoch == null) return
   754→      try {
   755→        stashDraft()
   756→        // startTakeaway semantics with pushes suppressed; pops the
   757→        // 'takeaway' stash (toast when items come back).
   758→        withNavSuppression(() => startTakeaway())
   759→      } finally {
   760→        releaseNavRestore(epoch)
   761→      }
   762→    }
   763→  }
   764→
   765→  // Always dispatch nav events to the handler of the LATEST render (single
   766→  // subscription created below, no stale closures).
   767→  useEffect(() => {
   768→    navHandlerRef.current = handleNavEvent
   769→  })
   770→
   771→  // Single subscription for the component's lifetime (idempotent handler).
   772→  useEffect(() => {
   773→    const unsubscribe = onNav((nav) => navHandlerRef.current(nav))
   774→    return unsubscribe
   775→  }, [])
   776→
   777→  // Mount-time deep-link restore: refresh/bookmark on '#/pos/order/N',
   778→  // '#/pos/table/N' or '#/pos/takeaway' reopens that screen (pushes are
   779→  // suppressed — the hash already points here). Floor is the default state.
   780→  useEffect(() => {
   781→    const nav = currentNav()
   782→    if (nav && nav.view === 'pos' && nav.sub) navHandlerRef.current(nav)
   783→  }, [])
   784→
   785→  // Re-assert the deep hash when this view becomes active again: page.tsx
   786→  // pushes a shallow '#/pos' when returning from another view, and this
   787→  // replaces it with the real deep hash of the preserved order screen.
   788→  useEffect(() => {
   789→    const wasActive = prevActiveRef.current
   790→    prevActiveRef.current = active
   791→    // Fire only on the hidden → visible flip (wasActive false, now true)
   792→    // while an order screen is preserved.
   793→    if (wasActive || !active || mode !== 'order') return
   794→    const sub = activeOrderId
   795→      ? `order/${activeOrderId}`
   796→      : selectedTable?.id != null
   797→        ? `table/${selectedTable.id}`
   798→        : 'takeaway'
   799→    replaceNav('pos', sub)
   800→  }, [active, mode, activeOrderId, selectedTable?.id])
   801→
   802→  // ── Render ───────────────────────────────────────────────────────
   803→  if (mode === 'tables') {
   804→    return (
   805→      <>
   806→        <TableSelect
   807→          onSelectTable={selectTable}
   808→          onTakeaway={startTakeaway}
   809→          onOpenTakeawayOrder={openTakeawayOrder}
   810→          transferOrderId={transferOrderId}
   811→          onTransferDone={handleTransferDone}
   812→          onSeatParty={handleSeatParty}
   813→          onSettleDeferred={handleSettleDeferred}
   814→        />
   815→        {/* Guests quick dialog — shown BEFORE entering order mode on a free
   816→            table (or after a Seat Party selection). */}
   817→        <GuestsDialog
   818→          open={guestsDialogOpen}
   819→          value={guestsValue}
   820→          pending={updateGuests.isPending}
   821→          onValueChange={setGuestsValue}
   822→          onConfirm={confirmGuests}
   823→          onCancel={cancelGuestsDialog}
   824→        />
   825→        {/* Deferred-check settlement (chips on the floor) — same payment modal
   826→            as order mode; hides its defer button for already-deferred orders. */}
   827→        {payOrder && (
   828→          <PaymentModal
   829→            order={payOrder}
   830→            open={payOpen}
   831→            onOpenChange={(o) => {
   832→              setPayOpen(o)
   833→              if (!o) setPayOrder(null)
   834→            }}
   835→            onSuccess={(updated, closed) => void handlePaySuccess(updated, closed)}
   836→            onDeferred={(deferredOrder) => void handleOrderDeferred(deferredOrder)}
   837→          />
   838→        )}
   839→        {/* Receipt after a full payment — the screen resets to the floor while
   840→            the receipt stays on top (independent of the payment modal). */}
   841→        {receiptOrder && (
   842→          <ReceiptModal
   843→            order={receiptOrder}
   844→            open={!!receiptOrder}
   845→            onOpenChange={(o) => {
   846→              if (!o) setReceiptOrder(null)
   847→            }}
   848→            onClose={() => setReceiptOrder(null)}
   849→          />
   850→        )}
   851→      </>
   852→    )
   853→  }
   854→
   855→  return (
   856→    <div className="flex h-full min-h-[560px] flex-col">
   857→      {/* ── Top bar (Odoo order chrome) ── */}
   858→      <header className="flex min-h-16 shrink-0 flex-wrap items-center gap-2 border-b border-[#E2E2E0] bg-white px-3 py-2 shadow-sm sm:px-4">
   859→        <Button
   860→          variant="ghost"
   861→          className="h-11 rounded-xl px-3 text-[#714B67] hover:bg-[#714B67]/10"
   862→          onClick={resetToTables}
   863→          title={t('pos.backToTables')}
   864→        >
   865→          <ChevronLeft className="size-5 rtl:rotate-180" />
   866→          <span>{t('pos.title')}</span>
   867→        </Button>
   868→        <div className="flex min-w-0 flex-1 flex-wrap items-center gap-x-2 gap-y-1">
   869→          <span className="truncate text-xl font-bold">{selectedTable?.name ?? t('pos.order')}</span>
   870→          {mergedExtraCount > 0 && (
   871→            <Badge className="gap-1 border-violet-300 bg-violet-100 text-violet-800 hover:bg-violet-100">
   872→              <Combine className="size-3" aria-hidden />
   873→              {t('pos.mergedTablesBadge', { n: mergedExtraCount })}
   874→            </Badge>
   875→          )}
   876→          {order && (
   877→            <Badge className="bg-[#714B67] text-white hover:bg-[#714B67]">
   878→              {t('common.order')} #{order.id}
   879→            </Badge>
   880→          )}
   881→          {order && (
   882→            <span className="text-xs tabular-nums text-muted-foreground" data-tick={elapsedTick}>
   883→              {elapsedSince(order.createdAt)}
   884→            </span>
   885→          )}
   886→          {preparingCount > 0 && (
   887→            <Badge className="border-amber-300 bg-amber-100 text-amber-800 hover:bg-amber-100">
   888→              {t('pos.preparingCount', { n: preparingCount })}
   889→            </Badge>
   890→          )}
   891→          {readyCount > 0 && (
   892→            <Badge className="border-emerald-300 bg-emerald-100 text-emerald-800 hover:bg-emerald-100">
   893→              {t('pos.readyCount', { n: readyCount })}
   894→            </Badge>
   895→          )}
   896→        </div>
   897→        <div className="flex items-center gap-2">
   898→          {/* Guests: N — edit while drafting (sent with the order) or on the live order */}
   899→          <Button
   900→            variant="outline"
   901→            className="h-11 rounded-xl border-[#714B67]/40 text-[#714B67] hover:bg-[#714B67]/10 hover:text-[#714B67]"
   902→            onClick={openGuestsEdit}
   903→            title={t('pos.editGuests')}
   904→          >
   905→            <Users />
   906→            <span className="hidden sm:inline">{t('common.guests')}:</span>
   907→            <span className="font-semibold tabular-nums">{order ? order.guests : guestsDraft}</span>
   908→          </Button>
   909→          <Button
   910→            variant="outline"
   911→            className="h-11 rounded-xl border-[#714B67]/40 text-[#714B67] hover:bg-[#714B67]/10 hover:text-[#714B67]"
   912→            disabled={!order}
   913→            onClick={startTransfer}
   914→            title={order ? t('pos.transferHint') : t('pos.transferDisabledHint')}
   915→          >
   916→            <ArrowLeftRight />
   917→            <span className="hidden md:inline">{t('pos.transfer')}</span>
   918→          </Button>
   919→          {order && canCancel && (
   920→            <Button
   921→              variant="outline"
   922→              className="h-11 rounded-xl border-destructive/40 text-destructive hover:bg-destructive/10 hover:text-destructive"
   923→              onClick={() => setCancelOpen(true)}
   924→            >
   925→              <Ban />
   926→              <span className="hidden md:inline">{t('common.cancel')}</span>
   927→            </Button>
   928→          )}
   929→        </div>
   930→      </header>
   931→
   932→      {/* ── Body: product grid + cart ── */}
   933→      <div className="flex min-h-0 flex-1 flex-col md:flex-row">
   934→        <ProductGrid products={products} onAdd={addProduct} className="min-h-0 flex-1" />
   935→        <div className="flex max-h-[60vh] w-full shrink-0 flex-col border-t border-[#E2E2E0] bg-white md:max-h-none md:w-[380px] md:border-l md:border-t-0 lg:w-[420px]">
   936→          <CartPanel
   937→            order={order}
   938→            orderLoading={orderLoading}
   939→            draft={draft}
   940→            table={selectedTable ?? { id: null, name: t('pos.order') }}
   941→            onDraftChange={setDraft}
   942→            onSend={() => void sendToKitchen()}
   943→            onPay={() => void handlePay()}
   944→            onPrintCheck={() => void handlePrintCheck()}
   945→            onCancel={() => setCancelOpen(true)}
   946→            canCancel={canCancel}
   947→            sending={sending}
   948→            userRole={user?.role ?? ''}
   949→          />
   950→        </div>
   951→      </div>
   952→
   953→      {/* ── Guests edit dialog (order mode) ── */}
   954→      <GuestsDialog
   955→        open={guestsDialogOpen}
   956→        value={guestsValue}
   957→        pending={updateGuests.isPending}
   958→        onValueChange={setGuestsValue}
   959→        onConfirm={confirmGuests}
   960→        onCancel={cancelGuestsDialog}
   961→      />
   962→
   963→      {/* ── Cancel order confirmation ── */}
   964→      <AlertDialog open={cancelOpen} onOpenChange={setCancelOpen}>
   965→        <AlertDialogContent>
   966→          <AlertDialogHeader>
   967→            <AlertDialogTitle>{t('pos.cancelTitle')}</AlertDialogTitle>
   968→            <AlertDialogDescription>
   969→              {t('pos.cancelDesc', {
   970→                order: order?.id ?? 0,
   971→                table: selectedTable?.name ?? t('pos.order'),
   972→                total: formatCurrency(order?.totalAmount ?? 0),
   973→                items: order?.items.length ?? 0,
   974→              })}
   975→            </AlertDialogDescription>
   976→          </AlertDialogHeader>
   977→          <AlertDialogFooter>
   978→            <AlertDialogCancel disabled={cancelMutation.isPending}>
   979→              {t('pos.keepOrder')}
   980→            </AlertDialogCancel>
   981→            <AlertDialogAction
   982→              className="bg-destructive text-white hover:bg-destructive/90"
   983→              disabled={cancelMutation.isPending}
   984→              onClick={(e) => {
   985→                e.preventDefault()
   986→                if (order) cancelMutation.mutate(order.id)
   987→              }}
   988→            >
   989→              {cancelMutation.isPending && <Loader2 className="animate-spin" />}{' '}
   990→              {t('pos.cancelOrder')}
   991→            </AlertDialogAction>
   992→          </AlertDialogFooter>
   993→        </AlertDialogContent>
   994→      </AlertDialog>
   995→
   996→      {/* ── Split payment ── */}
   997→      {payOrder && (
   998→        <PaymentModal
   999→          order={payOrder}
  1000→          open={payOpen}
  1001→          onOpenChange={(o) => {
  1002→            setPayOpen(o)
  1003→            if (!o) setPayOrder(null)
  1004→          }}
  1005→          onSuccess={(updated, closed) => void handlePaySuccess(updated, closed)}
  1006→          onDeferred={(deferredOrder) => void handleOrderDeferred(deferredOrder)}
  1007→        />
  1008→      )}
  1009→
  1010→      {/* ── Pre-payment guest check (from the cart panel) ── */}
  1011→      {checkOrder && (
  1012→        <CheckModal
  1013→          key={checkOrder.id}
  1014→          order={checkOrder}
  1015→          open={checkOpen}
  1016→          onOpenChange={(o) => {
  1017→            setCheckOpen(o)
  1018→            if (!o) setCheckOrder(null)
  1019→          }}
  1020→        />
  1021→      )}
  1022→
  1023→      {/* ── Receipt — shown after an order is fully paid, independent of the payment modal ── */}
  1024→      {receiptOrder && (
  1025→        <ReceiptModal
  1026→          order={receiptOrder}
  1027→          open={!!receiptOrder}
  1028→          onOpenChange={(o) => {
  1029→            if (!o) setReceiptOrder(null)
  1030→          }}
  1031→          onClose={() => setReceiptOrder(null)}
  1032→        />
  1033→      )}
  1034→    </div>
  1035→  )
  1036→}

  1038→/** Quick guests picker — stepper + 1-8 chips (min MIN_GUESTS, max MAX_GUESTS). */
  1039→function GuestsDialog({
  1040→  open,
  1041→  value,
  1042→  pending,
