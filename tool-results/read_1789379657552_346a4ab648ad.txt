   418→   418→    const sig = modifierSignature(modifiers)
   419→   419→    setDraft((prev) => {

   421→   421→      const existing = prev.find(
   422→   422→        (d) =>
   423→   423→          d.productId === p.id &&
   424→   424→          !d.notes &&
   425→   425→          d.course === course &&
   426→   426→          modifierSignature(d.modifiers) === sig,
   427→   427→      )
   428→   428→      if (existing) {
   429→   429→        return prev.map((d) =>
   430→   430→          d.key === existing.key ? { ...d, quantity: round2(d.quantity + quantity) } : d,
   431→   431→        )
   432→   432→      }
   433→   433→      return [
   434→   434→        ...prev,
   435→   435→        {
   436→   436→          key: newDraftKey(),
   437→   437→          productId: p.id,
   438→   438→          name: p.name,
   439→   439→          nameAr: p.nameAr ?? null,
   440→   440→          price: p.price,
   441→   441→          quantity,
   442→   442→          notes: '',
   443→   443→          course,
   444→   444→          modifiers,
   445→   445→        },
   446→   446→      ]
   447→   447→    })
   448→   448→  }
   449→   449→
   450→   450→  const addProduct = (p: Product) => {
   451→   451→    // Products with (active) option groups open the customize sheet; the
   452→   452→    // rest keep the classic one-tap add.
   453→   453→    if ((p.modifierGroups ?? []).some((g) => g.active && g.modifiers.some((m) => m.active))) {

   465→   465→  const draftToPayload = (d: DraftItem[]) =>

   474→   474→  const sendToKitchen = async (): Promise<Order | null> => {
   475→   475→    if (draft.length === 0 || sending) return order
   476→   476→    setSending(true)

   478→   478→      let result: { order: Order }
   479→   479→      // Multi-table seating (Seat Party): create the order spanning ALL the

   607→   607→    setPayOpen(true)

   609→   609→
   610→   610→  // Defer completed inside the payment modal (modal closes itself + fires

   612→   612→  const handleOrderDeferred = async (_deferredOrder: Order) => {

   617→   617→  const cancelMutation = useMutation({

   629→   629→    !!order && !!user && (user.role === 'admin' || (order.userId != null && order.userId === user.id))
   630→   630→
   631→   631→  const preparingCount = order?.items.filter((i) => i.status === 'preparing').length ?? 0

   636→   636→  const mergedExtraCount = order

   638→   638→    : Math.max(0, (seatingTables?.length ?? 1) - 1)
   639→   639→
   640→   640→  // ── Round 7: popstate-driven restores (browser/OS back & forward) ──
   641→   641→
   642→   642→  /** Enter order mode for a live order — mirrors openTakeawayOrder /
   643→   643→   *  applySelectTable semantics. Hash-driven, so it never pushes history. */
   644→   644→  const restoreLiveOrder = (restored: Order) => {
   645→   645→    setSelectedTable(
   646→   646→      restored.table

   661→   661→
   662→   662→  /** Claim the right to run a deep restore for `target`. Returns the epoch, or
   663→   663→   *  null when the identical restore is already in flight (popstate +
   664→   664→   *  hashchange double fire). A different target supersedes the old one. */
   665→   665→  const claimNavRestore = (target: string): number | null => {
   666→   666→    if (navInFlightRef.current === target) return null
   667→   667→    navInFlightRef.current = target
   668→   668→    navEpochRef.current += 1
   669→   669→    return navEpochRef.current
   670→   670→  }
   671→   671→
   672→   672→  /** Release the restore guards — only the owning epoch may clear them. */
   673→   673→  const releaseNavRestore = (epoch: number) => {
   674→   674→    if (epoch !== navEpochRef.current) return
   675→   675→    navInFlightRef.current = null
   676→   676→  }
   677→   677→
   678→   678→  /** True when the hash still points at `target` (mid-flight re-check). */
   679→   679→  const hashStill = (target: string) => {
   680→   680→    const now = currentNav()
   681→   681→    return now != null && now.view === 'pos' && now.sub === target
   682→   682→  }
   683→   683→
   684→   684→  /** Failed restore → toast + a safe floor entry. */
   685→   685→  const cannotReopen = () => {
   686→   686→    toast.error(t('pos.cannotReopenToast'))
   687→   687→    replaceNav('pos')
   688→   688→    resetToTables()
   689→   689→  }
   690→   690→
   691→   691→  /** Fetch an order imperatively (null on error). */
   692→   692→  const fetchOrderOrNull = async (id: number): Promise<Order | null> => {
   693→   693→    try {
   694→   694→      const data = await fetcher<{ order: Order }>(`/api/orders/${id}`)
   695→   695→      return data.order ?? null
   696→   696→    } catch {
   697→   697→      return null
   698→   698→    }
   699→   699→  }
   700→   700→
   701→   701→  /** Restore the live-order screen for '#/pos/order/{id}'. */
   702→   702→  const restoreOrderById = async (id: number, target: string) => {
   703→   703→    const epoch = claimNavRestore(target)
   704→   704→    if (epoch == null) return
   705→   705→    stashDraft() // leaving the current context — never lose unsent items
   706→   706→    try {
   707→   707→      const fetched = await fetchOrderOrNull(id)
   708→   708→      // The user navigated again mid-flight → abort silently.
   709→   709→      if (epoch !== navEpochRef.current || !hashStill(target)) return
   710→   710→      if (fetched && fetched.status === 'open') {
   711→   711→        restoreLiveOrder(fetched)
   712→   712→      } else {
   713→   713→        cannotReopen() // closed / cancelled / merged away / fetch error
   714→   714→      }
   715→   715→    } finally {
   716→   716→      releaseNavRestore(epoch)
   717→   717→    }
   718→   718→  }
   719→   719→
   720→   720→  /** Restore the screen for '#/pos/table/{id}': its live order when the table
   721→   721→   *  got occupied meanwhile (hash fixed to order/{id}), or the free-table
   722→   722→   *  draft screen — applySelectTable pops any stashed draft for it. */
   723→   723→  const restoreTableById = async (id: number, target: string) => {
   724→   724→    const epoch = claimNavRestore(target)
   725→   725→    if (epoch == null) return
   726→   726→    stashDraft()
   727→   727→    try {
   728→   728→      // Same data source as the floor screen (TableSelect's floorplans query).
   729→   729→      const data = await fetcher<{ floorPlans: FloorPlan[] }>('/api/floorplans')
   730→   730→      if (epoch !== navEpochRef.current || !hashStill(target)) return
   731→   731→      queryClient.setQueryData(['floorplans'], data)
   732→   732→      const table =
   733→   733→        data.floorPlans.flatMap((fp) => fp.tables).find((tb) => tb.id === id) ?? null
   734→   734→      if (!table) {
   735→   735→        cannotReopen()
   736→   736→        return
   737→   737→      }
   738→   738→      if (table.openOrderId != null) {
   739→   739→        // Occupied meanwhile → its live order screen; fix the hash to it.
   740→   740→        const orderTarget = `order/${table.openOrderId}`
   741→   741→        replaceNav('pos', orderTarget)
   742→   742→        const fetched = await fetchOrderOrNull(table.openOrderId)
   743→   743→        if (epoch !== navEpochRef.current || !hashStill(orderTarget)) return
   744→   744→        if (fetched && fetched.status === 'open') restoreLiveOrder(fetched)
   745→   745→        else cannotReopen()
   746→   746→        return
   747→   747→      }
   748→   748→      // Free table → draft mode (restores are push-suppressed inside).
   749→   749→      withNavSuppression(() => applySelectTable(table, 2))
   750→   750→    } catch {
   751→   751→      // Floorplans fetch failed — only bail to the floor when the hash still
   752→   752→      // points at this target (never clobber another view's hash).
   753→   753→      if (epoch === navEpochRef.current && hashStill(target)) cannotReopen()
   754→   754→    } finally {
   755→   755→      releaseNavRestore(epoch)
   756→   756→    }
   757→   757→  }
   758→   758→
   759→   759→  /** onNav handler — browser back/forward + hand-edited hashes. Idempotent:
   760→   760→   *  navigating to the state we are already in is a no-op (checked against
   761→   761→   *  both the rendered state and the synchronous appliedNavRef mirror). */
   762→   762→  const handleNavEvent = (nav: NavHash | null) => {
   763→   763→    if (nav == null || nav.view !== 'pos') return // another view owns the hash
   764→   764→    const sub = nav.sub
   765→   765→    if (sub == null) {
   766→   766→      // Floor — leave order mode exactly like the in-app back button
   767→   767→      // (stash + reset + replaceNav('#/pos'), itself idempotent here).
   768→   768→      if (mode === 'order' || appliedNavRef.current != null) resetToTables()
   769→   769→      return
   770→   770→    }
   771→   771→    if (sub.startsWith('order/')) {
   772→   772→      const id = Number.parseInt(sub.slice('order/'.length), 10)
   773→   773→      if (!Number.isInteger(id) || id <= 0) return
   774→   774→      if (appliedNavRef.current === sub) return // idempotent re-entry
   775→   775→      if (mode === 'order' && activeOrderId === id) return
   776→   776→      void restoreOrderById(id, sub)
   777→   777→      return
   778→   778→    }
   779→   779→    if (sub.startsWith('table/')) {
   780→   780→      const id = Number.parseInt(sub.slice('table/'.length), 10)
   781→   781→      if (!Number.isInteger(id) || id <= 0) return
   782→   782→      if (appliedNavRef.current === sub) return
   783→   783→      if (mode === 'order' && selectedTable?.id === id && activeOrderId == null) return
   784→   784→      void restoreTableById(id, sub)
   785→   785→      return
   786→   786→    }
   787→   787→    if (sub === 'takeaway') {
   788→   788→      if (appliedNavRef.current === sub) return
   789→   789→      if (mode === 'order' && selectedTable?.id == null && activeOrderId == null) return
   790→   790→      const epoch = claimNavRestore(sub)
   791→   791→      if (epoch == null) return
   792→   792→      try {
   793→   793→        stashDraft()
   794→   794→        // startTakeaway semantics with pushes suppressed; pops the
   795→   795→        // 'takeaway' stash (toast when items come back).
   796→   796→        withNavSuppression(() => startTakeaway())
   797→   797→      } finally {
   798→   798→        releaseNavRestore(epoch)
   799→   799→      }
   800→   800→    }
   801→   801→  }
   802→   802→
   803→   803→  // Always dispatch nav events to the handler of the LATEST render (single
   804→   804→  // subscription created below, no stale closures).
   805→   805→  useEffect(() => {
   806→   806→    navHandlerRef.current = handleNavEvent
   807→   807→  })
   808→   808→
   809→   809→  // Single subscription for the component's lifetime (idempotent handler).
   810→   810→  useEffect(() => {
   811→   811→    const unsubscribe = onNav((nav) => navHandlerRef.current(nav))
   812→   812→    return unsubscribe
   813→   813→  }, [])
   814→   814→
   815→   815→  // Mount-time deep-link restore: refresh/bookmark on '#/pos/order/N',
   816→   816→  // '#/pos/table/N' or '#/pos/takeaway' reopens that screen (pushes are
   817→   817→  // suppressed — the hash already points here). Floor is the default state.
   818→   818→  useEffect(() => {
   819→   819→    const nav = currentNav()
   820→   820→    if (nav && nav.view === 'pos' && nav.sub) navHandlerRef.current(nav)
   821→   821→  }, [])
   822→   822→
   823→   823→  // Re-assert the deep hash when this view becomes active again: page.tsx
   824→   824→  // pushes a shallow '#/pos' when returning from another view, and this
   825→   825→  // replaces it with the real deep hash of the preserved order screen.
   826→   826→  useEffect(() => {
   827→   827→    const wasActive = prevActiveRef.current
   828→   828→    prevActiveRef.current = active
   829→   829→    // Fire only on the hidden → visible flip (wasActive false, now true)
   830→   830→    // while an order screen is preserved.
   831→   831→    if (wasActive || !active || mode !== 'order') return
   832→   832→    const sub = activeOrderId
   833→   833→      ? `order/${activeOrderId}`
   834→   834→      : selectedTable?.id != null
   835→   835→        ? `table/${selectedTable.id}`
   836→   836→        : 'takeaway'
   837→   837→    replaceNav('pos', sub)
   838→   838→  }, [active, mode, activeOrderId, selectedTable?.id])
   839→   839→

   842→   842→    return (

   845→   845→          onSelectTable={selectTable}
   846→   846→          onTakeaway={startTakeaway}
   847→   847→          onOpenTakeawayOrder={openTakeawayOrder}

   850→   850→          onSeatParty={handleSeatParty}
   851→   851→          onSettleDeferred={handleSettleDeferred}
   852→   852→          onMyShift={() => setMyShiftOpen(true)}
   853→   853→        />

   856→   856→        {/* Guests quick dialog — shown BEFORE entering order mode on a free
   857→   857→            table (or after a Seat Party selection). */}
   858→   858→        <GuestsDialog
   859→   859→          open={guestsDialogOpen}
   860→   860→          value={guestsValue}
   861→   861→          pending={updateGuests.isPending}
   862→   862→          onValueChange={setGuestsValue}
   863→   863→          onConfirm={confirmGuests}
