    44→    44→type TableSelectProps = {
    45→    45→  onSelectTable: (table: RestaurantTable) => void
    46→    46→  onTakeaway: () => void
    47→    47→  onOpenTakeawayOrder: (order: Order) => void

    59→    59→type ToolKind = 'transfer' | 'merge' | 'seat'

    63→    63→
    64→    64→/** Per-shape tile silhouette classes — Odoo 17-style tiles. The floor grid
    65→    65→ *  keeps its responsive layout (tiles fill the grid cell width); only the
    66→    66→ *  silhouette changes (square/round/rectangle/oval). The admin floor-plan
    67→    67→ *  editor mirrors these exact strings. */
    68→    68→const SHAPE_TILE_CLASSES: Record<string, string> = {
    69→    69→  square: 'aspect-square rounded-2xl',
    70→    70→  round: 'aspect-square rounded-full',
    71→    71→  rectangle: 'h-24 rounded-2xl',
    72→    72→  oval: 'h-24 rounded-full',
    73→    73→}
    74→    74→
    75→    75→function shapeClasses(shape: string): string {
    76→    76→  return SHAPE_TILE_CLASSES[shape] ?? SHAPE_TILE_CLASSES.square
    77→    77→}

    79→    79→export default function TableSelect({
    80→    80→  onSelectTable,
    81→    81→  onTakeaway,
    82→    82→  onOpenTakeawayOrder,
    83→    83→  transferOrderId,
    84→    84→  onTransferDone,
    85→    85→  onSeatParty,
    86→    86→  onSettleDeferred,
    87→    87→}: TableSelectProps) {
    88→    88→  const queryClient = useQueryClient()
    89→    89→  const { t } = useI18n()
    90→    90→
    91→    91→  const [floorIdx, setFloorIdx] = useState(0)
    92→    92→  // When pos-view passes transferOrderId (Transfer clicked on the order
    93→    93→  // screen), this component mounts fresh in transfer step 2 (destination
    94→    94→  // selection) — pos-view only sets it right before switching to tables view,
    95→    95→  // so mount-time initialization is sufficient.
    96→    96→  const [tool, setTool] = useState<ToolKind | null>(transferOrderId != null ? 'transfer' : null)
    97→    97→  const [source, setSource] = useState<SourcePick | null>(
    98→    98→    transferOrderId != null ? { orderId: transferOrderId } : null,
    99→    99→  )
   100→   100→  // Seat Party: the free tables tapped so far, in selection order (the first
   101→   101→  // one becomes the primary table of the merged seating).
   102→   102→  const [seatSelection, setSeatSelection] = useState<RestaurantTable[]>([])
   103→   103→  // Paid/deferred tile tap-to-clear confirmation dialog target.
   104→   104→  const [clearTarget, setClearTarget] = useState<RestaurantTable | null>(null)
   105→   105→
   106→   106→  const { data: floorPlanData, isLoading } = useQuery({
   107→   107→    queryKey: ['floorplans'],
   108→   108→    queryFn: () => fetcher<{ floorPlans: FloorPlan[] }>('/api/floorplans'),
   109→   109→    refetchInterval: 3000, // LIVE table statuses
   110→   110→  })
   111→   111→
   112→   112→  const { data: openOrdersData } = useQuery({
   113→   113→    queryKey: ['orders', 'open'],
   114→   114→    queryFn: () => fetcher<{ orders: Order[] }>('/api/orders?status=open'),
   115→   115→    refetchInterval: 3000,
   116→   116→  })
   117→   117→
   118→   118→  // Outstanding deferred checks (client left, payment pending) — settle chips.
   119→   119→  const { data: deferredOrdersData } = useQuery({
   120→   120→    queryKey: ['orders', 'deferred'],
   121→   121→    queryFn: () => fetcher<{ orders: Order[] }>('/api/orders?status=deferred'),
   122→   122→    refetchInterval: 3000,
   123→   123→    enabled: !tool,
   124→   124→  })
   125→   125→
   126→   126→  const floorPlans = floorPlanData?.floorPlans ?? []
   127→   127→  const openOrders = openOrdersData?.orders ?? []
   128→   128→  const deferredOrders = deferredOrdersData?.orders ?? []
   129→   129→  const takeawayOrders = useMemo(
   130→   130→    () => openOrders.filter((o) => o.tableId === null),
   131→   131→    [openOrders],
   132→   132→  )
   133→   133→
   134→   134→  const idx = floorPlans.length > 0 ? Math.min(floorIdx, floorPlans.length - 1) : 0
   135→   135→  const plan = floorPlans[idx] ?? null
   136→   136→
   137→   137→  const sourceOrder = source ? openOrders.find((o) => o.id === source.orderId) ?? null : null
   138→   138→  const sourceLabel = source ? sourceOrder?.table?.name ?? t('common.takeaway') : null
   139→   139→
   140→   140→  // ── Clear table (paid/deferred → free) ────────────────────────────
   141→   141→  const clearTableMutation = useMutation({
   142→   142→    mutationFn: (vars: { id: number; name: string }) =>
   143→   143→      apiFetch<{ table: RestaurantTable }>(`/api/tables/${vars.id}/clear`, {
   144→   144→        method: 'POST',
   145→   145→      }),
   146→   146→    onSuccess: async (_data, vars) => {
   147→   147→      toast.success(t('pos.tableClearedToast', { table: vars.name }))
   148→   148→      setClearTarget(null)
   149→   149→      await Promise.all([
   150→   150→        queryClient.invalidateQueries({ queryKey: ['floorplans'] }),
   151→   151→        queryClient.invalidateQueries({ queryKey: ['tables-status'] }),
   152→   152→        queryClient.invalidateQueries({ queryKey: ['orders'] }),
   153→   153→      ])
   154→   154→    },
   155→   155→    onError: (err: Error) => toast.error(err.message),
   156→   156→  })
   157→   157→
   158→   158→  // ── Transfer & merge mutations ───────────────────────────────────
   159→   159→  const invalidateAfterMove = async (...orderIds: number[]) => {
   160→   160→    await Promise.all([
   161→   161→      queryClient.invalidateQueries({ queryKey: ['floorplans'] }),
   162→   162→      queryClient.invalidateQueries({ queryKey: ['orders'] }),
   163→   163→      queryClient.invalidateQueries({ queryKey: ['orders', 'open'] }),
   164→   164→      queryClient.invalidateQueries({ queryKey: ['tables-status'] }),
   165→   165→      ...orderIds.map((id) => queryClient.invalidateQueries({ queryKey: ['pos-order', id] })),
   166→   166→    ])
   167→   167→  }
   168→   168→
   169→   169→  const transferMutation = useMutation({
   170→   170→    mutationFn: (vars: { orderId: number; tableId: number; tableName: string }) =>
   171→   171→      apiFetch<{ order: Order }>(`/api/orders/${vars.orderId}/transfer`, {
   172→   172→        method: 'POST',
   173→   173→        body: { tableId: vars.tableId },
   174→   174→      }),
   175→   175→    onSuccess: async (_data, vars) => {
   176→   176→      toast.success(t('pos.transferToast', { order: vars.orderId, table: vars.tableName }))
   177→   177→      await invalidateAfterMove(vars.orderId)
   178→   178→      exitTool()
   179→   179→      onTransferDone?.()
   180→   180→    },
   181→   181→    onError: (err: Error) => toast.error(err.message),
   182→   182→  })
   183→   183→
   184→   184→  const mergeMutation = useMutation({
   185→   185→    mutationFn: (vars: { sourceId: number; targetId: number; targetLabel: string }) =>
   186→   186→      apiFetch<{ order: Order }>(`/api/orders/${vars.targetId}/merge`, {
   187→   187→        method: 'POST',
   188→   188→        body: { sourceOrderId: vars.sourceId },
   189→   189→      }),
   190→   190→    onSuccess: async (_data, vars) => {
   191→   191→      toast.success(t('pos.mergeToast', { table: vars.targetLabel }))
   192→   192→      await invalidateAfterMove(vars.sourceId, vars.targetId)
   193→   193→      exitTool()
   194→   194→    },
   195→   195→    onError: (err: Error) => toast.error(err.message),
   196→   196→  })
   197→   197→
   198→   198→  const busy = transferMutation.isPending || mergeMutation.isPending
   199→   199→
   200→   200→  // ── Tool mode helpers ────────────────────────────────────────────
   201→   201→  const exitTool = () => {
   202→   202→    setTool(null)
   203→   203→    setSource(null)
   204→   204→    setSeatSelection([])
   205→   205→  }
   206→   206→
   207→   207→  const startTransferTool = () => {
   208→   208→    setTool('transfer')
   209→   209→    setSource(null)
   210→   210→    setSeatSelection([])
   211→   211→  }
   212→   212→
   213→   213→  const startMergeTool = () => {
   214→   214→    setTool('merge')
   215→   215→    setSource(null)
   216→   216→    setSeatSelection([])
   217→   217→  }
   218→   218→
   219→   219→  const startSeatTool = () => {
   220→   220→    setTool('seat')
   221→   221→    setSource(null)
   222→   222→    setSeatSelection([])
   223→   223→  }
   224→   224→
   225→   225→  // Paid/deferred tables hold no open order — they await a cleanup click.
   226→   226→  const isPaidOrDeferred = (t2: RestaurantTable) =>
   227→   227→    t2.status === 'paid' || t2.status === 'deferred'
   228→   228→
   229→   229→  const tableInteraction = (t2: RestaurantTable): TableInteraction => {
   230→   230→    if (!tool || busy) return busy ? 'ineligible' : 'normal'
   231→   231→    if (tool === 'seat') {
   232→   232→      // Only free tables can host a new party.
   233→   233→      return t2.status === 'free' && t2.openOrderId == null ? 'normal' : 'ineligible'
   234→   234→    }
   235→   235→    if (tool === 'transfer') {
   236→   236→      if (source == null) return t2.openOrderId != null ? 'eligible' : 'ineligible'
   237→   237→      // Destination: free/reserved tables with no open order (paid/deferred
   238→   238→      // tables await cleanup, they are not transfer destinations).
   239→   239→      return t2.openOrderId == null && t2.status !== 'occupied' && !isPaidOrDeferred(t2)
   240→   240→        ? 'eligible'
   241→   241→        : 'ineligible'
   242→   242→    }
   243→   243→    // Merge: source = any occupied table; target = another open order's table.
   244→   244→    if (source == null) return t2.openOrderId != null ? 'eligible' : 'ineligible'
   245→   245→    return t2.openOrderId != null && t2.openOrderId !== source.orderId ? 'eligible' : 'ineligible'
   246→   246→  }
   247→   247→
   248→   248→  const takeawayInteraction = (o: Order): TableInteraction => {
   249→   249→    if (!tool || busy) return busy ? 'ineligible' : 'normal'
   250→   250→    if (tool === 'seat') return 'ineligible'
   251→   251→    if (tool === 'transfer') {
   252→   252→      // Takeaway orders are valid transfer sources, never destinations.
   253→   253→      return source == null ? 'eligible' : 'ineligible'
   254→   254→    }
   255→   255→    if (source == null) return 'eligible'
   256→   256→    return o.id !== source.orderId ? 'eligible' : 'ineligible'
   257→   257→  }
   258→   258→
   259→   259→  const handleTableClick = (t2: RestaurantTable) => {
   260→   260→    if (busy) return
   261→   261→    if (tool === 'seat') {
   262→   262→      if (t2.status !== 'free' || t2.openOrderId != null) return
   263→   263→      setSeatSelection((prev) => {
   264→   264→        if (prev.some((s) => s.id === t2.id)) return prev.filter((s) => s.id !== t2.id)
   265→   265→        if (prev.length >= MAX_SEATING_TABLES) return prev // max 4 tables per seating
   266→   266→        return [...prev, t2]
   267→   267→      })
   268→   268→      return
   269→   269→    }
   270→   270→    if (tool === 'transfer') {
   271→   271→      if (source == null) {
   272→   272→        if (t2.openOrderId == null) return
   273→   273→        setSource({ orderId: t2.openOrderId })
   274→   274→        return
   275→   275→      }
   276→   276→      if (t2.openOrderId != null || t2.status === 'occupied' || isPaidOrDeferred(t2)) return
   277→   277→      transferMutation.mutate({ orderId: source.orderId, tableId: t2.id, tableName: t2.name })
   278→   278→      return
   279→   279→    }
   280→   280→    if (tool === 'merge') {
   281→   281→      if (source == null) {
   282→   282→        if (t2.openOrderId == null) return
   283→   283→        setSource({ orderId: t2.openOrderId })
   284→   284→        return
   285→   285→      }
   286→   286→      if (t2.openOrderId == null || t2.openOrderId === source.orderId) return
   287→   287→      mergeMutation.mutate({ sourceId: source.orderId, targetId: t2.openOrderId, targetLabel: t2.name })
   288→   288→      return
   289→   289→    }
   290→   290→    // Normal mode — paid/deferred tiles are the tap-to-clear affordance.
   291→   291→    if (isPaidOrDeferred(t2) && t2.openOrderId == null) {
   292→   292→      setClearTarget(t2)
   293→   293→      return
   294→   294→    }
   295→   295→    onSelectTable(t2)
   296→   296→  }
   297→   297→
   298→   298→  const handleTakeawayClick = (o: Order) => {
   299→   299→    if (busy) return
   300→   300→    if (tool === 'seat') return
   301→   301→    if (tool === 'transfer') {
   302→   302→      if (source == null) setSource({ orderId: o.id })
   303→   303→      return
   304→   304→    }
   305→   305→    if (tool === 'merge') {
   306→   306→      if (source == null) {
   307→   307→        setSource({ orderId: o.id })
   308→   308→      } else if (o.id !== source.orderId) {
   309→   309→        mergeMutation.mutate({
   310→   310→          sourceId: source.orderId,
   311→   311→          targetId: o.id,
   312→   312→          targetLabel: t('common.takeaway'),
   313→   313→        })
   314→   314→      }
   315→   315→      return
   316→   316→    }
   317→   317→    onOpenTakeawayOrder(o)
   318→   318→  }
   319→   319→
   320→   320→  // Seat Party: hand the selection (in order) to pos-view, then exit the tool.
   321→   321→  const continueSeat = () => {
   322→   322→    if (seatSelection.length === 0) return
   323→   323→    const selected = seatSelection
   324→   324→    exitTool()
   325→   325→    onSeatParty(selected)
   326→   326→  }
   327→   327→
   328→   328→  // ── Banner text (transfer/merge — seat has its own banner) ────────
   329→   329→  const bannerText = busy
   330→   330→    ? tool === 'transfer'
   331→   331→      ? t('pos.transferring')
   332→   332→      : t('pos.merging')
   333→   333→    : tool === 'transfer'
   334→   334→      ? source
   335→   335→        ? sourceLabel
   336→   336→          ? t('pos.transferDest', { order: source.orderId, label: sourceLabel })
   337→   337→          : t('pos.transferDestNoLabel', { order: source.orderId })
   338→   338→        : t('pos.transferPick')
   339→   339→      : source
   340→   340→        ? t('pos.mergeTarget', {
   341→   341→            source: sourceLabel ?? `#${source.orderId}`,
   342→   342→          })
   343→   343→        : t('pos.mergePick')
   344→   344→
   345→   345→  const tables = plan?.tables ?? []
   346→   346→
   347→   347→  // ── Hall stats (current floor) + Seat Party availability ─────────
   348→   348→  const hallFreeCount = tables.filter(
   349→   349→    (tb) => tb.status === 'free' && tb.openOrderId == null,
   350→   350→  ).length
   351→   351→  const hallOccupiedCount = tables.filter(
   352→   352→    (tb) => tb.status === 'occupied' || tb.openOrderId != null,
   353→   353→  ).length
   354→   354→  const hallSeats = tables.reduce((sum, tb) => sum + (tb.capacity ?? 0), 0)
   355→   355→  const hasFreeTable = hallFreeCount > 0
   356→   356→  const seatSeats = seatSelection.reduce((sum, tb) => sum + (tb.capacity ?? 0), 0)
   357→   357→
   358→   358→  return (
   359→   359→    <div className="rms-scroll h-full overflow-y-auto">
   360→   360→      <div className="mx-auto flex w-full max-w-6xl flex-col">
   361→   361→        {/* ── Control bar ── */}
   362→   362→        <div className="flex flex-wrap items-center justify-between gap-3 border-b border-[#E2E2E0] bg-white px-4 py-3">
   363→   363→          {/* Floor switcher */}
   364→   364→          <div className="flex items-center gap-2">
   365→   365→            <Button
   366→   366→              variant="outline"
   367→   367→              size="icon"
   368→   368→              className="h-11 w-11 rounded-xl"
   369→   369→              disabled={idx <= 0}
   370→   370→              onClick={() => setFloorIdx(idx - 1)}
   371→   371→              aria-label={t('pos.prevFloor')}
   372→   372→            >
   373→   373→              <ChevronLeft className="size-5 rtl:rotate-180" />
   374→   374→            </Button>
   375→   375→            <span className="min-w-[110px] text-center text-lg font-bold">
   376→   376→              {plan?.name ?? t('pos.floors')}
   377→   377→            </span>
   378→   378→            <Button
   379→   379→              variant="outline"
   380→   380→              size="icon"
   381→   381→              className="h-11 w-11 rounded-xl"
   382→   382→              disabled={idx >= floorPlans.length - 1}
   383→   383→              onClick={() => setFloorIdx(idx + 1)}
   384→   384→              aria-label={t('pos.nextFloor')}
   385→   385→            >
   386→   386→              <ChevronRight className="size-5 rtl:rotate-180" />
   387→   387→            </Button>
   388→   388→          </div>
   389→   389→
   390→   390→          {/* Actions */}
   391→   391→          <div className="flex flex-wrap items-center gap-2">
   392→   392→            <Button
   393→   393→              variant="outline"
   394→   394→              className="h-11 rounded-xl"
   395→   395→              disabled={openOrders.length === 0 || !!tool}
   396→   396→              onClick={startTransferTool}
   397→   397→            >
   398→   398→              <ArrowLeftRight className="text-[#714B67]" />
   399→   399→              <span className="hidden sm:inline">{t('pos.transfer')}</span>
   400→   400→            </Button>
   401→   401→            <Button
   402→   402→              variant="outline"
   403→   403→              className="h-11 rounded-xl"
   404→   404→              disabled={openOrders.length < 2 || !!tool}
   405→   405→              onClick={startMergeTool}
   406→   406→            >
   407→   407→              <Combine className="text-[#714B67]" />
   408→   408→              <span className="hidden sm:inline">{t('pos.merge')}</span>
   409→   409→            </Button>
   410→   410→            <Button
   411→   411→              variant="outline"
   412→   412→              className="h-11 rounded-xl border-violet-400 text-violet-700 hover:bg-violet-50 hover:text-violet-700"
   413→   413→              disabled={!hasFreeTable || !!tool}
   414→   414→              onClick={startSeatTool}
   415→   415→              title={t('pos.seatParty')}
   416→   416→            >
   417→   417→              <Armchair className="text-violet-700" />
   418→   418→              <span className="hidden sm:inline">{t('pos.seatParty')}</span>
   419→   419→            </Button>
   420→   420→            <Button
   421→   421→              className="h-11 rounded-xl bg-[#714B67] text-white hover:bg-[#714B67]/90"
   422→   422→              disabled={!!tool}
   423→   423→              onClick={onTakeaway}
   424→   424→            >
   425→   425→              <Plus /> {t('pos.newTakeaway')}
   426→   426→            </Button>
   427→   427→          </div>
   428→   428→        </div>
   429→   429→
   430→   430→        {/* ── Seat Party banner (violet) ── */}
   431→   431→        {tool === 'seat' ? (
   432→   432→          <div
   433→   433→            className="flex flex-wrap items-center gap-2 bg-violet-600 px-4 py-3 text-white sm:gap-3"
   434→   434→            role="status"
   435→   435→          >
   436→   436→            <Armchair className="size-5 shrink-0" aria-hidden />
   437→   437→            <div className="min-w-0 flex-1">
   438→   438→              <p className="text-sm font-semibold">{t('pos.seatPick')}</p>
   439→   439→              <p className="text-xs font-medium tabular-nums text-white/80">
   440→   440→                {t('pos.seatTables', { n: seatSelection.length, seats: seatSeats })}
   441→   441→              </p>
   442→   442→            </div>
   443→   443→            <Button

   667→   667→function TableTile({
