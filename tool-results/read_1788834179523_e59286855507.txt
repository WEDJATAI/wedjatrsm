    48→type CartPanelProps = {
    49→  order: Order | null
    50→  orderLoading?: boolean
    51→  draft: DraftItem[]
    52→  table: { id: number | null; name: string }
    53→  onDraftChange: (draft: DraftItem[]) => void

    64→const STATUS_CHIP: Record<string, string> = {
    65→  new: 'bg-zinc-100 text-zinc-500',
    66→  preparing: 'bg-amber-100 text-amber-700 animate-pulse',

    70→
    71→export default function CartPanel({
    72→  order,
    73→  orderLoading = false,
    74→  draft,
    75→  table,
    76→  onDraftChange,
    77→  onSend,
    78→  onPay,
    79→  onPrintCheck,
    80→  onCancel,
    81→  canCancel = false,
    82→  sending = false,
    83→  userRole,
    84→}: CartPanelProps) {
    85→  const queryClient = useQueryClient()
    86→  const { t, lang } = useI18n()
    87→  const orderId = order?.id ?? null
    88→
    89→  const [editing, setEditing] = useState<DraftItem | null>(null)
    90→  const [editQty, setEditQty] = useState('1')
    91→  const [editNotes, setEditNotes] = useState('')
    92→  const [editCourse, setEditCourse] = useState('main')
    93→
    94→  const [discountOpen, setDiscountOpen] = useState(false)
    95→  const [discountTab, setDiscountTab] = useState<'percent' | 'fixed'>('percent')
    96→  const [percentInput, setPercentInput] = useState('')
    97→  const [fixedInput, setFixedInput] = useState('')
    98→
    99→  // ── PIN-gated item deletion state ─────────────────────────────────
   100→  // Sent items may only be removed with the admin's 6-digit PIN; the
   101→  // dialog stays open on a wrong PIN so the user can retry.
   102→  const [pinDialogItem, setPinDialogItem] = useState<OrderItem | null>(null)
   103→  const [pinInput, setPinInput] = useState('')
   104→  const [pinWrong, setPinWrong] = useState(false)
   105→
   106→  // ── Item transfer ("Move items") state ────────────────────────────
   107→  // moveQty holds a per-row quantity to move (default = the row's full
   108→  // quantity when selected); a partial value splits the row on the server —
   109→  // the classic "wrong check" fix (e.g. move just 1 of 3 Koshari).
   110→  const [moveMode, setMoveMode] = useState(false)
   111→  const [moveSelected, setMoveSelected] = useState<Set<number>>(() => new Set())
   112→  const [moveQty, setMoveQty] = useState<Record<number, number>>({})
   113→  const [moveDialogOpen, setMoveDialogOpen] = useState(false)
   114→
   115→  const totals = computeCartTotals(order, draft)
   116→  const noItems = (order?.items.length ?? 0) + draft.length === 0
   117→
   118→  // Open orders (for the move-items target picker) — only fetched while the
   119→  // picker dialog is open; the query key matches the floor's live query.
   120→  const { data: openOrdersData, isLoading: openOrdersLoading } = useQuery({
   121→    queryKey: ['orders', 'open'],
   122→    queryFn: () => fetcher<{ orders: Order[] }>('/api/orders?status=open'),
   123→    enabled: moveDialogOpen,
   124→  })
   125→  const otherOpenOrders = (openOrdersData?.orders ?? []).filter((o) => o.id !== orderId)
   126→
   127→  // ── Server mutations (sent items) ─────────────────────────────────
   128→  const markServed = useMutation({
   129→    mutationFn: (itemId: number) =>
   130→      apiFetch<{ item: OrderItem }>(`/api/order-items/${itemId}`, {
   131→        method: 'PUT',
   132→        body: { status: 'served' },
   133→      }),
   134→    onSuccess: async (data) => {
   135→      toast.success(
   136→        t('pos.markedServedToast', {
   137→          name: data.item.product
   138→            ? localizedName(data.item.product.name, data.item.product.nameAr, lang)
   139→            : t('pos.item'),
   140→        }),
   141→      )
   142→      await queryClient.invalidateQueries({ queryKey: ['pos-order'] })
   143→      await queryClient.invalidateQueries({ queryKey: ['orders'] })
   144→    },
   145→    onError: (err: Error) => toast.error(err.message),
   146→  })
   147→
   148→  // Item removal is PIN-gated (PUT /api/orders/[id] { removeItemIds, removePin }):
   149→  // the dialog collects the 6-digit PIN and the API verifies it server-side
   150→  // (403 with a descriptive message on a wrong or missing PIN).
   151→  const removeItem = useMutation({
   152→    mutationFn: ({ itemId, pin }: { itemId: number; pin: string }) => {
   153→      if (orderId == null) throw new Error(t('pos.noActiveOrder'))
   154→      return apiFetch<{ order: Order }>(`/api/orders/${orderId}`, {
   155→        method: 'PUT',
   156→        body: { removeItemIds: [itemId], removePin: pin },
   157→      })
   158→    },
   159→    onSuccess: async ({ order: updated }) => {
   160→      queryClient.setQueryData(['pos-order', updated.id], { order: updated })
   161→      await queryClient.invalidateQueries({ queryKey: ['orders'] })
   162→      await queryClient.invalidateQueries({ queryKey: ['floorplans'] })
   163→      await queryClient.invalidateQueries({ queryKey: ['tables-status'] })
   164→      toast.success(t('pos.itemRemovedToast'))
   165→      closePinDialog()
   166→    },
   167→    onError: (err: Error) => {
   168→      // The API message is descriptive (403 wrong/missing PIN); the dialog
   169→      // stays open with the inline wrong-PIN hint so the user can retry.
   170→      toast.error(err.message)
   171→      setPinWrong(true)
   172→    },
   173→  })
   174→
   175→  // ── PIN dialog helpers ────────────────────────────────────────────
   176→  const openPinDialog = (item: OrderItem) => {
   177→    setPinDialogItem(item)
   178→    setPinInput('')
   179→    setPinWrong(false)
   180→  }
   181→
   182→  const closePinDialog = () => {
   183→    setPinDialogItem(null)
   184→    setPinInput('')
   185→    setPinWrong(false)
   186→  }
   187→
   188→  const confirmPin = () => {
   189→    if (!pinDialogItem || pinInput.length < DELETE_PIN_LENGTH || removeItem.isPending) return
   190→    removeItem.mutate({ itemId: pinDialogItem.id, pin: pinInput })
   191→  }
   192→
   193→  const applyDiscount = useMutation({
   194→    mutationFn: (discountAmount: number) => {
   195→      if (orderId == null) throw new Error(t('pos.noActiveOrder'))
   196→      return apiFetch<{ order: Order }>(`/api/orders/${orderId}`, {
   197→        method: 'PUT',
   198→        body: { discountAmount },
   199→      })
   200→    },
   201→    onSuccess: async ({ order: updated }) => {
   202→      queryClient.setQueryData(['pos-order', updated.id], { order: updated })
   203→      await queryClient.invalidateQueries({ queryKey: ['orders'] })
   204→      await queryClient.invalidateQueries({ queryKey: ['floorplans'] })
   205→      await queryClient.invalidateQueries({ queryKey: ['tables-status'] })
   206→      toast.success(t('pos.discountUpdatedToast'))
   207→      setDiscountOpen(false)
   208→    },
   209→    onError: (err: Error) => toast.error(err.message),
   210→  })
   211→
   212→  // ── Item transfer mutation (move sent items to another open order) ──
   213→  // Sends the per-row quantities so the server can split rows (partial moves).
   214→  const transferItems = useMutation({
   215→    mutationFn: (vars: {
   216→      sourceId: number
   217→      items: { id: number; quantity: number }[]
   218→      targetOrderId: number
   219→      targetLabel: string
   220→    }) =>
   221→      apiFetch<{ source: Order; target: Order }>(`/api/orders/${vars.sourceId}/transfer-items`, {
   222→        method: 'POST',
   223→        body: { items: vars.items, targetOrderId: vars.targetOrderId },
   224→      }),
   225→    onSuccess: async (_data, vars) => {
   226→      const totalUnitsMoved = round2(vars.items.reduce((n, entry) => n + entry.quantity, 0))
   227→      toast.success(
   228→        t('pos.itemsMovedToast', { n: totalUnitsMoved, target: vars.targetLabel }),
   229→      )
   230→      await Promise.all([
   231→        queryClient.invalidateQueries({ queryKey: ['orders'] }),
   232→        queryClient.invalidateQueries({ queryKey: ['orders', 'open'] }),
   233→        queryClient.invalidateQueries({ queryKey: ['floorplans'] }),
   234→        queryClient.invalidateQueries({ queryKey: ['tables-status'] }),
   235→        queryClient.invalidateQueries({ queryKey: ['pos-order', vars.sourceId] }),
   236→        queryClient.invalidateQueries({ queryKey: ['pos-order', vars.targetOrderId] }),
   237→      ])
   238→      exitMoveMode()
   239→    },
   240→    onError: (err: Error) => toast.error(err.message),
   241→  })
   242→
   243→  // ── Move-items helpers ───────────────────────────────────────────
   244→  const enterMoveMode = () => {
   245→    setMoveMode(true)
   246→    setMoveSelected(new Set())
   247→    setMoveQty({})
   248→  }
   249→
   250→  const exitMoveMode = () => {
   251→    setMoveMode(false)
   252→    setMoveSelected(new Set())
   253→    setMoveQty({})
   254→    setMoveDialogOpen(false)
   255→  }
   256→
   257→  // Toggling a row ON seeds its move quantity with the FULL row quantity;
   258→  // toggling OFF forgets the per-row quantity.
   259→  const toggleMoveItem = (item: OrderItem) => {
   260→    if (moveSelected.has(item.id)) {
   261→      setMoveSelected((prev) => {
   262→        const next = new Set(prev)
   263→        next.delete(item.id)
   264→        return next
   265→      })
   266→      setMoveQty((prev) => {
   267→        const next = { ...prev }
   268→        delete next[item.id]
   269→        return next
   270→      })
   271→    } else {
   272→      setMoveSelected((prev) => {
   273→        const next = new Set(prev)
   274→        next.add(item.id)
   275→        return next
   276→      })
   277→      setMoveQty((prev) => ({ ...prev, [item.id]: item.quantity }))
   278→    }
   279→  }
   280→
   281→  // Only ids that still exist on the (live-polled) order — stale ids are dropped.
   282→  const selectedMoveIds = (order?.items ?? []).filter((i) => moveSelected.has(i.id)).map((i) => i.id)
   283→  const moveCount = selectedMoveIds.length
   284→  const liveItemById = new Map((order?.items ?? []).map((i) => [i.id, i]))
   285→
   286→  // Effective units to move for a row: the stepper value when present,
   287→  // clamped against the LIVE row quantity (polling may have changed it).
   288→  const moveQtyFor = (id: number): number => {
   289→    const live = liveItemById.get(id)
   290→    if (!live) return 0
   291→    return round2(Math.min(moveQty[id] ?? live.quantity, live.quantity))
   292→  }
   293→
   294→  // Stepper handler — clamps between 1 and the live row quantity.
   295→  const changeMoveQty = (item: OrderItem, delta: number) => {
   296→    setMoveQty((prev) => {
   297→      const current = prev[item.id] ?? item.quantity
   298→      const next = Math.min(Math.max(round2(current + delta), 1), item.quantity)
   299→      return { ...prev, [item.id]: round2(next) }
   300→    })
   301→  }
   302→
   303→  // Total UNITS selected to move (CTA label + moved toast).
   304→  const moveUnits = round2(selectedMoveIds.reduce((n, id) => n + moveQtyFor(id), 0))
   305→
   306→  const pickMoveTarget = (target: Order) => {
   307→    if (orderId == null || moveCount === 0) return
   308→    transferItems.mutate({
   309→      sourceId: orderId,
   310→      items: selectedMoveIds.map((id) => ({ id, quantity: moveQtyFor(id) })),
   311→      targetOrderId: target.id,
   312→      targetLabel: target.table?.name ?? t('common.takeaway'),
   313→    })
   314→  }
   315→
   316→  // ── Draft helpers ─────────────────────────────────────────────────
   317→  const updateDraft = (key: string, patch: Partial<DraftItem>) =>
   318→    onDraftChange(draft.map((d) => (d.key === key ? { ...d, ...patch } : d)))
   319→
   320→  const changeQty = (d: DraftItem, delta: number) => {
   321→    const next = round2(d.quantity + delta)
   322→    if (next <= 0) onDraftChange(draft.filter((x) => x.key !== d.key))
   323→    else updateDraft(d.key, { quantity: next })
   324→  }
   325→
   326→  const openEdit = (d: DraftItem) => {
   327→    setEditing(d)
   328→    setEditQty(String(d.quantity))
   329→    setEditNotes(d.notes)
   330→    setEditCourse(d.course)
   331→  }
   332→
   333→  const saveEdit = () => {
   334→    if (!editing) return
   335→    const qty = round2(parseFloat(editQty))
   336→    if (!Number.isFinite(qty) || qty <= 0) return
   337→    updateDraft(editing.key, { quantity: qty, notes: editNotes.trim(), course: editCourse })
   338→    setEditing(null)
   339→  }
   340→
   341→  // Seed discount dialog inputs from the current order discount when opening.
   342→  const openDiscountDialog = () => {
   343→    const current = order?.discountAmount ?? 0
   344→    const subtotal = order?.subtotalAmount ?? 0
   345→    setPercentInput(current > 0 && subtotal > 0 ? String(round2((current / subtotal) * 100)) : '')
   346→    setFixedInput(current > 0 ? String(current) : '')
   347→    setDiscountOpen(true)
   348→  }
   349→
   350→  // Effective discount from the dialog inputs (clamped to [0, order subtotal]).
   351→  const orderSubtotal = round2(order?.subtotalAmount ?? 0)
   352→  const previewDiscount = (() => {
   353→    if (!order) return 0
   354→    const raw =
   355→      discountTab === 'percent'
   356→        ? (orderSubtotal * (parseFloat(percentInput) || 0)) / 100
   357→        : parseFloat(fixedInput) || 0
   358→    if (!Number.isFinite(raw) || raw < 0) return 0
   359→    return round2(Math.min(raw, orderSubtotal))
   360→  })()
   361→  const previewBase = round2(orderSubtotal - previewDiscount)
   362→  const previewVat = round2(previewBase * TAX_RATE)
   363→  const previewServiceTax = round2(previewBase * SERVICE_TAX_RATE)
   364→
   365→  return (
   366→    <div className="flex min-h-0 flex-1 flex-col">
   367→      {/* Header */}
   368→      <div className="flex shrink-0 items-center justify-between gap-2 border-b border-[#E2E2E0] px-4 py-3">
   369→        <div className="min-w-0">
   370→          <p className="truncate text-base font-bold">{table.name}</p>
   371→          {(order?.user?.name || userRole) && (
   372→            <p className="text-[11px] text-muted-foreground">
   373→              {t('pos.server')}:{' '}
   374→              {order?.user?.name ?? (userRole ? t(`role.${userRole}`) : '')}
   375→            </p>
   376→          )}
   377→        </div>
   378→        {order ? (
   379→          <Badge className="shrink-0 bg-[#714B67] text-white hover:bg-[#714B67]">
   380→            {t('common.order')} #{order.id}
   381→          </Badge>
   382→        ) : (
   383→          <Badge variant="secondary" className="shrink-0">
   384→            {t('pos.notSent')}
   385→          </Badge>
   386→        )}
   387→      </div>
   388→
   389→      {/* Body */}
   390→      <div className="rms-scroll min-h-0 flex-1 overflow-y-auto px-4">
   391→        {orderLoading && !order && draft.length === 0 ? (
   392→          <div className="space-y-3 py-4">
   393→            <Skeleton className="h-14 w-full" />
   394→            <Skeleton className="h-14 w-full" />
   395→            <Skeleton className="h-14 w-full" />
   396→          </div>
   397→        ) : noItems ? (
   398→          <div className="flex h-full min-h-[160px] flex-col items-center justify-center gap-2 py-8 text-muted-foreground">
   399→            <ShoppingBag className="size-10 opacity-30" />
   400→            <p className="text-sm">{t('pos.tapToAdd')}</p>
   401→          </div>
   402→        ) : (
   403→          <>
   404→            {/* Sent to kitchen */}
   405→            {order && order.items.length > 0 && (
   406→              <section className="py-2">
   407→                <div className="flex items-center justify-between gap-2">
   408→                  <h3 className="py-1 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   409→                    {t('pos.sentItems')} ({order.items.length})
   410→                  </h3>
   411→                  <div className="flex shrink-0 items-center gap-1">
   412→                    {!moveMode && (
   413→                      <Button
   414→                        variant="outline"
   415→                        size="sm"
   416→                        className="h-8 gap-1 px-2 text-xs border-[#714B67]/40 text-[#714B67] hover:bg-[#714B67]/10 hover:text-[#714B67]"
   417→                        title={t('pos.transferItems')}
   418→                        onClick={enterMoveMode}
   419→                      >
   420→                        <ArrowLeftRight className="size-3.5" /> {t('pos.transferItems')}
   421→                      </Button>
   422→                    )}
   423→                    {moveMode ? (
   424→                      <span className="text-[11px] text-muted-foreground">{t('pos.moveModeHint')}</span>
   425→                    ) : (
   426→                      canCancel && (
   427→                        <Button
   428→                          variant="ghost"
   429→                          size="sm"
   430→                          className="h-8 px-2 text-xs text-destructive hover:bg-destructive/10 hover:text-destructive"
   431→                          title={t('pos.cancelOrder')}
   432→                          onClick={onCancel}
   433→                        >
   434→                          <Ban className="size-3.5" /> {t('pos.cancelOrder')}
   435→                        </Button>
   436→                      )
   437→                    )}
   438→                  </div>
   439→                </div>
   440→                {order.items.map((item) => (
   441→                  <SentItemRow
   442→                    key={item.id}
   443→                    item={item}
   444→                    onServed={() => markServed.mutate(item.id)}
   445→                    onRemove={() => openPinDialog(item)}
   446→                    servedPending={markServed.isPending && markServed.variables === item.id}
   447→                    removePending={removeItem.isPending && removeItem.variables?.itemId === item.id}
   448→                    moveMode={moveMode}
   449→                    moveSelected={moveSelected.has(item.id)}
   450→                    onToggleMove={() => toggleMoveItem(item)}
   451→                    moveQty={moveMode && moveSelected.has(item.id) ? moveQtyFor(item.id) : undefined}
   452→                    onMoveQty={
   453→                      moveMode && moveSelected.has(item.id)
   454→                        ? (delta: number) => changeMoveQty(item, delta)
   455→                        : undefined
   456→                    }
   457→                  />
   458→                ))}
   459→              </section>
   460→            )}
   461→
   462→            {/* New items (draft) */}
   463→            {draft.length > 0 && (
   464→              <section className="py-2">
   465→                <h3 className="py-1 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   466→                  {t('pos.newItemsNotSent')} ({draft.length})
   467→                </h3>
   468→                {draft.map((d) => (
   469→                  <DraftRow
   470→                    key={d.key}
   471→                    item={d}
   472→                    onQty={(delta) => changeQty(d, delta)}
   473→                    onEdit={() => openEdit(d)}
   474→                    onRemove={() => onDraftChange(draft.filter((x) => x.key !== d.key))}
   475→                  />
   476→                ))}
   477→              </section>
   478→            )}
   479→          </>
   480→        )}
   481→      </div>
   482→
   483→      {/* Footer */}
   484→      <div className="shrink-0 space-y-1.5 border-t border-[#E2E2E0] p-4">
   485→        <SummaryRow label={t('money.subtotal')} value={formatCurrency(totals.subtotal)} />
   486→        <div className="flex items-center justify-between text-sm">
   487→          <span className="flex items-center gap-1 text-muted-foreground">
   488→            {t('money.discount')}
   489→            <Button
   490→              variant="ghost"
   491→              size="icon"
   492→              className="size-7 text-muted-foreground"
   493→              disabled={!order}
   494→              title={order ? t('pos.applyDiscount') : t('pos.sendFirst')}
   495→              onClick={openDiscountDialog}
   496→            >
   497→              <Pencil className="size-3.5" />
   498→            </Button>
   499→          </span>
   500→          <span className="tabular-nums text-muted-foreground">
   501→            − {formatCurrency(totals.discount)}
   502→          </span>
   503→        </div>
   504→        <SummaryRow label={t('money.tax')} value={formatCurrency(totals.tax)} />
   505→        <SummaryRow label={t('money.serviceTax')} value={formatCurrency(totals.serviceTax)} />
   506→        <div className="flex items-center justify-between border-t border-[#E2E2E0] pt-2">
   507→          <span className="text-sm font-semibold">{t('money.total')}</span>
   508→          <span className="text-lg font-bold tabular-nums">{formatCurrency(totals.total)}</span>
   509→        </div>
   510→        {order && order.paidAmount > 0 && (
   511→          <>
   512→            <SummaryRow
   513→              label={t('money.paid')}
   514→              value={formatCurrency(order.paidAmount)}
   515→              valueClassName="text-emerald-600"
   516→            />
   517→            <SummaryRow
   518→              label={t('money.remaining')}
