    63→/** Persist the delivery contact for this tab (survives a reload on
    64→ *  '#/pos/delivery' so the screen can be restored after a refresh). */
    65→function persistDeliveryInfo(info: { phone: string; address: string } | null): void {
    66→  try {
    67→    if (info) window.sessionStorage.setItem(DELIVERY_INFO_KEY, JSON.stringify(info))
    68→    else window.sessionStorage.removeItem(DELIVERY_INFO_KEY)
    69→  } catch {
    70→    // storage unavailable — in-memory state still covers the session
    71→  }
    72→}
    73→
    74→function readDeliveryInfo(): { phone: string; address: string } | null {
    75→  try {
    76→    const raw = window.sessionStorage.getItem(DELIVERY_INFO_KEY)
    77→    if (!raw) return null
    78→    const parsed: unknown = JSON.parse(raw)
    79→    if (parsed && typeof parsed === 'object' && typeof (parsed as { phone?: unknown }).phone === 'string') {
    80→      const address = (parsed as { address?: unknown }).address
    81→      return {
    82→        phone: (parsed as { phone: string }).phone,
    83→        address: typeof address === 'string' ? address : '',
    84→      }
    85→    }
    86→  } catch {
    87→    // corrupt entry — ignore
    88→  }
    89→  return null
    90→}

    95→/** R13 PWA: read the mirrored product catalog (undefined when never cached). */
    96→function readProductsCache(): { products: Product[] } | undefined {
    97→  if (typeof window === 'undefined') return undefined
    98→  try {
    99→    const raw = window.localStorage.getItem('rms-pos-products-cache')
   100→    if (!raw) return undefined
   101→    const parsed: unknown = JSON.parse(raw)
   102→    if (parsed && typeof parsed === 'object' && Array.isArray((parsed as { products?: unknown }).products)) {
   103→      return parsed as { products: Product[] }
   104→    }
   105→  } catch {
   106→    // corrupt cache — treat as absent
   107→  }
   108→  return undefined
   109→}

   841→  /** Claim the right to run a deep restore for `target`. Returns the epoch, or
   842→   *  null when the identical restore is already in flight (popstate +
   843→   *  hashchange double fire). A different target supersedes the old one. */
   844→  const claimNavRestore = (target: string): number | null => {
   845→    if (navInFlightRef.current === target) return null
   846→    navInFlightRef.current = target
   847→    navEpochRef.current += 1
   848→    return navEpochRef.current
   849→  }

   852→  const releaseNavRestore = (epoch: number) => {
   853→    if (epoch !== navEpochRef.current) return
   854→    navInFlightRef.current = null
   855→  }

   857→  /** True when the hash still points at `target` (mid-flight re-check). */

   861→  }

   881→  const restoreOrderById = async (id: number, target: string) => {

   885→    try {
   886→      const fetched = await fetchOrderOrNull(id)

   896→    }

   902→  const restoreTableById = async (id: number, target: string) => {
   903→    const epoch = claimNavRestore(target)
   904→    if (epoch == null) return
   905→    stashDraft()
   906→    try {
   907→      // Same data source as the floor screen (TableSelect's floorplans query).
   908→      const data = await fetcher<{ floorPlans: FloorPlan[] }>('/api/floorplans')
   909→      if (epoch !== navEpochRef.current || !hashStill(target)) return
   910→      queryClient.setQueryData(['floorplans'], data)
   911→      const table =
   912→        data.floorPlans.flatMap((fp) => fp.tables).find((tb) => tb.id === id) ?? null
   913→      if (!table) {
   914→        cannotReopen()
   915→        return
   916→      }
   917→      if (table.openOrderId != null) {
   918→        // Occupied meanwhile → its live order screen; fix the hash to it.
   919→        const orderTarget = `order/${table.openOrderId}`
   920→        replaceNav('pos', orderTarget)
   921→        const fetched = await fetchOrderOrNull(table.openOrderId)
   922→        if (epoch !== navEpochRef.current || !hashStill(orderTarget)) return
   923→        if (fetched && fetched.status === 'open') restoreLiveOrder(fetched)
   924→        else cannotReopen()
   925→        return
   926→      }
   927→      // Free table → draft mode (restores are push-suppressed inside).
   928→      withNavSuppression(() => applySelectTable(table, 2))
   929→    } catch {
   930→      // Floorplans fetch failed — only bail to the floor when the hash still
   931→      // points at this target (never clobber another view's hash).
   932→      if (epoch === navEpochRef.current && hashStill(target)) cannotReopen()
   933→    } finally {
   934→      releaseNavRestore(epoch)
   935→    }
   936→  }

   938→  /** onNav handler — browser back/forward + hand-edited hashes. Idempotent:
   939→   *  navigating to the state we are already in is a no-op (checked against
   940→   *  both the rendered state and the synchronous appliedNavRef mirror). */
   941→  const handleNavEvent = (nav: NavHash | null) => {
   942→    if (nav == null || nav.view !== 'pos') return // another view owns the hash
   943→    const sub = nav.sub
   944→    if (sub == null) {
   945→      // Floor — leave order mode exactly like the in-app back button
   946→      // (stash + reset + replaceNav('#/pos'), itself idempotent here).
   947→      if (mode === 'order' || appliedNavRef.current != null) resetToTables()
   948→      return
   949→    }
   950→    if (sub.startsWith('order/')) {
   951→      const id = Number.parseInt(sub.slice('order/'.length), 10)
   952→      if (!Number.isInteger(id) || id <= 0) return
   953→      if (appliedNavRef.current === sub) return // idempotent re-entry
   954→      if (mode === 'order' && activeOrderId === id) return
   955→      void restoreOrderById(id, sub)
   956→      return
   957→    }
   958→    if (sub.startsWith('table/')) {
   959→      const id = Number.parseInt(sub.slice('table/'.length), 10)
   960→      if (!Number.isInteger(id) || id <= 0) return
   961→      if (appliedNavRef.current === sub) return
   962→      if (mode === 'order' && selectedTable?.id === id && activeOrderId == null) return
   963→      void restoreTableById(id, sub)
   964→      return
   965→    }
   966→    if (sub === 'takeaway') {
   967→      if (appliedNavRef.current === sub) return
   968→      if (mode === 'order' && selectedTable?.id == null && activeOrderId == null && orderKindRef.current !== 'delivery') return
   969→      const epoch = claimNavRestore(sub)
   970→      if (epoch == null) return
   971→      try {
   972→        stashDraft()
   973→        // startTakeaway semantics with pushes suppressed; pops the
   974→        // 'takeaway' stash (toast when items come back).
   975→        withNavSuppression(() => startTakeaway())
   976→      } finally {
   977→        releaseNavRestore(epoch)
   978→      }
   979→    }
   980→    if (sub === 'delivery') {
   981→      if (appliedNavRef.current === sub) return
   982→      if (mode === 'order' && selectedTable?.id == null && activeOrderId == null && orderKindRef.current === 'delivery') return
   983→      const epoch = claimNavRestore(sub)
   984→      if (epoch == null) return
   985→      try {
   986→        stashDraft()
   987→        // The delivery contact lives in state (view switches) or in
   988→        // sessionStorage (full reload) — without it the screen cannot be
   989→        // rebuilt, so fall back to the floor with the standard toast.
   990→        const info = deliveryInfo ?? readDeliveryInfo()
   991→        if (!info) {
   992→          cannotReopen()
   993→          return
   994→        }
   995→        withNavSuppression(() => startDelivery(info.phone, info.address))
   996→      } finally {
   997→        releaseNavRestore(epoch)
   998→      }
   999→    }
  1000→  }

  1295→/** Quick guests picker — stepper + 1-8 chips (min MIN_GUESTS, max MAX_GUESTS). */
  1296→function GuestsDialog({
  1297→  open,
  1298→  value,
  1299→  pending,
  1300→  onValueChange,
  1301→  onConfirm,
  1302→  onCancel,
  1303→}: {
  1304→  open: boolean
  1305→  value: number
  1306→  pending: boolean
  1307→  onValueChange: (v: number) => void
  1308→  onConfirm: () => void
  1309→  onCancel: () => void
  1310→}) {
  1311→  const { t } = useI18n()
  1312→
  1313→  const step = (delta: number) => {
  1314→    const next = Math.min(MAX_GUESTS, Math.max(MIN_GUESTS, Math.round(value + delta)))
  1315→    onValueChange(next)
  1316→  }
  1317→
  1318→  return (
  1319→    <Dialog open={open} onOpenChange={(o) => !o && onCancel()}>
  1320→      <DialogContent className="sm:max-w-sm">
  1321→        <DialogHeader>
  1322→          <DialogTitle className="flex items-center gap-2">
  1323→            <Users className="size-5 text-[#714B67]" /> {t('pos.editGuests')}
  1324→          </DialogTitle>
  1325→          <DialogDescription>{t('pos.guestsDialogDesc')}</DialogDescription>
  1326→        </DialogHeader>
  1327→        <div className="space-y-4">
  1328→          <div className="flex items-center justify-center gap-4">
  1329→            <Button
  1330→              variant="outline"
  1331→              size="icon"
  1332→              className="size-12 rounded-xl"
  1333→              disabled={value <= MIN_GUESTS}
  1334→              onClick={() => step(-1)}
  1335→              aria-label={t('pos.fewerGuests')}
  1336→            >
  1337→              <Minus />
  1338→            </Button>
  1339→            <span className="w-16 text-center text-4xl font-bold tabular-nums">{value}</span>
  1340→            <Button
  1341→              variant="outline"
  1342→              size="icon"
  1343→              className="size-12 rounded-xl"
  1344→              disabled={value >= MAX_GUESTS}
  1345→              onClick={() => step(1)}
  1346→              aria-label={t('pos.moreGuests')}
  1347→            >
  1348→              <Plus />
  1349→            </Button>
  1350→          </div>
  1351→          <div className="grid grid-cols-4 gap-1.5">
  1352→            {GUEST_QUICK_CHIPS.map((n) => (
  1353→              <button
  1354→                key={n}
  1355→                type="button"
  1356→                onClick={() => onValueChange(n)}
  1357→                className={cn(
  1358→                  'h-11 rounded-full border text-sm font-semibold tabular-nums transition-colors',
  1359→                  value === n
  1360→                    ? 'border-[#714B67] bg-[#714B67] text-white'
  1361→                    : 'border-[#E2E2E0] bg-white text-stone-600 hover:border-[#714B67]/40',
  1362→                )}
  1363→              >
  1364→                {n}
  1365→              </button>
  1366→            ))}
  1367→          </div>
  1368→        </div>
  1369→        <DialogFooter>
  1370→          <Button variant="outline" onClick={onCancel} disabled={pending}>
  1371→            {t('common.cancel')}
  1372→          </Button>
  1373→          <Button
  1374→            className="bg-[#714B67] text-white hover:bg-[#714B67]/90"
  1375→            onClick={onConfirm}
  1376→            disabled={pending}
  1377→          >
  1378→            {pending ? <Loader2 className="animate-spin" /> : <Check />} {t('common.confirm')}
  1379→          </Button>
  1380→        </DialogFooter>
  1381→      </DialogContent>
  1382→    </Dialog>
  1383→  )
  1384→}
