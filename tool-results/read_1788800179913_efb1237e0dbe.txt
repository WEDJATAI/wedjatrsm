    72→type ProductForm = {
    73→  name: string
    74→  categoryId: string // '' = no category
    75→  price: string
    76→  cost: string
    77→  isSellable: boolean
    78→  isStockable: boolean
    79→  sku: string
    80→  imageUrl: string
    81→  lowStockThreshold: string
    82→  stock: string // create only
    83→}
    84→
    85→type ProductFormErrors = Partial<
    86→  Record<'name' | 'price' | 'cost' | 'lowStockThreshold' | 'stock', string>
    87→>
    88→
    89→const EMPTY_PRODUCT_FORM: ProductForm = {
    90→  name: '',
    91→  categoryId: '',
    92→  price: '',
    93→  cost: '',
    94→  isSellable: true,
    95→  isStockable: false,
    96→  sku: '',
    97→  imageUrl: '',
    98→  lowStockThreshold: '10',
    99→  stock: '',
   100→}
   101→
   102→function toProductForm(p: Product): ProductForm {
   103→  return {
   104→    name: p.name,
   105→    categoryId: p.categoryId === null ? '' : String(p.categoryId),
   106→    price: String(p.price),
   107→    cost: String(p.cost),
   108→    isSellable: p.isSellable,
   109→    isStockable: p.isStockable,
   110→    sku: p.sku ?? '',
   111→    imageUrl: p.imageUrl ?? '',
   112→    lowStockThreshold: String(p.lowStockThreshold),
   113→    stock: String(p.stock),
   114→  }
   115→}
   116→
   117→function validateProductForm(form: ProductForm, isCreate: boolean): ProductFormErrors {
   118→  const errors: ProductFormErrors = {}
   119→  if (form.name.trim() === '') errors.name = 'Name is required'
   120→  const price = Number(form.price)
   121→  if (form.price.trim() === '' || !Number.isFinite(price) || price < 0) {
   122→    errors.price = 'Enter a price of 0 or more'
   123→  }
   124→  if (form.cost.trim() !== '') {
   125→    const cost = Number(form.cost)
   126→    if (!Number.isFinite(cost) || cost < 0) errors.cost = 'Cost must be 0 or more'
   127→  }
   128→  if (form.isStockable) {
   129→    if (form.lowStockThreshold.trim() !== '') {
   130→      const threshold = Number(form.lowStockThreshold)
   131→      if (!Number.isFinite(threshold) || threshold < 0) {
   132→        errors.lowStockThreshold = 'Must be 0 or more'
   133→      }
   134→    }
   135→    if (isCreate && form.stock.trim() !== '') {
   136→      const stock = Number(form.stock)
   137→      if (!Number.isFinite(stock) || stock < 0) errors.stock = 'Must be 0 or more'
   138→    }
   139→  }
   140→  return errors
   141→}

   174→function stockColorClass(product: Product): string | undefined {
   175→  if (!product.isStockable) return undefined
   176→  if (product.stock <= product.lowStockThreshold) {
   177→    return 'font-medium text-rose-600 dark:text-rose-400'
   178→  }
   179→  if (product.stock <= product.lowStockThreshold * 2) {
   180→    return 'font-medium text-amber-600 dark:text-amber-500'
   181→  }
   182→  return undefined
   183→}
   184→
   185→function MarginCell({ product }: { product: Product }) {
   186→  if (product.price <= 0) return <span className="text-muted-foreground">—</span>
   187→  const margin = ((product.price - product.cost) / product.price) * 100
   188→  const color =
   189→    margin > 40
   190→      ? 'text-emerald-600 dark:text-emerald-400'
   191→      : margin >= 20
   192→        ? 'text-amber-600 dark:text-amber-500'
   193→        : 'text-rose-600 dark:text-rose-400'
   194→  return <span className={cn('text-xs font-medium tabular-nums', color)}>{Math.round(margin)}%</span>
   195→}
   196→
   197→function StockCell({ product }: { product: Product }) {
   198→  if (!product.isStockable) return <span className="text-muted-foreground">—</span>
   199→  return <span className={cn('tabular-nums', stockColorClass(product))}>{formatQty(product.stock)}</span>
   200→}

   272→export default function ProductsView() {

   274→
   275→  const [search, setSearch] = useState('')
   276→  const [categoryFilter, setCategoryFilter] = useState('all')
   277→  const [showInactive, setShowInactive] = useState(false)
   278→

   280→  const [editing, setEditing] = useState<Product | null>(null)
   281→  const [form, setForm] = useState<ProductForm>(EMPTY_PRODUCT_FORM)
   282→
   283→  const [deleteTarget, setDeleteTarget] = useState<Product | null>(null)
   284→
   285→  const productsQuery = useQuery({
   286→    queryKey: ['products', { all: showInactive, category: categoryFilter }],
   287→    queryFn: () =>
   288→      fetcher<{ products: Product[] }>(
   289→        `/api/products?${new URLSearchParams({
   290→          ...(showInactive ? { all: '1' } : {}),

   292→        }).toString()}`,
   293→      ),
   294→  })
   295→
   296→  const categoriesQuery = useQuery({
   297→    queryKey: ['categories', 'all'],
   298→    queryFn: () => fetcher<{ categories: Category[] }>('/api/categories?all=1'),
   299→  })
   300→
   301→  const categories = useMemo(
   302→    () =>

   304→        (a, b) => a.displayOrder - b.displayOrder || a.name.localeCompare(b.name),

   306→    [categoriesQuery.data],
   307→  )
   308→
   309→  const products = useMemo(() => {

   313→      q === ''
   314→        ? items
   315→        : items.filter(
   316→            (p) => p.name.toLowerCase().includes(q) || (p.sku ?? '').toLowerCase().includes(q),
   317→          )
   318→    return [...filtered].sort((a, b) => a.name.localeCompare(b.name))
   319→  }, [productsQuery.data, search])
   320→
   321→  const isCreate = editing === null
   322→  const errors = validateProductForm(form, isCreate)
   323→  const hasErrors = Object.values(errors).some(Boolean)
   324→
   325→  const saveMutation = useMutation({
   326→    mutationFn: async () => {
   327→      const payload: Record<string, unknown> = {
   328→        name: form.name.trim(),
   329→        price: Number(form.price),
   330→        cost: form.cost.trim() === '' ? 0 : Number(form.cost),
   331→        isSellable: form.isSellable,
   332→        isStockable: form.isStockable,
   333→      }
   334→      if (form.categoryId !== '') payload.categoryId = Number(form.categoryId)
   335→      if (form.sku.trim() !== '') payload.sku = form.sku.trim()
   336→      if (form.imageUrl.trim() !== '') payload.imageUrl = form.imageUrl.trim()
   337→      if (form.isStockable) {
   338→        payload.lowStockThreshold =
   339→          form.lowStockThreshold.trim() === '' ? 0 : Number(form.lowStockThreshold)
   340→        if (editing === null) {
   341→          payload.stock = form.stock.trim() === '' ? 0 : Number(form.stock)
   342→        }
   343→      }
   344→      return editing === null
   345→        ? apiFetch<{ product: Product }>('/api/products', { method: 'POST', body: payload })
   346→        : apiFetch<{ product: Product }>(`/api/products/${editing.id}`, {
   347→            method: 'PUT',
   348→            body: payload,
   349→          })
   350→    },
   351→    onSuccess: () => {
   352→      void queryClient.invalidateQueries({ queryKey: ['products'] })
   353→      toast.success(editing === null ? 'Product created' : 'Product updated')
   354→      setFormOpen(false)
   355→    },
   356→    onError: (err: Error) => toast.error(err.message),
   357→  })
   358→
   359→  const deleteMutation = useMutation({
   360→    mutationFn: (id: number) =>
   361→      apiFetch<{ ok: boolean }>(`/api/products/${id}`, { method: 'DELETE' }),
   362→    onSuccess: () => {
   363→      void queryClient.invalidateQueries({ queryKey: ['products'] })
   364→      toast.success('Product deactivated')
   365→      setDeleteTarget(null)
   366→    },
   367→    onError: (err: Error) => toast.error(err.message),
   368→  })
   369→
   370→  function openCreate() {
   371→    setEditing(null)
   372→    setForm(EMPTY_PRODUCT_FORM)
   373→    setFormOpen(true)
   374→  }
   375→
   376→  function openEdit(product: Product) {
   377→    setEditing(product)
   378→    setForm(toProductForm(product))
   379→    setFormOpen(true)
   380→  }
   381→
   382→  const hasActiveFilters = search.trim() !== '' || categoryFilter !== 'all'
   383→
   384→  return (
   385→    <div className="mx-auto w-full max-w-7xl space-y-6 p-4 sm:p-6">
   386→      {/* Header */}
   387→      <div className="flex flex-wrap items-start justify-between gap-4">
   388→        <div className="space-y-1">
   389→          <h1 className="text-2xl font-bold">Products</h1>
   390→          <p className="text-muted-foreground text-sm">
   391→            Menu items and ingredients — prices, costs, availability
   392→          </p>
   393→        </div>
   394→        <Button className="h-11" onClick={openCreate}>
   395→          <Plus /> New Product
   396→        </Button>
   397→      </div>
   398→
   399→      {/* Toolbar */}
   400→      <Card className="p-3">
   401→        <div className="flex flex-wrap items-center gap-2">
   402→          <div className="relative w-full sm:w-72">
   403→            <Search
   404→              className="pointer-events-none absolute top-1/2 left-3 size-4 -translate-y-1/2 text-muted-foreground"
   405→              aria-hidden
   406→            />
   407→            <Input
   408→              value={search}
   409→              onChange={(e) => setSearch(e.target.value)}
   410→              placeholder="Search name or SKU…"
   411→              className="h-11 pl-9"
   412→              aria-label="Search products"
   413→            />
   414→          </div>
   415→          <Select value={categoryFilter} onValueChange={setCategoryFilter}>
   416→            <SelectTrigger className="h-11 w-full sm:w-52" aria-label="Filter by category">
   417→              <SelectValue placeholder="All categories" />
   418→            </SelectTrigger>
   419→            <SelectContent>
   420→              <SelectItem value="all">All categories</SelectItem>
   421→              {categories.map((c) => (
   422→                <SelectItem
   423→                  key={c.id}
   424→                  value={String(c.id)}
   425→                  className={c.active ? undefined : 'text-muted-foreground'}
   426→                >
   427→                  {c.name}
   428→                  {c.active ? '' : ' (inactive)'}
   429→                </SelectItem>
   430→              ))}
   431→            </SelectContent>
   432→          </Select>
   433→          <div className="flex h-11 items-center gap-2 rounded-md border px-3">
   434→            <Switch
   435→              id="products-show-inactive"
   436→              checked={showInactive}
   437→              onCheckedChange={setShowInactive}
   438→            />
   439→            <Label htmlFor="products-show-inactive" className="cursor-pointer text-sm">
   440→              Show inactive
   441→            </Label>
   442→          </div>
   443→        </div>
   444→      </Card>
   445→
   446→      {/* Content */}
   447→      <Card className="p-4">
   448→        {productsQuery.isLoading ? (
   449→          <div className="space-y-2">
   450→            {Array.from({ length: 8 }, (_, i) => (
   451→              <div key={i} className="flex items-center gap-4 rounded-lg border p-3">
   452→                <div className="flex-1 space-y-2">
   453→                  <Skeleton className="h-4 w-44" />
   454→                  <Skeleton className="h-3 w-24" />
   455→                </div>
   456→                <Skeleton className="h-4 w-16" />
   457→                <Skeleton className="size-11 rounded-md" />
   458→              </div>
   459→            ))}
   460→          </div>
   461→        ) : productsQuery.isError ? (
   462→          <div className="flex flex-col items-center justify-center gap-3 py-12 text-center">
   463→            <TriangleAlert className="size-10 text-destructive" aria-hidden />

   483→              </p>

   487→                <Plus /> New Product

   491→        ) : (

   494→            <div className="hidden md:block">
   495→              <div className="rms-scroll max-h-[520px] overflow-y-auto">
   496→                <Table>
   497→                  <TableHeader>

   504→                      <TableHead className="hidden text-right md:table-cell">Stock</TableHead>

   546→                        <TableCell className="hidden md:table-cell">

   555→                </Table>
   556→              </div>
   557→            </div>
   558→
   559→            {/* Mobile card list */}
   560→            <div className="space-y-2 md:hidden">
   561→              {products.map((p) => (
   562→                <div key={p.id} className="flex items-center gap-3 rounded-lg border p-3">

   566→                      {!p.active && <ActiveBadge active={false} />}
   567→                    </div>

   570→                        <Badge variant="outline" className="px-1.5">{p.category.name}</Badge>

   580→                    </div>
   581→                  </div>
   582→                  <div className="shrink-0 text-right font-medium tabular-nums">
   583→                    {formatCurrency(p.price)}
   584→                  </div>
   585→                  <ProductRowActions product={p} onEdit={openEdit} onDelete={setDeleteTarget} />
   586→                </div>
   587→              ))}
   588→            </div>
   589→          </>
   590→        )}
   591→      </Card>
   592→
   593→      {/* Create / edit dialog */}
   594→      <Dialog open={formOpen} onOpenChange={setFormOpen}>
   595→        <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-lg">
   596→          <DialogHeader>
   597→            <DialogTitle>{isCreate ? 'New Product' : 'Edit Product'}</DialogTitle>
   598→            <DialogDescription>
   599→              {isCreate
   600→                ? 'Add a menu item or ingredient to the catalog.'
   601→                : 'Update details. Stock levels are managed from Inventory.'}
   602→            </DialogDescription>
   603→          </DialogHeader>
   604→
   605→          <div className="grid gap-4">
   606→            <div className="grid gap-2">
   607→              <Label htmlFor="product-name">Name *</Label>
   608→              <Input
   609→                id="product-name"
   610→                value={form.name}
   611→                onChange={(e) => setForm({ ...form, name: e.target.value })}
   612→                placeholder="e.g. Koshary"
   613→                className="h-11"
   614→                aria-invalid={errors.name ? true : undefined}
   615→              />
   616→              <FieldError message={errors.name} />
   617→            </div>
   618→
   619→            <div className="grid gap-2">
   620→              <Label htmlFor="product-category">Category</Label>
   621→              <Select
   622→                value={form.categoryId === '' ? 'none' : form.categoryId}
   623→                onValueChange={(v) => setForm({ ...form, categoryId: v === 'none' ? '' : v })}
   624→              >
   625→                <SelectTrigger id="product-category" className="h-11 w-full">
   626→                  <SelectValue placeholder="None" />
   627→                </SelectTrigger>
   628→                <SelectContent>
   629→                  <SelectItem value="none">None</SelectItem>
   630→                  {categories.map((c) => (
   631→                    <SelectItem
   632→                      key={c.id}
   633→                      value={String(c.id)}
   634→                      className={c.active ? undefined : 'text-muted-foreground'}
   635→                    >
   636→                      {c.name}
   637→                      {c.active ? '' : ' (inactive)'}
   638→                    </SelectItem>
   639→                  ))}
   640→                </SelectContent>
   641→              </Select>
   642→            </div>
   643→
   644→            <div className="grid grid-cols-2 gap-4">
   645→              <div className="grid gap-2">
   646→                <Label htmlFor="product-price">Price *</Label>
   647→                <Input
   648→                  id="product-price"
   649→                  type="number"
   650→                  step={0.01}
   651→                  min={0}
   652→                  inputMode="decimal"
   653→                  value={form.price}
   654→                  onChange={(e) => setForm({ ...form, price: e.target.value })}
   655→                  placeholder="0.00"
   656→                  className="h-11"
   657→                  aria-invalid={errors.price ? true : undefined}
   658→                />
   659→                <FieldError message={errors.price} />
   660→              </div>
   661→              <div className="grid gap-2">
   662→                <Label htmlFor="product-cost">Cost</Label>
   663→                <Input
   664→                  id="product-cost"
   665→                  type="number"
   666→                  step={0.01}
   667→                  min={0}
   668→                  inputMode="decimal"
   669→                  value={form.cost}
   670→                  onChange={(e) => setForm({ ...form, cost: e.target.value })}
   671→                  placeholder="0.00"
   672→                  className="h-11"
   673→                  aria-invalid={errors.cost ? true : undefined}
   674→                />
   675→                <FieldError message={errors.cost} />
   676→              </div>
   677→            </div>
   678→
   679→            <SwitchRow
   680→              id="product-sellable"
   681→              label="Sellable"
   682→              helper="Appears in POS"
   683→              checked={form.isSellable}
   684→              onCheckedChange={(checked) => setForm({ ...form, isSellable: checked })}
   685→            />
   686→            <SwitchRow
   687→              id="product-stockable"
   688→              label="Stock tracking"
   689→              helper="Track inventory stock"
   690→              checked={form.isStockable}
   691→              onCheckedChange={(checked) => setForm({ ...form, isStockable: checked })}
   692→            />
   693→
   694→            <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   695→              <div className="grid gap-2">
   696→                <Label htmlFor="product-sku">SKU</Label>
   697→                <Input
   698→                  id="product-sku"
   699→                  value={form.sku}
   700→                  onChange={(e) => setForm({ ...form, sku: e.target.value })}
   701→                  placeholder="e.g. ING-014"
   702→                  className="h-11"
   703→                />
   704→              </div>
   705→              <div className="grid gap-2">
   706→                <Label htmlFor="product-image">Image URL</Label>
   707→                <Input
   708→                  id="product-image"
   709→                  type="url"
   710→                  value={form.imageUrl}
   711→                  onChange={(e) => setForm({ ...form, imageUrl: e.target.value })}
   712→                  placeholder="https://…"
   713→                  className="h-11"
   714→                />
   715→              </div>
   716→            </div>
   717→
   718→            {form.isStockable && (
   719→              <div className="grid grid-cols-2 gap-4">
   720→                <div className="grid gap-2">
   721→                  <Label htmlFor="product-threshold">Low stock alert at</Label>
   722→                  <Input
   723→                    id="product-threshold"
   724→                    type="number"
   725→                    step={0.01}
   726→                    min={0}
   727→                    inputMode="decimal"
   728→                    value={form.lowStockThreshold}
   729→                    onChange={(e) => setForm({ ...form, lowStockThreshold: e.target.value })}
   730→                    className="h-11"
   731→                    aria-invalid={errors.lowStockThreshold ? true : undefined}
   732→                  />
   733→                  <FieldError message={errors.lowStockThreshold} />
   734→                </div>
   735→                {isCreate && (
   736→                  <div className="grid gap-2">
   737→                    <Label htmlFor="product-initial-stock">Initial stock</Label>
   738→                    <Input
   739→                      id="product-initial-stock"
   740→                      type="number"
   741→                      step={0.01}
   742→                      min={0}
   743→                      inputMode="decimal"
   744→                      value={form.stock}
   745→                      onChange={(e) => setForm({ ...form, stock: e.target.value })}
   746→                      placeholder="0"
   747→                      className="h-11"
   748→                      aria-invalid={errors.stock ? true : undefined}
   749→                    />
   750→                    <p className="text-muted-foreground text-xs">Sets opening stock (logged as purchase)</p>
   751→                    <FieldError message={errors.stock} />
   752→                  </div>
   753→                )}
   754→              </div>
   755→            )}
   756→          </div>
   757→
   758→          <DialogFooter>
   759→            <Button
   760→              variant="outline"
   761→              className="h-11"
   762→              onClick={() => setFormOpen(false)}
   763→              disabled={saveMutation.isPending}
   764→            >
   765→              Cancel
   766→            </Button>
   767→            <Button
   768→              className="h-11"
   769→              disabled={saveMutation.isPending || hasErrors}
   770→              onClick={() => saveMutation.mutate()}
   771→            >
   772→              {saveMutation.isPending ? 'Saving…' : isCreate ? 'Create Product' : 'Save Changes'}
   773→            </Button>
   774→          </DialogFooter>
   775→        </DialogContent>
   776→      </Dialog>
   777→
   778→      {/* Delete confirm */}
   779→      <AlertDialog open={deleteTarget !== null} onOpenChange={(open) => !open && setDeleteTarget(null)}>
   780→        <AlertDialogContent>
   781→          <AlertDialogHeader>
   782→            <AlertDialogTitle>Deactivate product?</AlertDialogTitle>
   783→            <AlertDialogDescription>
   784→              “{deleteTarget?.name}” will disappear from POS. It stays in history and can be
   785→              re-enabled later from “Show inactive”.
   786→            </AlertDialogDescription>
   787→          </AlertDialogHeader>
   788→          <AlertDialogFooter>
   789→            <AlertDialogCancel className="h-11">Cancel</AlertDialogCancel>
   790→            <Button
   791→              variant="destructive"
   792→              className="h-11"
   793→              disabled={deleteMutation.isPending}
   794→              onClick={() => {
   795→                if (deleteTarget) deleteMutation.mutate(deleteTarget.id)
   796→              }}
   797→            >
   798→              {deleteMutation.isPending ? 'Deactivating…' : 'Deactivate'}
   799→            </Button>
   800→          </AlertDialogFooter>
   801→        </AlertDialogContent>
   802→      </AlertDialog>
   803→    </div>
   804→  )
   805→}
