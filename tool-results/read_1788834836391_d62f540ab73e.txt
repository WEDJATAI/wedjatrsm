    73→type ProductForm = {
    74→  name: string
    75→  nameAr: string // '' = no Arabic name (clears it)
    76→  categoryId: string // '' = no category
    77→  price: string
    78→  cost: string
    79→  isSellable: boolean

    82→  imageUrl: string
    83→  lowStockThreshold: string

    87→type ProductFormErrors = Partial<
    88→  Record<'name' | 'price' | 'cost' | 'lowStockThreshold' | 'stock', string>
    89→>
    90→
    91→const EMPTY_PRODUCT_FORM: ProductForm = {
    92→  name: '',
    93→  nameAr: '',
    94→  categoryId: '',
    95→  price: '',
    96→  cost: '',
    97→  isSellable: true,
    98→  isStockable: false,
    99→  sku: '',
   100→  imageUrl: '',
   101→  lowStockThreshold: '10',
   102→  stock: '',
   103→}
   104→
   105→function toProductForm(p: Product): ProductForm {
   106→  return {
   107→    name: p.name,
   108→    nameAr: p.nameAr ?? '',
   109→    categoryId: p.categoryId === null ? '' : String(p.categoryId),
   110→    price: String(p.price),
   111→    cost: String(p.cost),
   112→    isSellable: p.isSellable,
   113→    isStockable: p.isStockable,
   114→    sku: p.sku ?? '',
   115→    imageUrl: p.imageUrl ?? '',
   116→    lowStockThreshold: String(p.lowStockThreshold),
   117→    stock: String(p.stock),
   118→  }
   119→}
   120→
   121→function validateProductForm(
   122→  form: ProductForm,
   123→  isCreate: boolean,
   124→  t: (key: string) => string,
   125→): ProductFormErrors {
   126→  const errors: ProductFormErrors = {}
   127→  if (form.name.trim() === '') errors.name = t('admin.nameRequired')
   128→  const price = Number(form.price)
   129→  if (form.price.trim() === '' || !Number.isFinite(price) || price < 0) {
   130→    errors.price = t('admin.priceMin')
   131→  }
   132→  if (form.cost.trim() !== '') {
   133→    const cost = Number(form.cost)
   134→    if (!Number.isFinite(cost) || cost < 0) errors.cost = t('admin.costMin')
   135→  }
   136→  if (form.isStockable) {
   137→    if (form.lowStockThreshold.trim() !== '') {
   138→      const threshold = Number(form.lowStockThreshold)
   139→      if (!Number.isFinite(threshold) || threshold < 0) {
   140→        errors.lowStockThreshold = t('admin.mustBe0')
   141→      }
   142→    }
   143→    if (isCreate && form.stock.trim() !== '') {
   144→      const stock = Number(form.stock)
   145→      if (!Number.isFinite(stock) || stock < 0) errors.stock = t('admin.mustBe0')
   146→    }
   147→  }
   148→  return errors
   149→}

   182→function stockColorClass(product: Product): string | undefined {
   183→  if (!product.isStockable) return undefined
   184→  if (product.stock <= product.lowStockThreshold) {
   185→    return 'font-medium text-rose-600 dark:text-rose-400'
   186→  }
   187→  if (product.stock <= product.lowStockThreshold * 2) {
   188→    return 'font-medium text-amber-600 dark:text-amber-500'
   189→  }
   190→  return undefined
   191→}
   192→
   193→function MarginCell({ product }: { product: Product }) {
   194→  if (product.price <= 0) return <span className="text-muted-foreground">—</span>
   195→  const margin = ((product.price - product.cost) / product.price) * 100
   196→  const color =
   197→    margin > 40
   198→      ? 'text-emerald-600 dark:text-emerald-400'
   199→      : margin >= 20
   200→        ? 'text-amber-600 dark:text-amber-500'
   201→        : 'text-rose-600 dark:text-rose-400'
   202→  return <span className={cn('text-xs font-medium tabular-nums', color)}>{Math.round(margin)}%</span>
   203→}
   204→
   205→function StockCell({ product }: { product: Product }) {
   206→  if (!product.isStockable) return <span className="text-muted-foreground">—</span>
   207→  return <span className={cn('tabular-nums', stockColorClass(product))}>{formatQty(product.stock)}</span>
   208→}

   262→/** Product display name — Arabic primary when the UI is Arabic (falls back to English). */
   263→function ProductDisplayName({
   264→  product,
   265→  lang,
   266→  missingArLabel,
   267→}: {
   268→  product: Product
   269→  lang: 'en' | 'ar'
   270→  missingArLabel: string
   271→}) {
   272→  const localized = localizedName(product.name, product.nameAr, lang)
   273→  const missingAr = lang === 'ar' && !(product.nameAr ?? '').trim()

   275→    <div className="flex flex-wrap items-center gap-2">
   276→      <span className="font-medium">{localized}</span>
   277→      {missingAr && <MissingArBadge label={missingArLabel} />}
   278→    </div>
   279→  )

   319→export default function ProductsView() {
   320→  const { t, lang } = useI18n()
   321→  const queryClient = useQueryClient()
   322→
   323→  const [search, setSearch] = useState('')
   324→  const [categoryFilter, setCategoryFilter] = useState('all')
   325→  const [showInactive, setShowInactive] = useState(false)
   326→
   327→  const [formOpen, setFormOpen] = useState(false)
   328→  const [editing, setEditing] = useState<Product | null>(null)
   329→  const [form, setForm] = useState<ProductForm>(EMPTY_PRODUCT_FORM)
   330→
   331→  const [deleteTarget, setDeleteTarget] = useState<Product | null>(null)
   332→
   333→  const productsQuery = useQuery({
   334→    queryKey: ['products', { all: showInactive, category: categoryFilter }],
   335→    queryFn: () =>
   336→      fetcher<{ products: Product[] }>(
   337→        `/api/products?${new URLSearchParams({
   338→          ...(showInactive ? { all: '1' } : {}),
   339→          ...(categoryFilter !== 'all' ? { category: categoryFilter } : {}),
   340→        }).toString()}`,
   341→      ),
   342→  })
   343→
   344→  const categoriesQuery = useQuery({
   345→    queryKey: ['categories', 'all'],
   346→    queryFn: () => fetcher<{ categories: Category[] }>('/api/categories?all=1'),
   347→  })
   348→
   349→  const categories = useMemo(
   350→    () =>
   351→      [...(categoriesQuery.data?.categories ?? [])].sort(
   352→        (a, b) => a.displayOrder - b.displayOrder || a.name.localeCompare(b.name),
   353→      ),
   354→    [categoriesQuery.data],
   355→  )
   356→
   357→  const products = useMemo(() => {
   358→    const items = productsQuery.data?.products ?? []
   359→    const q = search.trim().toLowerCase()
   360→    const filtered =
   361→      q === ''
   362→        ? items
   363→        : items.filter(
   364→            (p) => p.name.toLowerCase().includes(q) || (p.sku ?? '').toLowerCase().includes(q),
   365→          )
   366→    return [...filtered].sort((a, b) => a.name.localeCompare(b.name))
   367→  }, [productsQuery.data, search])
   368→
   369→  const isCreate = editing === null
   370→  const errors = validateProductForm(form, isCreate, t)
   371→  const hasErrors = Object.values(errors).some(Boolean)
   372→
   373→  const saveMutation = useMutation({
   374→    mutationFn: async () => {
   375→      const payload: Record<string, unknown> = {
   376→        name: form.name.trim(),
   377→        nameAr: form.nameAr.trim(),
   378→        price: Number(form.price),
   379→        cost: form.cost.trim() === '' ? 0 : Number(form.cost),
   380→        isSellable: form.isSellable,
   381→        isStockable: form.isStockable,
   382→      }
   383→      if (form.categoryId !== '') payload.categoryId = Number(form.categoryId)
   384→      if (form.sku.trim() !== '') payload.sku = form.sku.trim()
   385→      if (form.imageUrl.trim() !== '') payload.imageUrl = form.imageUrl.trim()
   386→      if (form.isStockable) {
   387→        payload.lowStockThreshold =
   388→          form.lowStockThreshold.trim() === '' ? 0 : Number(form.lowStockThreshold)
   389→        if (editing === null) {
   390→          payload.stock = form.stock.trim() === '' ? 0 : Number(form.stock)
   391→        }
   392→      }
   393→      return editing === null
   394→        ? apiFetch<{ product: Product }>('/api/products', { method: 'POST', body: payload })
   395→        : apiFetch<{ product: Product }>(`/api/products/${editing.id}`, {
   396→            method: 'PUT',
   397→            body: payload,
   398→          })
   399→    },
   400→    onSuccess: () => {
   401→      void queryClient.invalidateQueries({ queryKey: ['products'] })
   402→      toast.success(editing === null ? t('admin.productCreated') : t('admin.productUpdated'))
   403→      setFormOpen(false)
   404→    },
   405→    onError: (err: Error) => toast.error(err.message),
   406→  })
   407→
   408→  const deleteMutation = useMutation({
   409→    mutationFn: (id: number) =>
   410→      apiFetch<{ ok: boolean }>(`/api/products/${id}`, { method: 'DELETE' }),
   411→    onSuccess: () => {
   412→      void queryClient.invalidateQueries({ queryKey: ['products'] })
   413→      toast.success(t('admin.productDeactivated'))
   414→      setDeleteTarget(null)
   415→    },
   416→    onError: (err: Error) => toast.error(err.message),
   417→  })
   418→
   419→  function openCreate() {
   420→    setEditing(null)
   421→    setForm(EMPTY_PRODUCT_FORM)
   422→    setFormOpen(true)
   423→  }
   424→
   425→  function openEdit(product: Product) {
   426→    setEditing(product)
   427→    setForm(toProductForm(product))
   428→    setFormOpen(true)
   429→  }
   430→
   431→  const hasActiveFilters = search.trim() !== '' || categoryFilter !== 'all'
   432→
   433→  return (
   434→    <div className="mx-auto w-full max-w-7xl space-y-6 p-4 sm:p-6">
   435→      {/* Header */}
   436→      <div className="flex flex-wrap items-start justify-between gap-4">
   437→        <div className="space-y-1">
   438→          <h1 className="text-2xl font-bold">{t('nav.products')}</h1>
   439→          <p className="text-muted-foreground text-sm">{t('admin.productsSubtitle')}</p>
   440→        </div>
   441→        <Button className="h-11" onClick={openCreate}>
   442→          <Plus /> {t('admin.newProduct')}
   443→        </Button>
   444→      </div>
   445→
   446→      {/* Toolbar */}
   447→      <Card className="p-3">
   448→        <div className="flex flex-wrap items-center gap-2">
   449→          <div className="relative w-full sm:w-72">
   450→            <Search
   451→              className="pointer-events-none absolute top-1/2 start-3 size-4 -translate-y-1/2 text-muted-foreground"
   452→              aria-hidden
   453→            />
   454→            <Input
   455→              value={search}
   456→              onChange={(e) => setSearch(e.target.value)}
   457→              placeholder={t('admin.searchProduct')}
   458→              className="h-11 ps-9"
   459→              aria-label={t('admin.searchProduct')}
   460→            />
   461→          </div>
   462→          <Select value={categoryFilter} onValueChange={setCategoryFilter}>
   463→            <SelectTrigger className="h-11 w-full sm:w-52" aria-label={t('admin.allCategories')}>
   464→              <SelectValue placeholder={t('admin.allCategories')} />
   465→            </SelectTrigger>
   466→            <SelectContent>
   467→              <SelectItem value="all">{t('admin.allCategories')}</SelectItem>
   468→              {categories.map((c) => (
   469→                <SelectItem
   470→                  key={c.id}
   471→                  value={String(c.id)}
   472→                  className={c.active ? undefined : 'text-muted-foreground'}
   473→                >
   474→                  {c.name}
   475→                  {c.active ? '' : ` ${t('admin.inactiveSuffix')}`}
   476→                </SelectItem>
   477→              ))}
   478→            </SelectContent>
   479→          </Select>
   480→          <div className="flex h-11 items-center gap-2 rounded-md border px-3">
   481→            <Switch
   482→              id="products-show-inactive"
   483→              checked={showInactive}
   484→              onCheckedChange={setShowInactive}
   485→            />
   486→            <Label htmlFor="products-show-inactive" className="cursor-pointer text-sm">
   487→              {t('admin.showInactive')}
   488→            </Label>
   489→          </div>
   490→        </div>
   491→      </Card>
   492→
   493→      {/* Content */}
   494→      <Card className="p-4">
   495→        {productsQuery.isLoading ? (
   496→          <div className="space-y-2">
   497→            {Array.from({ length: 8 }, (_, i) => (
   498→              <div key={i} className="flex items-center gap-4 rounded-lg border p-3">
   499→                <div className="flex-1 space-y-2">
   500→                  <Skeleton className="h-4 w-44" />
   501→                  <Skeleton className="h-3 w-24" />
   502→                </div>
   503→                <Skeleton className="h-4 w-16" />
   504→                <Skeleton className="size-11 rounded-md" />
   505→              </div>
   506→            ))}
   507→          </div>
   508→        ) : productsQuery.isError ? (
   509→          <div className="flex flex-col items-center justify-center gap-3 py-12 text-center">
   510→            <TriangleAlert className="size-10 text-destructive" aria-hidden />
   511→            <div>
   512→              <p className="font-medium">{t('admin.loadProductsFailed')}</p>
   513→              <p className="text-muted-foreground text-sm">
   514→                {productsQuery.error?.message ?? t('common.error')}
   515→              </p>
   516→            </div>
   517→            <Button variant="outline" className="h-11" onClick={() => void productsQuery.refetch()}>
   518→              {t('common.retry')}
   519→            </Button>
   520→          </div>
   521→        ) : products.length === 0 ? (
   522→          <div className="flex flex-col items-center justify-center gap-3 py-12 text-center">
   523→            <PackageSearch className="size-10 text-muted-foreground/50" aria-hidden />
   524→            <div>
   525→              <p className="font-medium">{t('admin.noProducts')}</p>
   526→              <p className="text-muted-foreground text-sm">
   527→                {hasActiveFilters ? t('admin.noProductsFilters') : t('admin.noProductsCreate')}
   528→              </p>
   529→            </div>
   530→            {!hasActiveFilters && (
   531→              <Button className="h-11" onClick={openCreate}>
   532→                <Plus /> {t('admin.newProduct')}
   533→              </Button>
   534→            )}
   535→          </div>
   536→        ) : (
   537→          <>
   538→            {/* Desktop / tablet table */}
   539→            <div className="hidden md:block">
   540→              <div className="rms-scroll max-h-[520px] overflow-y-auto">
   541→                <Table>
   542→                  <TableHeader>
   543→                    <TableRow>
   544→                      <TableHead>{t('admin.product')}</TableHead>
   545→                      <TableHead className="hidden md:table-cell">{t('admin.category')}</TableHead>
   546→                      <TableHead className="text-end">{t('common.price')}</TableHead>
   547→                      <TableHead className="hidden text-end lg:table-cell">{t('common.cost')}</TableHead>
   548→                      <TableHead className="hidden lg:table-cell">{t('admin.margin')}</TableHead>
   549→                      <TableHead className="hidden text-end md:table-cell">{t('admin.colStock')}</TableHead>
   550→                      <TableHead className="hidden lg:table-cell">{t('admin.colFlags')}</TableHead>
   551→                      <TableHead className="hidden md:table-cell">{t('common.active')}</TableHead>
   552→                      <TableHead className="w-12 text-end">
   553→                        <span className="sr-only">{t('common.actions')}</span>
   554→                      </TableHead>
   555→                    </TableRow>
   556→                  </TableHeader>
   557→                  <TableBody>
   558→                    {products.map((p) => (
   559→                      <TableRow key={p.id} className="h-14">
   560→                        <TableCell>
   561→                          <ProductDisplayName product={p} lang={lang} missingArLabel={t('admin.missingAr')} />
   562→                          {lang === 'ar' && p.name !== localizedName(p.name, p.nameAr, lang) ? (
   563→                            <div className="text-muted-foreground text-xs">{p.name}</div>
   564→                          ) : null}
   565→                          {p.sku ? (
   566→                            <div className="font-mono text-muted-foreground text-xs">{p.sku}</div>
   567→                          ) : null}
   568→                        </TableCell>
   569→                        <TableCell className="hidden md:table-cell">
   570→                          {p.category ? (
   571→                            <Badge variant="outline">{p.category.name}</Badge>
   572→                          ) : (
   573→                            <span className="text-muted-foreground">—</span>
   574→                          )}
