    47→type DragState = {
    48→  table: RestaurantTable

    50→  startX: number

    52→  /** Offset (in canvas %) between the grab point and the table center. */

    60→const TABLE_STYLES: Record<string, { surface: string; dot: string }> = {
    61→  free: { surface: 'border-emerald-400 bg-emerald-50 text-emerald-900', dot: 'bg-emerald-500' },
    62→  occupied: { surface: 'border-amber-400 bg-amber-100 text-amber-950', dot: 'bg-amber-500' },
    63→  reserved: { surface: 'border-rose-400 bg-rose-50 text-rose-900', dot: 'bg-rose-500' },
    64→}
    65→
    66→/** Canvas tile sizing per table shape (6-c spec): round → circle, etc. */
    67→const SHAPE_TILE_CLASSES: Record<string, string> = {
    68→  square: 'h-20 w-24 rounded-xl',
    69→  round: 'h-24 w-24 rounded-full',
    70→  rectangle: 'h-24 w-32 rounded-xl',
    71→  oval: 'h-24 w-32 rounded-full',
    72→}
    73→
    74→function shapeTileClasses(shape: string): string {
    75→  return SHAPE_TILE_CLASSES[shape] ?? SHAPE_TILE_CLASSES.square
    76→}
    77→
    78→/** Tiny shape glyph used next to each Select option. */
    79→function ShapeGlyph({ shape, className }: { shape: string; className?: string }) {
    80→  const cls =
    81→    shape === 'round'
    82→      ? 'size-3.5 rounded-full'
    83→      : shape === 'oval'
    84→        ? 'h-3 w-5 rounded-full'
    85→        : shape === 'rectangle'
    86→          ? 'h-3 w-5 rounded-[3px]'
    87→          : 'size-3.5 rounded-[3px]'
    88→  return <span className={cn('inline-block shrink-0 border-2 border-current', cls, className)} aria-hidden />
    89→}
    90→
    91→const clampPercent = (value: number, min: number, max: number) =>
    92→  Math.min(max, Math.max(min, value))
    93→const round1 = (value: number) => Math.round(value * 10) / 10
    94→const toNumber = (value: string, fallback: number) => {
    95→  const parsed = Number(value)
    96→  return Number.isFinite(parsed) ? parsed : fallback
    97→}

    99→/** Floor plan manager — list of plans + drag-to-arrange table canvas. */

   104→  const dragRef = useRef<DragState | null>(null)
   105→  const [drag, setDrag] = useState<DragState | null>(null)

   107→  const [selectedPlanId, setSelectedPlanId] = useState<number | null>(null)
   108→
   109→  const [newPlanOpen, setNewPlanOpen] = useState(false)
   110→  const [newPlanName, setNewPlanName] = useState('')
   111→
   112→  const [addTableOpen, setAddTableOpen] = useState(false)
   113→  const [addTableName, setAddTableName] = useState('')
   114→  const [addTableCapacity, setAddTableCapacity] = useState('2')
   115→  const [addTableShape, setAddTableShape] = useState('square')
   116→
   117→  const [editOpen, setEditOpen] = useState(false)
   118→  const [editingTable, setEditingTable] = useState<RestaurantTable | null>(null)
   119→  const [editName, setEditName] = useState('')
   120→  const [editCapacity, setEditCapacity] = useState('2')
   121→  const [editShape, setEditShape] = useState('square')
   122→  const [editX, setEditX] = useState('50')
   123→  const [editY, setEditY] = useState('50')
   124→  const [editStatus, setEditStatus] = useState('free')

   126→  const { data, isLoading, isError, error } = useQuery({
   127→    queryKey: ['floorplans'],

   131→

   133→  const selectedPlan = floorPlans.find((plan) => plan.id === selectedPlanId) ?? floorPlans[0] ?? null
   134→  const activeTables = selectedPlan
   135→    ? selectedPlan.tables.filter((table) => table.active)

   137→
   138→  // ── Mutations ────────────────────────────────────────────────────────
   139→
   140→  const createPlanMutation = useMutation({
   141→    mutationFn: (name: string) =>
   142→      apiFetch<{ floorPlan: FloorPlan }>('/api/floorplans', { method: 'POST', body: { name } }),
   143→    onSuccess: ({ floorPlan }) => {
   144→      toast.success(t('admin.floorPlanCreated', { name: floorPlan.name }))
   145→      void queryClient.invalidateQueries({ queryKey: ['floorplans'] })
   146→      setSelectedPlanId(floorPlan.id)
   147→      setNewPlanOpen(false)
   148→      setNewPlanName('')

   150→    onError: (mutationError) =>
   151→      toast.error(mutationError.message || t('error.generic')),
   152→  })
   153→
   154→  const togglePlanMutation = useMutation({
   155→    mutationFn: ({ id, active }: { id: number; active: boolean }) =>
   156→      apiFetch(`/api/floorplans/${id}`, { method: 'PUT', body: { active } }),
   157→    onMutate: async ({ id, active }) => {
   158→      await queryClient.cancelQueries({ queryKey: ['floorplans'] })
   159→      const previous = queryClient.getQueryData<{ floorPlans: FloorPlan[] }>(['floorplans'])
   160→      if (previous) {
   161→        queryClient.setQueryData(['floorplans'], {
   162→          floorPlans: previous.floorPlans.map((plan) =>
   163→            plan.id === id ? { ...plan, active } : plan,
   164→          ),
   165→        })
   166→      }
   167→      return { previous }
   168→    },
   169→    onError: (mutationError, _variables, context) => {
   170→      toast.error(mutationError.message || t('error.generic'))
   171→      if (context?.previous) {
   172→        queryClient.setQueryData(['floorplans'], context.previous)
   173→      }
   174→    },
   175→    onSettled: () => {
   176→      void queryClient.invalidateQueries({ queryKey: ['floorplans'] })
   177→    },
   178→  })
   179→
   180→  const addTableMutation = useMutation({
   181→    mutationFn: (input: {
   182→      floorPlanId: number
   183→      name: string
   184→      capacity: number
   185→      shape: string
   186→      positionX: number
   187→      positionY: number
   188→    }) => apiFetch<{ table: RestaurantTable }>('/api/tables', { method: 'POST', body: input }),
   189→    onSuccess: ({ table }) => {
   190→      toast.success(t('admin.tableAdded', { name: table.name }))
   191→      void queryClient.invalidateQueries({ queryKey: ['floorplans'] })
   192→      setAddTableOpen(false)
   193→      setAddTableName('')
   194→      setAddTableCapacity('2')
   195→      setAddTableShape('square')
   196→    },
   197→    onError: (mutationError) =>
   198→      toast.error(mutationError.message || t('error.generic')),
   199→  })
   200→
   201→  const updateTableMutation = useMutation({
   202→    mutationFn: ({
   203→      id,
   204→      body,
   205→    }: {
   206→      id: number
   207→      body: Record<string, unknown>
   208→      successMessage?: string
   209→      closeDialog?: boolean
   210→    }) => apiFetch<{ table: RestaurantTable }>(`/api/tables/${id}`, { method: 'PUT', body }),
   211→    onSuccess: (_result, variables) => {
   212→      toast.success(variables.successMessage ?? t('admin.tableUpdated'))
   213→      void queryClient.invalidateQueries({ queryKey: ['floorplans'] })
   214→      if (variables.closeDialog) {
   215→        setEditOpen(false)
   216→        setEditingTable(null)

   219→    onError: (mutationError) =>
   220→      toast.error(mutationError.message || t('error.generic')),
   221→  })

   223→  // ── Table edit dialog ────────────────────────────────────────────────
   224→
   225→  function openTableDialog(table: RestaurantTable) {
   226→    setEditName(table.name)
   227→    setEditCapacity(String(table.capacity))
   228→    setEditShape(table.shape ?? 'square')
   229→    setEditX(String(round1(table.positionX)))
   230→    setEditY(String(round1(table.positionY)))
   231→    setEditStatus(table.status)
   232→    setEditingTable(table)
   233→    setEditOpen(true)
   234→  }

   236→  // ── Pointer-based dragging (mouse + touch) ───────────────────────────
   237→
   238→  function beginDrag(event: React.PointerEvent<HTMLDivElement>, table: RestaurantTable) {
   239→    if (event.pointerType === 'mouse' && event.button !== 0) return
   240→    const rect = canvasRef.current?.getBoundingClientRect()
   241→    if (!rect) return
   242→    event.preventDefault()
   243→    try {
   244→      event.currentTarget.setPointerCapture(event.pointerId)
   245→    } catch {
   246→      // Pointer already released — ignore.
   247→    }
   248→    const pointerX = ((event.clientX - rect.left) / rect.width) * 100
   249→    const pointerY = ((event.clientY - rect.top) / rect.height) * 100
   250→    const state: DragState = {
   251→      table,
   252→      pointerId: event.pointerId,
   253→      startX: event.clientX,
   254→      startY: event.clientY,
   255→      offX: pointerX - table.positionX,
   256→      offY: pointerY - table.positionY,
   257→      x: table.positionX,
   258→      y: table.positionY,
   259→      movedPx: 0,
   260→    }
   261→    dragRef.current = state
   262→    setDrag(state)
   263→  }
   264→
   265→  function moveDrag(event: React.PointerEvent<HTMLDivElement>) {
   266→    const current = dragRef.current
   267→    if (!current || current.pointerId !== event.pointerId) return
   268→    const rect = canvasRef.current?.getBoundingClientRect()
   269→    if (!rect) return
   270→    const pointerX = ((event.clientX - rect.left) / rect.width) * 100
   271→    const pointerY = ((event.clientY - rect.top) / rect.height) * 100
   272→    const next: DragState = {
   273→      ...current,
   274→      x: clampPercent(pointerX - current.offX, 2, 98),
   275→      y: clampPercent(pointerY - current.offY, 2, 98),
   276→      movedPx: Math.max(
   277→        current.movedPx,
   278→        Math.hypot(event.clientX - current.startX, event.clientY - current.startY),
   279→      ),
   280→    }
   281→    dragRef.current = next
   282→    setDrag(next)
   283→  }
   284→
   285→  function endDrag(event: React.PointerEvent<HTMLDivElement>, commit: boolean) {
   286→    const current = dragRef.current
   287→    if (!current || current.pointerId !== event.pointerId) return
   288→    dragRef.current = null
   289→    setDrag(null)
   290→    if (!commit) return
   291→    if (current.movedPx < 6) {
   292→      // Treated as a click → open the table editor.
   293→      openTableDialog(current.table)
   294→      return
   295→    }
   296→    updateTableMutation.mutate({
   297→      id: current.table.id,
   298→      body: { positionX: round1(current.x), positionY: round1(current.y) },
   299→      successMessage: t('admin.positionSaved'),
   300→    })
   301→  }
   302→
   303→  // ── Form submit handlers ─────────────────────────────────────────────
   304→
   305→  function handleCreatePlan() {
   306→    const name = newPlanName.trim()
   307→    if (!name) {
   308→      toast.error(t('admin.floorPlanNameRequired'))
   309→      return
   310→    }
   311→    createPlanMutation.mutate(name)
   312→  }
   313→
   314→  function handleAddTable() {
   315→    if (!selectedPlan) return
   316→    const name = addTableName.trim()
   317→    if (!name) {
   318→      toast.error(t('admin.tableNameRequired'))
   319→      return
   320→    }
   321→    const parsedCapacity = Number(addTableCapacity)
   322→    const capacity = Math.max(
   323→      1,
   324→      Math.round(Number.isFinite(parsedCapacity) && parsedCapacity > 0 ? parsedCapacity : 2),
   325→    )
   326→    // Spawn at a center-ish random spot (30-70%).
   327→    const positionX = round1(30 + Math.random() * 40)
   328→    const positionY = round1(30 + Math.random() * 40)
   329→    addTableMutation.mutate({
   330→      floorPlanId: selectedPlan.id,
   331→      name,
   332→      capacity,
   333→      shape: addTableShape,
   334→      positionX,
   335→      positionY,
   336→    })
   337→  }
   338→
   339→  function handleSaveTable() {
   340→    if (!editingTable) return
   341→    const name = editName.trim()
   342→    if (!name) {
   343→      toast.error(t('admin.tableNameRequired'))
   344→      return
   345→    }
   346→    const parsedCapacity = Number(editCapacity)
   347→    const capacity = Math.max(
   348→      1,
   349→      Math.round(
   350→        Number.isFinite(parsedCapacity) && parsedCapacity > 0 ? parsedCapacity : editingTable.capacity,
   351→      ),
   352→    )
   353→    const body: Record<string, unknown> = {
   354→      name,
   355→      capacity,
   356→      shape: editShape,
   357→      positionX: clampPercent(round1(toNumber(editX, editingTable.positionX)), 0, 100),
   358→      positionY: clampPercent(round1(toNumber(editY, editingTable.positionY)), 0, 100),
   359→    }
   360→    // 'occupied' is managed automatically by open orders — never send it.
   361→    if (editStatus === 'free' || editStatus === 'reserved') {
   362→      body.status = editStatus
   363→    }
   364→    updateTableMutation.mutate({
   365→      id: editingTable.id,
   366→      body,
   367→      successMessage: t('admin.tableUpdated'),
   368→      closeDialog: true,
   369→    })
   370→  }
   371→
   372→  function handleRemoveTable() {
   373→    if (!editingTable) return
   374→    updateTableMutation.mutate({
   375→      id: editingTable.id,
   376→      body: { active: false },
   377→      successMessage: t('admin.tableRemoved'),
   378→      closeDialog: true,
   379→    })
   380→  }
   381→
   382→  const creatingPlan = createPlanMutation.isPending
   383→  const addingTable = addTableMutation.isPending
   384→  const savingTable = updateTableMutation.isPending
   385→
   386→  return (
   387→    <div className="mx-auto max-w-7xl space-y-6 p-4 sm:p-6">
   388→      {/* Header */}
   389→      <div className="flex flex-wrap items-center justify-between gap-3">
   390→        <div>
   391→          <h1 className="text-2xl font-bold tracking-tight">{t('nav.floorplans')}</h1>
   392→          <p className="text-sm text-muted-foreground">{t('admin.floorplansSubtitle')}</p>
   393→        </div>
   394→        <Dialog open={newPlanOpen} onOpenChange={setNewPlanOpen}>
   395→          <DialogTrigger asChild>
   396→            <Button size="lg">
   397→              <Plus className="size-4" />
   398→              {t('admin.newFloorPlan')}
   399→            </Button>
   400→          </DialogTrigger>
   401→          <DialogContent className="sm:max-w-sm">
   402→            <DialogHeader>
   403→              <DialogTitle>{t('admin.newFloorPlanTitle')}</DialogTitle>
   404→              <DialogDescription>{t('admin.newFloorPlanDesc')}</DialogDescription>
   405→            </DialogHeader>
   406→            <div className="grid gap-1.5">
   407→              <Label htmlFor="plan-name">{t('common.name')}</Label>
   408→              <Input
   409→                id="plan-name"
   410→                value={newPlanName}
   411→                placeholder={t('admin.planNamePlaceholder')}
   412→                onChange={(event) => setNewPlanName(event.target.value)}
   413→                onKeyDown={(event) => {
   414→                  if (event.key === 'Enter') handleCreatePlan()
   415→                }}
   416→              />
   417→            </div>
   418→            <DialogFooter>
   419→              <Button variant="outline" onClick={() => setNewPlanOpen(false)}>
   420→                {t('common.cancel')}
   421→              </Button>
   422→              <Button onClick={handleCreatePlan} disabled={creatingPlan || !newPlanName.trim()}>
   423→                {creatingPlan ? <Loader2 className="size-4 animate-spin" /> : <Plus className="size-4" />}
   424→                {t('admin.create')}
   425→              </Button>
   426→            </DialogFooter>
   427→          </DialogContent>
   428→        </Dialog>
   429→      </div>
   430→
   431→      {isError && (
   432→        <div className="rounded-lg border border-destructive/40 bg-destructive/10 px-4 py-3 text-sm text-destructive">
   433→          {error instanceof Error ? error.message : t('admin.loadFloorplansFailed')}
   434→        </div>
   435→      )}
   436→
   437→      {isLoading ? (
   438→        <div className="grid gap-6 lg:grid-cols-[280px_1fr]">
   439→          <div className="space-y-2">
   440→            {Array.from({ length: 4 }).map((_, index) => (
   441→              <Skeleton key={index} className="h-16 w-full" />
   442→            ))}
   443→          </div>
   444→          <div className="space-y-3">
   445→            <Skeleton className="h-8 w-48" />
   446→            <Skeleton className="h-[460px] w-full rounded-xl" />
   447→          </div>
   448→        </div>
   449→      ) : floorPlans.length === 0 ? (
   450→        <Card className="flex flex-col items-center justify-center gap-3 border-dashed p-12 py-20 text-center">
   451→          <Map className="size-14 text-muted-foreground/40" />
   452→          <div className="space-y-1">
   453→            <p className="font-semibold">{t('admin.noFloorPlans')}</p>
   454→            <p className="text-sm text-muted-foreground">{t('admin.createFirstPlan')}</p>
   455→          </div>
   456→          <Button onClick={() => setNewPlanOpen(true)}>
   457→            <Plus className="size-4" />
   458→            {t('admin.newFloorPlan')}
   459→          </Button>
   460→        </Card>
   461→      ) : (
   462→        <div className="grid gap-6 lg:grid-cols-[280px_1fr]">
   463→          {/* Floor plan list */}
   464→          <div className="space-y-2 self-start">
   465→            <p className="px-1 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   466→              {t('admin.diningAreas')}
   467→            </p>
   468→            {floorPlans.map((plan) => {
   469→              const activeTableCount = plan.tables.filter((table) => table.active).length
   470→              const isSelected = selectedPlan !== null && plan.id === selectedPlan.id
   471→              return (
   472→                <Card
   473→                  key={plan.id}
   474→                  role="button"
   475→                  tabIndex={0}
   476→                  onClick={() => setSelectedPlanId(plan.id)}
   477→                  onKeyDown={(event) => {
   478→                    if (event.key === 'Enter' || event.key === ' ') {
   479→                      event.preventDefault()
   480→                      setSelectedPlanId(plan.id)
   481→                    }
   482→                  }}
   483→                  className={cn(
   484→                    'cursor-pointer p-3 transition-colors hover:bg-accent/40',
   485→                    isSelected && 'border-primary bg-primary/5',
   486→                    !plan.active && 'opacity-60',
   487→                  )}
   488→                >
   489→                  <div className="flex items-center justify-between gap-3">
   490→                    <div className="min-w-0">
   491→                      <p className="truncate text-sm font-semibold leading-tight">{plan.name}</p>
   492→                      <p className="mt-0.5 text-xs text-muted-foreground">
   493→                        {t('admin.tableCount', { count: activeTableCount })}
   494→                      </p>
   495→                    </div>
   496→                    <div className="flex shrink-0 items-center" onClick={(e) => e.stopPropagation()}>
   497→                      <Switch
   498→                        checked={plan.active}
   499→                        onCheckedChange={(checked) =>
   500→                          togglePlanMutation.mutate({ id: plan.id, active: checked })
   501→                        }
   502→                        aria-label={`${t('common.active')}: ${plan.name}`}
   503→                      />
   504→                    </div>
   505→                  </div>
   506→                </Card>
   507→              )
   508→            })}
   509→          </div>
   510→
   511→          {/* Canvas editor */}
   512→          {selectedPlan && (
   513→            <Card className="gap-4 p-4">
   514→              <div className="flex flex-wrap items-center justify-between gap-2">
   515→                <div className="min-w-0">
   516→                  <h2 className="truncate text-lg font-bold leading-tight">{selectedPlan.name}</h2>
   517→                  <p className="mt-0.5 flex items-center gap-1 text-xs text-muted-foreground">
   518→                    <MapPin className="size-3.5 shrink-0" />
   519→                    {t('admin.dragHint')}
   520→                  </p>
   521→                </div>
   522→                <Dialog open={addTableOpen} onOpenChange={setAddTableOpen}>
   523→                  <DialogTrigger asChild>
   524→                    <Button variant="outline">
   525→                      <Plus className="size-4" />
   526→                      {t('admin.addTable')}
   527→                    </Button>
   528→                  </DialogTrigger>
   529→                  <DialogContent className="sm:max-w-sm">
   530→                    <DialogHeader>
   531→                      <DialogTitle>{t('admin.addTableTitle')}</DialogTitle>
   532→                      <DialogDescription>{t('admin.addTableDesc')}</DialogDescription>
   533→                    </DialogHeader>
   534→                    <div className="grid gap-4">
   535→                      <div className="grid gap-1.5">
   536→                        <Label htmlFor="table-name">{t('common.name')}</Label>
   537→                        <Input
   538→                          id="table-name"
   539→                          value={addTableName}
   540→                          placeholder={t('admin.tableNamePlaceholder')}
   541→                          onChange={(event) => setAddTableName(event.target.value)}
   542→                          onKeyDown={(event) => {
   543→                            if (event.key === 'Enter') handleAddTable()
   544→                          }}
   545→                        />
   546→                      </div>
