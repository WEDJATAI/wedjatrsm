    60→type TableSelectProps = {
    61→  onSelectTable: (table: RestaurantTable) => void
    62→  onTakeaway: () => void
    63→  /** R11: start a table-less DELIVERY draft with the customer contact. */
    64→  onDelivery: (phone: string, address: string) => void
    65→  /** Open a table-less open order (takeaway OR delivery chip). */
    66→  onOpenTablelessOrder: (order: Order) => void
    67→  /** When set, start directly in transfer step 2 (destination selection) for this order. */
    68→  transferOrderId?: number | null
    69→  /** Called after a transfer completes (pos-view resets its order screen state). */
    70→  onTransferDone?: () => void
    71→  /** Seat Party tool result — the free tables selected for the party, in
    72→   *  selection order (primary table first). TableSelect exits the tool itself. */
    73→  onSeatParty: (tables: RestaurantTable[]) => void
    74→  /** Deferred-check chip clicked — pos-view opens the payment modal to settle it. */
    75→  onSettleDeferred: (order: Order) => void
    76→  /** "My shift" closeout — pos-view opens the per-server shift sheet (R8). */
    77→  onMyShift?: () => void
    78→}
    79→
    80→type ToolKind = 'transfer' | 'merge' | 'seat'
    81→type SourcePick = { orderId: number }
    82→
    83→type TableInteraction = 'normal' | 'eligible' | 'ineligible'
    84→
    85→/** Per-shape tile silhouette classes — Odoo 17-style tiles. The floor grid
    86→ *  keeps its responsive layout (tiles fill the grid cell width); only the
    87→ *  silhouette changes (square/round/rectangle/oval). The admin floor-plan
    88→ *  editor mirrors these exact strings. */
    89→const SHAPE_TILE_CLASSES: Record<string, string> = {
    90→  square: 'aspect-square rounded-2xl',
    91→  round: 'aspect-square rounded-full',
    92→  rectangle: 'h-24 rounded-2xl',
    93→  oval: 'h-24 rounded-full',
    94→}
    95→
    96→function shapeClasses(shape: string): string {
    97→  return SHAPE_TILE_CLASSES[shape] ?? SHAPE_TILE_CLASSES.square
    98→}

   100→export default function TableSelect({
   101→  onSelectTable,
   102→  onTakeaway,
   103→  onDelivery,
   104→  onOpenTablelessOrder,
   105→  transferOrderId,
   106→  onTransferDone,
   107→  onSeatParty,
   108→  onSettleDeferred,
   109→  onMyShift,
   110→}: TableSelectProps) {
   111→  const queryClient = useQueryClient()
   112→  const { t } = useI18n()
   113→
   114→  const [floorIdx, setFloorIdx] = useState(0)
   115→  // When pos-view passes transferOrderId (Transfer clicked on the order
   116→  // screen), this component mounts fresh in transfer step 2 (destination
   117→  // selection) — pos-view only sets it right before switching to tables view,
   118→  // so mount-time initialization is sufficient.
   119→  const [tool, setTool] = useState<ToolKind | null>(transferOrderId != null ? 'transfer' : null)
   120→  const [source, setSource] = useState<SourcePick | null>(
   121→    transferOrderId != null ? { orderId: transferOrderId } : null,
   122→  )
   123→  // Seat Party: the free tables tapped so far, in selection order (the first
   124→  // one becomes the primary table of the merged seating).
   125→  const [seatSelection, setSeatSelection] = useState<RestaurantTable[]>([])
   126→  // Paid/deferred tile tap-to-clear confirmation dialog target.
   127→  const [clearTarget, setClearTarget] = useState<RestaurantTable | null>(null)
   128→  // R9: pending AI guest-move suggestion opened for quick review.
   129→  const [aiMove, setAiMove] = useState<MovementCandidateDTO | null>(null)
   130→
   131→  // R11: new-delivery dialog — customer phone (required) + address.
   132→  const [deliveryOpen, setDeliveryOpen] = useState(false)
   133→  const [deliveryPhone, setDeliveryPhone] = useState('')
   134→  const [deliveryAddress, setDeliveryAddress] = useState('')
   135→
   136→  const deliveryPhoneOk = deliveryPhone.trim().length >= 5 && deliveryPhone.trim().length <= 20
   137→
   138→  const { data: floorPlanData, isLoading } = useQuery({
   139→    queryKey: ['floorplans'],
   140→    queryFn: () => fetcher<{ floorPlans: FloorPlan[] }>('/api/floorplans'),
   141→    refetchInterval: 3000, // LIVE table statuses
   142→  })
   143→
   144→  const { data: openOrdersData } = useQuery({
   145→    queryKey: ['orders', 'open'],
   146→    queryFn: () => fetcher<{ orders: Order[] }>('/api/orders?status=open'),
   147→    refetchInterval: 3000,
   148→  })
   149→
   150→  // Outstanding deferred checks (client left, payment pending) — settle chips.
   151→  const { data: deferredOrdersData } = useQuery({
   152→    queryKey: ['orders', 'deferred'],
   153→    queryFn: () => fetcher<{ orders: Order[] }>('/api/orders?status=deferred'),
   154→    refetchInterval: 3000,
   155→    enabled: !tool,
   156→  })
   157→
   158→  // R9: pending AI guest-move suggestions (POS floor chips, amber). Hidden
   159→  // during transfer/merge/seat tools to avoid conflicting interactions —
   160→  // poll errors stay silent (the chips just disappear until data returns).
   161→  const { data: aiMovementsData } = useQuery({
   162→    queryKey: ['vision-movements', 'pending'],
   163→    queryFn: () =>
   164→      fetcher<{ movements: MovementCandidateDTO[] }>('/api/vision/movements?status=pending'),
   165→    refetchInterval: 15000,
   166→    enabled: !tool,
   167→  })
   168→  const aiMovements = aiMovementsData?.movements ?? []
   169→
   170→  const floorPlans = floorPlanData?.floorPlans ?? []
   171→  const openOrders = openOrdersData?.orders ?? []
   172→  const deferredOrders = deferredOrdersData?.orders ?? []
   173→  const takeawayOrders = useMemo(
   174→    () => openOrders.filter((o) => o.tableId === null && o.orderType !== 'delivery'),
   175→    [openOrders],
   176→  )
   177→  // R11: open delivery orders get their own chips (contact + amount).
   178→  const deliveryOrders = useMemo(
   179→    () => openOrders.filter((o) => o.tableId === null && o.orderType === 'delivery'),
   180→    [openOrders],
   181→  )
   182→
   183→  const idx = floorPlans.length > 0 ? Math.min(floorIdx, floorPlans.length - 1) : 0
   184→  const plan = floorPlans[idx] ?? null
   185→
   186→  const sourceOrder = source ? openOrders.find((o) => o.id === source.orderId) ?? null : null
   187→  const sourceLabel = source ? sourceOrder?.table?.name ?? t('common.takeaway') : null
   188→
   189→  // ── Manual table turnover (two clicks): paid/deferred → bus → amber
   190→  // 'dirty' (needs cleaning) → clean → free. Audit-logged server-side.
   191→  const clearTableMutation = useMutation({
   192→    mutationFn: (vars: { id: number; name: string; phase: 'bus' | 'clean' }) =>
   193→      apiFetch<{ table: RestaurantTable }>(`/api/tables/${vars.id}/clear`, {
   194→        method: 'POST',
   195→        body: { action: vars.phase },
   196→      }),
   197→    onSuccess: async (_data, vars) => {
   198→      toast.success(
   199→        vars.phase === 'bus'
   200→          ? t('pos.tableBussedToast', { table: vars.name })
   201→          : t('pos.tableCleanedToast', { table: vars.name }),
   202→      )
   203→      setClearTarget(null)
   204→      await Promise.all([
   205→        queryClient.invalidateQueries({ queryKey: ['floorplans'] }),
   206→        queryClient.invalidateQueries({ queryKey: ['tables-status'] }),
   207→        queryClient.invalidateQueries({ queryKey: ['orders'] }),
   208→      ])
   209→    },
   210→    onError: (err: Error) => toast.error(err.message),
   211→  })
   212→
   213→  // ── Transfer & merge mutations ───────────────────────────────────
   214→  const invalidateAfterMove = async (...orderIds: number[]) => {
   215→    await Promise.all([
   216→      queryClient.invalidateQueries({ queryKey: ['floorplans'] }),
   217→      queryClient.invalidateQueries({ queryKey: ['orders'] }),
   218→      queryClient.invalidateQueries({ queryKey: ['orders', 'open'] }),
   219→      queryClient.invalidateQueries({ queryKey: ['tables-status'] }),
   220→      ...orderIds.map((id) => queryClient.invalidateQueries({ queryKey: ['pos-order', id] })),
   221→    ])
   222→  }
   223→
   224→  const transferMutation = useMutation({
   225→    mutationFn: (vars: { orderId: number; tableId: number; tableName: string }) =>
   226→      apiFetch<{ order: Order }>(`/api/orders/${vars.orderId}/transfer`, {
   227→        method: 'POST',
   228→        body: { tableId: vars.tableId },
   229→      }),
   230→    onSuccess: async (_data, vars) => {
   231→      toast.success(t('pos.transferToast', { order: vars.orderId, table: vars.tableName }))
   232→      await invalidateAfterMove(vars.orderId)
   233→      exitTool()
   234→      onTransferDone?.()
   235→    },
   236→    onError: (err: Error) => toast.error(err.message),
   237→  })
   238→
   239→  const mergeMutation = useMutation({
   240→    mutationFn: (vars: { sourceId: number; targetId: number; targetLabel: string }) =>
   241→      apiFetch<{ order: Order }>(`/api/orders/${vars.targetId}/merge`, {
   242→        method: 'POST',
   243→        body: { sourceOrderId: vars.sourceId },
   244→      }),
   245→    onSuccess: async (_data, vars) => {
   246→      toast.success(t('pos.mergeToast', { table: vars.targetLabel }))
   247→      await invalidateAfterMove(vars.sourceId, vars.targetId)
   248→      exitTool()
   249→    },
   250→    onError: (err: Error) => toast.error(err.message),
   251→  })
   252→
   253→  const busy = transferMutation.isPending || mergeMutation.isPending
   254→
   255→  // ── Tool mode helpers ────────────────────────────────────────────
   256→  const exitTool = () => {
   257→    setTool(null)
   258→    setSource(null)
   259→    setSeatSelection([])
   260→  }
   261→
   262→  const startTransferTool = () => {
   263→    setTool('transfer')
   264→    setSource(null)
   265→    setSeatSelection([])
   266→  }
   267→
   268→  const startMergeTool = () => {
   269→    setTool('merge')
   270→    setSource(null)
   271→    setSeatSelection([])
   272→  }
   273→
   274→  const startSeatTool = () => {
   275→    setTool('seat')
   276→    setSource(null)
   277→    setSeatSelection([])
   278→  }
   279→
   280→  // Paid/deferred/dirty tables hold no open order — they await the manual
   281→  // bussing/cleaning clicks (amber turnover flow).
   282→  const isCleanupState = (t2: RestaurantTable) =>
   283→    t2.status === 'paid' || t2.status === 'deferred' || t2.status === 'dirty'
   284→
   285→  const tableInteraction = (t2: RestaurantTable): TableInteraction => {
   286→    if (!tool || busy) return busy ? 'ineligible' : 'normal'
   287→    if (tool === 'seat') {
   288→      // Only free tables can host a new party.
   289→      return t2.status === 'free' && t2.openOrderId == null ? 'normal' : 'ineligible'
   290→    }
   291→    if (tool === 'transfer') {
   292→      if (source == null) return t2.openOrderId != null ? 'eligible' : 'ineligible'
   293→      // Destination: free/reserved tables with no open order (paid/deferred/
   294→      // dirty tables await their turnover clicks, they are not destinations).
   295→      return t2.openOrderId == null && t2.status !== 'occupied' && !isCleanupState(t2)
   296→        ? 'eligible'
   297→        : 'ineligible'
   298→    }
   299→    // Merge: source = any occupied table; target = another open order's table.
   300→    if (source == null) return t2.openOrderId != null ? 'eligible' : 'ineligible'
   301→    return t2.openOrderId != null && t2.openOrderId !== source.orderId ? 'eligible' : 'ineligible'
   302→  }
   303→
   304→  const takeawayInteraction = (o: Order): TableInteraction => {
   305→    if (!tool || busy) return busy ? 'ineligible' : 'normal'
   306→    if (tool === 'seat') return 'ineligible'
   307→    if (tool === 'transfer') {
   308→      // Takeaway orders are valid transfer sources, never destinations.
   309→      return source == null ? 'eligible' : 'ineligible'
   310→    }
   311→    if (source == null) return 'eligible'
   312→    return o.id !== source.orderId ? 'eligible' : 'ineligible'
   313→  }
   314→
   315→  const handleTableClick = (t2: RestaurantTable) => {
   316→    if (busy) return
   317→    if (tool === 'seat') {
   318→      if (t2.status !== 'free' || t2.openOrderId != null) return
   319→      setSeatSelection((prev) => {
   320→        if (prev.some((s) => s.id === t2.id)) return prev.filter((s) => s.id !== t2.id)
   321→        if (prev.length >= MAX_SEATING_TABLES) return prev // max 4 tables per seating
   322→        return [...prev, t2]
   323→      })
   324→      return
   325→    }
   326→    if (tool === 'transfer') {
   327→      if (source == null) {
   328→        if (t2.openOrderId == null) return
   329→        setSource({ orderId: t2.openOrderId })
   330→        return
   331→      }
   332→      if (t2.openOrderId != null || t2.status === 'occupied' || isCleanupState(t2)) return
   333→      transferMutation.mutate({ orderId: source.orderId, tableId: t2.id, tableName: t2.name })
   334→      return
   335→    }
   336→    if (tool === 'merge') {
   337→      if (source == null) {
   338→        if (t2.openOrderId == null) return
   339→        setSource({ orderId: t2.openOrderId })
   340→        return
   341→      }
   342→      if (t2.openOrderId == null || t2.openOrderId === source.orderId) return
   343→      mergeMutation.mutate({ sourceId: source.orderId, targetId: t2.openOrderId, targetLabel: t2.name })
   344→      return
   345→    }
   346→    // Normal mode — paid/deferred tiles open the bussing dialog (→ amber
   347→    // 'dirty'), dirty tiles the cleaning dialog (→ free).
   348→    if (isCleanupState(t2) && t2.openOrderId == null) {
   349→      setClearTarget(t2)
   350→      return
   351→    }
   352→    onSelectTable(t2)
   353→  }
   354→
   355→  /** Table-less chip click (takeaway OR delivery order) — opens the order,
   356→   *  or acts as a transfer/merge source while those tools are active. */
   357→  const handleTablelessClick = (o: Order) => {
   358→    if (busy) return
   359→    if (tool === 'seat') return
   360→    if (tool === 'transfer') {
   361→      if (source == null) setSource({ orderId: o.id })
   362→      return
   363→    }
   364→    if (tool === 'merge') {
   365→      if (source == null) {
   366→        setSource({ orderId: o.id })
   367→      } else if (o.id !== source.orderId) {
   368→        mergeMutation.mutate({
   369→          sourceId: source.orderId,
   370→          targetId: o.id,
   371→          targetLabel: o.orderType === 'delivery' ? t('pos.delivery') : t('common.takeaway'),
   372→        })
   373→      }
   374→      return
   375→    }
   376→    onOpenTablelessOrder(o)
   377→  }
   378→
   379→  /** R11: delivery dialog confirm — hand the contact to pos-view. */
   380→  const confirmDelivery = () => {
   381→    if (!deliveryPhoneOk) return
   382→    const phone = deliveryPhone.trim()
   383→    const address = deliveryAddress.trim().slice(0, 200)
   384→    setDeliveryOpen(false)
   385→    setDeliveryPhone('')
   386→    setDeliveryAddress('')
   387→    onDelivery(phone, address)
   388→  }
   389→
   390→  // Seat Party: hand the selection (in order) to pos-view, then exit the tool.
   391→  const continueSeat = () => {
   392→    if (seatSelection.length === 0) return
   393→    const selected = seatSelection
   394→    exitTool()
   395→    onSeatParty(selected)
   396→  }
   397→
   398→  // ── Banner text (transfer/merge — seat has its own banner) ────────
   399→  const bannerText = busy
   400→    ? tool === 'transfer'
   401→      ? t('pos.transferring')
   402→      : t('pos.merging')
   403→    : tool === 'transfer'
   404→      ? source
   405→        ? sourceLabel
   406→          ? t('pos.transferDest', { order: source.orderId, label: sourceLabel })
   407→          : t('pos.transferDestNoLabel', { order: source.orderId })
   408→        : t('pos.transferPick')
   409→      : source
   410→        ? t('pos.mergeTarget', {
   411→            source: sourceLabel ?? `#${source.orderId}`,
   412→          })
   413→        : t('pos.mergePick')
   414→
   415→  const tables = plan?.tables ?? []
   416→
   417→  // ── Hall stats (current floor) + Seat Party availability ─────────
   418→  const hallFreeCount = tables.filter(
   419→    (tb) => tb.status === 'free' && tb.openOrderId == null,
   420→  ).length
   421→  const hallOccupiedCount = tables.filter(
   422→    (tb) => tb.status === 'occupied' || tb.openOrderId != null,
   423→  ).length
   424→  const hallSeats = tables.reduce((sum, tb) => sum + (tb.capacity ?? 0), 0)
   425→  const hallDirtyCount = tables.filter(
   426→    (tb) => tb.status === 'dirty' && tb.openOrderId == null,
   427→  ).length
   428→  const hasFreeTable = hallFreeCount > 0
   429→  const seatSeats = seatSelection.reduce((sum, tb) => sum + (tb.capacity ?? 0), 0)
   430→
   431→  return (
   432→    <div className="rms-scroll h-full overflow-y-auto">
   433→      <div className="mx-auto flex w-full max-w-6xl flex-col">
   434→        {/* ── Control bar ── */}
   435→        <div className="flex flex-wrap items-center justify-between gap-3 border-b border-[#E2E2E0] bg-white px-4 py-3">
   436→          {/* Floor switcher */}
   437→          <div className="flex items-center gap-2">
   438→            <Button
   439→              variant="outline"
   440→              size="icon"
   441→              className="h-11 w-11 rounded-xl"
   442→              disabled={idx <= 0}
   443→              onClick={() => setFloorIdx(idx - 1)}
   444→              aria-label={t('pos.prevFloor')}
   445→            >
   446→              <ChevronLeft className="size-5 rtl:rotate-180" />
   447→            </Button>
   448→            <span className="min-w-[110px] text-center text-lg font-bold">
   449→              {plan?.name ?? t('pos.floors')}
   450→            </span>
   451→            <Button
   452→              variant="outline"
   453→              size="icon"
   454→              className="h-11 w-11 rounded-xl"
   455→              disabled={idx >= floorPlans.length - 1}
   456→              onClick={() => setFloorIdx(idx + 1)}
   457→              aria-label={t('pos.nextFloor')}
   458→            >
   459→              <ChevronRight className="size-5 rtl:rotate-180" />
   460→            </Button>
   461→          </div>
   462→
   463→          {/* Actions */}
   464→          <div className="flex flex-wrap items-center gap-2">
   465→            <Button
   466→              variant="outline"
   467→              className="h-11 rounded-xl"
   468→              disabled={openOrders.length === 0 || !!tool}
   469→              onClick={startTransferTool}
   470→            >
   471→              <ArrowLeftRight className="text-[#714B67]" />
   472→              <span className="hidden sm:inline">{t('pos.transfer')}</span>
   473→            </Button>
   474→            <Button
   475→              variant="outline"
   476→              className="h-11 rounded-xl"
   477→              disabled={openOrders.length < 2 || !!tool}
   478→              onClick={startMergeTool}
   479→            >
   480→              <Combine className="text-[#714B67]" />
   481→              <span className="hidden sm:inline">{t('pos.merge')}</span>
   482→            </Button>
   483→            <Button
   484→              variant="outline"
   485→              className="h-11 rounded-xl border-violet-400 text-violet-700 hover:bg-violet-50 hover:text-violet-700"
   486→              disabled={!hasFreeTable || !!tool}
   487→              onClick={startSeatTool}
   488→              title={t('pos.seatParty')}
   489→            >
   490→              <Armchair className="text-violet-700" />
   491→              <span className="hidden sm:inline">{t('pos.seatParty')}</span>
   492→            </Button>
   493→            <Button
   494→              variant="outline"
   495→              className="h-11 rounded-xl border-emerald-300 text-emerald-700 hover:bg-emerald-50 hover:text-emerald-700"
   496→              disabled={!!tool}
   497→              onClick={onMyShift}
   498→              title={t('pos.myShift')}
   499→            >
   500→              <Wallet className="text-emerald-700" />
   501→              <span className="hidden sm:inline">{t('pos.myShift')}</span>
   502→            </Button>
   503→            <Button
   504→              className="h-11 rounded-xl bg-[#714B67] text-white hover:bg-[#714B67]/90"
   505→              disabled={!!tool}
   506→              onClick={onTakeaway}
   507→            >
   508→              <Plus /> {t('pos.newTakeaway')}
   509→            </Button>
   510→            {/* R11: table-less delivery order (phone + address dialog) */}
   511→            <Button
   512→              variant="outline"
   513→              className="h-11 rounded-xl border-[#714B67]/40 text-[#714B67] hover:bg-[#714B67]/10 hover:text-[#714B67]"
   514→              disabled={!!tool}
   515→              onClick={() => setDeliveryOpen(true)}
   516→              title={t('pos.newDeliveryOrder')}
   517→            >
   518→              <Bike className="text-[#714B67]" />
   519→              <span className="hidden sm:inline">{t('pos.delivery')}</span>
   520→            </Button>
   521→          </div>
   522→        </div>
   523→
   524→        {/* ── Seat Party banner (violet) ── */}
   525→        {tool === 'seat' ? (
   526→          <div

   896→
   897→function TableTile({
   898→  table,
   899→  interaction,
   900→  selected,
   901→  onClick,
   902→  movement,
   903→  onMovementClick,
   904→}: {
   905→  table: RestaurantTable
   906→  interaction: TableInteraction
   907→  /** Seat Party: this free table is part of the current selection. */
   908→  selected?: boolean
   909→  onClick: () => void
   910→  /** R9: pending AI guest-move suggestion originating at this table. */
   911→  movement?: MovementCandidateDTO | null
   912→  onMovementClick?: () => void
   913→}) {
   914→  const { t } = useI18n()
   915→  const occupied = table.status === 'occupied' || table.openOrderId != null
   916→  const reserved = table.status === 'reserved' && table.openOrderId == null
   917→  // Round5/6: bill settled (awaiting bussing) / deferred check (client left)
   918→  // / dirty = bussed, amber, awaiting the cleaning click.
   919→  const paid = table.status === 'paid' && table.openOrderId == null
   920→  const deferred = table.status === 'deferred' && table.openOrderId == null
   921→  const dirty = table.status === 'dirty' && table.openOrderId == null
   922→  // Round/oval silhouettes get extra horizontal padding so the centered
   923→  // content stays inside the circle (overflow-hidden + truncation clip it).
   924→  const isRound = table.shape === 'round' || table.shape === 'oval'
   925→  // Corner badges: on circles the top corner is clipped by the silhouette,
   926→  // so they sit at the top-center edge instead.
   927→  const cornerBadgePos = isRound ? 'left-1/2 top-1 -translate-x-1/2' : 'end-1 top-1'
   928→
   929→  // Odoo duration "heat": occupied tables tint by how long guests are seated.
   930→  const mins =
   931→    occupied && table.openOrderSince
   932→      ? Math.max(0, (Date.now() - new Date(table.openOrderSince).getTime()) / 60000)
   933→      : 0
   934→
   935→  const heat = occupied
   936→    ? mins < 15
   937→      ? {
   938→          tile: 'border-emerald-200 bg-emerald-50',
   939→          text: 'text-emerald-900',
   940→          amount: 'text-emerald-700',
   941→          meta: 'text-emerald-600',
   942→        }
   943→      : mins < 45
   944→        ? {
   945→            tile: 'border-amber-200 bg-amber-50',
   946→            text: 'text-amber-900',
   947→            amount: 'text-amber-700',
   948→            meta: 'text-amber-600',
   949→          }
   950→        : mins < 90
   951→          ? {
   952→              tile: 'border-orange-300 bg-orange-100',
   953→              text: 'text-orange-900',
   954→              amount: 'text-orange-700',
   955→              meta: 'text-orange-600',
   956→            }
   957→          : {
   958→              tile: 'border-rose-300 bg-rose-100',
   959→              text: 'text-rose-900',
   960→              amount: 'text-rose-700',
   961→              meta: 'text-rose-600',
   962→            }
   963→    : null
   964→
   965→  return (
   966→    <button
   967→      type="button"
   968→      disabled={interaction === 'ineligible'}
   969→      onClick={onClick}
   970→      className={cn(
   971→        'relative flex min-w-0 flex-col items-center justify-center gap-1 overflow-hidden border p-3 text-center shadow-sm transition hover:shadow-md active:scale-[0.97]',
   972→        shapeClasses(table.shape),
   973→        isRound && 'px-4',
   974→        occupied && heat
   975→          ? cn(heat.tile, heat.text)
   976→          : paid
   977→            ? 'border-emerald-600 bg-emerald-600 text-white'
   978→            : deferred
   979→              ? 'border-violet-400 bg-violet-100 text-violet-900 ring-1 ring-violet-400'
   980→              : dirty
   981→                ? 'border-amber-500 bg-amber-200 text-amber-900 ring-2 ring-amber-400'
   982→                : reserved
   983→                  ? 'border-amber-300 bg-amber-50/70 text-amber-900 ring-1 ring-amber-400'
   984→                  : 'border-[#E2E2E0] bg-white text-stone-500',
   985→        interaction === 'ineligible' && 'cursor-not-allowed opacity-40',
   986→        interaction === 'eligible' && 'ring-2 ring-[#714B67] ring-offset-1',
   987→        selected && 'ring-2 ring-violet-600 ring-offset-1',
   988→      )}
   989→    >
   990→      {/* FREE — clean white tile with capacity */}
   991→      {!occupied && !reserved && !paid && !deferred && !dirty && (
   992→        <>
   993→          <p className="max-w-full truncate text-base font-bold leading-tight text-stone-500">
   994→            {table.name}
   995→          </p>
   996→          <p className="flex items-center gap-1 text-xs text-stone-400">
   997→            <Users className="size-3.5" aria-hidden />
   998→            {table.capacity} {t('common.seats')}
   999→          </p>
  1000→        </>
  1001→      )}
  1002→
  1003→      {/* OCCUPIED — duration "heat" tint, amount + guests + elapsed */}
  1004→      {occupied && (
  1005→        <>
  1006→          <p className="max-w-full truncate text-base font-bold leading-tight">{table.name}</p>
  1007→          <p className={cn('text-lg font-extrabold tabular-nums leading-none', heat?.amount)}>
  1008→            {formatCurrency(table.openOrderTotal ?? 0)}
  1009→          </p>
  1010→          <p className={cn('flex items-center justify-center gap-1.5 text-[11px]', heat?.meta)}>
