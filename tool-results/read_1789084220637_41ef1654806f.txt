    19→/** Round a number to 2 decimal places (money persistence boundary). */
    20→export function round2(value: number): number {
    21→  return Math.round(value * 100) / 100

    24→/** Parse a numeric route param, throwing a 400 ApiError when invalid. */
    25→export function parseId(value: string, label = 'id'): number {
    26→  const id = Number(value)
    27→  if (!Number.isInteger(id) || id <= 0) {
    28→    throw new ApiError(`Invalid ${label}`, 400)
    29→  }
    30→  return id

    33→/**
    34→ * Resolve the numeric user id from a session payload. NOTE: auth.getSessionUser
    35→ * currently returns the DB user row ({ id, ... }) while the declared
    36→ * SessionPayload shape uses `userId` — support both spellings so orders are
    37→ * always attributed to the logged-in user.
    38→ */
    39→export function sessionUserId(session: SessionPayload): number {
    40→  const alt = session as unknown as { userId?: number; id?: number }
    41→  const id = alt.userId ?? alt.id
    42→  if (id == null || !Number.isFinite(id)) {
    43→    throw new ApiError('Unauthorized', 401)
    44→  }
    45→  return id
    46→}

    91→/** Parse the selected_modifiers JSON column into a clean list (R8). */
    92→export function parseSelectedModifiers(raw: string | null | undefined): SelectedModifier[] | null {
    93→  if (!raw) return null
    94→  try {
    95→    const parsed: unknown = JSON.parse(raw)
    96→    if (!Array.isArray(parsed) || parsed.length === 0) return null
    97→    const list: SelectedModifier[] = []
    98→    for (const entry of parsed) {
    99→      const m = entry as { id?: unknown; name?: unknown; nameAr?: unknown; priceDelta?: unknown }
   100→      const id = Number(m.id)
   101→      const priceDelta = Number(m.priceDelta)
   102→      if (!Number.isInteger(id) || id <= 0 || !Number.isFinite(priceDelta)) continue
   103→      list.push({
   104→        id,
   105→        name: String(m.name ?? ''),
   106→        nameAr: m.nameAr == null ? null : String(m.nameAr),
   107→        priceDelta: round2(priceDelta),
   108→      })
   109→    }
   110→    return list.length > 0 ? list : null
   111→  } catch {
   112→    return null
   113→  }
   114→}

   132→/** Map a loaded order (with ORDER_INCLUDE relations) to the API `Order` payload. */
   133→export function serializeOrder(order: OrderWithRelations): Order {
   134→  const paidAmount = order.payments.reduce((sum, p) => sum + p.amount, 0)
   135→  const total = round2(order.totalAmount)
   136→  return {
   137→    id: order.id,
   138→    tableId: order.tableId,
   139→    table: order.table,
   140→    userId: order.userId,
   141→    user: order.user,
   142→    status: order.status,
   143→    subtotalAmount: round2(order.subtotalAmount),
   144→    totalAmount: total,
   145→    discountAmount: round2(order.discountAmount),
   146→    taxAmount: round2(order.taxAmount),
   147→    serviceTaxAmount: round2(order.serviceTaxAmount),
   148→    discountReason: order.discountReason,
   149→    clientName: order.clientName,
   150→    extraTableIds: parseExtraTableIds(order.extraTableIds),
   151→    paidAmount: round2(paidAmount),
   152→    remainingAmount: round2(total - paidAmount),
   153→    guests: order.guests,
   154→    createdAt: order.createdAt.toISOString(),
   155→    closedAt: order.closedAt ? order.closedAt.toISOString() : null,
   156→    items: order.items.map(serializeOrderItem),
   157→    payments: order.payments.map(serializePayment),
   158→  }

   170→/** Parse the JSON extraTableIds column into a clean id list. */
   171→export function parseExtraTableIds(raw: string | null | undefined): number[] {
   172→  if (!raw) return []
   173→  try {
   174→    const parsed: unknown = JSON.parse(raw)
   175→    if (!Array.isArray(parsed)) return []
   176→    return parsed
   177→      .map(Number)
   178→      .filter((id) => Number.isInteger(id) && id > 0)
   179→  } catch {
   180→    return []
   181→  }

   184→/** All table ids bound to an order: primary tableId + extraTableIds. */
   185→export function orderTableIds(order: { tableId: number | null; extraTableIds: string | null }): number[] {
   186→  const ids: number[] = []
   187→  if (order.tableId != null) ids.push(order.tableId)
   188→  for (const id of parseExtraTableIds(order.extraTableIds)) {
   189→    if (!ids.includes(id)) ids.push(id)
   190→  }
   191→  return ids
   192→}

   194→/**
   195→ * Find an OPEN order that references the table — either as its primary
   196→ * table OR as one of its merged-seating extra tables. Used to guard
   197→ * transfer/destination tables and table release logic.
   198→ */
   199→export async function findOpenOrderOnTable(
   200→  tableId: number,
   201→  excludeOrderId?: number,
   202→): Promise<{ id: number } | null> {
   203→  const candidates = await db.order.findMany({
   204→    where: {
   205→      status: 'open',
   206→      OR: [{ tableId }, { extraTableIds: { not: null } }],
   207→    },
   208→    select: { id: true, tableId: true, extraTableIds: true },
   209→  })
   210→  for (const order of candidates) {
   211→    if (excludeOrderId != null && order.id === excludeOrderId) continue
   212→    if (orderTableIds(order).includes(tableId)) return { id: order.id }
   213→  }
   214→  return null
   215→}
   216→
   217→/**
   218→ * Set every table of an order to a target status — each table only when
   219→ * no OTHER open order still references it (as primary or extra table).
   220→ * Tables that are already 'free' (cleaned) or 'dirty' (amber, bussed and
   221→ * awaiting the cleaning click) are never re-marked by a later settlement —
   222→ * the manual cleaning flow always moves forward.
   223→ */
   224→export async function setTablesStatusForOrder(
   225→  order: { id: number; tableId: number | null; extraTableIds: string | null },
   226→  target: 'free' | 'paid' | 'deferred' | 'occupied',
   227→): Promise<void> {
   228→  const tableIds = orderTableIds(order)
   229→  for (const tableId of tableIds) {
   230→    const otherOpen = await findOpenOrderOnTable(tableId, order.id)
   231→    if (otherOpen) continue
   232→    const table = await db.restaurantTable.findUnique({
   233→      where: { id: tableId },
   234→      select: { status: true },
   235→    })
   236→    if (!table) continue
   237→    // never re-dirty a table that has already been bussed ('dirty') or
   238→    // cleaned ('free') — later settlements keep the manual cleaning state
   239→    if (
   240→      (target === 'paid' || target === 'deferred') &&
   241→      (table.status === 'free' || table.status === 'dirty')
   242→    ) {
   243→      continue
   244→    }
   245→    if (table.status === target) continue
   246→    await db.restaurantTable.update({ where: { id: tableId }, data: { status: target } })
   247→  }
   248→}

   250→/**
   251→ * Recompute an order's money fields from its current items:
   252→ * subtotal = Σ qty × unitPrice, discount clamped to [0, subtotal],
   253→ * VAT = (subtotal − discount) × TAX_RATE (14%),
   254→ * service tax = (subtotal − discount) × SERVICE_TAX_RATE (12%),
   255→ * total = subtotal − discount + VAT + service tax.
   256→ * Persists the result and returns the serialized order.
   257→ */
   258→export async function recomputeTotals(orderId: number): Promise<Order> {
   259→  const order = await db.order.findUnique({ where: { id: orderId }, include: { items: true } })
   260→  if (!order) throw new ApiError('Order not found', 404)
   261→  const subtotal = round2(
   262→    order.items.reduce((sum, item) => sum + item.quantity * item.unitPrice, 0),
   263→  )
   264→  const discount = Math.min(Math.max(order.discountAmount, 0), subtotal)
   265→  const base = round2(subtotal - discount)
   266→  const tax = round2(base * TAX_RATE)
   267→  const serviceTax = round2(base * SERVICE_TAX_RATE)
   268→  const total = round2(base + tax + serviceTax)
   269→  const updated = await db.order.update({
   270→    where: { id: orderId },
   271→    data: {
   272→      subtotalAmount: subtotal,
   273→      discountAmount: round2(discount),
   274→      taxAmount: tax,
   275→      serviceTaxAmount: serviceTax,
   276→      totalAmount: total,
   277→    },
   278→    include: ORDER_INCLUDE,
   279→  })
   280→  return serializeOrder(updated)
   281→}

   285→export type ValidatedOrderItem = {
   286→  productId: number
   287→  quantity: number
   288→  unitPrice: number
   289→  notes: string | null
   290→  course: string
   291→  /** R8: option snapshots rebuilt from the DB (never trusting client data);
   292→   *  null when the item carries no options (today's behavior). */
   293→  selectedModifiers: SelectedModifier[] | null
   294→  /** R8: Σ snapshot priceDelta (0 when none) — unitPrice already includes it */
   295→  modifiersPriceDelta: number
   296→}
   297→
   298→/**
   299→ * Validate an `items` array from a request body: non-empty, each product
   300→ * must exist / be active / be sellable, quantity > 0 and course ∈ COURSES
   301→ * (default 'main'). unitPrice = product.price.
   302→ *
   303→ * R8: each item may carry `selectedModifiers` [{ id }] — every id must be an
   304→ * ACTIVE Modifier of an ACTIVE group ATTACHED to the item's product; group
   305→ * minSelect/maxSelect are enforced per item; duplicates are rejected. The
   306→ * snapshot (name/nameAr/priceDelta) is always rebuilt from DB rows, and
   307→ * unitPrice = round2(product.price + Σ deltas).
   308→ */
   309→export async function validateOrderItems(items: unknown): Promise<ValidatedOrderItem[]> {
   310→  if (!Array.isArray(items) || items.length === 0) {
   311→    throw new ApiError('Order must contain at least one item', 400)
   312→  }
   313→  const raw = items as {
   314→    productId?: unknown
   315→    quantity?: unknown
   316→    notes?: unknown
   317→    course?: unknown
   318→    selectedModifiers?: unknown
   319→  }[]
   320→
   321→  const productIds = new Set<number>()
   322→  for (const item of raw) {
   323→    const productId = Number(item?.productId)
   324→    const quantity = Number(item?.quantity)
   325→    if (!Number.isInteger(productId) || productId <= 0) {
   326→      throw new ApiError('Invalid productId in order items', 400)
   327→    }
   328→    if (!Number.isFinite(quantity) || quantity <= 0) {
   329→      throw new ApiError(`Invalid quantity for product ${productId}`, 400)
   330→    }
   331→    if (item?.selectedModifiers != null && !Array.isArray(item.selectedModifiers)) {
   332→      throw new ApiError('selectedModifiers must be an array', 400)
   333→    }
   334→    productIds.add(productId)
   335→  }
   336→
   337→  const products = await db.product.findMany({
   338→    where: { id: { in: Array.from(productIds) } },
   339→    select: { id: true, name: true, price: true, active: true, isSellable: true },
   340→  })
   341→  const productById = new Map(products.map((p) => [p.id, p]))
   342→
   343→  // ── R8 phase A: parse the requested option ids per item ────────────
   344→  // (integer ids > 0, no duplicates within one item — snapshots come later)
   345→  const requestedIdsPerItem: (number[] | null)[] = raw.map((item) => {
   346→    const list = item?.selectedModifiers
   347→    if (!Array.isArray(list) || list.length === 0) return null
   348→    const productId = Number(item.productId)
   349→    const productLabel = productById.get(productId)?.name ?? String(productId)
   350→    const ids: number[] = []
   351→    const seen = new Set<number>()
   352→    for (const entry of list) {
   353→      const id = Number((entry as { id?: unknown })?.id)
   354→      if (!Number.isInteger(id) || id <= 0) {
   355→        throw new ApiError(`Invalid option for product "${productLabel}"`, 400)
   356→      }
   357→      if (seen.has(id)) {
   358→        throw new ApiError(
   359→          `Duplicate option id ${id} for product "${productLabel}"`,
   360→          400,
   361→        )
   362→      }
   363→      seen.add(id)
   364→      ids.push(id)
   365→    }
   366→    return ids
   367→  })
   368→
   369→  // ── R8 phase B: load every referenced modifier (with its group) and the
   370→  // group attachments of every product — two queries for the whole payload.
   371→  const allModifierIds = new Set<number>()
   372→  for (const ids of requestedIdsPerItem) {
   373→    if (ids) for (const id of ids) allModifierIds.add(id)
   374→  }
   375→  type ModifierRow = {
   376→    id: number
   377→    name: string
   378→    nameAr: string | null
   379→    priceDelta: number
   380→    active: boolean
   381→    groupId: number
   382→    group: { id: number; name: string; active: boolean }
   383→  }
   384→  const [modifierRows, linkRows] = await Promise.all([
   385→    allModifierIds.size === 0
   386→      ? Promise.resolve<ModifierRow[]>([])
   387→      : db.modifier.findMany({
   388→          where: { id: { in: Array.from(allModifierIds) } },
   389→          select: {
   390→            id: true,
   391→            name: true,
   392→            nameAr: true,
   393→            priceDelta: true,
   394→            active: true,
   395→            groupId: true,
   396→            group: { select: { id: true, name: true, active: true } },
   397→          },
   398→        }),
   399→    db.productModifierGroup.findMany({
   400→      where: { productId: { in: Array.from(productIds) } },
   401→      include: { modifierGroup: true },
   402→    }),
   403→  ])
   404→  const modifierById = new Map(modifierRows.map((m) => [m.id, m]))
   405→  // ACTIVE groups attached per product (constraints apply to these only)
   406→  const attachedGroupsByProduct = new Map<number, typeof linkRows[number]['modifierGroup'][]>()
   407→  for (const link of linkRows) {
   408→    if (!link.modifierGroup.active) continue
   409→    const list = attachedGroupsByProduct.get(link.productId) ?? []
   410→    list.push(link.modifierGroup)
   411→    attachedGroupsByProduct.set(link.productId, list)
   412→  }
   413→
   414→  return raw.map((item, index) => {
   415→    const productId = Number(item.productId)
   416→    const product = productById.get(productId)
   417→    if (!product) throw new ApiError(`Product ${productId} not found`, 400)
   418→    if (!product.active || !product.isSellable) {
   419→      throw new ApiError(`Product "${product.name}" is not available`, 400)
   420→    }
   421→    const course = item.course == null ? 'main' : String(item.course)
   422→    if (!(COURSES as readonly string[]).includes(course)) {
   423→      throw new ApiError(`Invalid course "${course}"`, 400)
   424→    }
   425→
   426→    // ── R8 phase C: snapshot from DB rows + group constraint checks ──
   427→    const requestedIds = requestedIdsPerItem[index]
   428→    const attached = attachedGroupsByProduct.get(productId) ?? []
   429→    const attachedGroupIds = new Set(attached.map((g) => g.id))
   430→    const selected: SelectedModifier[] = []
   431→    const countByGroup = new Map<number, number>()
   432→    if (requestedIds) {
   433→      for (const id of requestedIds) {
   434→        const mod = modifierById.get(id)
   435→        if (
   436→          !mod ||
   437→          !mod.active ||
   438→          !mod.group.active ||
   439→          !attachedGroupIds.has(mod.groupId)
   440→        ) {
   441→          throw new ApiError(`Invalid option for product "${product.name}"`, 400)
   442→        }
   443→        countByGroup.set(mod.groupId, (countByGroup.get(mod.groupId) ?? 0) + 1)
   444→        selected.push({
   445→          id: mod.id,
   446→          name: mod.name,
   447→          nameAr: mod.nameAr,
   448→          priceDelta: round2(mod.priceDelta),
   449→        })
   450→      }
   451→    }
   452→    for (const group of attached) {
   453→      const count = countByGroup.get(group.id) ?? 0
   454→      if (count < group.minSelect) {
   455→        throw new ApiError(
   456→          `Option group "${group.name}" for product "${product.name}" requires at least ${group.minSelect} option(s)`,
   457→          400,
   458→        )
   459→      }
   460→      if (count > group.maxSelect) {
   461→        throw new ApiError(
   462→          `Option group "${group.name}" for product "${product.name}" allows at most ${group.maxSelect} option(s)`,
   463→          400,
   464→        )
   465→      }
   466→    }
   467→
   468→    const modifiersPriceDelta = round2(
   469→      selected.reduce((sum, m) => sum + m.priceDelta, 0),
   470→    )
   471→    return {
   472→      productId,
   473→      quantity: Number(item.quantity),
   474→      unitPrice: round2(product.price + modifiersPriceDelta),
   475→      notes: item.notes == null ? null : String(item.notes),
   476→      course,
   477→      selectedModifiers: selected.length > 0 ? selected : null,
   478→      modifiersPriceDelta,
   479→    }
   480→  })
   481→}

   483→// ─── Stock availability ─────────────────────────────────────────────
   484→
   485→/**
   486→ * Verify that adding `newItems` to an order keeps enough stock.
   487→ * Quantities of the SAME products already on the existing open order are
   488→ * subtracted (they were validated when added and are deducted only once).
   489→ * Checks both direct stockable product stock and, via the recipe (BOM),
   490→ * each ingredient's aggregated requirement.
   491→ */
   492→export async function checkStockAvailability(
   493→  newItems: { productId: number; quantity: number }[],
   494→  existingOrderId?: number,
   495→): Promise<void> {
   496→  // Net additional quantity per product
   497→  const needs = new Map<number, number>()
   498→  for (const item of newItems) {
   499→    const quantity = Number(item.quantity)
   500→    if (!Number.isFinite(quantity) || quantity <= 0) continue
   501→    needs.set(item.productId, (needs.get(item.productId) ?? 0) + quantity)
   502→  }
   503→  if (needs.size === 0) return
   504→
   505→  if (existingOrderId != null) {
   506→    const existing = await db.orderItem.findMany({
   507→      where: { orderId: existingOrderId, productId: { in: Array.from(needs.keys()) } },
   508→      select: { productId: true, quantity: true },
   509→    })
   510→    for (const row of existing) {
   511→      if (row.productId == null) continue
   512→      needs.set(row.productId, (needs.get(row.productId) ?? 0) - row.quantity)
   513→    }
   514→  }
   515→
   516→  const requiredProductIds = Array.from(needs.entries())
   517→    .filter(([, need]) => need > 0)
   518→    .map(([id]) => id)
   519→  if (requiredProductIds.length === 0) return
   520→
   521→  const products = await db.product.findMany({
   522→    where: { id: { in: requiredProductIds } },
   523→    include: { recipeFor: { include: { ingredient: true } } },
   524→  })
   525→
   526→  // Aggregated ingredient requirements across all dishes
   527→  const ingredientNeeds = new Map<number, { name: string; stock: number; need: number }>()
   528→
   529→  for (const product of products) {
   530→    const need = needs.get(product.id) ?? 0
   531→    if (product.isStockable) {
   532→      if (product.stock + MONEY_EPSILON < need) {
   533→        throw new ApiError(
   534→          `Insufficient stock for "${product.name}" (available ${round2(product.stock)}, needed ${round2(need)})`,
   535→          400,
   536→        )
   537→      }
   538→    }
   539→    for (const component of product.recipeFor) {
   540→      const ingredientNeed = component.quantity * need
   541→      const previous = ingredientNeeds.get(component.ingredientId)
   542→      ingredientNeeds.set(component.ingredientId, {
   543→        name: component.ingredient.name,
   544→        stock: component.ingredient.stock,
   545→        need: (previous?.need ?? 0) + ingredientNeed,
   546→      })
   547→    }
   548→  }
   549→
   550→  for (const info of ingredientNeeds.values()) {
   551→    if (info.stock + MONEY_EPSILON < info.need) {
   552→      throw new ApiError(
   553→        `Insufficient stock for "${info.name}" (available ${round2(info.stock)}, needed ${round2(info.need)})`,
   554→        400,
   555→      )
   556→    }
   557→  }
   558→}
   559→
   560→// ─── Inventory deduction on close ───────────────────────────────────
   561→
   562→/**
   563→ * Deduct inventory for a paid order — IDEMPOTENT: skipped when any
   564→ * InventoryTransaction with reason 'sale' already exists for the order.
   565→ * Deducts stockable product stock directly and each recipe ingredient
   566→ * (component.quantity × item.quantity), writing 'sale' transactions.
   567→ */
   568→export async function deductInventoryForOrder(orderId: number): Promise<void> {
   569→  const existing = await db.inventoryTransaction.findFirst({
   570→    where: { orderId, reason: 'sale' },
   571→    select: { id: true },
   572→  })
   573→  if (existing) return
   574→
   575→  const order = await db.order.findUnique({
   576→    where: { id: orderId },
   577→    include: {
   578→      items: {
   579→        include: { product: { include: { recipeFor: { include: { ingredient: true } } } } },
   580→      },
   581→    },
   582→  })
   583→  if (!order) return
   584→
   585→  await db.$transaction(async (tx) => {
   586→    for (const item of order.items) {
   587→      const product = item.product
   588→      if (!product) continue
   589→
   590→      if (product.isStockable) {
   591→        await tx.product.update({
   592→          where: { id: product.id },
   593→          data: { stock: { decrement: item.quantity } },
   594→        })
   595→        await tx.inventoryTransaction.create({
   596→          data: {
   597→            productId: product.id,
   598→            quantityChange: -item.quantity,
   599→            reason: 'sale',
   600→            orderId,
   601→          },
   602→        })
   603→      }
   604→
   605→      for (const component of product.recipeFor) {
   606→        const quantity = component.quantity * item.quantity
   607→        if (quantity === 0) continue
   608→        await tx.product.update({
   609→          where: { id: component.ingredientId },
   610→          data: { stock: { decrement: quantity } },
   611→        })
   612→        await tx.inventoryTransaction.create({
   613→          data: {
   614→            productId: component.ingredientId,
   615→            quantityChange: -quantity,
   616→            reason: 'sale',
   617→            orderId,
   618→          },
   619→        })
   620→      }
   621→    }
   622→  })
   623→}

   625→// ─── Close / table release ──────────────────────────────────────────
   626→
   627→/**
   628→ * Auto-close an order when fully paid: status 'paid' + closedAt, deduct
   629→ * inventory (idempotent). Works for OPEN and DEFERRED orders — a deferred
   630→ * check is settled by recording the remaining payments. After closing,
   631→ * the order's tables become 'paid' (bill settled, awaiting the bussing
   632→ * click) unless they were already bussed to 'dirty' or cleaned to 'free'.
   633→ * Returns whether this call closed the order.
   634→ */
   635→export async function closeOrderIfFullyPaid(orderId: number): Promise<{ closed: boolean }> {
   636→  const order = await db.order.findUnique({
   637→    where: { id: orderId },
   638→    include: { payments: true },
   639→  })
   640→  if (!order) throw new ApiError('Order not found', 404)
   641→  if (order.status !== 'open' && order.status !== 'deferred') return { closed: false }
   642→
   643→  const paidAmount = order.payments.reduce((sum, p) => sum + p.amount, 0)
   644→  if (paidAmount < order.totalAmount - MONEY_EPSILON) return { closed: false }
   645→
   646→  await db.order.update({
   647→    where: { id: orderId },
   648→    data: { status: 'paid', closedAt: new Date() },
   649→  })
   650→  await deductInventoryForOrder(orderId)
   651→
   652→  await setTablesStatusForOrder(order, 'paid')
   653→  return { closed: true }
   654→}

   656→/**
   657→ * Defer a check: order → status 'deferred' with the client's name. The
   658→ * client leaves; the order's tables switch to 'deferred' (tap to bus →
   659→ * amber 'dirty' → tap to clean → free) so the check stays outstanding
   660→ * but the seating is vacated.
   661→ */
   662→export async function deferOrder(
   663→  orderId: number,
   664→  clientName: string,
   665→): Promise<OrderWithRelations> {
   666→  const order = await db.order.findUnique({
   667→    where: { id: orderId },
   668→    include: { items: { select: { id: true } } },
   669→  })
   670→  if (!order) throw new ApiError('Order not found', 404)
   671→  if (order.status !== 'open') {
   672→    throw new ApiError('Only open orders can be deferred', 400)
   673→  }
   674→  if (order.items.length === 0) {
   675→    throw new ApiError('Cannot defer an empty order', 400)
   676→  }
   677→  const updated = await db.order.update({
   678→    where: { id: orderId },
   679→    data: { status: 'deferred', clientName },
   680→  })
   681→  await setTablesStatusForOrder(updated, 'deferred')
   682→  return db.order.findUniqueOrThrow({ where: { id: orderId }, include: ORDER_INCLUDE })
   683→}

   685→/** Free a table if no other open order references it (primary or extra). */
   686→export async function freeTableIfUnused(tableId: number, excludeOrderId?: number): Promise<void> {
   687→  const otherOpenOrder = await findOpenOrderOnTable(tableId, excludeOrderId)
   688→  if (!otherOpenOrder) {
   689→    await db.restaurantTable.update({ where: { id: tableId }, data: { status: 'free' } })
   690→  }
   691→}

   693→/**
   694→ * R9 vision: re-house an open order (primary + merged extras) at a single
   695→ * destination table — the exact semantics of POST /api/orders/[id]/transfer,
   696→ * extracted so the human-confirmed AI movement flow reuses them.
   697→ *
   698→ * Runs inside the caller's transaction: primary table moves, extras are
   699→ * released, previous tables are freed when no OTHER open order still
   700→ * references them, and the destination table is marked occupied.
   701→ */
   702→export async function rehouseOpenOrder(
   703→  tx: Prisma.TransactionClient,
   704→  order: { id: number; tableId: number | null; extraTableIds: string | null },
   705→  targetTableId: number,
   706→): Promise<void> {
   707→  const previousTableIds = orderTableIds(order)
   708→  await tx.order.update({
   709→    where: { id: order.id },
   710→    data: { tableId: targetTableId, extraTableIds: null },
   711→  })
   712→  for (const previousTableId of previousTableIds) {
   713→    if (previousTableId === targetTableId) continue
   714→    const stillOpen = await findOpenOrderOnTable(previousTableId, order.id)
   715→    if (!stillOpen) {
   716→      await tx.restaurantTable.update({
   717→        where: { id: previousTableId },
   718→        data: { status: 'free' },
   719→      })
   720→    }
   721→  }
   722→  await tx.restaurantTable.update({
   723→    where: { id: targetTableId },
   724→    data: { status: 'occupied' },
   725→  })
   726→}

   756→/**
   757→ * Serialize tables with POS polling extras. For every table bound to an
   758→ * OPEN order (primary OR merged-seating extra table) attach openOrderId,
   759→ * openOrderTotal, openOrderItemCount, openOrderSince, openOrderGuests and
   760→ * openOrderMerged, and force live status 'occupied'. Tables holding a
   761→ * DEFERRED check (status 'deferred') additionally get deferredClientName
   762→ * + deferredOrderId so the floor can show who owes the deferred bill.
   763→ * Open + deferred orders are fetched in TWO queries (tables referenced
   764→ * directly or via extraTableIds, parsed in memory).
   765→ */
   766→export async function serializeTablesWithOpenOrders(tables: TableRow[]): Promise<RestaurantTable[]> {
   767→  if (tables.length === 0) return []
   768→  const tableIds = tables.map((t) => t.id)
   769→  const [openOrders, deferredOrders] = await Promise.all([
   770→    db.order.findMany({
   771→      where: {
   772→        status: 'open',
   773→        OR: [{ tableId: { in: tableIds } }, { extraTableIds: { not: null } }],
   774→      },
   775→      orderBy: { createdAt: 'asc' },
   776→      select: {
   777→        id: true,
   778→        tableId: true,
   779→        totalAmount: true,
   780→        guests: true,
   781→        createdAt: true,
   782→        extraTableIds: true,
   783→        items: { select: { id: true, quantity: true, unitPrice: true } },
   784→      },
   785→    }),
   786→    db.order.findMany({
   787→      where: {
   788→        status: 'deferred',
   789→        OR: [{ tableId: { in: tableIds } }, { extraTableIds: { not: null } }],
   790→      },
   791→      orderBy: { createdAt: 'desc' },
   792→      select: {
   793→        id: true,
   794→        tableId: true,
   795→        clientName: true,
   796→        extraTableIds: true,
   797→        totalAmount: true,
   798→        payments: { select: { amount: true } },
   799→      },
   800→    }),
   801→  ])
   802→
   803→  // table id → first open order bound to it (primary or extra table)
   804→  const openOrderByTable = new Map<
   805→    number,
   806→    (typeof openOrders)[number] & { merged: boolean }
   807→  >()
   808→  for (const order of openOrders) {
   809→    const ids = orderTableIds(order)
   810→    const merged = ids.length > 1
   811→    for (const id of ids) {
   812→      if (!openOrderByTable.has(id)) openOrderByTable.set(id, { ...order, merged })
   813→    }
   814→  }
   815→
   816→  // table id → newest deferred order bound to it (for 'deferred' tables)
   817→  const deferredOrderByTable = new Map<
   818→    number,
   819→    { id: number; clientName: string | null; remaining: number }
   820→  >()
   821→  for (const order of deferredOrders) {
   822→    const paidAmount = order.payments.reduce((sum, p) => sum + p.amount, 0)
   823→    const info = {
   824→      id: order.id,
   825→      clientName: order.clientName,
   826→      remaining: round2(order.totalAmount - paidAmount),
   827→    }
   828→    for (const id of orderTableIds(order)) {
   829→      if (!deferredOrderByTable.has(id)) deferredOrderByTable.set(id, info)
   830→    }
   831→  }
   832→
   833→  return tables.map((table) => {
   834→    const base = serializeTable(table)
   835→    if (base.status === 'deferred') {
   836→      const deferred = deferredOrderByTable.get(table.id)
   837→      if (deferred) {
   838→        return {
   839→          ...base,
   840→          deferredClientName: deferred.clientName,
   841→          deferredOrderId: deferred.id,
   842→        }
   843→      }
   844→      return base
   845→    }
   846→    const open = openOrderByTable.get(table.id)
   847→    if (!open) return base
   848→    const itemCount = open.items.reduce((sum, item) => sum + item.quantity, 0)
   849→    return {
   850→      ...base,
   851→      status: 'occupied',
   852→      openOrderId: open.id,
   853→      openOrderTotal: round2(open.totalAmount),
   854→      openOrderItemCount: round2(itemCount),
   855→      openOrderSince: open.createdAt.toISOString(),
   856→      openOrderGuests: open.guests,
   857→      openOrderMerged: open.merged,
   858→    }
   859→  })
   860→}

   864→/**
   865→ * Serialize floor plans (with already-filtered table rows) attaching the
   866→ * same open-order polling extras as /api/tables/status.
   867→ */
   868→export async function serializeFloorPlans(plans: FloorPlanRow[]): Promise<FloorPlan[]> {
   869→  if (plans.length === 0) return []
   870→  const tablesWithExtras = await serializeTablesWithOpenOrders(plans.flatMap((p) => p.tables))
   871→  const extrasById = new Map(tablesWithExtras.map((t) => [t.id, t]))
   872→  return plans.map((plan) => ({
   873→    id: plan.id,
   874→    name: plan.name,
   875→    backgroundImage: plan.backgroundImage,
   876→    active: plan.active,
   877→    tables: plan.tables
   878→      .map((table) => extrasById.get(table.id))
   879→      .filter((table): table is RestaurantTable => table != null),
   880→  }))
   881→}
